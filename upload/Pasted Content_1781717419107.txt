    def generate_timesfm_features(self, history_matrix):
        """
        Converts 2D/3D historical matrices into future point predictions using TimesFM.
        Input shape: (num_entities, history_length)
        """
        # Ensure array is correctly formatted for the model
        data_arr = np.array(history_matrix, dtype=np.float32)
        
        # Handle truncating or padding to fit required context_len
        if data_arr.shape[1] > self.context_len:
            data_arr = data_arr[:, -self.context_len:]
        elif data_arr.shape[1] < self.context_len:
            pad_width = self.context_len - data_arr.shape[1]
            data_arr = np.pad(data_arr, ((0,0), (pad_width,0)), 'edge')

        # Run Zero-Shot Inference
        padding_mask = np.ones_like(data_arr, dtype=np.float32)
        forecast, _ = self.tfm.forecast(inputs=data_arr, freq=0, mask=padding_mask)
        
        # Extract mean point forecasts for the horizon target
        return forecast[:, 0]

    def build_master_feature_matrix(self, team_a_history, team_b_history, context_data):

    def compute_bet_signals(self, X_target, market_odds_a, market_odds_b):
        """
        Calculates pure win probabilities and identifies high-value positive expected value (+EV) bets.
        """
        probabilities = self.meta_learner.predict_proba(X_target)
        results = []
        
        for i, prob in enumerate(probabilities):
            prob_a = prob[1] # Probability Team A wins
            prob_b = prob[0] # Probability Team B wins
            
            # Convert decimal odds to implied probabilities
            implied_prob_a = 1 / market_odds_a[i]
            implied_prob_b = 1 / market_odds_b[i]
            
            # Calculate Expected Value (EV)
            ev_a = (prob_a * (market_odds_a[i] - 1)) - (prob_b)
            ev_b = (prob_b * (market_odds_b[i] - 1)) - (prob_a)
            
            results.append({
                'match_index': i,
                'team_a_win_probability': round(prob_a, 4),
                'team_b_win_probability': round(prob_b, 4),
                'team_a_expected_value': round(ev_a, 4),
                'team_b_expected_value': round(ev_b, 4),
                'action_signal': "BET TEAM A (+EV)" if ev_a > 0.02 else ("BET TEAM B (+EV)" if ev_b > 0.02 else "HOLD / NO EDGE")
            })
            
        return pd.DataFrame(results)

class FootballDataPipeline:

        Initializes the pipeline with Football-Data.org authentication.

    def fetch_recent_results(self, league_id: int = 2021) -> pd.DataFrame:
        """
        Fetches match data for a specific league (Default: 2021 is the English Premier League).
        """
        conn = http.client.HTTPConnection(self.host)
        # Fetching finished matches to build our historical database
        url = f"/v4/competitions/{league_id}/matches?status=FINISHED"

        matches = data.get('matches', [])
        parsed_matches = []
        
        for m in matches:
            parsed_matches.append({
                'match_id': m['id'],

                'home_team_name': m['homeTeam']['name'],
                'away_team_id': m['awayTeam']['id'],
                'away_team_name': m['awayTeam']['name'],
                'home_goals': m['score']['fullTime']['home'],
                'away_goals': m['score']['fullTime']['away']
            })
            
        df = pd.DataFrame(parsed_matches)
        df['date'] = pd.to_datetime(df['date'])
        return df.sort_values('date')

        # Filter matches where the target team played either home or away
        team_df = match_df[(match_df['home_team_id'] == target_team_id) | (match_df['away_team_id'] == target_team_id)]
        
        goals_series = []
        for _, row in team_df.iterrows():
            if row['home_team_id'] == target_team_id:
                goals_series.append(row['home_goals'])
            else:
                goals_series.append(row['away_goals'])
                
        goals_arr = np.array(goals_series, dtype=np.float32)
        
        # Automatic Shape Alignment for TimesFM Context Length Requirements
        if len(goals_arr) >= self.context_len:
            goals_arr = goals_arr[-self.context_len:]
        else:
            # Pad historical data arrays if a team has played fewer than 32 matches
            pad_size = self.context_len - len(goals_arr)
            goals_arr = np.pad(goals_arr, (pad_size, 0), 'edge')
            
        return goals_arr.reshape(1, -1) # Reshape to 2D matrix [1, context_len]

    def generate_nvidia_forecast(self, history_matrix: np.ndarray) -> float:
        """
        Executes rapid inference on NVIDIA hardware to output a scalar projection.
        """
        padding_mask = np.ones_like(history_matrix, dtype=np.float32)
        forecast, _ = self.tfm.forecast(inputs=history_matrix, freq=0, mask=padding_mask)
        return float(forecast[0, 0])

class AsianFootballDataEngine:
    def __init__(self, api_key: str):
        self.api_key = api_key
        # Host endpoint for API-Football (RapidAPI or direct)
        self.host = "v3.football.api-sports.io"
        self.headers = {
            'x-rapidapi-key': self.api_key,
            'x-rapidapi-host': self.host
        }

    def fetch_asian_league_history(self, league_id: int, season: int) -> pd.DataFrame:
        """
        Fetches match data for Asian competitions.
        Example IDs for API-Football: 
        - J1 League (Japan): 98
        - K League 1 (South Korea): 292
        - Saudi Pro League: 307
        """
        conn = http.client.HTTPSConnection(self.host)
        # Endpoint to fetch all fixtures from a given league and year
        url = f"/fixtures?league={league_id}&season={season}"
        
        conn.request('GET', url, None, self.headers)
        res = conn.getresponse()
        raw_data = json.loads(res.read().decode('utf-8'))
        
        fixtures = raw_data.get('response', [])
        parsed_fixtures = []
        
        for f in fixtures:
            # Only pull completed matches to ensure valid historical data
            if f['fixture']['status']['short'] == "FT":
                parsed_fixtures.append({
                    'date': f['fixture']['date'],
                    'home_id': f['teams']['home']['id'],
                    'home_name': f['teams']['home']['name'],
                    'away_id': f['teams']['away']['id'],
                    'away_name': f['teams']['away']['name'],
                    'home_goals': f['goals']['home'],
                    'away_goals': f['goals']['away']
                })
                
        df = pd.DataFrame(parsed_fixtures)
        if not df.empty:
            df['date'] = pd.to_datetime(df['date'])
            df = df.sort_values('date')
        return df

class FBrefScraperEngine:
    def __init__(self):
        """
        Initializes the web scraper database.
        FBref URLs are highly structured based on competition codes.
        """
        self.urls = {
            # World Cup 2026 / Historical Tournament Match Logs
            'world_cup': "https://fbref.com",
            # Saudi Pro League (Example of elite Asian League history)
            'saudi_pro_league': "https://fbref.com"
        }
        # Standard browser headers to ensure successful data retrieval
        self.headers = {'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)'}

    def scrape_competition_fixtures(self, competition_key: str) -> pd.DataFrame:
        """
        Scrapes match results and statistical histories directly from FBref.
        Automatically handles HTML parsing and isolates the core fixture tables.
        """
        if competition_key not in self.urls:
            raise ValueError(f"Competition key '{competition_key}' not configured.")
            
        target_url = self.urls[competition_key]
        print(f"[SCRAPER] Accessing target data stream: {target_url}")
        
        # Read all HTML tabular elements on the page directly into memory
        all_tables = pd.read_html(target_url)
        
        # On FBref schedule pages, the primary fixture data frame is always the first table
        raw_fixtures = all_tables[0]
        
        # Dynamic database columns clean-up pipeline
        clean_df = raw_fixtures.dropna(subset=['Home', 'Away', 'Score'])
        
        processed_data = []
        for _, row in clean_df.iterrows():
            score_str = str(row['Score'])
            # Ensure the game has actually been played and has a valid scoreline
            if '–' in score_str: 
                try:
                    home_goals, away_goals = map(int, score_str.split('–'))
                    processed_data.append({
                        'date': row.get('Date', pd.NaT),
                        'stage': row.get('Round', 'Regular Season'),
                        'home_team': row['Home'],
                        'away_team': row['Away'],
                        'home_goals': home_goals,
                        'away_goals': away_goals,
                        'attendance': row.get('Attendance', 0)
                    })
                except ValueError:
                    continue # Skip structural row splits or unplayed matches
                    
        df = pd.DataFrame(processed_data)
        df['date'] = pd.to_datetime(df['date'], errors='coerce')
        return df.sort_values('date').dropna(subset=['date'])
