class NVIDIAFoundationForecaster:
    def __init__(self, context_len: int = 32):
        self.context_len = context_len
        print("[INFO] Mounting Google TimesFM Foundation Weights onto GPU Execution Space...")
        self.tfm = TimesFm(
            context_len=self.context_len,
            horizon_len=1,
            input_dim=1,
            backend="gpu" if torch.cuda.is_available() else "cpu"
        )
        self.tfm.load_from_checkpoint(repo_id="google/timesfm-2.0-default")

    def build_team_vector(self, df: pd.DataFrame, team_name: str) -> np.ndarray:
        """
        Filters matches for a specific team and builds a chronological array.
        Applies edge padding if the historical data falls short of the 32-game context window.
        """
        team_matches = df[(df['home_team'] == team_name) | (df['away_team'] == team_name)]
        series = []
        for _, row in team_matches.iterrows():
            if row['home_team'] == team_name:
                series.append(row['home_goals'])
            else:
                series.append(row['away_goals'])
                
        arr = np.array(series, dtype=np.float32)
        if len(arr) == 0:
            return np.zeros((1, self.context_len), dtype=np.float32)
            
        if len(arr) >= self.context_len:
            arr = arr[-self.context_len:]
        else:
            pad_needed = self.context_len - len(arr)
            arr = np.pad(arr, (pad_needed, 0), 'edge')
            
        return arr.reshape(1, -1)

    def extract_point_forecast(self, history_matrix: np.ndarray) -> float:
        """
        Executes zero-shot model inference to project the base performance metric.
        """
        mask = np.ones_like(history_matrix, dtype=np.float32)
        forecast, _ = self.tfm.forecast(inputs=history_matrix, freq=0, mask=mask)
        return float(forecast[0, 0])
