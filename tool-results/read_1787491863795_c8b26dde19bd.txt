   267→   267→// UNDERSTAT ANALYTICAL ENRICHMENT (Phase 1: team xG/xGA only)
   268→   268→// Runs AFTER API-Sports sync so that canonical teams exist in DB.
   269→   269→// ═════════════════════════════════════════════════════════════════════════
   270→   270→
   271→   271→const UNDERSTAT_LEAGUES = ['PL', 'LIGA', 'SA', 'BL', 'L1'] as const
   272→   272→const UNDERSTAT_SEASON = '2024' // Understat uses calendar year for 24/25 season
   273→   273→
   274→   274→async function syncUnderstatAnalytics(): Promise<{
   275→   275→  resolved: number
   276→   276→  unresolved: number
   277→   277→  updated: number
   278→   278→}> {
   279→   279→  const result = { resolved: 0, unresolved: 0, updated: 0 }
   280→   280→
   281→   281→  for (const leagueCode of UNDERSTAT_LEAGUES) {
   282→   282→    checkTimeout()
   283→   283→    try {
   284→   284→      const understatTeams = await fetchUnderstatLeagueTeams(leagueCode, parseInt(UNDERSTAT_SEASON))
   285→   285→      if (understatTeams.length === 0) continue
   286→   286→
   287→   287→      const { resolved, unresolved } = await resolveUnderstatTeams(
   288→   288→        db, understatTeams, leagueCode, UNDERSTAT_SEASON,
   289→   289→      )
   290→   290→
   291→   291→      result.resolved += resolved.length
   292→   292→      result.unresolved += unresolved.length
   293→   293→
   294→   294→      // Upsert resolved teams' analytics
   295→   295→      for (const r of resolved) {
   296→   296→        checkTimeout()
   297→   297→        try {
   298→   298→          const ut = understatTeams.find(t => t.id === r.understatTeamId)
   299→   299→          if (!ut) continue
   300→   300→
   301→   301→          // Validate: xG must be non-negative
   302→   302→          const xg = parseFloat(String(ut.xG)) || 0
   303→   303→          const xga = parseFloat(String(ut.xGA)) || 0
   304→   304→          const npxg = parseFloat(String(ut.npxG)) || 0
   305→   305→          const npxga = parseFloat(String(ut.npxGA)) || 0
   306→   306→          if (xg < 0 || xga < 0 || npxg < 0 || npxga < 0) {
   307→   307→            console.warn(`[SYNC/Understat] Invalid xG values for ${r.matchedDbTeamName}: xG=${xg}, xGA=${xga}`)
   308→   308→            continue
   309→   309→          }
   310→   310→
   311→   311→          // Calculate per-game values (Understat xG is season total)
   312→   312→          const matchesPlayed = (ut.wins || 0) + (ut.draws || 0) + (ut.losses || 0)
   313→   313→          const xgPerGame = matchesPlayed > 0 ? Math.round((xg / matchesPlayed) * 100) / 100 : null
   314→   314→          const xgaPerGame = matchesPlayed > 0 ? Math.round((xga / matchesPlayed) * 100) / 100 : null
   315→   315→          const ppda = ut.ppda?.def > 0 ? Math.round((ut.ppda.att / ut.ppda.def) * 100) / 100 : null
   316→   316→          const deep = matchesPlayed > 0 ? Math.round((ut.deep / matchesPlayed) * 100) / 100 : null
   317→   317→          const ppdaAllowed = ut.ppda_allowed?.def > 0
   318→   318→            ? Math.round((ut.ppda_allowed.att / ut.ppda_allowed.def) * 100) / 100
   319→   319→            : null
   320→   320→          const deepAllowed = matchesPlayed > 0
   321→   321→            ? Math.round((ut.deep_allowed / matchesPlayed) * 100) / 100
   322→   322→            : null
   323→   323→
   324→   324→          await db.teamAnalytic.upsert({
   325→   325→            where: {
   326→   326→              teamId_source_season_leagueCode: {
   327→   327→                teamId: r.dbTeamId,
   328→   328→                source: 'understat',
   329→   329→                season: UNDERSTAT_SEASON,
   330→   330→                leagueCode,
   331→   331→              },
   332→   332→            },
   333→   333→            update: {
   334→   334→              xgPerGame,
   335→   335→              xgaPerGame,
   336→   336→              npxGPerGame: matchesPlayed > 0 ? Math.round((npxg / matchesPlayed) * 100) / 100 : null,
   337→   337→              npxgaPerGame: matchesPlayed > 0 ? Math.round((npxga / matchesPlayed) * 100) / 100 : null,
   338→   338→              ppda,
   339→   339→              ppdaAllowed,
   340→   340→              deep,
   341→   341→              deepAllowed,
   342→   342→              syncedAt: new Date(),
   343→   343→            },
   344→   344→            create: {
   345→   345→              teamId: r.dbTeamId,
   346→   346→              source: 'understat',
   347→   347→              season: UNDERSTAT_SEASON,
   348→   348→              leagueCode,
   349→   349→              xgPerGame,
   350→   350→              xgaPerGame,
   351→   351→              npxGPerGame: matchesPlayed > 0 ? Math.round((npxg / matchesPlayed) * 100) / 100 : null,
   352→   352→              npxgaPerGame: matchesPlayed > 0 ? Math.round((npxga / matchesPlayed) * 100) / 100 : null,
   353→   353→              ppda,
   354→   354→              ppdaAllowed,
   355→   355→              deep,
   356→   356→              deepAllowed,
   357→   357→              sourceTeamId: String(ut.id),
   358→   358→              sourceTeamName: r.understatTeamName,
   359→   359→              syncedAt: new Date(),
   360→   360→            },
   361→   361→          })
   362→   362→          result.updated++
   363→   363→        } catch (err) {
   364→   364→          console.warn(`[SYNC/Understat] Failed to upsert for ${r.matchedDbTeamName}:`, err)
   365→   365→        }
   366→   366→      }
   367→   367→
   368→   368→      // Log unresolved teams
   369→   369→      for (const u of unresolved) {
   370→   370→        console.warn(
   371→   371→          `[SYNC/Understat] UNRESOLVED: ${u.understatTeamName} ` +
   372→   372→          `(id=${u.understatTeamId}, league=${leagueCode})`,
   373→   373→        )
   374→   374→      }
   375→   375→    } catch (err: any) {
   376→   376→      if (err.message?.includes('timeout')) throw err
   377→   377→      console.warn(`[SYNC/Understat] Failed for ${leagueCode}:`, err.message)
   378→   378→    }
   379→   379→  }
   380→   380→
   381→   381→  await logSync('understat', 'team_analytics', 'success',
   382→   382→    result.resolved + result.unresolved, result.resolved, result.updated)
   383→   383→
   384→   384→  return result
   385→   385→}

   548→   548→async function fetchScoreboardForLeague(league: { espnId: string; name: string; code: string }) {
   549→   549→  const SITE_V2 = 'https://site.api.espn.com/apis/v2/sports/soccer'
   550→   550→  const url = `${SITE_V2}/${league.espnId}/scoreboard`
   551→   551→  try {
   552→   552→    const res = await fetch(url, { cache: 'no-store' })
   553→   553→    if (!res.ok) return []
   554→   554→    const data = await res.json()
   555→   555→    const events: any[] = []
   556→   556→    for (const event of data.events || []) {
   557→   557→      const comp = event.competitions?.[0]
   558→   558→      if (!comp) continue
   559→   559→      const competitors = comp.competitors || []
   560→   560→      if (competitors.length < 2) continue
   561→   561→      const home = competitors.find((c: any) => c.homeAway === 'home') || competitors[0]
   562→   562→      const away = competitors.find((c: any) => c.homeAway === 'away') || competitors[1]
   563→   563→      const statusDetail = comp.status?.type?.detail || ''
   564→   564→      const minuteMatch = statusDetail.match(/(\d+)\s*'/)
   565→   565→      events.push({
   566→   566→        id: event.id?.toString() || '',
   567→   567→        homeTeamId: home.team?.id?.toString() || '',
   568→   568→        homeTeamName: home.team?.displayName || 'Unknown',
   569→   569→        homeTeamAbbr: home.team?.abbreviation || '???',
   570→   570→        homeTeamLogo: home.team?.logo || '',
   571→   571→        homeTeamColor: home.team?.color || '#00e676',
   572→   572→        awayTeamId: away.team?.id?.toString() || '',
   573→   573→        awayTeamName: away.team?.displayName || 'Unknown',
   574→   574→        awayTeamAbbr: away.team?.abbreviation || '???',
   575→   575→        awayTeamLogo: away.team?.logo || '',
   576→   576→        awayTeamColor: away.team?.color || '#ffffff',
   577→   577→        homeScore: parseInt(home.score) || 0,
   578→   578→        awayScore: parseInt(away.score) || 0,
   579→   579→        status: comp.status?.type?.name || 'STATUS_SCHEDULED',
   580→   580→        date: event.date || '',
   581→   581→        venue: comp.venue?.fullName || '',
   582→   582→        competition: event.name || league.name,
   583→   583→        competitionCode: league.code,
   584→   584→        minute: minuteMatch ? parseInt(minuteMatch[1]) : undefined,
   585→   585→      })
   586→   586→    }
   587→   587→    return events
   588→   588→  } catch (err) {
   589→   589→    console.error(`[SYNC/ESPN] Failed to fetch ${league.espnId}:`, err)
   590→   590→    return []
   591→   591→  }
   592→   592→}

   950→   950→          source: 'espn',
   951→   951→          created: standingsResult.created,
   952→   952→          updated: 0,
   953→   953→        }
   954→   954→        await logSync('espn', 'standings', 'success', standingsResult.created, standingsResult.created, 0)
   955→   955→      } catch (err: any) {
   956→   956→        if (err.message?.includes('timeout')) {
   957→   957→          console.warn('[SYNC] ESPN standings timeout')
   958→   958→          await logSync('espn', 'standings', 'partial', 0, 0, 0, err.message)
   959→   959→        } else {
   960→   960→          console.error('[SYNC] ESPN standings failed:', err)
   961→   961→          await logSync('espn', 'standings', 'error', 0, 0, 0, String(err))
   962→   962→        }
   963→   963→      }
   964→   964→    }
   965→   965→
   966→   966→    summary.durationMs = elapsed()
   967→   967→    console.log(`[SYNC] Complete in ${summary.durationMs}ms`, JSON.stringify(summary.syncs))
   968→   968→
   969→   969→    return NextResponse.json(summary)
   970→   970→  } catch (error) {
   971→   971→    const duration = elapsed()
   972→   972→    console.error(`[SYNC] Fatal error after ${duration}ms:`, error)
   973→   973→    summary.success = false
   974→   974→    summary.durationMs = duration
   975→   975→    summary.error = error instanceof Error ? error.message : String(error)
   976→   976→    return NextResponse.json(summary, { status: 500 })
   977→   977→  }
   978→   978→}
