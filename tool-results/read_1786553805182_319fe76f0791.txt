    59→// ── Types ──────────────────────────────────────────────────────────────────────
    60→
    61→interface EnhancedPlayer extends Player {
    62→  teamName?: string
    63→  teamCode?: string
    64→  teamColor?: string
    65→  nationality?: string
    66→  appearances?: number
    67→  minutesPlayed?: number
    68→}
    69→// ── Helpers ────────────────────────────────────────────────────────────────────
    70→
    71→function generateRadarStats(p: EnhancedPlayer) {
    72→  return {
    73→    Pace: p.position === 'FWD' || p.position === 'MID' ? 75 : 55,
    74→    Shooting: p.position === 'FWD' ? 75 : 50,
    75→    Passing: p.position === 'MID' ? 75 : 55,
    76→    Defending: p.position === 'DEF' ? 75 : 45,
    77→    Physical: 65,
    78→    Dribbling: p.position === 'FWD' || p.position === 'MID' ? 72 : 50,
    79→  }
    80→}

    82→// ── Component ──────────────────────────────────────────────────────────────────
    83→
    84→export function PlayerView() {
    85→  const teams = useElasticoStore(s => s.teams)
    86→  const token = useElasticoStore(s => s.token)
    87→  const [players, setPlayers] = useState<EnhancedPlayer[]>([])
    88→  const [search, setSearch] = useState('')
    89→  const [positionFilter, setPositionFilter] = useState<string>('all')
    90→  const [teamFilter, setTeamFilter] = useState<string>('all')
    91→  const [sortBy, setSortBy] = useState<string>('rating')
    92→  const [selectedPlayer, setSelectedPlayer] = useState<EnhancedPlayer | null>(null)
    93→  const [comparePlayer, setComparePlayer] = useState<EnhancedPlayer | null>(null)
    94→  const [page, setPage] = useState(0)
    95→  const perPage = 12
    96→
    97→  // Fetch players from API
    98→  useEffect(() => {
    99→    async function fetchPlayers() {
   100→      try {
   101→        const headers: Record<string, string> = {}
   102→        if (token) headers['Authorization'] = `Bearer ${token}`
   103→        const params = new URLSearchParams()
   104→        if (sortBy) params.set('sortBy', sortBy)
   105→        if (positionFilter !== 'all') params.set('position', positionFilter)
   106→        if (search) params.set('search', search)
   107→        params.set('limit', '100')
   108→        const res = await fetch(`/api/players?${params}`, { headers })
   109→        if (res.ok) {
   110→          const data = await res.json()
   111→          if (data.players && data.players.length > 0) {
   112→            setPlayers(data.players.map((p: Record<string, unknown>) => ({
   113→              ...p,
   114→              teamName: (p.team as Record<string, unknown>)?.name as string || p.teamName,
   115→              teamCode: (p.team as Record<string, unknown>)?.code as string || p.teamCode,
   116→              teamColor: (p.team as Record<string, unknown>)?.primaryColor as string || '#00e676',
   117→            } as EnhancedPlayer)))
   118→            return
   119→          }
   120→        }
   121→      } catch {
   122→        // API error — players will remain empty
   123→      }
   124→    }
   125→    fetchPlayers()
   126→  }, [token, sortBy, positionFilter, search])
   127→
   128→  // Derived data
   129→  const filteredPlayers = useMemo(() => {
   130→    let result = [...players]
   131→    if (search) result = result.filter(p => p.name.toLowerCase().includes(search.toLowerCase()))
   132→    if (positionFilter !== 'all') result = result.filter(p => p.position === positionFilter)
   133→    if (teamFilter !== 'all') result = result.filter(p => p.teamName === teamFilter)
   134→    result.sort((a, b) => {
   135→      if (sortBy === 'rating') return b.rating - a.rating
   136→      if (sortBy === 'goals') return b.goals - a.goals
   137→      if (sortBy === 'assists') return b.assists - a.assists
   138→      if (sortBy === 'name') return a.name.localeCompare(b.name)
   139→      if (sortBy === 'age') return (a.age || 0) - (b.age || 0)
   140→      if (sortBy === 'marketValue') return (b.marketValue || 0) - (a.marketValue || 0)
   141→      return 0
   142→    })
   143→    return result
   144→  }, [players, search, positionFilter, teamFilter, sortBy])
   145→
   146→  const pagedPlayers = filteredPlayers.slice(page * perPage, (page + 1) * perPage)
   147→  const totalPages = Math.ceil(filteredPlayers.length / perPage)
   148→
   149→  const topScorers = useMemo(() => [...players].sort((a, b) => b.goals - a.goals).slice(0, 10), [players])
   150→  const topByValue = useMemo(() => [...players].filter(p => p.marketValue).sort((a, b) => (b.marketValue || 0) - (a.marketValue || 0)).slice(0, 10), [players])
   151→
   152→  const positionalGroups = useMemo(() => {
   153→    const groups: Record<string, EnhancedPlayer[]> = {}
   154→    for (const p of players) {
   155→      if (!groups[p.position]) groups[p.position] = []
   156→      groups[p.position].push(p)
   157→    }
   158→    return groups
   159→  }, [players])
   160→
   161→  const ageData = useMemo(() => {
   162→    const buckets: Record<string, number> = { '18-21': 0, '22-25': 0, '26-29': 0, '30-33': 0, '34+': 0 }
   163→    for (const p of players) {
   164→      const age = p.age || 25
   165→      if (age <= 21) buckets['18-21']++
   166→      else if (age <= 25) buckets['22-25']++
   167→      else if (age <= 29) buckets['26-29']++
   168→      else if (age <= 33) buckets['30-33']++
   169→      else buckets['34+']++
   170→    }
   171→    return Object.entries(buckets).map(([range, count]) => ({ range, count }))
   172→  }, [players])
   173→
   174→  const nationalities = useMemo(() => {
   175→    const map: Record<string, number> = {}
   176→    for (const p of players) {
   177→      const nat = p.nationality || 'Unknown'
   178→      map[nat] = (map[nat] || 0) + 1
   179→    }
   180→    return Object.entries(map).sort((a, b) => b[1] - a[1]).slice(0, 10)
   181→  }, [players])
   182→
   183→  const uniqueTeams = useMemo(() => [...new Set(players.map(p => p.teamName).filter(Boolean))], [players])
   184→
   185→  // Similar players
   186→  const similarPlayers = useMemo(() => {
   187→    if (!selectedPlayer) return []
   188→    const stats = generateRadarStats(selectedPlayer)
   189→    return players
   190→      .filter(p => p.id !== selectedPlayer.id && p.position === selectedPlayer.position)
   191→      .map(p => {
   192→        const ps = generateRadarStats(p)
   193→        const diff = Math.abs(stats.Pace - ps.Pace) + Math.abs(stats.Shooting - ps.Shooting) +
   194→          Math.abs(stats.Passing - ps.Passing) + Math.abs(stats.Defending - ps.Defending) +
   195→          Math.abs(stats.Physical - ps.Physical) + Math.abs(stats.Dribbling - ps.Dribbling)
   196→        return { player: p, similarity: 100 - Math.round(diff / 6) }
   197→      })
   198→      .sort((a, b) => b.similarity - a.similarity)
   199→      .slice(0, 5)
   200→  }, [selectedPlayer, players])
   201→
   202→  // CSV Export
   203→  const exportCSV = useCallback(() => {
   204→    const headers = ['Name', 'Position', 'Team', 'Age', 'Goals', 'Assists', 'Rating', 'Market Value (€M)', 'Yellow Cards', 'Red Cards', 'Nationality']
   205→    const rows = filteredPlayers.map(p => [
   206→      p.name, p.position, p.teamName || '', p.age || '', p.goals, p.assists, p.rating, p.marketValue || '', p.yellowCards, p.redCards, p.nationality || ''
   207→    ])
   208→    const csv = [headers.join(','), ...rows.map(r => r.map(v => `"${v}"`).join(','))].join('\n')
   209→    const blob = new Blob([csv], { type: 'text/csv' })
   210→    const url = URL.createObjectURL(blob)
   211→    const a = document.createElement('a')
   212→    a.href = url
   213→    a.download = 'elastico-players.csv'
   214→    a.click()
   215→    URL.revokeObjectURL(url)
   216→    toast.success('Players exported to CSV')
   217→  }, [filteredPlayers])
   218→
   219→  const getPositionColor = (pos: string) => {
   220→    switch (pos) {
   221→      case 'GK': return 'text-yellow-400 bg-yellow-400/10 border-yellow-400/30'
   222→      case 'DEF': return 'text-blue-400 bg-blue-400/10 border-blue-400/30'
   223→      case 'MID': return 'text-emerald-400 bg-emerald-400/10 border-emerald-400/30'
   224→      case 'FWD': return 'text-red-400 bg-red-400/10 border-red-400/30'
   225→      default: return ''
   226→    }
   227→  }
   228→
   229→  const getRatingColor = (r: number) => {
   230→    if (r >= 8.0) return 'text-emerald-400'
   231→    if (r >= 7.0) return 'text-emerald-300'
   232→    if (r >= 6.0) return 'text-yellow-400'
   233→    return 'text-red-400'
   234→  }
   235→
   236→  return (
   237→    <motion.div
   238→      initial={{ opacity: 0, y: 20 }}
   239→      animate={{ opacity: 1, y: 0 }}
   240→      className="space-y-6"
   241→    >
   242→      {/* Header */}
   243→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   244→        <div>
   245→          <h1 className="text-2xl font-bold flex items-center gap-2">
   246→            <Users className="text-primary" /> Player Analytics
   247→          </h1>
   248→          <p className="text-muted-foreground text-sm mt-1">Search, analyze, and compare football players</p>
   249→        </div>
   250→        <Button variant="outline" size="sm" onClick={exportCSV} className="gap-2">
   251→          <Download className="size-4" /> Export CSV
   252→        </Button>
   253→      </div>
   254→
   255→      {/* Search & Filters */}
   256→      <div className="flex flex-col sm:flex-row gap-3">
   257→        <div className="relative flex-1">
   258→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   259→          <Input
   260→            placeholder="Search players..."
   261→            value={search}
   262→            onChange={(e) => { setSearch(e.target.value); setPage(0) }}
   263→            className="pl-9"
   264→          />
   265→        </div>
   266→        <Select value={positionFilter} onValueChange={(v) => { setPositionFilter(v); setPage(0) }}>
   267→          <SelectTrigger className="w-full sm:w-36">
   268→            <SelectValue placeholder="Position" />
   269→          </SelectTrigger>
   270→          <SelectContent>
   271→            <SelectItem value="all">All Positions</SelectItem>
   272→            <SelectItem value="GK">GK</SelectItem>
   273→            <SelectItem value="DEF">DEF</SelectItem>
   274→            <SelectItem value="MID">MID</SelectItem>
   275→            <SelectItem value="FWD">FWD</SelectItem>
   276→          </SelectContent>
   277→        </Select>
   278→        <Select value={teamFilter} onValueChange={(v) => { setTeamFilter(v); setPage(0) }}>
   279→          <SelectTrigger className="w-full sm:w-40">
   280→            <SelectValue placeholder="Team" />
   281→          </SelectTrigger>
   282→          <SelectContent>
   283→            <SelectItem value="all">All Teams</SelectItem>
   284→            {uniqueTeams.map(t => (
   285→              <SelectItem key={t} value={t!}>{t}</SelectItem>
   286→            ))}
   287→          </SelectContent>
   288→        </Select>
   289→        <Select value={sortBy} onValueChange={setSortBy}>
   290→          <SelectTrigger className="w-full sm:w-36">
   291→            <SelectValue placeholder="Sort by" />
   292→          </SelectTrigger>
   293→          <SelectContent>
   294→            <SelectItem value="rating">Rating</SelectItem>
   295→            <SelectItem value="goals">Goals</SelectItem>
   296→            <SelectItem value="assists">Assists</SelectItem>
   297→            <SelectItem value="marketValue">Market Value</SelectItem>
   298→            <SelectItem value="age">Age</SelectItem>
   299→            <SelectItem value="name">Name</SelectItem>
   300→          </SelectContent>
   301→        </Select>
   302→      </div>
   303→
   304→      <Tabs defaultValue="grid" className="space-y-4">
   305→        <TabsList className="flex flex-wrap gap-1 h-auto bg-card/50 p-1 rounded-lg">
   306→          <TabsTrigger value="grid" className="text-xs">Player Grid</TabsTrigger>
   307→          <TabsTrigger value="leaderboard" className="text-xs">Top Scorers</TabsTrigger>
   308→          <TabsTrigger value="radar" className="text-xs">Radar Charts</TabsTrigger>
   309→          <TabsTrigger value="compare" className="text-xs">Compare</TabsTrigger>
   310→          <TabsTrigger value="value" className="text-xs">Market Value</TabsTrigger>
   311→          <TabsTrigger value="positions" className="text-xs">By Position</TabsTrigger>
   312→          <TabsTrigger value="age" className="text-xs">Age Dist.</TabsTrigger>
   313→          <TabsTrigger value="nationality" className="text-xs">Nationality</TabsTrigger>
   314→          <TabsTrigger value="cards" className="text-xs">Cards</TabsTrigger>
   315→          <TabsTrigger value="subs" className="text-xs">Substitutions</TabsTrigger>
   316→          <TabsTrigger value="similar" className="text-xs">Similar</TabsTrigger>
   317→          <TabsTrigger value="form" className="text-xs">Form</TabsTrigger>
   318→        </TabsList>
   319→
   320→        {/* Player Cards Grid */}
   321→        <TabsContent value="grid">
   322→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4">
   323→            {pagedPlayers.map((player) => (
   324→              <motion.div
   325→                key={player.id}
   326→                whileHover={{ scale: 1.02, y: -4 }}
   327→                whileTap={{ scale: 0.98 }}
   328→                className="cursor-pointer"
   329→                onClick={() => setSelectedPlayer(player)}
   330→              >
   331→                <Card className="glass-card h-full overflow-hidden">
   332→                  <div className="h-2" style={{ backgroundColor: player.teamColor || '#00e676' }} />
   333→                  <CardContent className="p-4">
   334→                    <div className="flex items-start gap-3">
   335→                      <div className="w-14 h-14 rounded-full bg-muted flex items-center justify-center text-lg font-bold shrink-0 border-2 border-border">
   336→                        {player.number}
   337→                      </div>
   338→                      <div className="min-w-0 flex-1">
   339→                        <h3 className="text-sm font-semibold truncate">{player.name}</h3>
   340→                        <div className="flex items-center gap-2 mt-1">
   341→                          <Badge variant="outline" className={cn('text-[10px]', getPositionColor(player.position))}>
   342→                            {player.position}
   343→                          </Badge>
   344→                          <span className="text-[10px] text-muted-foreground truncate">{player.teamName}</span>
   345→                        </div>
   346→                      </div>
   347→                      <div className={cn('text-xl font-bold', getRatingColor(player.rating))}>
   348→                        {player.rating.toFixed(1)}
   349→                      </div>
   350→                    </div>
   351→                    <div className="grid grid-cols-3 gap-2 mt-4">
   352→                      <div className="text-center">
   353→                        <div className="text-lg font-bold text-primary">{player.goals}</div>
   354→                        <div className="text-[10px] text-muted-foreground">Goals</div>
   355→                      </div>
   356→                      <div className="text-center">
   357→                        <div className="text-lg font-bold text-primary">{player.assists}</div>
   358→                        <div className="text-[10px] text-muted-foreground">Assists</div>
   359→                      </div>
   360→                      <div className="text-center">
   361→                        <div className="text-lg font-bold text-primary">{player.yellowCards + player.redCards}</div>
   362→                        <div className="text-[10px] text-muted-foreground">Cards</div>
   363→                      </div>
   364→                    </div>
   365→                    {player.marketValue && (
   366→                      <div className="mt-3 pt-3 border-t border-border/50 flex items-center gap-1">
   367→                        <TrendingUp className="size-3 text-muted-foreground" />
   368→                        <span className="text-xs text-muted-foreground">€{player.marketValue}M</span>
   369→                      </div>
   370→                    )}
   371→                  </CardContent>
   372→                </Card>
   373→              </motion.div>
   374→            ))}
   375→          </div>
   376→          {/* Pagination */}
   377→          {totalPages > 1 && (
   378→            <div className="flex items-center justify-center gap-2 mt-4">
   379→              <Button variant="outline" size="icon" className="h-8 w-8" onClick={() => setPage(Math.max(0, page - 1))} disabled={page === 0}>
   380→                <ChevronLeft className="size-4" />
   381→              </Button>
   382→              <span className="text-sm text-muted-foreground">{page + 1} / {totalPages}</span>
   383→              <Button variant="outline" size="icon" className="h-8 w-8" onClick={() => setPage(Math.min(totalPages - 1, page + 1))} disabled={page >= totalPages - 1}>
   384→                <ChevronRight className="size-4" />
   385→              </Button>
   386→            </div>
   387→          )}
   388→        </TabsContent>
   389→
   390→        {/* Top Scorers Leaderboard */}
   391→        <TabsContent value="leaderboard">
   392→          <Card className="glass-card">
   393→            <CardHeader className="pb-2">
   394→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   395→                <Star className="size-4 text-yellow-400" /> Top Scorers Leaderboard
   396→              </CardTitle>
   397→            </CardHeader>
   398→            <CardContent>
   399→              <div className="max-h-96 overflow-y-auto">
   400→                <Table>
   401→                  <TableHeader>
   402→                    <TableRow>
   403→                      <TableHead className="w-10">#</TableHead>
   404→                      <TableHead>Player</TableHead>
   405→                      <TableHead className="text-center">Goals</TableHead>
   406→                      <TableHead className="text-center">Assists</TableHead>
   407→                      <TableHead className="text-center">Mins</TableHead>
   408→                      <TableHead className="text-center">G/90</TableHead>
   409→                      <TableHead className="text-center">Rating</TableHead>
   410→                    </TableRow>
   411→                  </TableHeader>
   412→                  <TableBody>
   413→                    {topScorers.map((p, i) => (
   414→                      <TableRow key={p.id} className="cursor-pointer hover:bg-accent/50" onClick={() => setSelectedPlayer(p)}>
   415→                        <TableCell className="font-bold text-muted-foreground">{i + 1}</TableCell>
   416→                        <TableCell>
   417→                          <div className="flex items-center gap-2">
   418→                            <span className="text-sm font-medium">{p.name}</span>
   419→                            <Badge variant="outline" className={cn('text-[9px]', getPositionColor(p.position))}>{p.position}</Badge>
   420→                          </div>
   421→                          <span className="text-[10px] text-muted-foreground">{p.teamName}</span>
   422→                        </TableCell>
   423→                        <TableCell className="text-center font-bold text-primary">{p.goals}</TableCell>
   424→                        <TableCell className="text-center">{p.assists}</TableCell>
   425→                        <TableCell className="text-center text-muted-foreground">{p.minutesPlayed || (p.appearances || 0) * 90}</TableCell>
   426→                        <TableCell className="text-center font-medium">
   427→                          {((p.goals / ((p.minutesPlayed || (p.appearances || 0) * 90) / 90))).toFixed(2)}
   428→                        </TableCell>
   429→                        <TableCell className={cn('text-center font-bold', getRatingColor(p.rating))}>{p.rating.toFixed(1)}</TableCell>
   430→                      </TableRow>
   431→                    ))}
   432→                  </TableBody>
   433→                </Table>
   434→              </div>
   435→            </CardContent>
   436→          </Card>
   437→        </TabsContent>
   438→
   439→        {/* Player Radar Chart */}
   440→        <TabsContent value="radar">
   441→          <Card className="glass-card">
   442→            <CardHeader className="pb-2">
   443→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   444→                <Award className="size-4 text-primary" /> Player Radar Charts
   445→              </CardTitle>
   446→            </CardHeader>
   447→            <CardContent>
   448→              <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
   449→                {topScorers.slice(0, 6).map((player) => {
   450→                  const stats = generateRadarStats(player)
   451→                  const radarData = Object.entries(stats).map(([k, v]) => ({ stat: k, value: v }))
   452→                  return (
   453→                    <Card key={player.id} className="glass-card border-border/30">
   454→                      <CardContent className="p-3">
   455→                        <h4 className="text-sm font-semibold text-center mb-1">{player.name}</h4>
   456→                        <p className="text-[10px] text-center text-muted-foreground mb-2">{player.position} · {player.teamName}</p>
   457→                        <div className="h-48">
   458→                          <ResponsiveContainer width="100%" height="100%">
   459→                            <RadarChart data={radarData} cx="50%" cy="50%" outerRadius="65%">
   460→                              <PolarGrid stroke="hsl(var(--border))" />
   461→                              <PolarAngleAxis dataKey="stat" tick={{ fontSize: 9 }} stroke="hsl(var(--muted-foreground))" />
   462→                              <PolarRadiusAxis tick={{ fontSize: 8 }} stroke="hsl(var(--muted-foreground))" domain={[0, 100]} />
   463→                              <Radar dataKey="value" stroke="#00e676" fill="#00e676" fillOpacity={0.15} strokeWidth={2} />
   464→                            </RadarChart>
   465→                          </ResponsiveContainer>
   466→                        </div>
   467→                      </CardContent>
   468→                    </Card>
   469→                  )
   470→                })}
   471→              </div>
   472→            </CardContent>
   473→          </Card>
   474→        </TabsContent>
   475→
