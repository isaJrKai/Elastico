    45→    45→function computeEloProb(homeElo: number, awayElo: number, outcome: string): number {
    46→    46→  const expectedHome = 1 / (1 + Math.pow(10, (awayElo - homeElo) / 400))
    47→    47→  if (outcome === 'home') return Math.round(expectedHome * 100)
    48→    48→  if (outcome === 'away') return Math.round((1 - expectedHome) * 100 * 0.75)
    49→    49→  return Math.round(100 - expectedHome * 100 - (1 - expectedHome) * 100 * 0.75)
    50→    50→}

   113→   113→// Fallback: generate shot positions from match events when no deep data source
   114→   114→function generateShotsFromEvents(events: MatchEvent[], homeXg: number, awayXg: number): ShotMapPoint[] {
   115→   115→  const shots: ShotMapPoint[] = []
   116→   116→  const homeGoals = events.filter(e => e.type === 'goal' && e.team === 'home')
   117→   117→  const awayGoals = events.filter(e => e.type === 'goal' && e.team === 'away')
   118→   118→  for (let i = 0; i < Math.ceil(homeXg * 3); i++) {
   119→   119→    shots.push({ x: 15 + (i * 17 % 30), y: 10 + (i * 23 % 80), team: 'home', goal: i < homeGoals.length })
   120→   120→  }
   121→   121→  for (let i = 0; i < Math.ceil(awayXg * 3); i++) {
   122→   122→    shots.push({ x: 55 + (i * 17 % 30), y: 10 + (i * 23 % 80), team: 'away', goal: i < awayGoals.length })
   123→   123→  }
   124→   124→  return shots
   125→   125→}
