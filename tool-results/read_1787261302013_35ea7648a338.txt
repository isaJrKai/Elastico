   158→function NvidiaApiStatusBadge({ token }: { token?: string }) {
   159→  const [status, setStatus] = React.useState<'loading' | 'connected' | 'disconnected'>('loading')
   160→
   161→  React.useEffect(() => {
   162→    fetch('/api/chat', {
   163→      method: 'POST',
   164→      headers: {
   165→        'Content-Type': 'application/json',
   166→        ...(token ? { Authorization: `Bearer ${token}` } : {}),
   167→      },
   168→      body: JSON.stringify({ message: '__status_check__', model: 'pro', stream: false }),
   169→    })
   170→      .then((res) => res.json())
   171→      .then((data) => {
   172→        if (data.model === 'pro') {
   173→          setStatus('connected')
   174→        } else if (data.model === 'mock-fallback') {
   175→          setStatus('disconnected')
   176→        } else {
   177→          setStatus('disconnected')
   178→        }
   179→      })
   180→      .catch(() => setStatus('disconnected'))
   181→  }, [])
   182→
   183→  if (status === 'loading') {
   184→    return (
   185→      <Badge variant="outline" className="gap-1.5 text-xs border-border">
   186→        <Loader2 className="size-3 animate-spin text-muted-foreground" />
   187→        Checking...
   188→      </Badge>
   189→    )
   190→  }
   191→
   192→  if (status === 'connected') {
   193→    return (
   194→      <Badge variant="outline" className="gap-1.5 text-xs border-emerald-500/30 text-emerald-400 bg-emerald-500/5">
   195→        <CheckCircle2 className="size-3" />
   196→        Connected
   197→      </Badge>
   198→    )
   199→  }
   200→
   201→  return (
   202→    <Badge variant="outline" className="gap-1.5 text-xs border-amber-500/30 text-amber-400 bg-amber-500/5">
   203→      <XCircle className="size-3" />
   204→      Not Configured
   205→    </Badge>
   206→  )
   207→}

   257→function BandwidthSection() {
   258→  const [bandwidth, setBandwidth] = useState<{ totalKB: number; requests: number; avgBytesPerRequest: number } | null>(null)
   259→  const { isOffline, isRegistered, isStandalone, clearCache } = usePWA()
   260→
   261→  useEffect(() => {
   262→    // Poll bandwidth tracker every 2 seconds
   263→    const load = async () => {
   264→      try {
   265→        const mod = await import('@/lib/compressed-data-stream')
   266→        setBandwidth(mod.BandwidthTracker.summary())
   267→      } catch {}
   268→    }
   269→    load()
   270→    const interval = setInterval(load, 2000)
   271→    return () => clearInterval(interval)
   272→  }, [])
   273→
   274→  return (
   275→    <Card className="glass-card border-border">
   276→      <CardHeader className="pb-4">
   277→        <SectionHeader
   278→          icon={Radio}
   279→          title="Bandwidth & Offline"
   280→          description="PWA caching & data usage monitoring"
   281→        />
   282→      </CardHeader>
   283→      <CardContent className="space-y-4">
   284→        {/* PWA Status Grid */}
   285→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   286→          <StatusPill
   287→            label="Network"
   288→            value={isOffline ? 'Offline' : 'Online'}
   289→            active={!isOffline}
   290→            icon={isOffline ? WifiOff : Wifi}
   291→          />
   292→          <StatusPill
   293→            label="App Cached"
   294→            value={isRegistered ? 'Yes' : 'No'}
   295→            active={isRegistered}
   296→            icon={Smartphone}
   297→          />
   298→          <StatusPill
   299→            label="Standalone"
   300→            value={isStandalone ? 'Yes' : 'No'}
   301→            active={isStandalone}
   302→            icon={Monitor}
   303→          />
   304→          <StatusPill
   305→            label="Data Used"
   306→            value={bandwidth ? `${bandwidth.totalKB.toFixed(1)} KB` : '—'}
   307→            active={bandwidth ? bandwidth.totalKB < 100 : true}
   308→            icon={Zap}
   309→          />
   310→        </div>
   311→
   312→        {/* Bandwidth Details */}
   313→        {bandwidth && (
   314→          <div className="rounded-lg border border-border bg-secondary/30 p-4 space-y-3">
   315→            <div className="flex items-center justify-between text-sm">
   316→              <span className="text-muted-foreground">API Requests</span>
   317→              <span className="font-mono text-foreground">{bandwidth.requests}</span>
   318→            </div>
   319→            <div className="flex items-center justify-between text-sm">
   320→              <span className="text-muted-foreground">Avg per Request</span>
   321→              <span className="font-mono text-foreground">{bandwidth.avgBytesPerRequest} B</span>
   322→            </div>
   323→            <div className="flex items-center justify-between text-sm">
   324→              <span className="text-muted-foreground">Total Today</span>
   325→              <span className={cn(
   326→                'font-mono font-medium',
   327→                bandwidth.totalKB < 50 ? 'text-emerald-400' : (bandwidth.totalKB ?? 0) < 200 ? 'text-amber-400' : 'text-red-400'
   328→              )}>
   329→                {(bandwidth.totalKB ?? 0).toFixed(1)} KB
   330→              </span>
   331→            </div>
   332→          </div>
   333→        )}
   334→
   335→        {/* Compression Features */}
   336→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   337→          {[
   338→            { label: 'Null Stripping', desc: 'Removes 30-60% empty fields' },
   339→            { label: 'Key Compaction', desc: '2-char keys in compact mode' },
   340→            { label: 'Service Worker', desc: '0 MB data on app reopen' },
   341→            { label: 'Diff Updates', desc: 'Only changed data sent' },
   342→          ].map((f) => (
   343→            <div key={f.label} className="flex items-center gap-2.5 rounded-md border border-border bg-secondary/20 px-3 py-2">
   344→              <CheckCircle2 className="size-3.5 text-primary shrink-0" />
   345→              <div>
   346→                <p className="text-xs font-medium text-foreground">{f.label}</p>
   347→                <p className="text-[10px] text-muted-foreground">{f.desc}</p>
   348→              </div>
   349→            </div>
   350→          ))}
   351→        </div>
   352→
   353→        {/* Clear Cache Button */}
   354→        {isRegistered && (
   355→          <Button
   356→            variant="outline"
   357→            size="sm"
   358→            className="w-full border-border text-xs"
   359→            onClick={async () => {
   360→              await clearCache()
   361→              toast.success('Cache Cleared', { description: 'App will re-download fresh data on next load.' })
   362→            }}
   363→          >
   364→            <Trash2 className="size-3.5 mr-2" />
   365→            Clear Offline Cache & Re-download

   370→          <Info className="mt-0.5 size-4 shrink-0 text-primary/70" />
   371→          <p className="text-[11px] leading-relaxed text-muted-foreground">
   372→            After the first visit, ELASTICO caches itself (~5 MB) to your device storage.
   373→            Subsequent opens load instantly from your phone — using <strong className="text-foreground">0 MB mobile data</strong>.
   374→            Live match updates use compressed diff payloads under 5 KB per cycle.
   375→          </p>
