    32→function StatCard({ title, value, icon: Icon, change, color = 'emerald', suffix = '' }: { title: string; value: string | number; icon: any; change?: number; color?: string; suffix?: string }) {
    33→  const isPositive = change && change >= 0
    34→  return (
    35→    <Card className="glass-card glass-card-hover animate-fade-in-up">
    36→      <CardContent className="p-4">
    37→        <div className="flex items-center justify-between">
    38→          <div className="space-y-1">
    39→            <p className="text-xs text-muted-foreground uppercase tracking-wider">{title}</p>
    40→            <p className="text-2xl font-bold">{value}{suffix}</p>
    41→            {change !== undefined && (
    42→              <div className={`flex items-center gap-1 text-xs ${isPositive ? 'text-emerald-400' : 'text-red-400'}`}>
    43→                {isPositive ? <ArrowUpRight className="h-3 w-3" /> : <ArrowDownRight className="h-3 w-3" />}
    44→                <span>{Math.abs(change)}%</span>
    45→                <span className="text-muted-foreground">vs last period</span>
    46→              </div>
    47→            )}
    48→          </div>
    49→          <div className={`p-3 rounded-xl ${color === 'emerald' ? 'bg-emerald-500/10 text-emerald-400' : color === 'gold' ? 'bg-yellow-500/10 text-yellow-400' : color === 'red' ? 'bg-red-500/10 text-red-400' : 'bg-blue-500/10 text-blue-400'}`}>
    50→            <Icon className="h-5 w-5" />
    51→          </div>
    52→        </div>
    53→      </CardContent>
    54→    </Card>
    55→  )
    56→}

    58→function AdminGauge({ value, max = 100, label, color = '#00e676' }: { value: number; max?: number; label: string; color?: string }) {
    59→  const pct = Math.min((value / max) * 100, 100)
    60→  return (
    61→    <Card className="glass-card animate-fade-in-up">
    62→      <CardContent className="p-4">
    63→        <div className="flex flex-col items-center">
    64→          <div className="relative w-20 h-10 overflow-hidden rounded-t-full">
    65→            <div className="absolute inset-0 bg-muted/50" />
    66→            <div className="absolute bottom-0 left-0 right-0 transition-all duration-1000" style={{ height: `${pct}%`, background: color }} />
    67→          </div>
    68→          <div className="text-center">
    69→            <p className="text-lg font-bold">{value}{max === 100 ? '%' : ''}</p>
    70→            <p className="text-xs text-muted-foreground">{label}</p>
    71→          </div>
    72→        </div>
    73→      </CardContent>
    74→    </Card>
    75→  )
    76→}
