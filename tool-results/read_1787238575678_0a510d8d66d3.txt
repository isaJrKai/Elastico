     1→"""LSTM Deep Learning Prediction Engine for Football Match Outcomes.

     4→team performance vectors to predict match outcomes (home / draw / away).
     5→
     6→Architecture highlights:
     7→- Bidirectional LSTM captures temporal patterns in recent form.
     8→- Attention mechanism learns which timesteps matter most.
     9→- FC head with batch normalisation and dropout for regularisation.
    10→- Training uses Adam, ReduceLROnPlateau, gradient clipping, and early stopping.
    11→
    12→Capabilities:
    13→- Sequence construction from raw match history dicts.
    14→- Full training loop with validation monitoring.
    15→- Calibrated softmax probabilities that sum to 1.0.

    17→- Model persistence via ``torch.save`` / ``joblib`` for metadata.

   140→# ═══════════════════════════════════════════════════════════════════════════════
   141→# FootballLSTM — the neural network
   142→# ═══════════════════════════════════════════════════════════════════════════════
   143→
   144→
   145→class FootballLSTM(nn.Module):
   146→    """Bidirectional LSTM with attention for football match outcome prediction.
   147→
   148→    Architecture::
   149→
   150→        Input (batch, seq_len, input_size)
   151→        ├─► Home LSTM ─► Attention ─► home_context (batch, hidden*2)
   152→        └─► Away LSTM ─► Attention ─► away_context (batch, hidden*2)
   153→        Concat ─► FC(256) ─► BN ─► ReLU ─► Dropout
   154→                      ─► FC(128) ─► BN ─► ReLU ─► Dropout
   155→                      ─► FC(3)
   156→
   157→    Output: 3-class logits (home / draw / away).
   158→    """
   159→
   160→    def __init__(self, config: LSTMConfig):
   161→        super().__init__()
   162→        self.config = config
   163→
   164→        bidir_hidden = config.hidden_size * 2
   165→
   166→        # --- Shared (but separately-run) LSTM for each team ---
   167→        self.lstm = nn.LSTM(
   168→            input_size=config.input_size,
   169→            hidden_size=config.hidden_size,
   170→            num_layers=config.num_layers,
   171→            batch_first=True,
   172→            bidirectional=True,
   173→            dropout=config.dropout if config.num_layers > 1 else 0.0,
   174→        )
   175→
   176→        # --- Attention ---
   177→        self.attention = nn.Linear(bidir_hidden, 1)
   178→
   179→        # --- FC head ---
   180→        fc_input = bidir_hidden * 2  # concat home + away contexts
   181→        self.fc = nn.Sequential(
   182→            nn.Linear(fc_input, 128),
   183→            nn.BatchNorm1d(128),
   184→            nn.ReLU(),
   185→            nn.Dropout(config.dropout),
   186→            nn.Linear(128, 64),
   187→            nn.BatchNorm1d(64),
   188→            nn.ReLU(),
   189→            nn.Dropout(config.dropout),
   190→            nn.Linear(64, _N_CLASSES),
   191→        )
   192→
   193→        self._init_weights()
   194→
   195→    # ------------------------------------------------------------------
   196→    # Weight initialisation (Xavier uniform for linear, orthogonal for LSTM)
   197→    # ------------------------------------------------------------------
   198→
   199→    def _init_weights(self) -> None:
   200→        for name, param in self.named_parameters():
   201→            if "lstm" in name and "weight_ih" in name:
   202→                nn.init.xavier_uniform_(param)
   203→            elif "lstm" in name and "weight_hh" in name:
   204→                nn.init.orthogonal_(param)
   205→            elif "lstm" in name and "bias" in name:
   206→                nn.init.zeros_(param)
   207→                # Set forget-gate bias to 1 for stable long-term memory
   208→                n = param.size(0)
   209→                start, end = n // 4, n // 2
   210→                nn.init.constant_(param.data[start:end], 1.0)
   211→            elif "weight" in name and param.ndim >= 2:
   212→                nn.init.xavier_uniform_(param)
   213→            elif "bias" in name:
   214→                nn.init.zeros_(param)
   215→
   216→    # ------------------------------------------------------------------
   217→    # Forward
   218→    # ------------------------------------------------------------------
   219→
   220→    def _attend(self, lstm_out: torch.Tensor) -> torch.Tensor:
   221→        """Softmax attention over timesteps.
   222→
   223→        Parameters
   224→        ----------
   225→        lstm_out : (batch, seq_len, hidden*2)
   226→
   227→        Returns
   228→        -------
   229→        context : (batch, hidden*2)
   230→        """
   231→        attn_logits = self.attention(lstm_out)  # (batch, seq_len, 1)
   232→        attn_weights = torch.softmax(attn_logits, dim=1)  # (batch, seq_len, 1)
   233→        context = torch.sum(attn_weights * lstm_out, dim=1)  # (batch, hidden*2)
   234→        return context
   235→
   236→    def forward(self, x_home: torch.Tensor, x_away: torch.Tensor) -> torch.Tensor:
   237→        """Forward pass.
   238→
   239→        Parameters
   240→        ----------
   241→        x_home : (batch, seq_len, input_size)
   242→        x_away : (batch, seq_len, input_size)
   243→
   244→        Returns
   245→        -------
   246→        logits : (batch, 3)
   247→        """
   248→        # Encode home team sequence
   249→        home_out, _ = self.lstm(x_home)  # (batch, seq_len, hidden*2)
   250→        home_ctx = self._attend(home_out)  # (batch, hidden*2)
   251→
   252→        # Encode away team sequence (reusing the same LSTM weights)
   253→        away_out, _ = self.lstm(x_away)  # (batch, seq_len, hidden*2)
   254→        away_ctx = self._attend(away_out)  # (batch, hidden*2)
   255→
   256→        # Concatenate contexts
   257→        combined = torch.cat([home_ctx, away_ctx], dim=1)  # (batch, hidden*4)
   258→
   259→        return self.fc(combined)  # (batch, 3)
   260→

   304→    # ==================================================================
   305→    # Sequence Building
   306→    # ==================================================================
   307→
   308→    def build_sequences(
   309→        self,
   310→        matches: list[dict[str, Any]],
   311→        feature_cols: list[str],
   312→    ) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
   313→        """Build genuine temporal sequences for LSTM training.
   314→
   315→        For each match, constructs sequences of past match performance
   316→        for both home and away teams.  Each timestep in the sequence
   317→        represents ONE past match with features computed from data
   318→        available AT THAT POINT IN TIME (no future leakage).
   319→
   320→        The feature_cols should typically be ``LSTM_SEQUENCE_FEATURES``
   321→        from the canonical schema (15 features).  These are a subset of
   322→        the full 50-feature XGBoost schema — selected for being meaningful
   323→        as per-match performance snapshots.
   324→
   325→        Parameters
   326→        ----------
   327→        matches : list[dict]
   328→            Match records sorted chronologically.  Each dict must contain
   329→            at least ``"date"``, ``"home_team"``, ``"away_team"``,
   330→            ``"home_goals"``, ``"away_goals"``.
   331→        feature_cols : list[str]
   332→            Column names to extract per timestep.  Should match
   333→            ``LSTM_SEQUENCE_FEATURES`` from the canonical schema.
   334→
   335→        Returns
   336→        -------
   337→        X_home : np.ndarray, shape (N, seq_len, n_features)
   338→        X_away : np.ndarray, shape (N, seq_len, n_features)
   339→        y : np.ndarray, shape (N,), dtype int32
   340→            Labels: 0 = home win, 1 = draw, 2 = away win.
   341→
   342→        Notes
   343→        -----
   344→        - Matches that lack enough history for EITHER team are dropped.
   345→        - No zero-padding or vector repetition is performed.
   346→        - Each timestep's features are computed from rolling stats up to
   347→          that point, NOT from future data.
   348→        """
   349→        self.feature_cols = list(feature_cols)
   350→        seq_len = self.config.sequence_length
   351→        n_features = len(feature_cols)
   352→
   353→        # Import the feature engineer to compute per-match features
   354→        from src.ml.features import FeatureEngineer
   355→
   356→        # Use the FeatureEngineer to compute rolling features per team
   357→        # at each point in time.
   358→        engineer = FeatureEngineer()
   359→
   360→        # First pass: build a per-team chronological list of match records
   361→        # and compute the 50-feature vector for each match using rolling
   362→        # stats available at that point.
   363→        #
   364→        # We process all matches in order, building up team histories.
   365→        # For each match, we compute the full 50-feature vector for both
   366→        # teams, then extract the 15 LSTM features from it.
   367→        #
   368→        # This gives us a per-team timeline of feature vectors, where
   369→        # each vector reflects that team's state AT that point in time.
   370→
   371→        from collections import defaultdict
   372→        from datetime import datetime as _dt
   373→
   374→        # team -> list of (date, feature_dict_50)
   375→        team_feature_timeline: dict[str, list[tuple]] = defaultdict(list)
   376→        # team -> list of (date, home_goals, away_goals, result_char) — for rolling stats
   377→        team_history: dict[str, list[tuple]] = defaultdict(list)
   378→        team_home_history: dict[str, list[tuple]] = defaultdict(list)
   379→        team_away_history: dict[str, list[tuple]] = defaultdict(list)
   380→
   381→        # Store per-match computed 50-feature vectors for both teams
   382→        # match_idx -> (home_feature_dict, away_feature_dict)
   383→        match_features: dict[int, tuple[dict[str, float], dict[str, float]]] = {}
   384→
   385→        def _parse_date(value: Any) -> Any:
   386→            if isinstance(value, _dt):
   387→                return value
   388→            if isinstance(value, str):
   389→                for fmt in ("%Y-%m-%d", "%Y-%m-%dT%H:%M:%S", "%d/%m/%Y"):
   390→                    try:
   391→                        return _dt.strptime(value, fmt)
   392→                    except ValueError:
   393→                        continue
   394→            return None
   395→
   396→        # Process matches in chronological order
   397→        for match_idx, match in enumerate(matches):
   398→            home_team = str(match["home_team"])
   399→            away_team = str(match["away_team"])
   400→            hg = int(match.get("home_goals", 0))
   401→            ag = int(match.get("away_goals", 0))
   402→
   403→            if hg > ag:
   404→                home_result, away_result = "W", "L"
   405→            elif hg < ag:
   406→                home_result, away_result = "L", "W"
   407→            else:
   408→                home_result, away_result = "D", "D"
   409→
   410→            match_date = _parse_date(match.get("date")) or _dt.now()
   411→
   412→            # Build stats from rolling history (same logic as FeatureEngineer)
   413→            home_stats = engineer._stats_from_history(
   414→                team_history[home_team],
   415→                team_home_history[home_team],
   416→                team_away_history[home_team],
   417→                window=10,
   418→            )
   419→            away_stats = engineer._stats_from_history(
   420→                team_history[away_team],
   421→                team_home_history[away_team],
   422→                team_away_history[away_team],
   423→                window=10,
   424→            )
   425→
   426→            # Set last-match date for rest-day calculation
   427→            if team_history[home_team]:
   428→                home_stats["last_match_date"] = team_history[home_team][-1][0].isoformat()
   429→            if team_history[away_team]:
   430→                away_stats["last_match_date"] = team_history[away_team][-1][0].isoformat()
   431→
   432→            # Build 50-feature vectors for both teams
   433→            home_feats = engineer.build_match_features(
   434→                match_data={"date": match_date},
   435→                home_stats=home_stats,
   436→                away_stats=away_stats,
   437→                h2h_data=None,  # H2H is match-specific, not per-team
   438→                odds=None,
   439→            )
   440→            away_feats = engineer.build_match_features(
   441→                match_data={"date": match_date},
   442→                home_stats=away_stats,
   443→                away_stats=home_stats,  # swap perspective for away team
   444→                h2h_data=None,
   445→                odds=None,
   446→            )
   447→
   448→            match_features[match_idx] = (home_feats, away_feats)
   449→
   450→            # Add to team timelines — the home_feats represent the home
   451→            # team's state, away_feats represent the away team's state
   452→            team_feature_timeline[home_team].append((match_date, home_feats))
   453→            team_feature_timeline[away_team].append((match_date, away_feats))
   454→
   455→            # Update rolling histories for future matches
   456→            team_history[home_team].append((match_date, hg, ag, home_result))
   457→            team_history[away_team].append((match_date, ag, hg, away_result))
   458→            team_home_history[home_team].append((match_date, hg, ag, home_result))
   459→            team_away_history[away_team].append((match_date, ag, hg, away_result))
   460→
   461→        # Second pass: for each match, extract sequences from timelines
   462→        X_home_list: list[np.ndarray] = []
   463→        X_away_list: list[np.ndarray] = []
   464→        y_list: list[int] = []
   465→
   466→        for match_idx, match in enumerate(matches):
   467→            home_team = str(match["home_team"])
   468→            away_team = str(match["away_team"])
   469→            hg = int(match.get("home_goals", 0))
   470→            ag = int(match.get("away_goals", 0))
   471→
   472→            # Result label
   473→            if hg > ag:
   474→                label = 0  # home win
   475→            elif hg < ag:
   476→                label = 2  # away win
   477→            else:
   478→                label = 1  # draw
   479→
   480→            home_timeline = team_feature_timeline[home_team]
   481→            away_timeline = team_feature_timeline[away_team]
   482→
   483→            # The current match is the LAST entry in each timeline.
   484→            # We need seq_len entries BEFORE the current match.
   485→            if len(home_timeline) < seq_len + 1:
   486→                continue
   487→            if len(away_timeline) < seq_len + 1:
   488→                continue
   489→
   490→            # Extract the last seq_len entries BEFORE the current match
   491→            # Timeline is chronological, current match is at index -1
   492→            home_seq_features = []
   493→            for date, feats in home_timeline[-(seq_len + 1):-1]:
   494→                vec = [float(feats.get(c, 0.0)) for c in feature_cols]
   495→                vec = np.nan_to_num(vec, nan=0.0, posinf=0.0, neginf=0.0)
   496→                home_seq_features.append(vec)
   497→
   498→            away_seq_features = []
   499→            for date, feats in away_timeline[-(seq_len + 1):-1]:
   500→                vec = [float(feats.get(c, 0.0)) for c in feature_cols]
   501→                vec = np.nan_to_num(vec, nan=0.0, posinf=0.0, neginf=0.0)
   502→                away_seq_features.append(vec)
   503→
   504→            home_seq = np.array(home_seq_features, dtype=np.float32)
   505→            away_seq = np.array(away_seq_features, dtype=np.float32)
   506→
   507→            if home_seq.shape != (seq_len, n_features):
   508→                continue
   509→            if away_seq.shape != (seq_len, n_features):
   510→                continue
   511→
   512→            X_home_list.append(home_seq)
   513→            X_away_list.append(away_seq)
   514→            y_list.append(label)
   515→
   516→        if not X_home_list:
   517→            logger.warning(
   518→                "build_sequences produced 0 samples.  Check data length vs sequence_length=%d.",
   519→                seq_len,
   520→            )
   521→            return (
   522→                np.empty((0, seq_len, n_features), dtype=np.float32),
   523→                np.empty((0, seq_len, n_features), dtype=np.float32),
   524→                np.empty(0, dtype=np.int32),
   525→            )
   526→
   527→        X_home = np.stack(X_home_list, axis=0)  # (N, seq_len, n_features)
   528→        X_away = np.stack(X_away_list, axis=0)
   529→        y = np.array(y_list, dtype=np.int32)
   530→
   531→        logger.info(
   532→            "Built GENUINE TEMPORAL sequences: %d samples, seq_len=%d, n_features=%d",
   533→            len(y), seq_len, n_features,
   534→        )
   535→        return X_home, X_away, y
   536→
   537→    # ==================================================================
   538→    # Training
   539→    # ==================================================================
   540→
   541→    def train(
   542→        self,
   543→        X_home: np.ndarray,
   544→        X_away: np.ndarray,
   545→        y: np.ndarray,
   546→        X_val_home: np.ndarray | None = None,
   547→        X_val_away: np.ndarray | None = None,
   548→        y_val: np.ndarray | None = None,
   549→    ) -> dict[str, Any]:
   550→        """Train the LSTM model.
   551→
   552→        Parameters
   553→        ----------
   554→        X_home, X_away : np.ndarray, shape (N, seq_len, input_size)
   555→            Training sequences for home and away teams.
   556→        y : np.ndarray, shape (N,), dtype int
   557→            Labels (0=home, 1=draw, 2=away).
   558→        X_val_home, X_val_away, y_val : optional
   559→            Validation set.  When provided, early stopping monitors
   560→            validation loss.
   561→
   562→        Returns
   563→        -------
   564→        dict
   565→            Training metrics: epochs_run, best_epoch, train_loss_best,
   566→            val_loss_best (if validation), learning_rates, etc.
   567→        """
   568→        # --- Validate inputs ---
   569→        X_home = np.asarray(X_home, dtype=np.float32)
   570→        X_away = np.asarray(X_away, dtype=np.float32)
   571→        y = np.asarray(y, dtype=np.int64)
   572→
   573→        # Auto-detect input_size from data if not set
   574→        if X_home.ndim == 3 and X_home.shape[2] != self.config.input_size:
   575→            logger.info(
   576→                "input_size mismatch: config=%d, data=%d. Updating config.",
   577→                self.config.input_size,
   578→                X_home.shape[2],
   579→            )
   580→            self.config.input_size = X_home.shape[2]
   581→
   582→        # Set random seed for reproducibility
   583→        torch.manual_seed(self.config.random_seed)
   584→        if torch.cuda.is_available():
   585→            torch.cuda.manual_seed_all(self.config.random_seed)
   586→
   587→        # --- Instantiate model ---
   588→        self.model = FootballLSTM(self.config).to(self.device)
   589→
   590→        # --- Data loaders ---
   591→        train_dataset = _MatchSequenceDataset(X_home, X_away, y)
   592→        train_loader = torch.utils.data.DataLoader(
   593→            train_dataset,
   594→            batch_size=self.config.batch_size,
   595→            shuffle=True,
   596→            drop_last=False,
   597→            num_workers=0,
   598→            pin_memory=self.device.type == "cuda",
   599→        )
   600→
   601→        has_val = (
   602→            X_val_home is not None
   603→            and X_val_away is not None
   604→            and y_val is not None
   605→        )
   606→
   607→        if has_val:
   608→            X_val_home = np.asarray(X_val_home, dtype=np.float32)
   609→            X_val_away = np.asarray(X_val_away, dtype=np.float32)
   610→            y_val = np.asarray(y_val, dtype=np.int64)
   611→
   612→            val_dataset = _MatchSequenceDataset(X_val_home, X_val_away, y_val)
   613→            val_loader = torch.utils.data.DataLoader(
   614→                val_dataset,
   615→                batch_size=self.config.batch_size,
   616→                shuffle=False,
   617→                drop_last=False,
   618→                num_workers=0,
   619→                pin_memory=self.device.type == "cuda",
   620→            )
   621→
   622→        # --- Optimiser, loss, scheduler ---
   623→        optimizer = torch.optim.Adam(
   624→            self.model.parameters(),
   625→            lr=self.config.learning_rate,
   626→            weight_decay=self.config.weight_decay,
   627→        )
   628→        criterion = nn.CrossEntropyLoss()
   629→        scheduler = torch.optim.lr_scheduler.ReduceLROnPlateau(
   630→            optimizer,
   631→            mode="min",
   632→            factor=self.config.lr_scheduler_factor,
   633→            patience=self.config.lr_scheduler_patience,
   634→            min_lr=self.config.lr_scheduler_min_lr,
   635→            verbose=False,
   636→        )
   637→
   638→        # --- Training loop ---
   639→        cfg = self.config
   640→        best_val_loss = float("inf")
   641→        best_train_loss = float("inf")
   642→        best_epoch = 0
   643→        epochs_no_improve = 0
   644→        train_loss_history: list[float] = []
   645→        val_loss_history: list[float] = []
   646→        lr_history: list[float] = []
   647→
   648→        self._best_state = None
   649→
   650→        for epoch in range(1, cfg.epochs + 1):
   651→            # --- Train ---
   652→            self.model.train()
   653→            epoch_loss = 0.0
   654→            n_batches = 0
   655→
   656→            for batch_xh, batch_xa, batch_y in train_loader:
   657→                batch_xh = batch_xh.to(self.device)
   658→                batch_xa = batch_xa.to(self.device)
   659→                batch_y = batch_y.to(self.device)
   660→
   661→                optimizer.zero_grad()
   662→                logits = self.model(batch_xh, batch_xa)  # (B, 3)
   663→                loss = criterion(logits, batch_y)
   664→
   665→                loss.backward()
   666→                torch.nn.utils.clip_grad_norm_(
   667→                    self.model.parameters(), max_norm=cfg.grad_clip_max_norm
   668→                )
   669→                optimizer.step()
   670→
   671→                epoch_loss += loss.item()
   672→                n_batches += 1
   673→
   674→            avg_train_loss = epoch_loss / max(n_batches, 1)
   675→            train_loss_history.append(avg_train_loss)
   676→            lr_history.append(optimizer.param_groups[0]["lr"])
   677→
   678→            if avg_train_loss < best_train_loss:
   679→                best_train_loss = avg_train_loss
   680→
   681→            # --- Validate ---
   682→            if has_val:
   683→                val_loss = self._eval_loss(val_loader, criterion)
   684→                val_loss_history.append(val_loss)
   685→                scheduler.step(val_loss)
   686→
   687→                if val_loss < best_val_loss:
   688→                    best_val_loss = val_loss
   689→                    best_epoch = epoch
   690→                    epochs_no_improve = 0
   691→                    # Deepcopy best model state
   692→                    self._best_state = copy.deepcopy(self.model.state_dict())
   693→                else:
   694→                    epochs_no_improve += 1
   695→
   696→                if epoch % 10 == 0 or epoch == 1:
   697→                    logger.info(
   698→                        "Epoch %3d/%d  train_loss=%.4f  val_loss=%.4f  lr=%.2e",
   699→                        epoch,
   700→                        cfg.epochs,
   701→                        avg_train_loss,
   702→                        val_loss,
   703→                        optimizer.param_groups[0]["lr"],
   704→                    )
   705→
   706→                # Early stopping
   707→                if (
   708→                    cfg.early_stopping_patience > 0
   709→                    and epochs_no_improve >= cfg.early_stopping_patience
   710→                ):
   711→                    logger.info(
   712→                        "Early stopping at epoch %d (no improvement for %d epochs)",
   713→                        epoch,
   714→                        epochs_no_improve,
   715→                    )
   716→                    break
   717→            else:
   718→                # No validation: track training loss only
   719→                scheduler.step(avg_train_loss)
   720→                if avg_train_loss < best_val_loss:
   721→                    best_val_loss = avg_train_loss
   722→                    best_epoch = epoch
   723→                    epochs_no_improve = 0
   724→                    self._best_state = copy.deepcopy(self.model.state_dict())
   725→                else:
   726→                    epochs_no_improve += 1
   727→
   728→                if epoch % 10 == 0 or epoch == 1:
   729→                    logger.info(
   730→                        "Epoch %3d/%d  train_loss=%.4f  lr=%.2e",
   731→                        epoch,
   732→                        cfg.epochs,
   733→                        avg_train_loss,
   734→                        optimizer.param_groups[0]["lr"],
   735→                    )
   736→
   737→                if (
   738→                    cfg.early_stopping_patience > 0
   739→                    and epochs_no_improve >= cfg.early_stopping_patience
   740→                ):
   741→                    logger.info(
   742→                        "Early stopping at epoch %d (no improvement for %d epochs)",
   743→                        epoch,
   744→                        epochs_no_improve,
   745→                    )
   746→                    break
   747→
   748→        # --- Restore best weights ---
   749→        if self._best_state is not None:
   750→            self.model.load_state_dict(self._best_state)
   751→            logger.info("Restored best model weights from epoch %d", best_epoch)
   752→
   753→        self.model.eval()
   754→        self.is_trained = True
   755→
   756→        metrics: dict[str, Any] = {
   757→            "epochs_run": epoch,
   758→            "best_epoch": best_epoch,
   759→            "train_loss_best": best_train_loss,
   760→            "train_loss_final": train_loss_history[-1] if train_loss_history else None,
   761→            "n_samples_train": int(len(y)),
   762→            "n_samples_val": int(len(y_val)) if has_val else 0,
   763→            "device": str(self.device),
   764→            "loss_history": train_loss_history,
   765→            "lr_history": lr_history,
   766→        }
   767→
   768→        if has_val and val_loss_history:
   769→            metrics["val_loss_best"] = best_val_loss
   770→            metrics["val_loss_final"] = val_loss_history[-1]
   771→
   772→        logger.info(
   773→            "LSTM training complete – epochs=%d, best_epoch=%d, best_train_loss=%.4f%s",
   774→            metrics["epochs_run"],
   775→            best_epoch,
   776→            best_train_loss,
   777→            f", best_val_loss={best_val_loss:.4f}" if has_val else "",
   778→        )
   779→
   780→        return metrics
   781→
   782→    # ------------------------------------------------------------------
   783→    # Validation loss helper
   784→    # ------------------------------------------------------------------
   785→
   786→    def _eval_loss(
   787→        self,
   788→        loader: torch.utils.data.DataLoader,
   789→        criterion: nn.Module,
   790→    ) -> float:
   791→        """Compute average loss over a data loader (no gradient)."""
   792→        self.model.eval()
   793→        total_loss = 0.0
   794→        n_batches = 0
   795→        with torch.no_grad():
   796→            for batch_xh, batch_xa, batch_y in loader:
   797→                batch_xh = batch_xh.to(self.device)
   798→                batch_xa = batch_xa.to(self.device)
   799→                batch_y = batch_y.to(self.device)
   800→                logits = self.model(batch_xh, batch_xa)
   801→                loss = criterion(logits, batch_y)
   802→                total_loss += loss.item()
   803→                n_batches += 1
   804→        return total_loss / max(n_batches, 1)
   805→
   806→    # ==================================================================
   807→    # Prediction
   808→    # ==================================================================
   809→
   810→    def predict(
   811→        self,
   812→        X_home: np.ndarray,
   813→        X_away: np.ndarray,
   814→    ) -> dict[str, Any]:
   815→        """Predict match outcome probabilities.
   816→
   817→        Parameters
   818→        ----------
   819→        X_home : np.ndarray, shape (N, seq_len, input_size) or (seq_len, input_size)
   820→        X_away : np.ndarray, same shape as X_home
   821→
   822→        Returns
   823→        -------
   824→        dict
   825→            For a **single** sample: ``{home_win, draw, away_win,
   826→            predicted_class, confidence}``.  Probabilities are softmax-
   827→            normalised and sum to exactly 1.0.
   828→        """
   829→        if not self.is_trained or self.model is None:
   830→            raise RuntimeError("Model has not been trained or loaded yet.")
   831→
   832→        X_home = np.asarray(X_home, dtype=np.float32)
   833→        X_away = np.asarray(X_away, dtype=np.float32)
   834→
   835→        # Ensure 3-D
   836→        if X_home.ndim == 2:
   837→            X_home = X_home[np.newaxis, ...]
   838→        if X_away.ndim == 2:
   839→            X_away = X_away[np.newaxis, ...]
   840→
   841→        dataset = _SingleMatchDataset(X_home, X_away)
   842→        loader = torch.utils.data.DataLoader(
   843→            dataset, batch_size=len(dataset), shuffle=False
   844→        )
   845→
   846→        self.model.eval()
   847→        all_probs: list[np.ndarray] = []
   848→
   849→        with torch.no_grad():
   850→            for batch_xh, batch_xa in loader:
   851→                batch_xh = batch_xh.to(self.device)
   852→                batch_xa = batch_xa.to(self.device)
   853→                logits = self.model(batch_xh, batch_xa)  # (B, 3)
   854→                probs = torch.softmax(logits, dim=1).cpu().numpy()
   855→                all_probs.append(probs)
   856→
   857→        probs_matrix = np.concatenate(all_probs, axis=0)  # (N, 3)
   858→
   859→        # For the common single-sample case, return a flat dict
   860→        if probs_matrix.shape[0] == 1:
   861→            p = probs_matrix[0]
   862→            # Normalise to exactly 1.0
   863→            p = p / p.sum()
   864→            pred_idx = int(np.argmax(p))
   865→            return {
   866→                "home_win": float(p[0]),
   867→                "draw": float(p[1]),
   868→                "away_win": float(p[2]),
   869→                "predicted_class": _CLASS_IDX_TO_LABEL[pred_idx],
   870→                "confidence": float(p[pred_idx]),
   871→            }
   872→
   873→        # Batch case: return list of dicts
   874→        results = []
   875→        for i in range(probs_matrix.shape[0]):
   876→            p = probs_matrix[i]
   877→            p = p / p.sum()
   878→            pred_idx = int(np.argmax(p))
   879→            results.append(
   880→                {
   881→                    "home_win": float(p[0]),
   882→                    "draw": float(p[1]),
   883→                    "away_win": float(p[2]),
   884→                    "predicted_class": _CLASS_IDX_TO_LABEL[pred_idx],
   885→                    "confidence": float(p[pred_idx]),
   886→                }
   887→            )
   888→        return {"batch": results}
   889→
   890→    # ==================================================================
   891→    # Evaluation
   892→    # ==================================================================
   893→
   894→    def evaluate(
   895→        self,
   896→        X_home: np.ndarray,
   897→        X_away: np.ndarray,
   898→        y: np.ndarray,
   899→    ) -> dict[str, Any]:
   900→        """Evaluate the model on a labelled dataset.
   901→
   902→        Computes: accuracy, per-class precision / recall / F1, macro
   903→        averages, Brier score, log-loss, and confusion matrix.
   904→
   905→        Parameters
   906→        ----------
   907→        X_home, X_away : np.ndarray, shape (N, seq_len, input_size)
   908→        y : np.ndarray, shape (N,), dtype int
   909→
   910→        Returns
   911→        -------
   912→        dict
   913→            Comprehensive evaluation metrics.
   914→        """
   915→        if not self.is_trained or self.model is None:
   916→            raise RuntimeError("Model has not been trained or loaded yet.")
   917→
   918→        X_home = np.asarray(X_home, dtype=np.float32)
   919→        X_away = np.asarray(X_away, dtype=np.float32)
   920→        y = np.asarray(y, dtype=np.int64)
   921→
   922→        if X_home.ndim == 2:
   923→            X_home = X_home[np.newaxis, ...]
   924→        if X_away.ndim == 2:
   925→            X_away = X_away[np.newaxis, ...]
   926→
   927→        dataset = _MatchSequenceDataset(X_home, X_away, y)
   928→        loader = torch.utils.data.DataLoader(
   929→            dataset, batch_size=self.config.batch_size, shuffle=False
   930→        )
   931→
   932→        self.model.eval()
   933→        all_probs: list[np.ndarray] = []
   934→        all_labels: list[int] = []
   935→
   936→        with torch.no_grad():
   937→            for batch_xh, batch_xa, batch_y in loader:
   938→                batch_xh = batch_xh.to(self.device)
   939→                batch_xa = batch_xa.to(self.device)
   940→                logits = self.model(batch_xh, batch_xa)
   941→                probs = torch.softmax(logits, dim=1).cpu().numpy()
   942→                all_probs.append(probs)
   943→                all_labels.extend(batch_y.numpy().tolist())
   944→
   945→        proba = np.concatenate(all_probs, axis=0)  # (N, 3)
   946→        y_arr = np.array(all_labels, dtype=np.int64)
   947→        preds = np.argmax(proba, axis=1)
   948→
   949→        n = len(y_arr)
   950→
   951→        # --- Overall accuracy ---
   952→        correct = int(np.sum(preds == y_arr))
   953→        accuracy = correct / n
   954→
   955→        # --- Log loss ---
   956→        eps = 1e-15
   957→        clipped = np.clip(proba, eps, 1.0 - eps)
   958→        eye = np.eye(_N_CLASSES)[y_arr]
   959→        log_loss_val = float(
   960→            -np.mean(np.sum(eye * np.log(clipped), axis=1))
   961→        )
   962→
   963→        # --- Brier score (multi-class) ---
   964→        brier = float(np.mean(np.sum((proba - eye) ** 2, axis=1)))
   965→
   966→        # --- Per-class metrics ---
   967→        class_metrics: dict[str, dict[str, float]] = {}
   968→        for cls_idx, cls_label in _CLASS_IDX_TO_LABEL.items():
   969→            tp = int(np.sum((preds == cls_idx) & (y_arr == cls_idx)))
   970→            fp = int(np.sum((preds == cls_idx) & (y_arr != cls_idx)))
   971→            fn = int(np.sum((preds != cls_idx) & (y_arr == cls_idx)))
   972→
   973→            precision = tp / (tp + fp) if (tp + fp) > 0 else 0.0
   974→            recall = tp / (tp + fn) if (tp + fn) > 0 else 0.0
   975→            f1 = (
   976→                2 * precision * recall / (precision + recall)
   977→                if (precision + recall) > 0
   978→                else 0.0
   979→            )
   980→
   981→            class_metrics[cls_label] = {
   982→                "precision": precision,
   983→                "recall": recall,
   984→                "f1": f1,
   985→                "support": int(np.sum(y_arr == cls_idx)),
   986→            }
   987→
   988→        # Macro averages
   989→        macro_precision = float(
   990→            np.mean([m["precision"] for m in class_metrics.values()])
   991→        )
   992→        macro_recall = float(
   993→            np.mean([m["recall"] for m in class_metrics.values()])
   994→        )
   995→        macro_f1 = float(
   996→            np.mean([m["f1"] for m in class_metrics.values()])
   997→        )
   998→
   999→        # --- Confusion matrix ---
  1000→        cm = np.zeros((_N_CLASSES, _N_CLASSES), dtype=int)
  1001→        for true, pred in zip(y_arr, preds):
  1002→            cm[int(true)][int(pred)] += 1
  1003→
  1004→        # --- Calibration: expected calibration error (ECE) ---
  1005→        ece = self._compute_ece(proba, y_arr, n_bins=10)
  1006→
  1007→        return {
  1008→            "n_samples": n,
  1009→            "accuracy": accuracy,
  1010→            "log_loss": log_loss_val,
  1011→            "brier_score": brier,
  1012→            "ece": ece,
  1013→            "macro_precision": macro_precision,
  1014→            "macro_recall": macro_recall,
  1015→            "macro_f1": macro_f1,
  1016→            "per_class": class_metrics,
  1017→            "confusion_matrix": cm.tolist(),
  1018→        }
  1019→
  1020→    # ------------------------------------------------------------------
  1021→    # Expected Calibration Error
  1022→    # ------------------------------------------------------------------
  1023→
  1024→    @staticmethod
  1025→    def _compute_ece(
  1026→        proba: np.ndarray, y: np.ndarray, n_bins: int = 10
  1027→    ) -> float:
  1028→        """Compute Expected Calibration Error.
  1029→
  1030→        Bins predictions by confidence and measures the gap between
  1031→        average confidence and average accuracy in each bin.
  1032→        """
  1033→        confidences = np.max(proba, axis=1)
  1034→        predictions = np.argmax(proba, axis=1)
  1035→        accuracies = (predictions == y).astype(np.float32)
  1036→
  1037→        bin_boundaries = np.linspace(0.0, 1.0, n_bins + 1)
  1038→        ece = 0.0
  1039→        for i in range(n_bins):
  1040→            lo, hi = bin_boundaries[i], bin_boundaries[i + 1]
  1041→            if i == n_bins - 1:
  1042→                mask = (confidences >= lo) & (confidences <= hi)
  1043→            else:
  1044→                mask = (confidences >= lo) & (confidences < hi)
  1045→            if mask.sum() == 0:
  1046→                continue
  1047→            avg_conf = confidences[mask].mean()
  1048→            avg_acc = accuracies[mask].mean()
  1049→            ece += (mask.sum() / len(y)) * abs(avg_conf - avg_acc)
  1050→        return float(ece)
  1051→

  1053→    # Persistence
  1054→    # ==================================================================
  1055→
  1056→    def save(self, name: str = "lstm_v1") -> str:
  1057→        """Save model weights and config to disk.
  1058→
  1059→        Creates:
  1060→        - ``<model_dir>/<name>.pt`` — PyTorch state dict
  1061→        - ``<model_dir>/<name>_meta.joblib`` — config + feature columns
  1062→
  1063→        Parameters
  1064→        ----------
  1065→        name : str
  1066→            Model identifier (without extension).
  1067→
  1068→        Returns
  1069→        -------
  1070→        str
  1071→            Absolute path to the saved model file.
  1072→        """
  1073→        if not self.is_trained or self.model is None:
  1074→            raise RuntimeError("Cannot save – model has not been trained yet.")
  1075→
  1076→        self.model_dir.mkdir(parents=True, exist_ok=True)
  1077→
  1078→        # PyTorch weights
  1079→        model_path = self.model_dir / f"{name}.pt"
  1080→        torch.save(self.model.state_dict(), str(model_path))
