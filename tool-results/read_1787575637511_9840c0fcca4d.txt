   126→   126→        if (res.status === 401) throw new Error('Sign in to load model predictions.')

   128→   128→        throw new Error(body.error || `Prediction engine returned ${res.status}`)
   129→   129→      }
   130→   130→      const data = await res.json()
   131→   131→      setComputePredictions(data.predictions || [])
   132→   132→      setComputeState('success')

   134→   134→      setComputeError(err instanceof Error ? err.message : 'Failed to compute predictions.')
   135→   135→      setComputeState('error')
   136→   136→    }

   141→   141→    setViewState('loading')
   142→   142→    setErrorMessage('')
   143→   143→
   144→   144→    try {
   145→   145→      const headers: Record<string, string> = {}
   146→   146→      if (token) headers['Authorization'] = `Bearer ${token}`
   147→   147→
   148→   148→      // Fetch predictions
   149→   149→      const res = await fetch('/api/predictions', { headers })

   151→   151→        const body = await res.json().catch(() => ({}))
   152→   152→        throw new Error(body.error || `Request failed (${res.status})`)
   153→   153→      }
   154→   154→      const data = await res.json()
   155→   155→      const fetched: PredictionRecord[] = Array.isArray(data) ? data : data.predictions || []
   156→   156→      setPredictions(fetched)
   157→   157→
   158→   158→      // Fetch leaderboard
   159→   159→      const lbRes = await fetch('/api/leaderboard')

   165→   165→        setLeaderboardCount(entries.length)

   168→   168→      // Determine view state
   169→   169→      if (fetched.length === 0) {
   170→   170→        setViewState('empty')
   171→   171→      } else {
   172→   172→        setViewState('success')
   173→   173→      }
   174→   174→    } catch (err) {
   175→   175→      const msg = err instanceof Error ? err.message : 'Failed to load predictions. Check your connection and try again.'
   176→   176→      setErrorMessage(msg)
   177→   177→      setViewState('error')
   178→   178→    }
   179→   179→  }, [token, user?.id])

   196→   196→  // Separate predictions: resolved (match played) vs pending
   197→   197→  const resolvedPredictions = useMemo(
   198→   198→    () => predictions.filter(p => p.actualHomeGoals != null && p.actualAwayGoals != null),
   199→   199→    [predictions],
   200→   200→  )
   201→   201→  const pendingPredictions = useMemo(
   202→   202→    () => predictions.filter(p => p.actualHomeGoals == null || p.actualAwayGoals == null),
   203→   203→    [predictions],

   220→   220→    if (sortBy === 'model') arr.sort((a, b) => a.model.localeCompare(b.model))

   223→   223→
   224→   224→  // Stats from DB user record
   225→   225→  const accuracy = user?.predictionAccuracy ?? 0
   226→   226→  const streak = user?.predictionStreak ?? 0

   230→   230→  // Accuracy by Model — computed from real resolved predictions

   236→   236→      if (p.isCorrect) map[p.model].correct++

   238→   238→    return Object.entries(map)
   239→   239→      .map(([model, { total, correct }]) => ({
   240→   240→        model,
   241→   241→        accuracy: total > 0 ? Math.round((correct / total) * 100) : 0,
   242→   242→        total,

   244→   244→      .sort((a, b) => b.accuracy - a.accuracy)

   282→   282→    const homeGoals = parseInt(scores.home, 10)
   283→   283→    const awayGoals = parseInt(scores.away, 10)

   285→   285→      toast.error('Invalid score', { description: 'Scores must be non-negative integers.' })
   286→   286→      return
   287→   287→    }

   289→   289→    const storeMatch = matchMap[matchId]
   290→   290→    setSubmittingMatchId(matchId)

   292→   292→      const res = await fetch('/api/predictions', {
   293→   293→        method: 'POST',
   294→   294→        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
   295→   295→        body: JSON.stringify({
   296→   296→          matchId,
   297→   297→          predictedHomeGoals: homeGoals,
   298→   298→          predictedAwayGoals: awayGoals,
   299→   299→          predictedOutcome: choice,
   300→   300→          confidence: quickConfidence,
   301→   301→          model: 'user',
   302→   302→          homeTeam: storeMatch?.homeTeam?.name ?? null,
   303→   303→          awayTeam: storeMatch?.awayTeam?.name ?? null,
   304→   304→          competition: storeMatch?.competition ?? null,
   305→   305→          matchDate: storeMatch?.date ?? null,
   306→   306→        }),
   307→   307→      })
   308→   308→      if (!res.ok) {
   309→   309→        const body = await res.json().catch(() => ({}))
   310→   310→        throw new Error(body.error || `Submission failed (${res.status})`)
   311→   311→      }
   312→   312→      toast.success('Prediction submitted!')
   313→   313→      fetchData() // refresh
   314→   314→    } catch (err) {
   315→   315→      const msg = err instanceof Error ? err.message : 'Failed to submit prediction'
   316→   316→      toast.error('Prediction failed', { description: msg })
   317→   317→    } finally {
   318→   318→      setSubmittingMatchId(null)
   319→   319→    }
   320→   320→  }, [token, quickScores, quickConfidence, matchMap, fetchData])
   321→   321→
   322→   322→  const updateScore = useCallback((matchId: string, side: 'home' | 'away', value: string) => {

   329→   329→        away: side === 'away' ? cleaned : (prev[matchId]?.away ?? ''),
   330→   330→      },
   331→   331→    }))
   332→   332→  }, [])
   333→   333→
   334→   334→  // Check which upcoming matches already have predictions
   335→   335→  const predictedMatchIds = useMemo(() => new Set(predictions.map(p => p.matchId)), [predictions])
   336→   336→
   337→   337→  // ── Render: LOADING ─────────────────────────────────────────────────────
   338→   338→
   339→   339→  if (viewState === 'loading') {
   340→   340→    return (
   341→   341→      <div className="space-y-5 animate-fade-in-up">
   342→   342→        <div className="flex items-center gap-3">
   343→   343→          <Skeleton className="size-10 rounded-xl" />
   344→   344→          <div className="flex-1 space-y-2">
   345→   345→            <Skeleton className="h-4 w-48" />
   346→   346→            <Skeleton className="h-3 w-72" />
   347→   347→          </div>
   348→   348→        </div>

   355→   355→        <Skeleton className="h-64 rounded-xl" />

   372→   372→          </div>
   373→   373→        </div>
   374→   374→        <DataState
   375→   375→          type="error"
   376→   376→          message={errorMessage || 'Failed to load predictions'}
   377→   377→          onRetry={fetchData}
   378→   378→        />
   379→   379→      </div>
   380→   380→    )
   381→   381→  }
   382→   382→
   383→   383→  // ── Render: EMPTY ───────────────────────────────────────────────────────
   384→   384→
   385→   385→  if (viewState === 'empty') {
   386→   386→    return (

   388→   388→        {/* Header */}
   389→   389→        <div className="flex items-center gap-3">
   390→   390→          <div className="flex size-10 items-center justify-center rounded-xl bg-primary/15">
   391→   391→            <Target className="size-5 text-primary" />
   392→   392→          </div>
   393→   393→          <div className="flex-1">

   397→   397→        </div>

   401→   401→          {[
   402→   402→            { icon: Target, label: 'Accuracy', value: `${accuracy}%`, color: 'text-primary' },

   421→   421→        {/* Quick Predict — still visible when empty so user can start */}
