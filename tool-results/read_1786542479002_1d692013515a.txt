   360→   360→      if (token) headers['Authorization'] = `Bearer ${token}`
   361→   361→
   362→   362→      // Map activeTab to football-data.org status
   363→   363→      let statusParam = 'FINISHED'
   364→   364→      if (activeTab === 'upcoming') statusParam = 'SCHEDULED,TIMED'
   365→   365→      else if (activeTab === 'live') statusParam = 'IN_PLAY,PAUSED'
   366→   366→      else if (activeTab === 'all') statusParam = 'FINISHED'
   367→   367→
   368→   368→      const leagueParam = leagueFilter === 'all' ? 'PL' : leagueFilter
   369→   369→      const res = await fetch(`/api/football-data?action=matches&competition=${leagueParam}&status=${statusParam}`, { headers })
   370→   370→      const data: any = await res.json()
   371→   371→      const matches = (data.data || []).map((m: any) => ({
   372→   372→        ...m,
   373→   373→        id: m.id || `fd-${Math.random().toString(36).slice(2)}`,
   374→   374→        competition: m.competition,
   375→   375→        homeTeam: m.homeTeam,
   376→   376→        awayTeam: m.awayTeam,
   377→   377→        homeScore: m.homeScore,
   378→   378→        awayScore: m.awayScore,
   379→   379→        status: m.status,
   380→   380→        date: m.date,
   381→   381→        matchday: m.matchday,
   382→   382→      }))
   383→   383→      setMatchesData(matches)
   384→   384→    } catch (err) {
   385→   385→      console.error('Failed to fetch matches:', err)
   386→   386→    } finally {
   387→   387→      setLoading(false)
   388→   388→      setLastRefresh(new Date())
   389→   389→    }
   390→   390→  }, [activeTab, leagueFilter, searchQuery, stageFilter, groupFilter, token])
   391→   391→
   392→   392→  useEffect(() => { fetchMatches() }, [fetchMatches])
   393→   393→
   394→   394→  // Auto-refresh live
   395→   395→  useEffect(() => {
   396→   396→    const iv = setInterval(() => { if (activeTab === 'live') fetchMatches() }, 15000)
   397→   397→    return () => clearInterval(iv)
   398→   398→  }, [activeTab, fetchMatches])
   399→   399→
   400→   400→  // Sort
   401→   401→  const sortedMatches = useMemo(() => {
   402→   402→    const arr = [...matchesData]
   403→   403→    switch (sortBy) {
   404→   404→      case 'elo': return arr.sort((a, b) => Math.abs((b.homeEloBefore ?? 0) - (b.awayEloBefore ?? 0)) - Math.abs((a.homeEloBefore ?? 0) - (a.awayEloBefore ?? 0)))
   405→   405→      case 'xg': return arr.sort((a, b) => (b.homeXg + b.awayXg) - (a.homeXg + a.awayXg))
   406→   406→      default: return arr
   407→   407→    }
   408→   408→  }, [matchesData, sortBy])
   409→   409→
   410→   410→  const allMatches = matchesData
   411→   411→  const liveCount = allMatches.filter((m) => m.status === 'live' || m.status === 'halftime').length
   412→   412→  const availableStages = useMemo(() => STAGES.filter((s) => allMatches.some((m) => m.stage === s)), [allMatches])
   413→   413→  const availableGroups = useMemo(() => GROUPS.filter((g) => allMatches.some((m) => m.group === g)), [allMatches])
   414→   414→
   415→   415→  return (
   416→   416→    <div className="flex flex-col gap-4 animate-fade-in-up">
   417→   417→      {/* Header */}
   418→   418→      <div className="flex items-center gap-3">
   419→   419→        <div className="flex size-10 items-center justify-center rounded-xl bg-primary/15"><Zap className="size-5 text-primary" /></div>
   420→   420→        <div>
   421→   421→          <h1 className="text-2xl font-bold tracking-tight">Matches</h1>
   422→   422→          <p className="text-sm text-muted-foreground">Browse, filter, and predict match outcomes</p>
   423→   423→        </div>
   424→   424→      </div>
   425→   425→
   426→   426→      {/* Filter Bar */}
   427→   427→      <div className="glass-card-premium rounded-xl p-4 space-y-3">
   428→   428→        <div className="flex flex-col sm:flex-row gap-3">
   429→   429→          <Select value={leagueFilter} onValueChange={setLeagueFilter}>
   430→   430→            <SelectTrigger className="h-9 w-full sm:w-[180px] bg-muted/50 border-border text-sm">
   431→   431→              <Trophy className="size-3.5 mr-1.5 text-muted-foreground" /><SelectValue placeholder="League" />
   432→   432→            </SelectTrigger>
   433→   433→            <SelectContent className="glass-card border-border max-h-[280px]">
   434→   434→              {LEAGUES.map((l) => <SelectItem key={l.code} value={l.code}>{l.name}</SelectItem>)}
   435→   435→            </SelectContent>
   436→   436→          </Select>
   437→   437→          <div className="relative flex-1">
   438→   438→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   439→   439→            <Input placeholder="Search by team name..." value={searchQuery} onChange={(e) => setSearchQuery(e.target.value)} className="pl-9 h-9 bg-muted/50 border-border text-sm" />
   440→   440→          </div>
   441→   441→          <Select value={stageFilter} onValueChange={setStageFilter}>
   442→   442→            <SelectTrigger className="h-9 w-full sm:w-[160px] bg-muted/50 border-border text-sm">
   443→   443→              <Filter className="size-3.5 mr-1.5 text-muted-foreground" /><SelectValue placeholder="Stage" />
   444→   444→            </SelectTrigger>
   445→   445→            <SelectContent className="glass-card border-border">
   446→   446→              <SelectItem value="all">All Stages</SelectItem>
   447→   447→              {availableStages.map((s) => <SelectItem key={s} value={s}>{s}</SelectItem>)}
   448→   448→            </SelectContent>
   449→   449→          </Select>
   450→   450→          <Select value={groupFilter} onValueChange={setGroupFilter}>
   451→   451→            <SelectTrigger className="h-9 w-full sm:w-[120px] bg-muted/50 border-border text-sm"><SelectValue placeholder="Group" /></SelectTrigger>
   452→   452→            <SelectContent className="glass-card border-border">
   453→   453→              <SelectItem value="all">All Groups</SelectItem>
   454→   454→              {availableGroups.map((g) => <SelectItem key={g} value={g}>Group {g}</SelectItem>)}
   455→   455→            </SelectContent>
   456→   456→          </Select>
   457→   457→          <Select value={sortBy} onValueChange={(v) => setSortBy(v as SortKey)}>
   458→   458→            <SelectTrigger className="h-9 w-full sm:w-[130px] bg-muted/50 border-border text-sm">
   459→   459→              <ArrowUpDown className="size-3.5 mr-1.5 text-muted-foreground" /><SelectValue placeholder="Sort" />
   460→   460→            </SelectTrigger>
   461→   461→            <SelectContent className="glass-card border-border">
   462→   462→              <SelectItem value="date">By Date</SelectItem>
   463→   463→              <SelectItem value="elo">By ELO Diff</SelectItem>
   464→   464→              <SelectItem value="xg">By xG Total</SelectItem>
   465→   465→            </SelectContent>
   466→   466→          </Select>
   467→   467→          <Button variant="outline" size="icon" className="h-9 w-9 shrink-0 border-border bg-muted/50 hover:bg-accent" onClick={() => fetchMatches()} disabled={loading} aria-label="Refresh">
   468→   468→            <RefreshCw className={cn('size-4', loading && 'animate-spin')} />
   469→   469→          </Button>
   470→   470→        </div>
   471→   471→        {/* Group quick-buttons */}
   472→   472→        <div className="flex items-center gap-1.5 flex-wrap">
   473→   473→          <span className="text-[10px] text-muted-foreground mr-1">Groups:</span>
   474→   474→          {GROUPS.map((g) => (
   475→   475→            <button key={g} onClick={() => setGroupFilter(groupFilter === g ? 'all' : g)}
   476→   476→              className={cn('flex size-7 items-center justify-center rounded-md text-[10px] font-bold transition-all',
   477→   477→                groupFilter === g ? 'bg-primary/20 text-primary ring-glow-emerald' : 'bg-muted/30 text-muted-foreground hover:bg-muted/50 hover:text-foreground')}>
   478→   478→              {g}
   479→   479→            </button>
   480→   480→          ))}
   481→   481→        </div>
   482→   482→      </div>
   483→   483→
   484→   484→      {/* Tabs */}
   485→   485→      <Tabs value={activeTab} onValueChange={(v) => setActiveTab(v as MatchTab)} className="w-full">
   486→   486→        <TabsList className="glass-card w-full h-10 bg-muted/30 p-1 rounded-lg">
   487→   487→          {STATUS_TABS.map((tab) => {
   488→   488→            const count = tab.value === 'live' ? liveCount : allMatches.length
   489→   489→            return (
   490→   490→              <TabsTrigger key={tab.value} value={tab.value} className="flex-1 h-8 text-xs font-semibold rounded-md transition-all gap-1.5 data-[state=active]:bg-primary/15 data-[state=active]:text-primary data-[state=active]:shadow-sm">
   491→   491→                {tab.value === 'live' && liveCount > 0 && (
   492→   492→                  <span className="relative flex size-1.5"><span className="absolute inline-flex size-full animate-ping rounded-full bg-red-400 opacity-75" /><span className="relative inline-flex size-1.5 rounded-full bg-red-500" /></span>
   493→   493→                )}
   494→   494→                {tab.label}
   495→   495→                {count > 0 && <span className="ml-0.5 text-[10px] opacity-60">({count})</span>}
   496→   496→              </TabsTrigger>
   497→   497→            )
   498→   498→          })}
   499→   499→        </TabsList>
   500→   500→
   501→   501→        {STATUS_TABS.map((tab) => (
   502→   502→          <TabsContent key={tab.value} value={tab.value} className="mt-4">
   503→   503→            {loading && sortedMatches.length === 0 && (
   504→   504→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   505→   505→                {Array.from({ length: 6 }).map((_, i) => <MatchCardSkeleton key={i} />)}
   506→   506→              </div>
   507→   507→            )}
   508→   508→
   509→   509→            {!loading && sortedMatches.length === 0 && (
   510→   510→              <div className="flex flex-col items-center justify-center py-16 text-center">
   511→   511→                <div className="size-16 rounded-full bg-muted/50 flex items-center justify-center mb-4"><Eye className="size-7 text-muted-foreground" /></div>
   512→   512→                <h3 className="text-sm font-semibold mb-1">No {tab.value === 'all' ? '' : tab.value} matches found</h3>
   513→   513→                <p className="text-xs text-muted-foreground max-w-[260px]">
   514→   514→                  {searchQuery || stageFilter !== 'all' || groupFilter !== 'all' ? 'Try adjusting your filters.' : 'No matches available. Check back later!'}
   515→   515→                </p>
   516→   516→                {(searchQuery || stageFilter !== 'all' || groupFilter !== 'all') && (
   517→   517→                  <Button variant="outline" size="sm" className="mt-4 h-8 text-xs border-border" onClick={() => { setLeagueFilter('PL'); setSearchQuery(''); setStageFilter('all'); setGroupFilter('all') }}>Clear Filters</Button>
   518→   518→                )}
   519→   519→              </div>
   520→   520→            )}
   521→   521→
   522→   522→            {sortedMatches.length > 0 && (
   523→   523→              <>
   524→   524→                <div className="flex items-center justify-between mb-2 px-1">
   525→   525→                  <span className="text-[11px] text-muted-foreground">{sortedMatches.length} match{sortedMatches.length !== 1 ? 'es' : ''}{loading && ' (refreshing...)'}</span>
   526→   526→                  <span className="text-[10px] text-muted-foreground/50">Updated {lastRefresh.toLocaleTimeString()}</span>
   527→   527→                </div>
   528→   528→                <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   529→   529→                  {sortedMatches.map((match, idx) => (
   530→   530→                    <div key={match.id} style={{ animationDelay: `${Math.min(idx * 50, 300)}ms` }}><MatchCard match={match} /></div>
   531→   531→                  ))}
   532→   532→                </div>
   533→   533→              </>
   534→   534→            )}
   535→   535→          </TabsContent>
   536→   536→        ))}
   537→   537→      </Tabs>
   538→   538→    </div>
