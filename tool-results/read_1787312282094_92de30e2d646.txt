     1→'use client'
     2→
     3→import React, { useState, useEffect, useCallback, useMemo } from 'react'
     4→import { useElasticoStore } from '@/store/use-elastico-store'
     5→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
     6→import { Badge } from '@/components/ui/badge'
     7→import { Button } from '@/components/ui/button'
     8→import { Skeleton } from '@/components/ui/skeleton'
     9→import { Input } from '@/components/ui/input'
    10→import { DataState } from '@/components/elastico/primitives'
    11→import { Label } from '@/components/ui/label'
    12→import { Slider } from '@/components/ui/slider'
    13→import {
    14→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    15→} from '@/components/ui/select'
    16→import {
    17→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    18→} from '@/components/ui/table'
    19→import {
    20→  BarChart, Bar, XAxis, YAxis, CartesianGrid, Tooltip as RTooltip, ResponsiveContainer, Cell,
    21→} from 'recharts'
    22→import { axisProps, cartesianGridProps, tooltipContentStyle, tooltipLabelStyle, chartColor } from '@/lib/chart-theme'
    23→import {
    24→  Target, ArrowUpDown, Flame, CheckCircle2, XCircle, Crown, Download,
    25→  Trophy, Zap, TrendingUp, Send, AlertCircle,
    26→} from 'lucide-react'
    27→import { cn } from '@/lib/utils'
    28→import { generateCSV } from '@/lib/export'
    29→import { toast } from 'sonner'
    30→
    31→// ── Types ──────────────────────────────────────────────────────────────────
    32→
    33→/** Shape returned by GET /api/predictions */
    34→interface PredictionRecord {
    35→  id: string
    36→  matchId: string
    37→  homeTeam: string
    38→  awayTeam: string
    39→  competition: string
    40→  predictedHomeGoals: number
    41→  predictedAwayGoals: number
    42→  predictedOutcome: string
    43→  confidence: number
    44→  model: string
    45→  actualHomeGoals: number | null
    46→  actualAwayGoals: number | null
    47→  isCorrect: boolean | null
    48→  points: number
    49→  matchDate: string | null
    50→  createdAt: string
    51→}
    52→
    53→interface LeaderboardEntry {
    54→  rank: number
    55→  id: string
    56→  name: string | null
    57→  predictionAccuracy: number
    58→  predictionStreak: number
    59→  bestStreak: number
    60→  totalPredictions: number
    61→}
    62→
    63→type ViewState = 'loading' | 'error' | 'empty' | 'success'
    64→
    65→// ═══════════════════════════════════════════════════════════════════════════════
    66→
    67→export default function PredictionsView() {
    68→  const user = useElasticoStore(s => s.user)
    69→  const token = useElasticoStore(s => s.token)
    70→  const matches = useElasticoStore(s => s.matches)
    71→
    72→  // ── State ────────────────────────────────────────────────────────────────
    73→  const [predictions, setPredictions] = useState<PredictionRecord[]>([])
    74→  const [leaderboardPos, setLeaderboardPos] = useState<number>(0)
    75→  const [leaderboardCount, setLeaderboardCount] = useState<number>(0)
    76→  const [viewState, setViewState] = useState<ViewState>('loading')
    77→  const [errorMessage, setErrorMessage] = useState('')
    78→  const [sortBy, setSortBy] = useState<'date' | 'confidence' | 'model'>('date')
    79→  const [modelFilter, setModelFilter] = useState('all')
    80→  const [resultFilter, setResultFilter] = useState<'all' | 'correct' | 'incorrect' | 'pending'>('all')
    81→
    82→  // Quick Predict state
    83→  const [quickConfidence, setQuickConfidence] = useState(50)
    84→  const [submittingMatchId, setSubmittingMatchId] = useState<string | null>(null)
    85→  // Per-match score inputs: { [matchId]: { home: number, away: number } }
    86→  const [quickScores, setQuickScores] = useState<Record<string, { home: string; away: string }>>({})
    87→
    88→  // ── Data Fetching ────────────────────────────────────────────────────────
    89→  const fetchData = useCallback(async () => {
    90→    setViewState('loading')
    91→    setErrorMessage('')
    92→
    93→    try {
    94→      const headers: Record<string, string> = {}
    95→      if (token) headers['Authorization'] = `Bearer ${token}`
    96→
    97→      // Fetch predictions
    98→      const res = await fetch('/api/predictions', { headers })
    99→      if (!res.ok) {
   100→        const body = await res.json().catch(() => ({}))
   101→        throw new Error(body.error || `Request failed (${res.status})`)
   102→      }
   103→      const data = await res.json()
   104→      const fetched: PredictionRecord[] = Array.isArray(data) ? data : data.predictions || []
   105→      setPredictions(fetched)
   106→
   107→      // Fetch leaderboard
   108→      const lbRes = await fetch('/api/leaderboard')
   109→      if (lbRes.ok) {
   110→        const lbData = await lbRes.json()
   111→        const entries: LeaderboardEntry[] = Array.isArray(lbData) ? lbData : lbData.leaderboard || []
   112→        const pos = entries.findIndex((e: LeaderboardEntry) => e.id === user?.id)
   113→        setLeaderboardPos(pos >= 0 ? pos + 1 : 0)
   114→        setLeaderboardCount(entries.length)
   115→      }
   116→
   117→      // Determine view state
   118→      if (fetched.length === 0) {
   119→        setViewState('empty')
   120→      } else {
   121→        setViewState('success')
   122→      }
   123→    } catch (err) {
   124→      const msg = err instanceof Error ? err.message : 'Failed to load predictions. Check your connection and try again.'
   125→      setErrorMessage(msg)
   126→      setViewState('error')
   127→    }
   128→  }, [token, user?.id])
   129→
   130→  useEffect(() => { fetchData() }, [fetchData])
   131→
   132→  // ── Derived data ─────────────────────────────────────────────────────────
   133→
   134→  // Match map from store for cross-referencing (status, team colors)
   135→  const matchMap = useMemo(() => {
   136→    const map: Record<string, (typeof matches)[number]> = {}
   137→    for (const m of matches) map[m.id] = m
   138→    return map
   139→  }, [matches])
   140→
   141→  const upcomingMatches = useMemo(() => matches.filter(m => m.status === 'upcoming'), [matches])
   142→
   143→  // Separate predictions: resolved (match played) vs pending
   144→  const resolvedPredictions = useMemo(
   145→    () => predictions.filter(p => p.actualHomeGoals != null && p.actualAwayGoals != null),
   146→    [predictions],
   147→  )
   148→  const pendingPredictions = useMemo(
   149→    () => predictions.filter(p => p.actualHomeGoals == null || p.actualAwayGoals == null),
   150→    [predictions],
   151→  )
   152→
   153→  // User's model values (for filter dropdown)
   154→  const usedModels = useMemo(() => {
   155→    const set = new Set<string>()
   156→    for (const p of predictions) set.add(p.model)
   157→    return Array.from(set).sort()
   158→  }, [predictions])
   159→
   160→  const filteredResolved = useMemo(() => {
   161→    let arr = [...resolvedPredictions]
   162→    if (modelFilter !== 'all') arr = arr.filter(p => p.model === modelFilter)
   163→    if (resultFilter === 'correct') arr = arr.filter(p => p.isCorrect === true)
   164→    if (resultFilter === 'incorrect') arr = arr.filter(p => p.isCorrect === false)
   165→    if (sortBy === 'date') arr.sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   166→    if (sortBy === 'confidence') arr.sort((a, b) => b.confidence - a.confidence)
   167→    if (sortBy === 'model') arr.sort((a, b) => a.model.localeCompare(b.model))
   168→    return arr
   169→  }, [resolvedPredictions, modelFilter, resultFilter, sortBy])
   170→
   171→  // Stats from DB user record
   172→  const accuracy = user?.predictionAccuracy ?? 0
   173→  const streak = user?.predictionStreak ?? 0
   174→  const bestStreak = user?.bestStreak ?? 0
   175→  const totalPredictions = user?.totalPredictions ?? 0
   176→
   177→  // Accuracy by Model — computed from real resolved predictions
   178→  const accuracyByModel = useMemo(() => {
   179→    const map: Record<string, { total: number; correct: number }> = {}
   180→    for (const p of resolvedPredictions) {
   181→      if (!map[p.model]) map[p.model] = { total: 0, correct: 0 }
   182→      map[p.model].total++
   183→      if (p.isCorrect) map[p.model].correct++
   184→    }
   185→    return Object.entries(map)
   186→      .map(([model, { total, correct }]) => ({
   187→        model,
   188→        accuracy: total > 0 ? Math.round((correct / total) * 100) : 0,
   189→        total,
   190→      }))
   191→      .sort((a, b) => b.accuracy - a.accuracy)
   192→  }, [resolvedPredictions])
   193→
   194→  // ── Actions ──────────────────────────────────────────────────────────────
   195→
   196→  const handleExport = useCallback(() => {
   197→    if (filteredResolved.length === 0) {
   198→      toast.error('Nothing to export', { description: 'No resolved predictions to export.' })
   199→      return
   200→    }
   201→    const data = filteredResolved.map(p => ({
   202→      Date: p.matchDate ? new Date(p.matchDate).toLocaleDateString() : new Date(p.createdAt).toLocaleDateString(),
   203→      Match: `${p.homeTeam} vs ${p.awayTeam}`,
   204→      Competition: p.competition,
   205→      Prediction: p.predictedOutcome,
   206→      'Predicted Score': `${p.predictedHomeGoals}-${p.predictedAwayGoals}`,
   207→      'Actual Score': `${p.actualHomeGoals ?? '?'}-${p.actualAwayGoals ?? '?'}`,
   208→      Confidence: `${p.confidence}%`,
   209→      Model: p.model,
   210→      Result: p.isCorrect === true ? 'Correct' : p.isCorrect === false ? 'Incorrect' : 'Pending',
   211→      Points: p.points,
   212→    }))
   213→    generateCSV(data, 'elastico-predictions')
   214→    toast.success('Exported!', { description: `${data.length} predictions exported as CSV` })
   215→  }, [filteredResolved])
   216→
   217→  const handleQuickPredict = useCallback(async (matchId: string, choice: string) => {
   218→    if (!token) {
   219→      toast.error('Not authenticated')
   220→      return
   221→    }
   222→
   223→    const scores = quickScores[matchId]
   224→    if (!scores || scores.home === '' || scores.away === '') {
   225→      toast.error('Score required', { description: 'Enter your predicted score for both teams.' })
   226→      return
   227→    }
   228→
   229→    const homeGoals = parseInt(scores.home, 10)
   230→    const awayGoals = parseInt(scores.away, 10)
   231→    if (isNaN(homeGoals) || isNaN(awayGoals) || homeGoals < 0 || awayGoals < 0) {
   232→      toast.error('Invalid score', { description: 'Scores must be non-negative integers.' })
   233→      return
   234→    }
   235→
   236→    const storeMatch = matchMap[matchId]
   237→    setSubmittingMatchId(matchId)
   238→    try {
   239→      const res = await fetch('/api/predictions', {
   240→        method: 'POST',
   241→        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
   242→        body: JSON.stringify({
   243→          matchId,
   244→          predictedHomeGoals: homeGoals,
   245→          predictedAwayGoals: awayGoals,
   246→          predictedOutcome: choice,
   247→          confidence: quickConfidence,
   248→          model: 'user',
   249→          homeTeam: storeMatch?.homeTeam?.name ?? null,
   250→          awayTeam: storeMatch?.awayTeam?.name ?? null,
   251→          competition: storeMatch?.competition ?? null,
   252→          matchDate: storeMatch?.date ?? null,
   253→        }),
   254→      })
   255→      if (!res.ok) {
   256→        const body = await res.json().catch(() => ({}))
   257→        throw new Error(body.error || `Submission failed (${res.status})`)
   258→      }
   259→      toast.success('Prediction submitted!')
   260→      fetchData() // refresh
   261→    } catch (err) {
   262→      const msg = err instanceof Error ? err.message : 'Failed to submit prediction'
   263→      toast.error('Prediction failed', { description: msg })
   264→    } finally {
   265→      setSubmittingMatchId(null)
   266→    }
   267→  }, [token, quickScores, quickConfidence, matchMap, fetchData])
   268→
   269→  const updateScore = useCallback((matchId: string, side: 'home' | 'away', value: string) => {
   270→    // Allow only digits
   271→    const cleaned = value.replace(/[^0-9]/g, '')
   272→    setQuickScores(prev => ({
   273→      ...prev,
   274→      [matchId]: {
   275→        home: side === 'home' ? cleaned : (prev[matchId]?.home ?? ''),
   276→        away: side === 'away' ? cleaned : (prev[matchId]?.away ?? ''),
   277→      },
   278→    }))
   279→  }, [])
   280→
   281→  // Check which upcoming matches already have predictions
   282→  const predictedMatchIds = useMemo(() => new Set(predictions.map(p => p.matchId)), [predictions])
   283→
   284→  // ── Render: LOADING ─────────────────────────────────────────────────────
   285→
   286→  if (viewState === 'loading') {
   287→    return (
   288→      <div className="space-y-5 animate-fade-in-up">
   289→        <div className="flex items-center gap-3">
   290→          <Skeleton className="size-10 rounded-xl" />
   291→          <div className="flex-1 space-y-2">
   292→            <Skeleton className="h-4 w-48" />
   293→            <Skeleton className="h-3 w-72" />
   294→          </div>
   295→        </div>
   296→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   297→          {Array.from({ length: 4 }).map((_, i) => (
   298→            <Skeleton key={i} className="h-20 rounded-xl" />
   299→          ))}
   300→        </div>
   301→        <Skeleton className="h-72 rounded-xl" />
   302→        <Skeleton className="h-64 rounded-xl" />
   303→      </div>
   304→    )
   305→  }
   306→
   307→  // ── Render: ERROR ───────────────────────────────────────────────────────
   308→
   309→  if (viewState === 'error') {
   310→    return (
   311→      <div className="space-y-5 animate-fade-in-up">
   312→        <div className="flex items-center gap-3">
   313→          <div className="flex size-10 items-center justify-center rounded-xl bg-primary/15">
   314→            <Target className="size-5 text-primary" />
   315→          </div>
   316→          <div>
   317→            <h2 className="text-base font-bold text-primary">Predictions</h2>
   318→            <p className="text-sm text-muted-foreground">Track, analyze, and improve your prediction performance</p>
   319→          </div>
   320→        </div>
   321→        <DataState
   322→          type="error"
   323→          message={errorMessage || 'Failed to load predictions'}
   324→          onRetry={fetchData}
   325→        />
   326→      </div>
   327→    )
   328→  }
   329→
   330→  // ── Render: EMPTY ───────────────────────────────────────────────────────
   331→
   332→  if (viewState === 'empty') {
   333→    return (
   334→      <div className="space-y-5 animate-fade-in-up">
   335→        {/* Header */}
   336→        <div className="flex items-center gap-3">
   337→          <div className="flex size-10 items-center justify-center rounded-xl bg-primary/15">
   338→            <Target className="size-5 text-primary" />
   339→          </div>
   340→          <div className="flex-1">
   341→            <h2 className="text-base font-bold text-primary">Predictions</h2>
   342→            <p className="text-sm text-muted-foreground">Track, analyze, and improve your prediction performance</p>
   343→          </div>
   344→        </div>
   345→
   346→        {/* Stats (all zeros — honest) */}
   347→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   348→          {[
   349→            { icon: Target, label: 'Accuracy', value: `${accuracy}%`, color: 'text-primary' },
   350→            { icon: Zap, label: 'Total Predictions', value: totalPredictions, color: 'text-cyan-400' },
   351→            { icon: Flame, label: 'Current Streak', value: streak, color: 'text-orange-400' },
   352→            { icon: Crown, label: 'Best Streak', value: bestStreak, color: 'text-amber-400' },
   353→          ].map(s => (
   354→            <Card key={s.label} className="glass-card-premium card-hover-lift rounded-xl">
   355→              <CardContent className="flex items-center gap-3 p-4">
   356→                <div className="flex size-9 shrink-0 items-center justify-center rounded-lg bg-muted/50">
   357→                  <s.icon className={cn('size-4', s.color)} />
   358→                </div>
   359→                <div>
   360→                  <p className="text-lg font-bold leading-tight">{s.value}</p>
   361→                  <p className="text-[10px] text-muted-foreground">{s.label}</p>
   362→                </div>
   363→              </CardContent>
   364→            </Card>
   365→          ))}
   366→        </div>
   367→
   368→        {/* Quick Predict — still visible when empty so user can start */}
   369→        {upcomingMatches.length > 0 && (
   370→          <Card className="glass-card-premium rounded-xl">
   371→            <CardHeader className="pb-3">
   372→              <CardTitle className="text-sm font-bold flex items-center gap-2">
   373→                <Send className="size-4 text-primary" />
   374→                Quick Predict — Upcoming Matches
   375→              </CardTitle>
   376→            </CardHeader>
   377→            <CardContent className="space-y-3">
   378→              <div className="flex items-center gap-3 px-1 pb-2">
   379→                <Label className="text-xs text-muted-foreground whitespace-nowrap">Your confidence:</Label>
   380→                <Slider
   381→                  value={[quickConfidence]}
   382→                  onValueChange={([v]) => setQuickConfidence(v)}
   383→                  min={10}
   384→                  max={100}
   385→                  step={5}
   386→                  className="flex-1"
   387→                />
   388→                <span className="text-xs font-mono text-primary tabular-nums w-8 text-right">{quickConfidence}%</span>
   389→              </div>
   390→              {upcomingMatches.slice(0, 4).map((m) => (
   391→                <QuickPredictRow
   392→                  key={m.id}
   393→                  match={m}
   394→                  scores={quickScores[m.id] ?? { home: '', away: '' }}
   395→                  onScoreChange={(side, val) => updateScore(m.id, side, val)}
   396→                  onPredict={(choice) => handleQuickPredict(m.id, choice)}
   397→                  isSubmitting={submittingMatchId === m.id}
   398→                  alreadyPredicted={predictedMatchIds.has(m.id)}
   399→                />
   400→              ))}
   401→            </CardContent>
   402→          </Card>
   403→        )}
   404→
   405→        <DataState
   406→          type="empty"
   407→          message="No predictions yet. Use Quick Predict above or the Prediction Engine to get started."
   408→          actionLabel="Refresh"
   409→          actionOnClick={fetchData}
   410→        />
   411→      </div>
   412→    )
   413→  }
   414→
   415→  // ── Render: SUCCESS ─────────────────────────────────────────────────────
   416→
   417→  return (
   418→    <div className="space-y-5 animate-fade-in-up">
   419→      {/* Header */}
   420→      <div className="flex items-center gap-3">
   421→        <div className="flex size-10 items-center justify-center rounded-xl bg-primary/15">
   422→          <Target className="size-5 text-primary" />
   423→        </div>
   424→        <div className="flex-1">
   425→          <h2 className="text-base font-bold text-primary">Predictions</h2>
   426→          <p className="text-sm text-muted-foreground">Track, analyze, and improve your prediction performance</p>
   427→        </div>
   428→        <Button
   429→          variant="outline"
   430→          size="sm"
   431→          className="gap-1.5 border-border text-xs"
   432→          onClick={handleExport}
   433→          disabled={filteredResolved.length === 0}
   434→        >
   435→          <Download className="size-3.5" />Export CSV
   436→        </Button>
   437→      </div>
   438→
   439→      {/* ── ACCURACY STATS CARDS ── */}
   440→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   441→        {[
   442→          { icon: Target, label: 'Accuracy', value: `${accuracy}%`, color: 'text-primary' },
   443→          { icon: Zap, label: 'Total Predictions', value: totalPredictions, color: 'text-cyan-400' },
   444→          { icon: Flame, label: 'Current Streak', value: streak, color: 'text-orange-400' },
   445→          { icon: Crown, label: 'Best Streak', value: bestStreak, color: 'text-amber-400' },
   446→        ].map(s => (
   447→          <Card key={s.label} className="glass-card-premium card-hover-lift rounded-xl">
   448→            <CardContent className="flex items-center gap-3 p-4">
   449→              <div className="flex size-9 shrink-0 items-center justify-center rounded-lg bg-muted/50">
   450→                <s.icon className={cn('size-4', s.color)} />
   451→              </div>
   452→              <div>
   453→                <p className="text-lg font-bold leading-tight">{s.value}</p>
   454→                <p className="text-[10px] text-muted-foreground">{s.label}</p>
   455→              </div>
   456→            </CardContent>
   457→          </Card>
   458→        ))}
   459→      </div>
   460→
   461→      {/* ── LEADERBOARD POSITION ── */}
   462→      {leaderboardPos > 0 && (
   463→        <Card className="glass-card-premium rounded-xl border-amber-500/20">
   464→          <CardContent className="flex items-center gap-4 p-4">
   465→            <div className="flex size-10 items-center justify-center rounded-xl bg-amber-500/15">
   466→              <Trophy className="size-5 text-amber-400" />
   467→            </div>
   468→            <div>
   469→              <p className="text-sm font-semibold">Leaderboard Position</p>
   470→              <p className="text-2xl font-black text-amber-400">#{leaderboardPos}</p>
   471→            </div>
   472→            <div className="ml-auto text-right">
   473→              <p className="text-xs text-muted-foreground">
   474→                of {leaderboardCount} predictor{leaderboardCount !== 1 ? 's' : ''}
   475→              </p>
   476→            </div>
   477→          </CardContent>
   478→        </Card>
   479→      )}
   480→
   481→      {/* ── QUICK PREDICT PANEL ── */}
   482→      {upcomingMatches.length > 0 && (
   483→        <Card className="glass-card-premium rounded-xl">
   484→          <CardHeader className="pb-3">
   485→            <CardTitle className="text-sm font-bold flex items-center gap-2">
   486→              <Send className="size-4 text-primary" />
   487→              Quick Predict — Upcoming Matches
   488→            </CardTitle>
   489→          </CardHeader>
   490→          <CardContent className="space-y-3">
   491→            {/* Confidence slider */}
   492→            <div className="flex items-center gap-3 px-1 pb-2">
   493→              <Label className="text-xs text-muted-foreground whitespace-nowrap">Your confidence:</Label>
   494→              <Slider
   495→                value={[quickConfidence]}
   496→                onValueChange={([v]) => setQuickConfidence(v)}
   497→                min={10}
   498→                max={100}
   499→                step={5}
   500→                className="flex-1"
   501→              />
   502→              <span className="text-xs font-mono text-primary tabular-nums w-8 text-right">{quickConfidence}%</span>
   503→            </div>
   504→            {upcomingMatches.slice(0, 6).map((m) => (
   505→              <QuickPredictRow
   506→                key={m.id}
   507→                match={m}
   508→                scores={quickScores[m.id] ?? { home: '', away: '' }}
   509→                onScoreChange={(side, val) => updateScore(m.id, side, val)}
   510→                onPredict={(choice) => handleQuickPredict(m.id, choice)}
   511→                isSubmitting={submittingMatchId === m.id}
