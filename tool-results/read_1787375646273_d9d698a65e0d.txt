    58→function StatCard({ title, value, icon: Icon, change, color = 'emerald', suffix = '', dataClass }: {
    59→  title: string; value: string | number; icon: React.ElementType; change?: number; color?: string; suffix?: string; dataClass?: 'REAL' | 'DERIVED' | 'DEMO' | 'SIMULATION' | 'BUG' | 'MIXED'
    60→}) {
    61→  const isPositive = change !== undefined && change >= 0
    62→  const colorMap: Record<string, string> = {
    63→    emerald: 'bg-emerald-500/10 text-emerald-400',
    64→    gold: 'bg-yellow-500/10 text-yellow-400',
    65→    red: 'bg-red-500/10 text-red-400',
    66→    blue: 'bg-cyan-500/10 text-cyan-400',
    67→    purple: 'bg-purple-500/10 text-purple-400',
    68→    orange: 'bg-orange-500/10 text-orange-400',
    69→  }
    70→  return (
    71→    <Card className="glass-card glass-card-hover animate-fade-in-up">
    72→      <CardContent className="p-4">
    73→        <div className="flex items-center justify-between">
    74→          <div className="space-y-1 min-w-0">
    75→            <p className="text-xs text-muted-foreground uppercase tracking-wider flex items-center gap-1.5">
    76→              <Icon className="h-3 w-3" />
    77→              {title}
    78→            </p>
    79→            <div className="flex items-center gap-1.5">
    80→              <p className="text-2xl font-bold truncate">{value}{suffix}</p>
    81→              {dataClass && <StatusBadge variant="dataclass" value={dataClass} />}
    82→            </div>
    83→            {change !== undefined && (
    84→              <div className={`flex items-center gap-1 text-xs ${isPositive ? 'text-emerald-400' : 'text-red-400'}`}>
    85→                {isPositive ? <ArrowUpRight className="h-3 w-3" /> : <ArrowDownRight className="h-3 w-3" />}
    86→                <span>{Math.abs(change)}%</span>
    87→                <span className="text-muted-foreground">vs last period</span>
    88→              </div>
    89→            )}
    90→          </div>
    91→          <div className={`p-3 rounded-xl shrink-0 ${colorMap[color] || colorMap.emerald}`}>
    92→            <Icon className="h-5 w-5" />
    93→          </div>
    94→        </div>
    95→      </CardContent>
    96→    </Card>
    97→  )
    98→}
