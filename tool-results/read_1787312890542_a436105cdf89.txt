    86→function statusColor(status: string) {
    87→  if (status === 'PASS' || status === 'HEALED' || status === 'INTACT' || status === 'HEALTHY' || status === 'ONLINE' || status === 'STABLE' || status === 'ok') return '#00e676'
    88→  if (status === 'WARNING' || status === 'DRIFT_DETECTED' || status === 'QUARANTINED' || status === 'MODIFIED' || status === 'DEGRADED' || status === 'UNSTABLE' || status === 'warning') return '#eab308'
    89→  return '#ef4444'
    90→}
    91→
    92→function statusBgClass(status: string) {
    93→  const c = statusColor(status)
    94→  if (c === '#00e676') return 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30'
    95→  if (c === '#eab308') return 'bg-yellow-500/15 text-yellow-400 border-yellow-500/30'
    96→  return 'bg-red-500/15 text-red-400 border-red-500/30'
    97→}

    99→function CircGauge({ value, max = 100, size = 120, label, strokeColor }: {
   100→  value: number; max?: number; size?: number; label: string; strokeColor?: string
   101→}) {
   102→  const pct = Math.min(value / max, 1)
   103→  const r = (size - 16) / 2
   104→  const circ = 2 * Math.PI * r
   105→  const offset = circ * (1 - pct)
   106→  const color = strokeColor || (pct > 0.7 ? '#00e676' : pct > 0.4 ? '#eab308' : '#ef4444')
   107→
   108→  return (
   109→    <div className="flex flex-col items-center gap-2">
   110→      <svg width={size} height={size} className="drop-shadow-lg">
   111→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="oklch(0.15 0.02 260)" strokeWidth={8} />
   112→        <circle
   113→          cx={size / 2} cy={size / 2} r={r} fill="none"
   114→          stroke={color} strokeWidth={8} strokeLinecap="round"
   115→          strokeDasharray={circ} strokeDashoffset={offset}
   116→          transform={`rotate(-90 ${size / 2} ${size / 2})`}
   117→          className="transition-all duration-1000 ease-out"
   118→        />
   119→        <text x={size / 2} y={size / 2 - 4} textAnchor="middle" className="fill-foreground text-2xl font-bold" dominantBaseline="middle">
   120→          {Math.round(value)}
   121→        </text>
   122→        <text x={size / 2} y={size / 2 + 18} textAnchor="middle" className="fill-muted-foreground text-xs" dominantBaseline="middle">
   123→          / {max}
   124→        </text>
   125→      </svg>
   126→      <span className="text-xs text-muted-foreground font-medium">{label}</span>
   127→    </div>
   128→  )
   129→}

   143→async function checkServiceHealth(url: string, timeoutMs = 5000): Promise<{ ok: boolean; timeMs: number; data?: any }> {
   144→  const start = Date.now()
   145→  try {
   146→    const controller = new AbortController()
   147→    const timer = setTimeout(() => controller.abort(), timeoutMs)
   148→    const res = await fetch(url, { signal: controller.signal })
   149→    clearTimeout(timer)
   150→    const data = await res.json().catch(() => null)
   151→    return { ok: res.ok, timeMs: Date.now() - start, data }
   152→  } catch {
   153→    return { ok: false, timeMs: Date.now() - start }
   154→  }
   155→}
