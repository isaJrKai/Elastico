    88→    88→// ── Demo Tactical Profiles ─────────────────────────────────────────────
    89→    89→
    90→    90→function generateDemoProfile(team: { style: string | null; xgPerGame: number | null; xgaPerGame: number | null; possession: number; pressIntensity: number; passAccuracy: number }, seed: number) {
    91→    91→  const hash = (s: string) => s.split('').reduce((a, c) => a + c.charCodeAt(0), 0)
    92→    92→  const base = hash(team.style || 'balanced') + seed
    93→    93→  const jitter = (i: number, range: number) => ((base * (i + 1) * 7) % range)
    94→    94→  // Use safe defaults when xG is null (no fabrication)
    95→    95→  const xg = team.xgPerGame ?? 1.3
    96→    96→  const xga = team.xgaPerGame ?? 1.1
    97→    97→
    98→    98→  return {
    99→    99→    attack: Math.min(95, Math.max(30, xg * 25 + jitter(0, 20) - 10)),
   100→   100→    midfield: Math.min(95, Math.max(30, team.possession * 0.7 + team.passAccuracy * 0.2 + jitter(1, 15) - 7)),
   101→   101→    defense: Math.min(95, Math.max(30, 100 - xga * 30 + jitter(2, 15) - 7)),
   102→   102→    pressing: Math.min(95, Math.max(20, team.pressIntensity + jitter(3, 20) - 10)),
   103→   103→    possession: Math.min(95, Math.max(20, team.possession + jitter(4, 10) - 5)),
   104→   104→    setPiece: Math.min(95, Math.max(25, 40 + jitter(5, 40))),
   105→   105→  }
   106→   106→}

   110→   110→function FormationPitch({
   111→   111→  formation,
   112→   112→  teamColor,
   113→   113→  mirror,
   114→   114→  players,
   115→   115→}: {
   116→   116→  formation: string
   117→   117→  teamColor: string
   118→   118→  mirror?: boolean
   119→   119→  players?: { name: string; number: number; position: string }[]
   120→   120→}) {
   121→   121→  const fmt = FORMATIONS[formation]
   122→   122→  if (!fmt) return null
   123→   123→
   124→   124→  return (
   125→   125→    <div className="relative w-full aspect-[3/4] rounded-lg border border-border/60 overflow-hidden bg-emerald-950/30">
   126→   126→      {/* Pitch markings */}
   127→   127→      <div className="absolute inset-2 border border-white/10 rounded-sm" />
   128→   128→      <div className="absolute top-1/2 left-2 right-2 border-t border-white/10" />
   129→   129→      <div className="absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 size-24 rounded-full border border-white/10" />
   130→   130→      <div className="absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 size-1 rounded-full bg-white/20" />
   131→   131→      {/* Penalty areas */}
   132→   132→      <div className="absolute top-2 left-1/2 -translate-x-1/2 w-[40%] h-[16%] border border-white/10 border-t-0 rounded-b-md" />
   133→   133→      <div className="absolute bottom-2 left-1/2 -translate-x-1/2 w-[40%] h-[16%] border border-white/10 border-b-0 rounded-t-md" />
   134→   134→      {/* Goal areas */}
   135→   135→      <div className="absolute top-2 left-1/2 -translate-x-1/2 w-[20%] h-[8%] border border-white/10 border-t-0" />
   136→   136→      <div className="absolute bottom-2 left-1/2 -translate-x-1/2 w-[20%] h-[8%] border border-white/10 border-b-0" />
   137→   137→
   138→   138→      {/* Player dots */}
   139→   139→      {fmt.positions.map((pos, i) => {
   140→   140→        const x = mirror ? 100 - pos.x : pos.x
   141→   141→        const y = pos.y
   142→   142→        const player = players?.[i]
   143→   143→        return (
   144→   144→          <div
   145→   145→            key={i}
   146→   146→            className="absolute flex flex-col items-center -translate-x-1/2 -translate-y-1/2 group"
   147→   147→            style={{ left: `${x}%`, top: `${y}%` }}
   148→   148→          >
   149→   149→            <div
   150→   150→              className="size-5 rounded-full border-2 border-white/40 shadow-md flex items-center justify-center transition-transform hover:scale-125"
   151→   151→              style={{ backgroundColor: teamColor }}
   152→   152→            >
   153→   153→              <span className="text-[8px] font-bold text-white leading-none">
   154→   154→                {player?.number || (i === 0 ? '1' : '')}
   155→   155→              </span>
   156→   156→            </div>
   157→   157→            {player && (
   158→   158→              <span className="text-[8px] text-white/70 mt-0.5 whitespace-nowrap font-medium opacity-0 group-hover:opacity-100 transition-opacity">
   159→   159→                {player.name}
   160→   160→              </span>
   161→   161→            )}
   162→   162→            {!player && (
   163→   163→              <span className="text-[8px] text-white/40 mt-0.5 font-mono">
   164→   164→                {pos.role}
   165→   165→              </span>
   166→   166→            )}
   167→   167→          </div>
   168→   168→        )
   169→   169→      })}
   170→   170→    </div>
   171→   171→  )
   172→   172→}

   194→   194→        setAwayTeamId(m.awayTeamId)
