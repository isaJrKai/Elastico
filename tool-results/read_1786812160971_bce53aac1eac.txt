    44→function getEloProb(eloA: number, eloB: number) {
    45→  const eA = 1 / (1 + Math.pow(10, (eloB - eloA) / 400))
    46→  return { home: +(eA * 100).toFixed(1), away: +((1 - eA) * 100).toFixed(1), draw: +(Math.min(eA, 1 - eA) * 30).toFixed(1) }
    47→}
