   380→// ── POST /admin/train-ai ────────────────────────────────────────────────────
   381→adminRouter.post("/train-ai", adminMiddleware, (req: Request, res: Response) => {
   382→  try {
   383→    const finishedMatches = db.matches.filter(m => m.status === "finished");
   384→    let ratingsAdjustmentsCount = 0;
   385→    let totalPredictionsCount = 0;
   386→    let alignmentRatioSum = 0;
   387→    let processedMatchesCount = 0;
   388→
   389→    const learningRate = db.settings?.ai_learning_rate || 0.15;
   390→
   391→    finishedMatches.forEach(m => {
   392→      const vH = m.votes_home || 0;
   393→      const vD = m.votes_draw || 0;
   394→      const vA = m.votes_away || 0;
   395→      const totalMatchVotes = vH + vD + vA;
   396→
   397→      if (totalMatchVotes === 0) return;
   398→
   399→      // Filter extreme biased attempts to contaminate AI Continuous Learning ELO bounds
   400→      const maxShare = Math.max(vH, vD, vA) / totalMatchVotes;
   401→      if (maxShare > 0.88 && totalMatchVotes >= 5) {
   402→        return; // Skip skewed match prediction profile
   403→      }
   404→
   405→      totalPredictionsCount += totalMatchVotes;
   406→      processedMatchesCount++;
   407→
   408→      // Determine actual outcome
   409→      let outcome: "home" | "draw" | "away" = "draw";
   410→      if (m.home_score > m.away_score) outcome = "home";
   411→      else if (m.away_score > m.home_score) outcome = "away";
   412→
   413→      // Calculate accuracy ratio
   414→      let alignmentFactor = 0.33;
   415→      if (outcome === "home") alignmentFactor = vH / totalMatchVotes;
   416→      else if (outcome === "away") alignmentFactor = vA / totalMatchVotes;
   417→      else alignmentFactor = vD / totalMatchVotes;
   418→
   419→      alignmentRatioSum += alignmentFactor;
   420→
   421→      const homeTeam = db.teams.find(t => t.id === m.home_team_id);
   422→      const awayTeam = db.teams.find(t => t.id === m.away_team_id);
   423→
   424→      if (homeTeam && awayTeam) {
   425→        // Adjust squad ratings to feed the continuous mathematical loop
   426→        const adjustmentDelta = Math.round(30 * learningRate * (alignmentFactor - 0.33));
   427→        if (adjustmentDelta !== 0) {
   428→          if (outcome === "home") {
   429→            homeTeam.elo_rating += adjustmentDelta;
   430→            awayTeam.elo_rating -= Math.round(adjustmentDelta / 2);
   431→          } else if (outcome === "away") {
   432→            awayTeam.elo_rating += adjustmentDelta;
   433→            homeTeam.elo_rating -= Math.round(adjustmentDelta / 2);
   434→          } else {
   435→            // Draw alignment
   436→            homeTeam.elo_rating += Math.round(adjustmentDelta / 3);
   437→            awayTeam.elo_rating += Math.round(adjustmentDelta / 3);
   438→          }
   439→          ratingsAdjustmentsCount++;
   440→        }
   441→      }
   442→    });
   443→
   444→    db.save();
   445→
   446→    res.json({
   447→      message: "AI Knowledge Training completed successfully!",
   448→      metrics: {
   449→        processed_matches: processedMatchesCount,
   450→        total_predictions_analyzed: totalPredictionsCount,
   451→        average_user_alignment: processedMatchesCount > 0 ? (alignmentRatioSum / processedMatchesCount) : 0,
   452→        ratings_adjustments_applied: ratingsAdjustmentsCount,
   453→        current_global_learning_rate: learningRate
   454→      }
   455→    });
   456→  } catch (err: any) {
   457→    res.status(500).json({ error: "Failed to process AI model training sequence." });
   458→  }
