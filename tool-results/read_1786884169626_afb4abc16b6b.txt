    29→// Simple deterministic hash for probability estimation
    30→function hashCode(str: string): number {
    31→  let hash = 0
    32→  for (let i = 0; i < str.length; i++) {
    33→    hash = ((hash << 5) - hash) + str.charCodeAt(i)
    34→    hash |= 0
    35→  }
    36→  return hash
    37→}
