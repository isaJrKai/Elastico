   140→def build_team_sequences(df_complete):
   141→    """Build team-centric temporal sequences for every match.
   142→
   143→    For each target match at time T:
   144→      1. Find all past matches for HomeTeam where Date < T
   145→      2. Find all past matches for AwayTeam where Date < T
   146→      3. Take the last SEQ_LEN matches for each team
   147→      4. Extract 15 team-centric features per timestep
   148→      5. If either team has < SEQ_LEN past matches, SKIP the target match
   149→
   150→    Returns:
   151→        X_home: np.ndarray (N_valid, SEQ_LEN, 15)
   152→        X_away: np.ndarray (N_valid, SEQ_LEN, 15)
   153→        y:      np.ndarray (N_valid,) int32
   154→        indices: list[int] — row indices into df_complete for valid matches
   155→        skipped: int — number of matches skipped
   156→    """
   157→    print('Building temporal sequences...')
   158→
   159→    # Build per-team match index: team -> [(date, row_idx, is_home), ...]
   160→    team_matches = {}
   161→    for idx, row in df_complete.iterrows():
   162→        ht = row['HomeTeam']
   163→        at = row['AwayTeam']
   164→        dt = row['Date']
   165→        for team, is_home in [(ht, True), (at, False)]:
   166→            if team not in team_matches:
   167→                team_matches[team] = []
   168→            team_matches[team].append((dt, idx, is_home))
   169→
   170→    # Sort each team's matches chronologically
   171→    for team in team_matches:
   172→        team_matches[team].sort(key=lambda x: x[0])
   173→
   174→    # For each target match, build sequences
   175→    X_home_list = []
   176→    X_away_list = []
   177→    y_list = []
   178→    valid_indices = []
   179→    skipped = 0
   180→
   181→    for idx, row in df_complete.iterrows():
   182→        ht = row['HomeTeam']
   183→        at = row['AwayTeam']
   184→        target_date = row['Date']
   185→
   186→        # Get past matches for home team (Date < target_date)
   187→        home_past = [(dt, ri, ih) for dt, ri, ih in team_matches[ht]
   188→                     if dt < target_date and ri != idx]
   189→        # Get past matches for away team
   190→        away_past = [(dt, ri, ih) for dt, ri, ih in team_matches[at]
   191→                     if dt < target_date and ri != idx]
   192→
   193→        # Check sufficient history
   194→        if len(home_past) < SEQ_LEN or len(away_past) < SEQ_LEN:
   195→            skipped += 1
   196→            continue
   197→
   198→        # Take last SEQ_LEN matches for each team
   199→        home_seq_raw = home_past[-SEQ_LEN:]
   200→        away_seq_raw = away_past[-SEQ_LEN:]
   201→
   202→        # Build feature matrices
   203→        home_features = np.zeros((SEQ_LEN, N_SEQ_FEATURES), dtype=np.float32)
   204→        away_features = np.zeros((SEQ_LEN, N_SEQ_FEATURES), dtype=np.float32)
   205→
   206→        for t, (dt, ri, is_home) in enumerate(home_seq_raw):
   207→            past_row = df_complete.loc[ri]
   208→            feat = extract_team_features(past_row, is_home)
   209→            home_features[t] = feat
   210→
   211→        for t, (dt, ri, is_home) in enumerate(away_seq_raw):
   212→            past_row = df_complete.loc[ri]
   213→            feat = extract_team_features(past_row, is_home)
   214→            away_features[t] = feat
   215→
   216→        # Replace any NaN/inf with 0
   217→        home_features = np.nan_to_num(home_features, nan=0.0, posinf=0.0, neginf=0.0)
   218→        away_features = np.nan_to_num(away_features, nan=0.0, posinf=0.0, neginf=0.0)
   219→
   220→        X_home_list.append(home_features)
   221→        X_away_list.append(away_features)
   222→        y_list.append(int(row['result_encoded']))
   223→        valid_indices.append(idx)
   224→
   225→    X_home = np.stack(X_home_list, axis=0)
   226→    X_away = np.stack(X_away_list, axis=0)
   227→    y = np.array(y_list, dtype=np.int32)
   228→
   229→    print(f'  Total matches: {len(df_complete)}')
   230→    print(f'  Valid sequences: {len(y)}')
   231→    print(f'  Skipped (insufficient history): {skipped}')
   232→    print(f'  Coverage: {len(y)/len(df_complete)*100:.1f}%')
   233→
   234→    return X_home, X_away, y, valid_indices, skipped

   237→def extract_team_features(row, is_home):
   238→    """Extract 15 team-centric features from a match row.
   239→
   240→    Parameters:
   241→        row: DataFrame row (from training_dataset.parquet)
   242→        is_home: True if the team was the home team in this match
   243→
   244→    Returns:
   245→        np.ndarray of shape (15,) — team-centric feature vector
   246→    """
   247→    feat = np.zeros(N_SEQ_FEATURES, dtype=np.float32)
   248→    for i, (home_col, away_col) in enumerate(TEAM_FEATURE_MAP):
   249→        col = home_col if is_home else away_col
   250→        feat[i] = float(row.get(col, 0.0))
   251→    feat[14] = 1.0 if is_home else 0.0  # is_home_flag
   252→    return feat

   256→# Phase 4D — NORMALIZATION
   257→# ═══════════════════════════════════════════════════════════════════════
   258→
   259→class Normalizer:
   260→    """StandardScaler fit on training data only."""
   261→
   262→    def __init__(self):
   263→        self.mean = None
   264→        self.std = None
   265→        self.n_features = None
   266→
   267→    def fit(self, X):
   268→        """Fit on training data. X shape: (N, seq_len, n_features) or (N, n_features)."""
   269→        if X.ndim == 3:
   270→            X_flat = X.reshape(-1, X.shape[-1])
   271→        else:
   272→            X_flat = X
   273→        self.n_features = X_flat.shape[1]
   274→        self.mean = X_flat.mean(axis=0)
   275→        self.std = X_flat.std(axis=0)
   276→        # Avoid division by zero
   277→        self.std[self.std < 1e-8] = 1.0
   278→        return self
   279→
   280→    def transform(self, X):
   281→        """Transform using stored parameters."""
   282→        if X.ndim == 3:
   283→            shape = X.shape
   284→            X_flat = X.reshape(-1, self.n_features)
   285→            X_norm = (X_flat - self.mean) / self.std
   286→            return X_norm.reshape(shape)
   287→        return (X - self.mean) / self.std
   288→
   289→    def to_dict(self):
   290→        return {'mean': self.mean.tolist(), 'std': self.std.tolist(),
   291→                'n_features': self.n_features}
   292→
   293→    @classmethod
   294→    def from_dict(cls, d):
   295→        n = cls()
   296→        n.mean = np.array(d['mean'])
   297→        n.std = np.array(d['std'])
   298→        n.n_features = d['n_features']
   299→        return n

   302→# ═══════════════════════════════════════════════════════════════════════
   303→# Phase 4F — LSTM ARCHITECTURE
   304→# ═══════════════════════════════════════════════════════════════════════
   305→#
   306→# Inspected existing lstm_engine.py FootballLSTM (hidden=64, layers=2,
   307→# BiLSTM, attention, FC: 256→128→64→3).
   308→#
   309→# Changes from existing architecture:
   310→#   1. REDUCED hidden_size: 64 → 48
   311→#      Reason: ~4000 training sequences. 64*2*2=256 FC input with
   312→#      128→64→3 head = ~50K params just in FC. Too many for this data size.
   313→#      48*2*2=192 FC input with 64→3 head = ~14K params. More appropriate.
   314→#   2. REDUCED FC head: 256→128→64→3 → 192→64→3
   315→#      Reason: Removed one FC layer. Less overfitting risk.
   316→#   3. INCREASED dropout: 0.3 → 0.4
   317→#      Reason: XGBoost showed severe overfitting. LSTM with sequences
   318→#      will overfit even more without strong regularization.
   319→#   4. PRESERVED: BiLSTM, attention, batch norm, gradient clipping,
   320→#      LR scheduling, early stopping, Xavier/orthogonal init.
   321→#
   322→
   323→import torch
   324→import torch.nn as nn
   325→
   326→
   327→class FootballLSTM_v1(nn.Module):
   328→    """Two-branch BiLSTM with attention for football prediction.
   329→
   330→    Architecture:
   331→        Home seq (B, T, 15) → BiLSTM → Attention → home_ctx (B, 96)
   332→        Away seq (B, T, 15) → BiLSTM → Attention → away_ctx (B, 96) [shared weights]
   333→        Concat → FC(192→64) → BN → ReLU → Dropout → FC(64→3)
   334→
   335→    Changes from lstm_engine.py v1:
   336→      hidden_size: 64 → 48 (fewer params for ~4K sequences)
   337→      FC head: 256→128→64→3 → 192→64→3 (one fewer layer)
   338→      dropout: 0.3 → 0.4 (stronger regularization)
   339→    """
   340→
   341→    def __init__(self, input_size=15, hidden_size=48, num_layers=2,
   342→                 dropout=0.4, n_classes=3):
   343→        super().__init__()
   344→        self.input_size = input_size
   345→        self.hidden_size = hidden_size
   346→        self.num_layers = num_layers
   347→        self.dropout_rate = dropout
   348→        bidir_hidden = hidden_size * 2  # 96
   349→
   350→        # Shared BiLSTM
   351→        self.lstm = nn.LSTM(
   352→            input_size=input_size,
   353→            hidden_size=hidden_size,
   354→            num_layers=num_layers,
   355→            batch_first=True,
   356→            bidirectional=True,
   357→            dropout=dropout if num_layers > 1 else 0.0,
   358→        )
   359→
   360→        # Attention
   361→        self.attention = nn.Linear(bidir_hidden, 1)
   362→
   363→        # FC head
   364→        fc_input = bidir_hidden * 2  # 192
   365→        self.fc = nn.Sequential(
   366→            nn.Linear(fc_input, 64),
   367→            nn.BatchNorm1d(64),
   368→            nn.ReLU(),
   369→            nn.Dropout(dropout),
   370→            nn.Linear(64, n_classes),
   371→        )
   372→
   373→        self._init_weights()
   374→
   375→    def _init_weights(self):
   376→        for name, param in self.named_parameters():
   377→            if 'lstm' in name and 'weight_ih' in name:
   378→                nn.init.xavier_uniform_(param)
   379→            elif 'lstm' in name and 'weight_hh' in name:
   380→                nn.init.orthogonal_(param)
   381→            elif 'lstm' in name and 'bias' in name:
   382→                nn.init.zeros_(param)
   383→                n = param.size(0)
   384→                start, end = n // 4, n // 2
   385→                nn.init.constant_(param.data[start:end], 1.0)
   386→            elif 'weight' in name and param.ndim >= 2:
   387→                nn.init.xavier_uniform_(param)
   388→            elif 'bias' in name:
   389→                nn.init.zeros_(param)
   390→
   391→    def _attend(self, lstm_out):
   392→        """Softmax attention: (B, T, H*2) → (B, H*2)"""
   393→        attn_logits = self.attention(lstm_out)       # (B, T, 1)
   394→        attn_weights = torch.softmax(attn_logits, dim=1)  # (B, T, 1)
   395→        context = torch.sum(attn_weights * lstm_out, dim=1)  # (B, H*2)
   396→        return context
   397→
   398→    def forward(self, x_home, x_away):
   399→        """
   400→        x_home: (B, T, 15)  x_away: (B, T, 15)
   401→        Returns: logits (B, 3)
   402→        """
   403→        home_out, _ = self.lstm(x_home)
   404→        home_ctx = self._attend(home_out)
   405→
   406→        away_out, _ = self.lstm(x_away)
   407→        away_ctx = self._attend(away_out)
   408→
   409→        combined = torch.cat([home_ctx, away_ctx], dim=1)
   410→        return self.fc(combined)

   418→def train_lstm(model, train_loader, val_loader, config):
   419→    """Train LSTM with early stopping, gradient clipping, LR scheduling."""
   420→    device = torch.device('cpu')
   421→    model = model.to(device)
   422→
   423→    # Class weights for imbalance
   424→    y_all = []
   425→    for _, _, yb in train_loader:
   426→        y_all.append(yb.numpy())
   427→    y_all = np.concatenate(y_all)
   428→    counts = np.bincount(y_all, minlength=3).astype(float)
   429→    total = counts.sum()
   430→    class_weights = torch.tensor(total / (3 * counts), dtype=torch.float32).to(device)
   431→    print(f'  Class weights: {class_weights.tolist()}')
   432→
   433→    optimizer = torch.optim.Adam(model.parameters(), lr=config['lr'],
   434→                                  weight_decay=config['weight_decay'])
   435→    scheduler = torch.optim.lr_scheduler.ReduceLROnPlateau(
   436→        optimizer, factor=config['lr_factor'], patience=config['lr_patience'],
   437→        min_lr=config['lr_min'], mode='min')
   438→    criterion = nn.CrossEntropyLoss(weight=class_weights)
   439→
   440→    best_val_loss = float('inf')
   441→    best_state = None
   442→    patience_counter = 0
   443→    history = {'train_loss': [], 'val_loss': [],
   444→               'train_acc': [], 'val_acc': [], 'lr': []}
   445→
   446→    for epoch in range(config['max_epochs']):
   447→        # --- Train ---
   448→        model.train()
   449→        train_loss_sum = 0.0
   450→        train_correct = 0
   451→        train_total = 0
   452→        for xh, xa, yb in train_loader:
   453→            xh, xa, yb = xh.to(device), xa.to(device), yb.to(device)
   454→            optimizer.zero_grad()
   455→            logits = model(xh, xa)
   456→            loss = criterion(logits, yb)
   457→            loss.backward()
   458→            torch.nn.utils.clip_grad_norm_(model.parameters(), config['grad_clip'])
   459→            optimizer.step()
   460→            train_loss_sum += loss.item() * yb.size(0)
   461→            train_correct += (logits.argmax(1) == yb).sum().item()
   462→            train_total += yb.size(0)
   463→
   464→        train_loss = train_loss_sum / train_total
   465→        train_acc = train_correct / train_total
   466→
   467→        # --- Validate ---
   468→        model.eval()
   469→        val_loss_sum = 0.0
   470→        val_correct = 0
   471→        val_total = 0
   472→        with torch.no_grad():
   473→            for xh, xa, yb in val_loader:
   474→                xh, xa, yb = xh.to(device), xa.to(device), yb.to(device)
   475→                logits = model(xh, xa)
   476→                loss = criterion(logits, yb)
   477→                val_loss_sum += loss.item() * yb.size(0)
   478→                val_correct += (logits.argmax(1) == yb).sum().item()
   479→                val_total += yb.size(0)
   480→
   481→        val_loss = val_loss_sum / val_total
   482→        val_acc = val_correct / val_total
   483→        current_lr = optimizer.param_groups[0]['lr']
   484→
   485→        scheduler.step(val_loss)
   486→
   487→        history['train_loss'].append(train_loss)
   488→        history['val_loss'].append(val_loss)
   489→        history['train_acc'].append(train_acc)
   490→        history['val_acc'].append(val_acc)
   491→        history['lr'].append(current_lr)
   492→
   493→        if (epoch + 1) % 10 == 0 or epoch == 0:
   494→            print(f'    Epoch {epoch+1:3d}: train_loss={train_loss:.4f} train_acc={train_acc:.4f} '
   495→                  f'val_loss={val_loss:.4f} val_acc={val_acc:.4f} lr={current_lr:.6f}')
   496→
   497→        # Early stopping
   498→        if val_loss < best_val_loss:
   499→            best_val_loss = val_loss
   500→            best_state = copy.deepcopy(model.state_dict())
   501→            patience_counter = 0
   502→        else:
   503→            patience_counter += 1
   504→            if patience_counter >= config['patience']:
   505→                print(f'    Early stopping at epoch {epoch+1}')
   506→                break
   507→
   508→    # Restore best model
   509→    if best_state is not None:
   510→        model.load_state_dict(best_state)
   511→
   512→    return model, history

   516→# Phase 4I — EVALUATION
   517→# ═══════════════════════════════════════════════════════════════════════
   518→
   519→def compute_ece(y_true, y_prob, n_bins=15):
   520→    """Expected Calibration Error."""
   521→    n_classes = y_prob.shape[1]
   522→    ece = 0.0
   523→    for c in range(n_classes):
   524→        y_binary = (y_true == c).astype(int)
   525→        prob_c = y_prob[:, c]
   526→        edges = np.linspace(0, 1, n_bins + 1)
   527→        for i in range(n_bins):
   528→            mask = (prob_c >= edges[i]) & (prob_c < edges[i + 1])
   529→            if i == n_bins - 1:
   530→                mask = (prob_c >= edges[i]) & (prob_c <= edges[i + 1])
   531→            n_bin = mask.sum()
   532→            if n_bin == 0:
   533→                continue
   534→            ece += (n_bin / len(y_true)) * abs(prob_c[mask].mean() - y_binary[mask].mean())
   535→    return ece / n_classes
   536→
   537→
   538→def compute_metrics(y_true, y_proba, y_pred, label='set'):
   539→    """Full metrics suite."""
   540→    n = len(y_true)
   541→    m = {'label': label, 'n_samples': n}
   542→    m['accuracy'] = float(accuracy_score(y_true, y_pred))
   543→    try:
   544→        m['log_loss'] = float(log_loss(y_true, y_proba))
   545→    except Exception:
   546→        m['log_loss'] = float('nan')
   547→    n_classes = y_proba.shape[1]
   548→    y_onehot = np.zeros_like(y_proba)
   549→    for i in range(n_classes):
   550→        y_onehot[:, i] = (y_true == i).astype(float)
   551→    briers = []
   552→    for i in range(n_classes):
   553→        try:
   554→            briers.append(brier_score_loss(y_onehot[:, i], y_proba[:, i]))
   555→        except Exception:
   556→            briers.append(float('nan'))
   557→    m['brier_score'] = float(np.nanmean(briers))
   558→    prec, rec, f1, support = precision_recall_fscore_support(
   559→        y_true, y_pred, labels=[0, 1, 2], zero_division=0)
   560→    for i, name in enumerate(['Home', 'Draw', 'Away']):
   561→        m[f'precision_{name}'] = float(prec[i])
   562→        m[f'recall_{name}'] = float(rec[i])
   563→        m[f'f1_{name}'] = float(f1[i])
   564→        m[f'support_{name}'] = int(support[i])
   565→    m['precision_macro'] = float(np.mean(prec))
   566→    m['recall_macro'] = float(np.mean(rec))
   567→    m['f1_macro'] = float(np.mean(f1))
   568→    m['confusion_matrix'] = confusion_matrix(y_true, y_pred, labels=[0, 1, 2]).tolist()
   569→    m['ece'] = float(compute_ece(y_true, y_proba))
   570→    return m
   571→
   572→
   573→def predict_proba(model, X_home, X_away, batch_size=256):
   574→    """Get probability predictions from model."""
   575→    model.eval()
   576→    device = torch.device('cpu')
   577→    all_proba = []
   578→    with torch.no_grad():
   579→        for i in range(0, len(X_home), batch_size):
   580→            xh = torch.as_tensor(X_home[i:i+batch_size], dtype=torch.float32).to(device)
   581→            xa = torch.as_tensor(X_away[i:i+batch_size], dtype=torch.float32).to(device)
   582→            logits = model(xh, xa)
   583→            proba = torch.softmax(logits, dim=1).cpu().numpy()
   584→            all_proba.append(proba)
   585→    return np.concatenate(all_proba, axis=0)
   586→

   588→def compute_breakdown(df_subset, y_true, y_proba, y_pred, group_col):
   589→    """Compute metrics by group (season/league)."""
   590→    results = []
   591→    for g in sorted(df_subset[group_col].unique()):
   592→        mask = df_subset[group_col] == g
   593→        if mask.sum() < 20:
   594→            continue
   595→        yt = y_true[mask]
   596→        yp = y_pred[mask]
   597→        ypr = y_proba[mask]
   598→        m = {'group': str(g), 'n': int(mask.sum()),
   599→             'accuracy': float(accuracy_score(yt, yp))}
   600→        try:
   601→            m['log_loss'] = float(log_loss(yt, ypr))
   602→        except Exception:
   603→            m['log_loss'] = float('nan')
   604→        m['ece'] = float(compute_ece(yt, ypr))
   605→        prec, rec, f1, _ = precision_recall_fscore_support(yt, yp, labels=[0,1,2], zero_division=0)
   606→        m['f1_home'] = float(f1[0])
   607→        m['f1_draw'] = float(f1[1])
   608→        m['f1_away'] = float(f1[2])
   609→        results.append(m)
   610→    return results

   614→# Phase 4J — XGBOOST COMPARISON

   617→def compute_prediction_correlation(proba_a, proba_b, y_true):
   618→    """Correlation between two models' predictions."""
   619→    p_a = proba_a.flatten()
   620→    p_b = proba_b.flatten()
   621→    corr = float(np.corrcoef(p_a, p_b)[0, 1])
   622→    pred_a = np.argmax(proba_a, axis=1)
   623→    pred_b = np.argmax(proba_b, axis=1)
   624→    agreement = float((pred_a == pred_b).mean())
   625→    both_correct = float(((pred_a == y_true) & (pred_b == y_true)).mean())
   626→    # When they disagree, who is right more often?
   627→    disagree_mask = pred_a != pred_b
   628→    if disagree_mask.sum() > 0:
   629→        a_right = (pred_a[disagree_mask] == y_true[disagree_mask]).mean()
   630→        b_right = (pred_b[disagree_mask] == y_true[disagree_mask]).mean()
   631→    else:
   632→        a_right = b_right = 0.0
   633→    return {
   634→        'probability_correlation': corr,
   635→        'prediction_agreement': agreement,
   636→        'both_correct_rate': both_correct,
   637→        'disagree_rate': float(disagree_mask.mean()),
   638→        'lstm_right_on_disagree': float(a_right),
   639→        'xgb_right_on_disagree': float(b_right),
   640→    }

   643→def train_xgb_football_only(df_train, df_test, football_features):
   644→    """Train a football-only XGBoost on the same temporal split for direct comparison."""
   645→    import xgboost as xgb
   646→
   647→    X_train = df_train[football_features].values.astype(np.float32)
   648→    y_train = df_train['result_encoded'].values.astype(np.int32)
   649→    X_test = df_test[football_features].values.astype(np.float32)
   650→    y_test = df_test['result_encoded'].values.astype(np.int32)
   651→
   652→    # Filter complete cases
   653→    train_mask = ~np.isnan(X_train).any(axis=1)
   654→    test_mask = ~np.isnan(X_test).any(axis=1)
   655→    X_train, y_train = X_train[train_mask], y_train[train_mask]
   656→    X_test, y_test = X_test[test_mask], y_test[test_mask]
   657→
   658→    counts = np.bincount(y_train, minlength=3)
   659→    total = counts.sum()
   660→    sw = {i: total / (3 * c) for i, c in enumerate(counts) if c > 0}
   661→    sample_weights = np.array([sw.get(y, 1.0) for y in y_train])
   662→
   663→    dtrain = xgb.DMatrix(X_train, label=y_train, weight=sample_weights,
   664→                          feature_names=football_features)
   665→    dtest = xgb.DMatrix(X_test, label=y_test, feature_names=football_features)
   666→
   667→    params = {'objective': 'multi:softprob', 'num_class': 3,
   668→              'eval_metric': 'mlogloss', 'max_depth': 5, 'eta': 0.05,
   669→              'subsample': 0.8, 'colsample_bytree': 0.8,
   670→              'min_child_weight': 5, 'gamma': 0.1,
   671→              'lambda': 1.5, 'alpha': 0.1, 'seed': 42, 'nthread': 4}
   672→
   673→    bst = xgb.train(params, dtrain, num_boost_round=1000,
   674→                    evals=[(dtrain, 'train'), (dtest, 'test')],
   675→                    early_stopping_rounds=100, verbose_eval=False)
   676→
   677→    proba = bst.predict(dtest)
   678→    pred = np.argmax(proba, axis=1)
   679→    return proba, pred, y_test
