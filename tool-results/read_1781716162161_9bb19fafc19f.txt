    53→  matchId: string
    54→  predictedHomeGoals: number
    55→  predictedAwayGoals: number
    56→  predictedOutcome: string
    57→  confidence: number
    58→  model: string
    59→  isCorrect: boolean | null
    60→  points: number
    61→  createdAt: string
    62→  match: {
    63→    homeTeam: { name: string; code: string; primaryColor: string } | null

    70→}

    73→  id: string
    74→  homeTeam: { name: string; code: string; primaryColor: string } | null
    75→  awayTeam: { name: string; code: string; primaryColor: string } | null
    76→  date: string | null
    77→  stage: string
    78→}
    79→

    81→type SortDir = 'asc' | 'desc'

    84→  elo: '#00e676',
    85→  poisson: '#00b4d8',
    86→  dixon_coles: '#ffd700',
    87→  monte_carlo: '#a855f7',
    88→  user: '#ff4757',
    89→}
    90→
    91→const MODEL_LABELS: Record<string, string> = {
    92→  elo: 'ELO',
    93→  poisson: 'Poisson',
    94→  dixon_coles: 'Dixon-Coles',
    95→  monte_carlo: 'Monte Carlo',
    96→  user: 'User',
    97→}

    99→// ── Custom Tooltip ─────────────────────────────────────────────────────────────
   100→
   101→function ChartTooltip({
   102→  active,
   103→  payload,
   104→  label,
   105→}: {
   106→  active?: boolean
   107→  payload?: Array<{ value: number; name: string; color: string }>
   108→  label?: string
   109→}) {
   110→  if (!active || !payload?.length) return null
   111→  return (
   112→    <div className="glass-card rounded-lg p-3 text-sm shadow-lg border border-border/50">
   113→      <p className="text-muted-foreground text-xs mb-1">{label}</p>
   114→      {payload.map((entry, i) => (
   115→        <p key={i} className="font-semibold" style={{ color: entry.color }}>
   116→          {entry.name}: {entry.value.toFixed(1)}%
   117→        </p>
   118→      ))}
   119→    </div>
   120→  )
   121→}

   124→// PREDICTIONS VIEW
   125→// ═══════════════════════════════════════════════════════════════════════════════
   126→
   127→export default function PredictionsView() {
   128→  const { user, token } = useElasticoStore()
   129→
   130→  // ── State ─────────────────────────────────────────────────────────────────
   131→
   132→  const [predictions, setPredictions] = useState<PredictionRecord[]>([])
   133→  const [upcomingMatches, setUpcomingMatches] = useState<UpcomingMatch[]>([])
   134→  const [loading, setLoading] = useState(true)
   135→  const [sortField, setSortField] = useState<SortField>('date')
   136→  const [sortDir, setSortDir] = useState<SortDir>('desc')
   137→
   138→  // Make prediction form state
   139→  const [selectedMatch, setSelectedMatch] = useState<string>('')
   140→  const [homeGoals, setHomeGoals] = useState(1)
   141→  const [awayGoals, setAwayGoals] = useState(0)
   142→  const [outcome, setOutcome] = useState<string>('')
   143→  const [confidence, setConfidence] = useState(50)
   144→  const [model, setModel] = useState<string>('user')
   145→  const [submitting, setSubmitting] = useState(false)
   146→  const [submitMessage, setSubmitMessage] = useState<string | null>(null)
   147→
   148→  // ── Computed Stats ─────────────────────────────────────────────────────────
   149→
   150→  const totalPredictions = user?.totalPredictions || 0
   151→  const correctPredictions = user?.correctPredictions || 0
   152→  const accuracyPct = totalPredictions > 0 ? Math.round((correctPredictions / totalPredictions) * 100) : 0
   153→  const currentStreak = user?.predictionStreak || 0
   154→
   155→  // Model accuracy breakdown
   156→  const modelAccuracyMap = React.useMemo(() => {
   157→    const map: Record<string, { total: number; correct: number }> = {}
   158→    for (const p of predictions) {
   159→      if (!map[p.model]) map[p.model] = { total: 0, correct: 0 }
   160→      map[p.model].total++
   161→      if (p.isCorrect === true) map[p.model].correct++
   162→    }
   163→    return map
   164→  }, [predictions])
   165→
   166→  const modelChartData = React.useMemo(() => {
   167→    return Object.entries(modelAccuracyMap).map(([key, val]) => ({
   168→      name: MODEL_LABELS[key] || key,
   169→      accuracy: val.total > 0 ? (val.correct / val.total) * 100 : 0,
   170→      color: MODEL_COLORS[key] || '#666',
   171→    }))
   172→  }, [modelAccuracyMap])
   173→
   174→  // ── Data Fetching ──────────────────────────────────────────────────────────
   175→
   176→  const fetchPredictions = useCallback(async () => {
   177→    try {
   178→      const res = await fetch('/api/matches?status=finished&limit=50', {
   179→        headers: token ? { Authorization: `Bearer ${token}` } : undefined,
   180→      })
   181→      const data = await res.json()
   182→      // Predictions are embedded in user profile, we fetch finished matches and
   183→      // reconstruct from user stats — in a real app, there would be a GET /api/predictions
   184→      // For now we use the matches to demonstrate
   185→      setPredictions([])
   186→    } catch (err) {
   187→      console.error('Failed to fetch predictions:', err)
   188→    }
   189→  }, [token])
   190→
   191→  const fetchUpcoming = useCallback(async () => {
   192→    try {
   193→      const res = await fetch('/api/matches?status=upcoming&limit=20')
   194→      const data = await res.json()
   195→      setUpcomingMatches(
   196→        (data.matches || []).map((m: Record<string, unknown>) => ({
   197→          id: m.id,
   198→          homeTeam: m.homeTeam,
   199→          awayTeam: m.awayTeam,
   200→          date: m.date,
   201→          stage: m.stage,
   202→        })),
   203→      )
   204→    } catch (err) {
   205→      console.error('Failed to fetch upcoming:', err)
   206→    }
   207→  }, [])
   208→
   209→  useEffect(() => {
   210→    const load = async () => {
   211→      setLoading(true)
   212→      await Promise.all([fetchPredictions(), fetchUpcoming()])
   213→      setLoading(false)
   214→    }
   215→    load()
   216→  }, [fetchPredictions, fetchUpcoming])
   217→
   218→  // ── Sorting ────────────────────────────────────────────────────────────────
   219→
   220→  const toggleSort = (field: SortField) => {
   221→    if (sortField === field) {
   222→      setSortDir((d) => (d === 'asc' ? 'desc' : 'asc'))
   223→    } else {
   224→      setSortField(field)
   225→      setSortDir('desc')
   226→    }
   227→  }
   228→
   229→  const sortedPredictions = React.useMemo(() => {
   230→    return [...predictions].sort((a, b) => {
   231→      let cmp = 0
   232→      switch (sortField) {
   233→        case 'date':
   234→          cmp = new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime()
   235→          break
   236→        case 'confidence':
   237→          cmp = a.confidence - b.confidence
   238→          break
   239→        case 'result':
   240→          cmp = (a.isCorrect === true ? 1 : 0) - (b.isCorrect === true ? 1 : 0)
   241→          break
   242→      }
   243→      return sortDir === 'asc' ? cmp : -cmp
   244→    })
   245→  }, [predictions, sortField, sortDir])
   246→
   247→  // ── Submit Prediction ──────────────────────────────────────────────────────
   248→
   249→  const handleSubmit = async () => {
   250→    if (!selectedMatch || !outcome) return
   251→    setSubmitting(true)
   252→    setSubmitMessage(null)
   253→
   254→    try {
   255→      const res = await fetch('/api/predictions', {
   256→        method: 'POST',
   257→        headers: {
   258→          'Content-Type': 'application/json',
   259→          ...(token ? { Authorization: `Bearer ${token}` } : {}),
   260→        },
   261→        body: JSON.stringify({
   262→          matchId: selectedMatch,
   263→          predictedHomeGoals: homeGoals,
   264→          predictedAwayGoals: awayGoals,
   265→          predictedOutcome: outcome,
   266→          confidence,
   267→          model,
   268→        }),
   269→      })
   270→
   271→      if (res.ok) {
   272→        setSubmitMessage('Prediction submitted successfully!')
   273→        setSelectedMatch('')
   274→        setHomeGoals(1)
   275→        setAwayGoals(0)
   276→        setOutcome('')
   277→        setConfidence(50)
   278→        setModel('user')
   279→      } else {
   280→        const err = await res.json()
   281→        setSubmitMessage(err.error || 'Failed to submit prediction')
   282→      }
   283→    } catch {
   284→      setSubmitMessage('Network error. Please try again.')
   285→    } finally {
   286→      setSubmitting(false)
   287→      setTimeout(() => setSubmitMessage(null), 4000)
   288→    }
   289→  }
   290→
   291→  // Auto-set outcome based on score
   292→  useEffect(() => {
   293→    if (homeGoals > awayGoals) setOutcome('home')
   294→    else if (homeGoals < awayGoals) setOutcome('away')
   295→    else setOutcome('draw')
   296→  }, [homeGoals, awayGoals])
   297→
   298→  // ── Render: Accuracy Gauge ─────────────────────────────────────────────────
   299→
   300→  const gaugeGradient = `conic-gradient(
   301→    #00e676 0% ${accuracyPct}%,
   302→    oklch(0.2 0.03 260) ${accuracyPct}% 100%
   303→  )`
   304→
   305→  // ═══════════════════════════════════════════════════════════════════════════
   306→  return (
   307→    <div className="space-y-6 animate-fade-in-up">
   308→      {/* ── Page Header ──────────────────────────────────────────────────── */}
   309→      <div className="flex items-center gap-3">
   310→        <div className="flex size-10 items-center justify-center rounded-xl bg-primary/15">
   311→          <Target className="size-5 text-primary" />
   312→        </div>
   313→        <div>
   314→          <h1 className="text-2xl font-bold tracking-tight">Predictions</h1>
   315→          <p className="text-sm text-muted-foreground">
   316→            Track your prediction accuracy and make new predictions
   317→          </p>
   318→        </div>
   319→      </div>
   320→
   321→      {/* ── Accuracy Overview + Model Comparison ─────────────────────────── */}
   322→      <div className="grid grid-cols-1 gap-6 lg:grid-cols-3">
   323→        {/* ── Accuracy Gauge Card ─────────────────────────────────────────── */}
   324→        <Card className="glass-card border-border/50">
   325→          <CardHeader className="pb-2">
   326→            <CardTitle className="text-sm font-medium text-muted-foreground">
   327→              Accuracy Overview
   328→            </CardTitle>
   329→          </CardHeader>
   330→          <CardContent className="flex flex-col items-center gap-4 pt-2">
   331→            {/* Circular gauge */}
   332→            <div className="relative">
   333→              <div
   334→                className="flex size-40 items-center justify-center rounded-full"
   335→                style={{ background: gaugeGradient }}
   336→              >
   337→                <div className="flex size-32 flex-col items-center justify-center rounded-full bg-card">
   338→                  <span className="text-4xl font-bold text-primary">{accuracyPct}%</span>
   339→                  <span className="text-xs text-muted-foreground">Accuracy</span>
   340→                </div>
   341→              </div>
   342→            </div>
   343→
   344→            {/* Stats row */}
   345→            <div className="grid w-full grid-cols-2 gap-3">
   346→              <div className="rounded-lg bg-muted/50 p-3 text-center">
   347→                <p className="text-lg font-bold">{totalPredictions}</p>
   348→                <p className="text-xs text-muted-foreground">Total</p>
   349→              </div>
   350→              <div className="rounded-lg bg-muted/50 p-3 text-center">
   351→                <p className="text-lg font-bold text-emerald-400">{correctPredictions}</p>
   352→                <p className="text-xs text-muted-foreground">Correct</p>
   353→              </div>
   354→              <div className="rounded-lg bg-muted/50 p-3 text-center">
   355→                <p className="text-lg font-bold text-primary">{accuracyPct}%</p>
   356→                <p className="text-xs text-muted-foreground">Accuracy</p>
   357→              </div>
   358→              <div className="rounded-lg bg-muted/50 p-3 text-center">
   359→                <div className="flex items-center justify-center gap-1">
   360→                  <Flame className="size-3.5 text-amber-400" />
   361→                  <p className="text-lg font-bold text-amber-400">{currentStreak}</p>
   362→                </div>
   363→                <p className="text-xs text-muted-foreground">Streak</p>
   364→              </div>
   365→            </div>
   366→          </CardContent>
   367→        </Card>
   368→
   369→        {/* ── Prediction History Table ─────────────────────────────────────── */}
   370→        <Card className="glass-card border-border/50 lg:col-span-2">
   371→          <CardHeader className="pb-2">
   372→            <div className="flex items-center justify-between">
   373→              <CardTitle className="text-sm font-medium text-muted-foreground">
   374→                Prediction History
   375→              </CardTitle>
   376→              {totalPredictions > 0 && (
   377→                <Badge variant="outline" className="text-xs border-primary/30 text-primary">
   378→                  {totalPredictions} predictions
   379→                </Badge>
   380→              )}
   381→            </div>
   382→          </CardHeader>
   383→          <CardContent>
   384→            {loading ? (
   385→              <div className="space-y-2">
   386→                {Array.from({ length: 5 }).map((_, i) => (
   387→                  <Skeleton key={i} className="h-10 w-full" />
   388→                ))}
   389→              </div>
   390→            ) : sortedPredictions.length === 0 ? (
   391→              <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">
   392→                <Target className="mb-3 size-10 opacity-30" />
   393→                <p className="text-sm font-medium">No predictions yet</p>
   394→                <p className="text-xs mt-1">Start predicting matches to see your history</p>
   395→              </div>
   396→            ) : (
   397→              <div className="max-h-96 overflow-y-auto rounded-lg border border-border/50">
   398→                <Table>
   399→                  <TableHeader>
   400→                    <TableRow className="hover:bg-transparent">
   401→                      <TableHead className="text-xs">Match</TableHead>
   402→                      <TableHead className="text-xs">Pred.</TableHead>
   403→                      <TableHead className="text-xs">Actual</TableHead>
   404→                      <TableHead
   405→                        className="text-xs cursor-pointer select-none"
   406→                        onClick={() => toggleSort('result')}
   407→                      >
   408→                        <div className="flex items-center gap-1">
   409→                          Result <ArrowUpDown className="size-3" />
   410→                        </div>
   411→                      </TableHead>
   412→                      <TableHead
   413→                        className="text-xs cursor-pointer select-none"
   414→                        onClick={() => toggleSort('confidence')}
   415→                      >
   416→                        <div className="flex items-center gap-1">
   417→                          Conf. <ArrowUpDown className="size-3" />
   418→                        </div>
   419→                      </TableHead>
   420→                      <TableHead className="text-xs">Model</TableHead>
   421→                      <TableHead className="text-xs text-right">Pts</TableHead>
   422→                    </TableRow>
   423→                  </TableHeader>
   424→                  <TableBody>
   425→                    {sortedPredictions.map((p) => {
   426→                      const homeName = p.match?.homeTeam?.code || 'TBD'
   427→                      const awayName = p.match?.awayTeam?.code || 'TBD'
   428→                      const isCorrect = p.isCorrect === true
   429→                      const isWrong = p.isCorrect === false
   430→
   431→                      return (
   432→                        <TableRow key={p.id}>
   433→                          <TableCell className="text-xs font-medium">
   434→                            {homeName} vs {awayName}
   435→                          </TableCell>
   436→                          <TableCell className="text-xs">
   437→                            <span className="text-primary font-semibold">
   438→                              {p.predictedHomeGoals}-{p.predictedAwayGoals}
   439→                            </span>
   440→                          </TableCell>
   441→                          <TableCell className="text-xs">
   442→                            {p.match ? `${p.match.homeScore}-${p.match.awayScore}` : '—'}
   443→                          </TableCell>
   444→                          <TableCell>
   445→                            {isCorrect && (
   446→                              <CheckCircle2 className="size-4 text-emerald-400" />
   447→                            )}
   448→                            {isWrong && (
   449→                              <XCircle className="size-4 text-red-400" />
   450→                            )}
   451→                            {p.isCorrect === null && (
   452→                              <span className="text-xs text-muted-foreground">Pending</span>
   453→                            )}
   454→                          </TableCell>
   455→                          <TableCell className="text-xs">{p.confidence}%</TableCell>
   456→                          <TableCell>
   457→                            <Badge
   458→                              variant="outline"
   459→                              className="text-[10px] px-1.5 py-0"
   460→                              style={{
   461→                                borderColor: MODEL_COLORS[p.model] || '#666',
   462→                                color: MODEL_COLORS[p.model] || '#666',
   463→                              }}
   464→                            >
   465→                              {MODEL_LABELS[p.model] || p.model}
   466→                            </Badge>
   467→                          </TableCell>
   468→                          <TableCell className="text-xs text-right font-semibold">
   469→                            <span className={p.points > 0 ? 'text-amber-400' : 'text-muted-foreground'}>
   470→                              {p.points > 0 ? `+${p.points}` : '0'}
   471→                            </span>
   472→                          </TableCell>
   473→                        </TableRow>
   474→                      )
   475→                    })}
   476→                  </TableBody>
   477→                </Table>
   478→              </div>
   479→            )}
   480→          </CardContent>
   481→        </Card>
   482→      </div>
   483→
   484→      {/* ── Model Comparison + Make Prediction ───────────────────────────── */}
   485→      <div className="grid grid-cols-1 gap-6 lg:grid-cols-2">
   486→        {/* ── Model Comparison ─────────────────────────────────────────────── */}
   487→        <Card className="glass-card border-border/50">
   488→          <CardHeader className="pb-2">
   489→            <div className="flex items-center gap-2">
   490→              <Brain className="size-4 text-primary" />
   491→              <CardTitle className="text-sm font-medium text-muted-foreground">
   492→                Model Comparison
   493→              </CardTitle>
   494→            </div>
   495→          </CardHeader>
   496→          <CardContent>
   497→            {modelChartData.length === 0 ? (
   498→              <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">
   499→                <BarChart3 className="mb-3 size-10 opacity-30" />
   500→                <p className="text-sm font-medium">No model data yet</p>
   501→                <p className="text-xs mt-1">Predict matches to compare model accuracy</p>
   502→              </div>
   503→            ) : (
   504→              <div className="h-64">
   505→                <ResponsiveContainer width="100%" height="100%">
   506→                  <BarChart data={modelChartData} layout="vertical" margin={{ left: 0, right: 20, top: 5, bottom: 5 }}>
   507→                    <CartesianGrid strokeDasharray="3 3" stroke="oklch(0.2 0.03 260)" />
   508→                    <XAxis type="number" domain={[0, 100]} tick={{ fill: 'oklch(0.6 0 0)', fontSize: 11 }} />
   509→                    <YAxis type="category" dataKey="name" width={80} tick={{ fill: 'oklch(0.8 0 0)', fontSize: 11 }} />
   510→                    <RTooltip content={<ChartTooltip />} />
   511→                    <Bar dataKey="accuracy" name="Accuracy" radius={[0, 4, 4, 0]} barSize={24}>
   512→                      {modelChartData.map((entry, index) => (
   513→                        <Cell key={index} fill={entry.color} fillOpacity={0.85} />
   514→                      ))}
   515→                    </Bar>
   516→                  </BarChart>
   517→                </ResponsiveContainer>
   518→              </div>
   519→            )}
   520→
   521→            {/* Model legend */}
   522→            {modelChartData.length > 0 && (
   523→              <div className="mt-4 flex flex-wrap gap-3">
   524→                {modelChartData.map((m) => (
   525→                  <div key={m.name} className="flex items-center gap-1.5 text-xs text-muted-foreground">
   526→                    <div className="size-2.5 rounded-full" style={{ backgroundColor: m.color }} />
   527→                    <span>{m.name}</span>
   528→                  </div>
   529→                ))}
   530→              </div>
   531→            )}
   532→          </CardContent>
   533→        </Card>
   534→
   535→        {/* ── Make Prediction Panel ────────────────────────────────────────── */}
   536→        <Card className="glass-card border-border/50">
   537→          <CardHeader className="pb-2">
   538→            <div className="flex items-center gap-2">
   539→              <Send className="size-4 text-primary" />
   540→              <CardTitle className="text-sm font-medium text-muted-foreground">
   541→                Make a Prediction
   542→              </CardTitle>
   543→            </div>
   544→          </CardHeader>
   545→          <CardContent className="space-y-5">
   546→            {/* Match Selector */}
   547→            <div className="space-y-1.5">
   548→              <Label className="text-xs text-muted-foreground">Select Match</Label>
   549→              <Select value={selectedMatch} onValueChange={setSelectedMatch}>
   550→                <SelectTrigger className="w-full bg-muted/50 border-border/50">
   551→                  <SelectValue placeholder="Choose an upcoming match..." />
   552→                </SelectTrigger>
   553→                <SelectContent>
   554→                  {upcomingMatches.map((m) => (
   555→                    <SelectItem key={m.id} value={m.id}>
   556→                      <span className="text-xs">
   557→                        {(m.homeTeam?.name || 'TBD')} vs {(m.awayTeam?.name || 'TBD')}
   558→                        {m.stage && (
   559→                          <span className="ml-2 text-muted-foreground">({m.stage})</span>
   560→                        )}
   561→                      </span>
   562→                    </SelectItem>
   563→                  ))}
   564→                  {upcomingMatches.length === 0 && (
   565→                    <SelectItem value="_none" disabled>
   566→                      No upcoming matches
   567→                    </SelectItem>
   568→                  )}
   569→                </SelectContent>
   570→              </Select>
   571→            </div>
   572→
   573→            {/* Score Prediction */}
   574→            <div className="space-y-1.5">
   575→              <Label className="text-xs text-muted-foreground">Predicted Score</Label>
   576→              <div className="flex items-center justify-center gap-6 rounded-lg border border-border/50 bg-muted/30 p-4">
   577→                {/* Home goals */}
   578→                <div className="flex flex-col items-center gap-1">
   579→                  <Button
   580→                    variant="ghost"
   581→                    size="icon"

   584→                  >
   585→                    <Plus className="size-3.5" />
   586→                  </Button>
   587→                  <span className="text-3xl font-bold tabular-nums">{homeGoals}</span>
   588→                  <Button
   589→                    variant="ghost"
   590→                    size="icon"
   591→                    className="size-7 rounded-full bg-destructive/10 text-destructive hover:bg-destructive/20"
   592→                    onClick={() => setHomeGoals((g) => Math.max(0, g - 1))}
   593→                  >
   594→                    <Minus className="size-3.5" />
   595→                  </Button>
   596→                </div>
   597→
   598→                <span className="text-xl font-bold text-muted-foreground">—</span>
   599→
   600→                {/* Away goals */}
   601→                <div className="flex flex-col items-center gap-1">
   602→                  <Button
   603→                    variant="ghost"
   604→                    size="icon"
   605→                    className="size-7 rounded-full bg-primary/10 text-primary hover:bg-primary/20"
   606→                    onClick={() => setAwayGoals((g) => Math.max(0, g + 1))}
   607→                  >
   608→                    <Plus className="size-3.5" />
   609→                  </Button>
   610→                  <span className="text-3xl font-bold tabular-nums">{awayGoals}</span>
   611→                  <Button
   612→                    variant="ghost"
   613→                    size="icon"
   614→                    className="size-7 rounded-full bg-destructive/10 text-destructive hover:bg-destructive/20"
   615→                    onClick={() => setAwayGoals((g) => Math.max(0, g - 1))}
   616→                  >
   617→                    <Minus className="size-3.5" />
   618→                  </Button>
   619→                </div>
   620→              </div>
   621→            </div>
   622→
   623→            {/* Outcome Selector */}
   624→            <div className="space-y-1.5">
   625→              <Label className="text-xs text-muted-foreground">Predicted Outcome</Label>
   626→              <div className="grid grid-cols-3 gap-2">
   627→                {(['home', 'draw', 'away'] as const).map((o) => {
   628→                  const labels = { home: 'Home Win', draw: 'Draw', away: 'Away Win' }
   629→                  const isActive = outcome === o
   630→                  return (
   631→                    <button
   632→                      key={o}
   633→                      onClick={() => setOutcome(o)}
   634→                      className={`rounded-lg border px-3 py-2 text-xs font-medium transition-all ${
   635→                        isActive
   636→                          ? 'border-primary bg-primary/15 text-primary shadow-sm shadow-primary/10'
   637→                          : 'border-border/50 bg-muted/30 text-muted-foreground hover:bg-muted/50'
   638→                      }`}
   639→                    >
   640→                      {labels[o]}
   641→                    </button>
   642→                  )
   643→                })}
   644→              </div>
   645→            </div>
   646→
   647→            {/* Confidence Slider */}
   648→            <div className="space-y-2">
   649→              <div className="flex items-center justify-between">
   650→                <Label className="text-xs text-muted-foreground">Confidence</Label>
   651→                <span className="text-sm font-bold text-primary tabular-nums">{confidence}%</span>
   652→              </div>
   653→              <Slider
   654→                value={[confidence]}
   655→                onValueChange={([v]) => setConfidence(v)}
   656→                min={1}
   657→                max={100}
   658→                step={1}
   659→              />
   660→              <div className="flex justify-between text-[10px] text-muted-foreground">
   661→                <span>Low (1%)</span>
   662→                <span>High (100%)</span>
   663→              </div>
   664→            </div>
   665→
   666→            {/* Model Selector */}
   667→            <div className="space-y-1.5">
   668→              <Label className="text-xs text-muted-foreground">Prediction Model</Label>
   669→              <Select value={model} onValueChange={setModel}>
   670→                <SelectTrigger className="w-full bg-muted/50 border-border/50">
   671→                  <SelectValue />
   672→                </SelectTrigger>
   673→                <SelectContent>
   674→                  {Object.entries(MODEL_LABELS).map(([key, label]) => (
   675→                    <SelectItem key={key} value={key}>
   676→                      <div className="flex items-center gap-2">
   677→                        <div
   678→                          className="size-2.5 rounded-full"
   679→                          style={{ backgroundColor: MODEL_COLORS[key] }}
   680→                        />
   681→                        <span className="text-xs">{label}</span>
   682→                      </div>
   683→                    </SelectItem>
   684→                  ))}
   685→                </SelectContent>
   686→              </Select>
   687→            </div>
   688→
   689→            {/* Submit Message */}
   690→            {submitMessage && (
   691→              <div
   692→                className={`rounded-lg border px-3 py-2 text-xs ${
   693→                  submitMessage.includes('success')
   694→                    ? 'border-emerald-500/30 bg-emerald-500/10 text-emerald-400'
   695→                    : 'border-red-500/30 bg-red-500/10 text-red-400'
   696→                }`}
   697→              >
   698→                {submitMessage}
   699→              </div>
   700→            )}
   701→
   702→            {/* Submit Button */}
   703→            <Button
   704→              className="w-full bg-primary text-primary-foreground hover:bg-primary/90"
   705→              onClick={handleSubmit}
   706→              disabled={!selectedMatch || !outcome || submitting}
   707→            >
   708→              {submitting ? (
   709→                <span className="flex items-center gap-2">
   710→                  <span className="size-4 animate-spin rounded-full border-2 border-current border-t-transparent" />
   711→                  Submitting...
   712→                </span>
   713→              ) : (
   714→                <span className="flex items-center gap-2">
   715→                  <Send className="size-4" />
   716→                  Submit Prediction
   717→                </span>
   718→              )}
   719→            </Button>
   720→          </CardContent>
   721→        </Card>
   722→      </div>
   723→    </div>
