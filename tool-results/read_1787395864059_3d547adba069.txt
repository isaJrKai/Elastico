    51→async function syncFixturesFromApiSports(): Promise<{
    52→  created: number
    53→  updated: number
    54→  processed: number
    55→}> {
    56→  const result = { created: 0, updated: 0, processed: 0 }
    57→
    58→  try {
    59→    // 1. Live fixtures (1 API call — covers all leagues)
    60→    const liveFixtures = await fetchLiveFixtures()
    61→    console.log(`[SYNC/API-Sports] Got ${liveFixtures.length} live fixtures`)
    62→
    63→    for (const f of liveFixtures) {
    64→      checkTimeout()
    65→      await upsertApiSportsFixture(f, result)
    66→    }
    67→
    68→    // 2. Today's fixtures from top 5 leagues (5 API calls)
    69→    const top5 = AS_LEAGUES.slice(0, 5) // PL, LIGA, SA, BL, L1
    70→    for (const league of top5) {
    71→      checkTimeout()
    72→      try {
    73→        const fixtures = await fetchFixtures(league.id)
    74→        for (const f of fixtures) {
    75→          checkTimeout()
    76→          await upsertApiSportsFixture(f, result)
    77→        }
    78→      } catch (err: any) {
    79→        if (err.message?.includes('timeout')) throw err
    80→        console.warn(`[SYNC/API-Sports] Failed fixtures for ${league.code}:`, err.message)
    81→      }
    82→    }
    83→  } catch (err: any) {
    84→    if (err.message?.includes('timeout')) throw err
    85→    console.warn('[SYNC/API-Sports] Fixtures failed:', err.message || err)
    86→  }
    87→
    88→  return result

   421→async function fetchScoreboardForLeague(league: { espnId: string; name: string; code: string }) {
   422→  const SITE_V2 = 'https://site.api.espn.com/apis/v2/sports/soccer'
   423→  const url = `${SITE_V2}/${league.espnId}/scoreboard`
   424→  try {
   425→    const res = await fetch(url, { cache: 'no-store' })
   426→    if (!res.ok) return []
   427→    const data = await res.json()
   428→    const events: any[] = []
   429→    for (const event of data.events || []) {
   430→      const comp = event.competitions?.[0]
   431→      if (!comp) continue
   432→      const competitors = comp.competitors || []
   433→      if (competitors.length < 2) continue
   434→      const home = competitors.find((c: any) => c.homeAway === 'home') || competitors[0]
   435→      const away = competitors.find((c: any) => c.homeAway === 'away') || competitors[1]
   436→      const statusDetail = comp.status?.type?.detail || ''
   437→      const minuteMatch = statusDetail.match(/(\d+)\s*'/)
   438→      events.push({
   439→        id: event.id?.toString() || '',
   440→        homeTeamId: home.team?.id?.toString() || '',
   441→        homeTeamName: home.team?.displayName || 'Unknown',
   442→        homeTeamAbbr: home.team?.abbreviation || '???',
   443→        homeTeamLogo: home.team?.logo || '',
   444→        homeTeamColor: home.team?.color || '#00e676',
   445→        awayTeamId: away.team?.id?.toString() || '',
   446→        awayTeamName: away.team?.displayName || 'Unknown',
   447→        awayTeamAbbr: away.team?.abbreviation || '???',
   448→        awayTeamLogo: away.team?.logo || '',
   449→        awayTeamColor: away.team?.color || '#ffffff',
   450→        homeScore: parseInt(home.score) || 0,
   451→        awayScore: parseInt(away.score) || 0,
   452→        status: comp.status?.type?.name || 'STATUS_SCHEDULED',
   453→        date: event.date || '',
   454→        venue: comp.venue?.fullName || '',
   455→        competition: event.name || league.name,
   456→        competitionCode: league.code,
   457→        minute: minuteMatch ? parseInt(minuteMatch[1]) : undefined,
   458→      })
   459→    }
   460→    return events
   461→  } catch (err) {
   462→    console.error(`[SYNC/ESPN] Failed to fetch ${league.espnId}:`, err)
   463→    return []
   464→  }
   465→}

   602→            teamCode: entry.code,
   603→            teamLogo: entry.logo,
   604→            competition: leagueName,
   605→            rank: entry.rank,
   606→            played: entry.played,
   607→            wins: entry.wins,
   608→            draws: entry.draws,
   609→            losses: entry.losses,
   610→            goalsFor: entry.goalsFor,
   611→            goalsAgainst: entry.goalsAgainst,
   612→            goalDiff: entry.goalDiff,
   613→            points: entry.points,
   614→            form: entry.form || null,
   615→            homeRecord: entry.homeRecord || null,
   616→            source: 'espn',
   617→            lastSyncedAt: new Date(),
   618→          },
   619→          create: {
   620→            teamName: entry.team,
   621→            teamCode: entry.code,
   622→            teamLogo: entry.logo,
   623→            competition: leagueName,
   624→            competitionCode: code,
   625→            season,
   626→            rank: entry.rank,
   627→            played: entry.played,
   628→            wins: entry.wins,
   629→            draws: entry.draws,
   630→            losses: entry.losses,
   631→            goalsFor: entry.goalsFor,
   632→            goalsAgainst: entry.goalsAgainst,
   633→            goalDiff: entry.goalDiff,
   634→            points: entry.points,
   635→            form: entry.form || null,
   636→            homeRecord: entry.homeRecord || null,
   637→            source: 'espn',
   638→          },
   639→        })
   640→        totalCreated++
   641→      } catch (err) {
   642→        console.warn(`[SYNC/Standings] Failed to upsert ${entry.team} in ${code}:`, err)
   643→      }
   644→    }
   645→  }
   646→
   647→  return { created: totalCreated }
   648→}

   652→async function logSync(
   653→  source: string,
   654→  action: string,
   655→  status: string,
   656→  processed: number,
   657→  created: number,
   658→  updated: number,
   659→  errorMessage?: string
   660→) {
   661→  try {
   662→    await db.syncLog.create({
   663→      data: {
   664→        source,
   665→        action,
   666→        status,
   667→        recordsProcessed: processed,
   668→        recordsCreated: created,
   669→        recordsUpdated: updated,
   670→        durationMs: elapsed(),
   671→        errorMessage: errorMessage || null,
   672→      },
   673→    })
   674→  } catch (err) {
   675→    console.error('[SYNC] Failed to write SyncLog:', err)
   676→  }
   677→}

   679→// ═══════════════════════════════════════════════════════════════════════════════
   680→// MAIN HANDLER
   681→// ═══════════════════════════════════════════════════════════════════════════════
   682→
   683→export async function GET(req: NextRequest) {
   684→  // 1. Validate cron secret
   685→  if (!validateCron(req)) {
   686→    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
   687→  }
   688→
   689→  const summary: any = {
   690→    success: true,
   691→    syncs: {
   692→      fixtures: { source: 'none', created: 0, updated: 0 },
   693→      standings: { source: 'none', created: 0, updated: 0 },
   694→      teams: { source: 'none', created: 0, updated: 0 },
   695→      players: { source: 'none', created: 0, updated: 0 },
   696→    },
   697→    durationMs: 0,
   698→  }
   699→
   700→  try {
   701→    // ═══ 2. API-Sports: Fixtures + Standings + Teams + Players ═══
   702→    let apiSportsAvailable = false
   703→
   704→    // 2a. Fixtures (live + today's top 5 leagues)
   705→    try {
   706→      const fixResult = await syncFixturesFromApiSports()
   707→      if (fixResult.processed > 0) {
   708→        apiSportsAvailable = true
   709→        summary.syncs.fixtures = {
   710→          source: 'api-sports',
   711→          created: fixResult.created,
   712→          updated: fixResult.updated,
   713→        }
   714→        await logSync('api-sports', 'fixtures', 'success', fixResult.processed, fixResult.created, fixResult.updated)
   715→      }
   716→    } catch (err: any) {
   717→      if (err.message?.includes('timeout')) {
   718→        console.warn('[SYNC] API-Sports fixtures timeout')
   719→        await logSync('api-sports', 'fixtures', 'partial', 0, 0, 0, err.message)
   720→      } else {
   721→        console.warn('[SYNC] API-Sports fixtures failed:', err.message)
   722→      }
   723→    }
   724→
   725→    // 2b. Standings (top 10 leagues — only if API-Sports is working)
   726→    if (apiSportsAvailable) {
   727→      try {
   728→        const stResult = await syncStandingsFromApiSports()
   729→        summary.syncs.standings = {
   730→          source: 'api-sports',
   731→          created: stResult.created,
   732→          updated: stResult.updated,
   733→        }
   734→        await logSync('api-sports', 'standings', 'success', stResult.created, stResult.created, stResult.updated)
   735→      } catch (err: any) {
   736→        if (err.message?.includes('timeout')) {
   737→          console.warn('[SYNC] API-Sports standings timeout')
   738→          await logSync('api-sports', 'standings', 'partial', 0, 0, 0, err.message)
   739→        } else {
   740→          console.warn('[SYNC] API-Sports standings failed:', err.message)
   741→        }
   742→      }
   743→    }
   744→
   745→    // 2c. Teams (top 5 leagues — venue info, country, etc.)
   746→    if (apiSportsAvailable) {
   747→      try {
   748→        const tmResult = await syncTeamsFromApiSports()
   749→        summary.syncs.teams = {
   750→          source: 'api-sports',
   751→          created: tmResult.created,
   752→          updated: tmResult.updated,
   753→        }
   754→        await logSync('api-sports', 'teams', 'success', tmResult.created + tmResult.updated, tmResult.created, tmResult.updated)
   755→      } catch (err: any) {
   756→        if (err.message?.includes('timeout')) {
   757→          console.warn('[SYNC] API-Sports teams timeout')
   758→        } else {
   759→          console.warn('[SYNC] API-Sports teams failed:', err.message)
   760→        }
   761→      }
   762→    }
   763→
   764→    // 2d. Player stats (top scorers, top 5 leagues)
   765→    if (apiSportsAvailable) {
   766→      try {
   767→        const plResult = await syncPlayersFromApiSports()
   768→        summary.syncs.players = {
   769→          source: 'api-sports',
   770→          created: plResult.created,
   771→          updated: plResult.updated,
   772→        }
   773→        await logSync('api-sports', 'players', 'success', plResult.created + plResult.updated, plResult.created, plResult.updated)
   774→      } catch (err: any) {
   775→        if (err.message?.includes('timeout')) {
   776→          console.warn('[SYNC] API-Sports players timeout')
   777→        } else {
   778→          console.warn('[SYNC] API-Sports players failed:', err.message)
   779→        }
   780→      }
   781→    }
   782→
   783→    // ═══ 3. ESPN Fallback — only if API-Sports didn't return fixtures ═══
   784→    if (!apiSportsAvailable) {
   785→      try {
   786→        const espnResult = await syncFixturesFromESPN()
   787→        summary.syncs.fixtures = {
   788→          source: 'espn',
   789→          created: espnResult.created,
   790→          updated: espnResult.updated,
   791→        }
   792→        await logSync('espn', 'fixtures', 'success', espnResult.processed, espnResult.created, espnResult.updated)
   793→      } catch (err: any) {
   794→        console.error('[SYNC] ESPN fixtures failed:', err)
   795→        await logSync('espn', 'fixtures', 'error', 0, 0, 0, String(err))
   796→        summary.syncs.fixtures.error = String(err)
   797→      }
   798→
   799→      // ESPN Standings fallback
   800→      try {
   801→        const standingsResult = await syncStandingsFromESPN()
   802→        summary.syncs.standings = {
   803→          source: 'espn',
   804→          created: standingsResult.created,
   805→          updated: 0,
   806→        }
   807→        await logSync('espn', 'standings', 'success', standingsResult.created, standingsResult.created, 0)
   808→      } catch (err: any) {
   809→        if (err.message?.includes('timeout')) {
   810→          console.warn('[SYNC] ESPN standings timeout')
   811→          await logSync('espn', 'standings', 'partial', 0, 0, 0, err.message)
   812→        } else {
   813→          console.error('[SYNC] ESPN standings failed:', err)
   814→          await logSync('espn', 'standings', 'error', 0, 0, 0, String(err))
   815→        }
   816→      }
   817→    }
   818→
   819→    summary.durationMs = elapsed()
   820→    console.log(`[SYNC] Complete in ${summary.durationMs}ms`, JSON.stringify(summary.syncs))
   821→
   822→    return NextResponse.json(summary)
   823→  } catch (error) {
   824→    const duration = elapsed()
   825→    console.error(`[SYNC] Fatal error after ${duration}ms:`, error)
   826→    summary.success = false
   827→    summary.durationMs = duration
   828→    summary.error = error instanceof Error ? error.message : String(error)
   829→    return NextResponse.json(summary, { status: 500 })
   830→  }
   831→}
