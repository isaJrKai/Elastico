   140→   140→# ═══════════════════════════════════════════════════════════════════════════════
   141→   141→# FootballLSTM — the neural network
   142→   142→# ═══════════════════════════════════════════════════════════════════════════════
   143→   143→
   144→   144→
   145→   145→class FootballLSTM(nn.Module):
   146→   146→    """Bidirectional LSTM with attention for football match outcome prediction.
   147→   147→
   148→   148→    Architecture::
   149→   149→
   150→   150→        Input (batch, seq_len, input_size)
   151→   151→        ├─► Home LSTM ─► Attention ─► home_context (batch, hidden*2)
   152→   152→        └─► Away LSTM ─► Attention ─► away_context (batch, hidden*2)
   153→   153→        Concat ─► FC(256) ─► BN ─► ReLU ─► Dropout
   154→   154→                      ─► FC(128) ─► BN ─► ReLU ─► Dropout
   155→   155→                      ─► FC(3)
   156→   156→
   157→   157→    Output: 3-class logits (home / draw / away).
   158→   158→    """
   159→   159→
   160→   160→    def __init__(self, config: LSTMConfig):
   161→   161→        super().__init__()
   162→   162→        self.config = config
   163→   163→
   164→   164→        bidir_hidden = config.hidden_size * 2
   165→   165→
   166→   166→        # --- Shared (but separately-run) LSTM for each team ---
   167→   167→        self.lstm = nn.LSTM(
   168→   168→            input_size=config.input_size,
   169→   169→            hidden_size=config.hidden_size,
   170→   170→            num_layers=config.num_layers,
   171→   171→            batch_first=True,
   172→   172→            bidirectional=True,
   173→   173→            dropout=config.dropout if config.num_layers > 1 else 0.0,
   174→   174→        )
   175→   175→
   176→   176→        # --- Attention ---
   177→   177→        self.attention = nn.Linear(bidir_hidden, 1)
   178→   178→
   179→   179→        # --- FC head ---
   180→   180→        fc_input = bidir_hidden * 2  # concat home + away contexts
   181→   181→        self.fc = nn.Sequential(
   182→   182→            nn.Linear(fc_input, 128),
   183→   183→            nn.BatchNorm1d(128),
   184→   184→            nn.ReLU(),
   185→   185→            nn.Dropout(config.dropout),
   186→   186→            nn.Linear(128, 64),
   187→   187→            nn.BatchNorm1d(64),
   188→   188→            nn.ReLU(),
   189→   189→            nn.Dropout(config.dropout),
   190→   190→            nn.Linear(64, _N_CLASSES),
   191→   191→        )
   192→   192→
   193→   193→        self._init_weights()
   194→   194→
   195→   195→    # ------------------------------------------------------------------
   196→   196→    # Weight initialisation (Xavier uniform for linear, orthogonal for LSTM)
   197→   197→    # ------------------------------------------------------------------
   198→   198→
   199→   199→    def _init_weights(self) -> None:
   200→   200→        for name, param in self.named_parameters():
   201→   201→            if "lstm" in name and "weight_ih" in name:
   202→   202→                nn.init.xavier_uniform_(param)
   203→   203→            elif "lstm" in name and "weight_hh" in name:
   204→   204→                nn.init.orthogonal_(param)
   205→   205→            elif "lstm" in name and "bias" in name:
   206→   206→                nn.init.zeros_(param)
   207→   207→                # Set forget-gate bias to 1 for stable long-term memory
   208→   208→                n = param.size(0)
   209→   209→                start, end = n // 4, n // 2
   210→   210→                nn.init.constant_(param.data[start:end], 1.0)
   211→   211→            elif "weight" in name and param.ndim >= 2:
   212→   212→                nn.init.xavier_uniform_(param)
   213→   213→            elif "bias" in name:
   214→   214→                nn.init.zeros_(param)
   215→   215→
   216→   216→    # ------------------------------------------------------------------
   217→   217→    # Forward
   218→   218→    # ------------------------------------------------------------------
   219→   219→
   220→   220→    def _attend(self, lstm_out: torch.Tensor) -> torch.Tensor:
   221→   221→        """Softmax attention over timesteps.
   222→   222→
   223→   223→        Parameters
   224→   224→        ----------
   225→   225→        lstm_out : (batch, seq_len, hidden*2)
   226→   226→
   227→   227→        Returns
   228→   228→        -------
   229→   229→        context : (batch, hidden*2)
   230→   230→        """
   231→   231→        attn_logits = self.attention(lstm_out)  # (batch, seq_len, 1)
   232→   232→        attn_weights = torch.softmax(attn_logits, dim=1)  # (batch, seq_len, 1)
   233→   233→        context = torch.sum(attn_weights * lstm_out, dim=1)  # (batch, hidden*2)
   234→   234→        return context
   235→   235→
   236→   236→    def forward(self, x_home: torch.Tensor, x_away: torch.Tensor) -> torch.Tensor:
   237→   237→        """Forward pass.
   238→   238→
   239→   239→        Parameters
   240→   240→        ----------
   241→   241→        x_home : (batch, seq_len, input_size)
   242→   242→        x_away : (batch, seq_len, input_size)
   243→   243→
   244→   244→        Returns
   245→   245→        -------
   246→   246→        logits : (batch, 3)
   247→   247→        """
   248→   248→        # Encode home team sequence
   249→   249→        home_out, _ = self.lstm(x_home)  # (batch, seq_len, hidden*2)
   250→   250→        home_ctx = self._attend(home_out)  # (batch, hidden*2)
   251→   251→
   252→   252→        # Encode away team sequence (reusing the same LSTM weights)
   253→   253→        away_out, _ = self.lstm(x_away)  # (batch, seq_len, hidden*2)
   254→   254→        away_ctx = self._attend(away_out)  # (batch, hidden*2)
   255→   255→
   256→   256→        # Concatenate contexts
   257→   257→        combined = torch.cat([home_ctx, away_ctx], dim=1)  # (batch, hidden*4)
   258→   258→
   259→   259→        return self.fc(combined)  # (batch, 3)
   260→   260→

   265→   265→
   266→   266→
   267→   267→class LSTMEngine:
   268→   268→    """Full LSTM training and inference engine.
   269→   269→
   270→   270→    Handles:
   271→   271→    - Sequence dataset creation from match history
   272→   272→    - Training with early stopping and LR scheduling
   273→   273→    - Evaluation with classification + calibration metrics
   274→   274→    - Model save / load
   275→   275→    - Single-match and batch inference
   276→   276→
   277→   277→    Usage
   278→   278→    -----
   279→   279→    >>> engine = LSTMEngine()
   280→   280→    >>> X_h, X_a, y = engine.build_sequences(matches, feature_cols)

   282→   282→    >>> pred = engine.predict(X_h[:1], X_a[:1])

   284→   284→    """
   285→   285→
   286→   286→    def __init__(

   288→   288→        config: LSTMConfig | None = None,
   289→   289→        model_dir: str = "saved_models",

   291→   291→        self.config = config or LSTMConfig()
   292→   292→        self.model_dir = Path(model_dir)
   293→   293→        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
   294→   294→        self.model: Optional[FootballLSTM] = None
   295→   295→        self.is_trained: bool = False
   296→   296→        self.feature_cols: list[str] = []
   297→   297→

   299→   299→        self.model_dir.mkdir(parents=True, exist_ok=True)

   301→   301→        # Best model state (for early stopping restoration)

   304→   304→    # ==================================================================
   305→   305→    # Sequence Building
   306→   306→    # ==================================================================
   307→   307→
   308→   308→    def build_sequences(
   309→   309→        self,
   310→   310→        matches: list[dict[str, Any]],
   311→   311→        feature_cols: list[str],
   312→   312→    ) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
   313→   313→        """Build genuine temporal sequences for LSTM training.
   314→   314→
   315→   315→        For each match, constructs sequences of past match performance
   316→   316→        for both home and away teams.  Each timestep in the sequence
   317→   317→        represents ONE past match with features computed from data
   318→   318→        available AT THAT POINT IN TIME (no future leakage).
   319→   319→
   320→   320→        The feature_cols should typically be ``LSTM_SEQUENCE_FEATURES``
   321→   321→        from the canonical schema (15 features).  These are a subset of
   322→   322→        the full 50-feature XGBoost schema — selected for being meaningful
   323→   323→        as per-match performance snapshots.
   324→   324→
   325→   325→        Parameters
   326→   326→        ----------
   327→   327→        matches : list[dict]
   328→   328→            Match records sorted chronologically.  Each dict must contain
   329→   329→            at least ``"date"``, ``"home_team"``, ``"away_team"``,
   330→   330→            ``"home_goals"``, ``"away_goals"``.
   331→   331→        feature_cols : list[str]
   332→   332→            Column names to extract per timestep.  Should match
   333→   333→            ``LSTM_SEQUENCE_FEATURES`` from the canonical schema.
   334→   334→
   335→   335→        Returns
   336→   336→        -------
   337→   337→        X_home : np.ndarray, shape (N, seq_len, n_features)
   338→   338→        X_away : np.ndarray, shape (N, seq_len, n_features)
   339→   339→        y : np.ndarray, shape (N,), dtype int32
   340→   340→            Labels: 0 = home win, 1 = draw, 2 = away win.
   341→   341→
   342→   342→        Notes
   343→   343→        -----
   344→   344→        - Matches that lack enough history for EITHER team are dropped.
   345→   345→        - No zero-padding or vector repetition is performed.
   346→   346→        - Each timestep's features are computed from rolling stats up to
   347→   347→          that point, NOT from future data.
   348→   348→        """
   349→   349→        self.feature_cols = list(feature_cols)
   350→   350→        seq_len = self.config.sequence_length
   351→   351→        n_features = len(feature_cols)
   352→   352→
   353→   353→        # Import the feature engineer to compute per-match features
   354→   354→        from src.ml.features import FeatureEngineer
   355→   355→
   356→   356→        # Use the FeatureEngineer to compute rolling features per team
   357→   357→        # at each point in time.
   358→   358→        engineer = FeatureEngineer()
   359→   359→
   360→   360→        # First pass: build a per-team chronological list of match records
   361→   361→        # and compute the 50-feature vector for each match using rolling
   362→   362→        # stats available at that point.
   363→   363→        #
   364→   364→        # We process all matches in order, building up team histories.
   365→   365→        # For each match, we compute the full 50-feature vector for both
   366→   366→        # teams, then extract the 15 LSTM features from it.
   367→   367→        #
   368→   368→        # This gives us a per-team timeline of feature vectors, where
   369→   369→        # each vector reflects that team's state AT that point in time.
   370→   370→
   371→   371→        from collections import defaultdict
   372→   372→        from datetime import datetime as _dt
   373→   373→
   374→   374→        # team -> list of (date, feature_dict_50)
   375→   375→        team_feature_timeline: dict[str, list[tuple]] = defaultdict(list)
   376→   376→        # team -> list of (date, home_goals, away_goals, result_char) — for rolling stats
   377→   377→        team_history: dict[str, list[tuple]] = defaultdict(list)
   378→   378→        team_home_history: dict[str, list[tuple]] = defaultdict(list)
   379→   379→        team_away_history: dict[str, list[tuple]] = defaultdict(list)
   380→   380→
   381→   381→        # Store per-match computed 50-feature vectors for both teams
   382→   382→        # match_idx -> (home_feature_dict, away_feature_dict)
   383→   383→        match_features: dict[int, tuple[dict[str, float], dict[str, float]]] = {}
   384→   384→
   385→   385→        def _parse_date(value: Any) -> Any:
   386→   386→            if isinstance(value, _dt):
   387→   387→                return value
   388→   388→            if isinstance(value, str):
   389→   389→                for fmt in ("%Y-%m-%d", "%Y-%m-%dT%H:%M:%S", "%d/%m/%Y"):
   390→   390→                    try:
   391→   391→                        return _dt.strptime(value, fmt)
   392→   392→                    except ValueError:
   393→   393→                        continue
   394→   394→            return None
   395→   395→
   396→   396→        # Process matches in chronological order
   397→   397→        for match_idx, match in enumerate(matches):
   398→   398→            home_team = str(match["home_team"])
   399→   399→            away_team = str(match["away_team"])
   400→   400→            hg = int(match.get("home_goals", 0))
   401→   401→            ag = int(match.get("away_goals", 0))
   402→   402→
   403→   403→            if hg > ag:
   404→   404→                home_result, away_result = "W", "L"
   405→   405→            elif hg < ag:
   406→   406→                home_result, away_result = "L", "W"
   407→   407→            else:
   408→   408→                home_result, away_result = "D", "D"
   409→   409→
   410→   410→            match_date = _parse_date(match.get("date")) or _dt.now()
   411→   411→
   412→   412→            # Build stats from rolling history (same logic as FeatureEngineer)
   413→   413→            home_stats = engineer._stats_from_history(
   414→   414→                team_history[home_team],
   415→   415→                team_home_history[home_team],
   416→   416→                team_away_history[home_team],
   417→   417→                window=10,
   418→   418→            )
   419→   419→            away_stats = engineer._stats_from_history(
   420→   420→                team_history[away_team],
   421→   421→                team_home_history[away_team],
   422→   422→                team_away_history[away_team],
   423→   423→                window=10,
   424→   424→            )
   425→   425→
   426→   426→            # Set last-match date for rest-day calculation
   427→   427→            if team_history[home_team]:
   428→   428→                home_stats["last_match_date"] = team_history[home_team][-1][0].isoformat()
   429→   429→            if team_history[away_team]:
   430→   430→                away_stats["last_match_date"] = team_history[away_team][-1][0].isoformat()
   431→   431→
   432→   432→            # Build 50-feature vectors for both teams
   433→   433→            home_feats = engineer.build_match_features(
   434→   434→                match_data={"date": match_date},
   435→   435→                home_stats=home_stats,
   436→   436→                away_stats=away_stats,
   437→   437→                h2h_data=None,  # H2H is match-specific, not per-team
   438→   438→                odds=None,
   439→   439→            )
   440→   440→            away_feats = engineer.build_match_features(
   441→   441→                match_data={"date": match_date},
   442→   442→                home_stats=away_stats,
   443→   443→                away_stats=home_stats,  # swap perspective for away team
   444→   444→                h2h_data=None,
   445→   445→                odds=None,
   446→   446→            )
   447→   447→
   448→   448→            match_features[match_idx] = (home_feats, away_feats)
   449→   449→
   450→   450→            # Add to team timelines — the home_feats represent the home
   451→   451→            # team's state, away_feats represent the away team's state
   452→   452→            team_feature_timeline[home_team].append((match_date, home_feats))
   453→   453→            team_feature_timeline[away_team].append((match_date, away_feats))
   454→   454→
   455→   455→            # Update rolling histories for future matches
   456→   456→            team_history[home_team].append((match_date, hg, ag, home_result))
   457→   457→            team_history[away_team].append((match_date, ag, hg, away_result))
   458→   458→            team_home_history[home_team].append((match_date, hg, ag, home_result))
   459→   459→            team_away_history[away_team].append((match_date, ag, hg, away_result))
   460→   460→
   461→   461→        # Second pass: for each match, extract sequences from timelines
   462→   462→        X_home_list: list[np.ndarray] = []
   463→   463→        X_away_list: list[np.ndarray] = []
   464→   464→        y_list: list[int] = []
   465→   465→
   466→   466→        for match_idx, match in enumerate(matches):
   467→   467→            home_team = str(match["home_team"])
   468→   468→            away_team = str(match["away_team"])
   469→   469→            hg = int(match.get("home_goals", 0))
   470→   470→            ag = int(match.get("away_goals", 0))
   471→   471→
   472→   472→            # Result label
   473→   473→            if hg > ag:
   474→   474→                label = 0  # home win
   475→   475→            elif hg < ag:
   476→   476→                label = 2  # away win
   477→   477→            else:
   478→   478→                label = 1  # draw
   479→   479→
   480→   480→            home_timeline = team_feature_timeline[home_team]
   481→   481→            away_timeline = team_feature_timeline[away_team]
   482→   482→
   483→   483→            # The current match is the LAST entry in each timeline.
   484→   484→            # We need seq_len entries BEFORE the current match.
   485→   485→            if len(home_timeline) < seq_len + 1:
   486→   486→                continue
   487→   487→            if len(away_timeline) < seq_len + 1:
   488→   488→                continue
   489→   489→
   490→   490→            # Extract the last seq_len entries BEFORE the current match
   491→   491→            # Timeline is chronological, current match is at index -1
   492→   492→            home_seq_features = []
   493→   493→            for date, feats in home_timeline[-(seq_len + 1):-1]:
   494→   494→                vec = [float(feats.get(c, 0.0)) for c in feature_cols]
   495→   495→                vec = np.nan_to_num(vec, nan=0.0, posinf=0.0, neginf=0.0)
   496→   496→                home_seq_features.append(vec)
   497→   497→
   498→   498→            away_seq_features = []
   499→   499→            for date, feats in away_timeline[-(seq_len + 1):-1]:
   500→   500→                vec = [float(feats.get(c, 0.0)) for c in feature_cols]
   501→   501→                vec = np.nan_to_num(vec, nan=0.0, posinf=0.0, neginf=0.0)
   502→   502→                away_seq_features.append(vec)
   503→   503→
   504→   504→            home_seq = np.array(home_seq_features, dtype=np.float32)
   505→   505→            away_seq = np.array(away_seq_features, dtype=np.float32)
   506→   506→
   507→   507→            if home_seq.shape != (seq_len, n_features):
   508→   508→                continue
   509→   509→            if away_seq.shape != (seq_len, n_features):
   510→   510→                continue
   511→   511→
   512→   512→            X_home_list.append(home_seq)
   513→   513→            X_away_list.append(away_seq)
   514→   514→            y_list.append(label)
   515→   515→
   516→   516→        if not X_home_list:
   517→   517→            logger.warning(
   518→   518→                "build_sequences produced 0 samples.  Check data length vs sequence_length=%d.",
   519→   519→                seq_len,
   520→   520→            )
   521→   521→            return (
   522→   522→                np.empty((0, seq_len, n_features), dtype=np.float32),
   523→   523→                np.empty((0, seq_len, n_features), dtype=np.float32),
   524→   524→                np.empty(0, dtype=np.int32),
   525→   525→            )
   526→   526→
   527→   527→        X_home = np.stack(X_home_list, axis=0)  # (N, seq_len, n_features)
   528→   528→        X_away = np.stack(X_away_list, axis=0)
   529→   529→        y = np.array(y_list, dtype=np.int32)
   530→   530→
   531→   531→        logger.info(
   532→   532→            "Built GENUINE TEMPORAL sequences: %d samples, seq_len=%d, n_features=%d",
   533→   533→            len(y), seq_len, n_features,
   534→   534→        )
   535→   535→        return X_home, X_away, y
   536→   536→
   537→   537→    # ==================================================================
   538→   538→    # Training
   539→   539→    # ==================================================================
   540→   540→
   541→   541→    def train(
   542→   542→        self,
   543→   543→        X_home: np.ndarray,
   544→   544→        X_away: np.ndarray,
   545→   545→        y: np.ndarray,
   546→   546→        X_val_home: np.ndarray | None = None,
   547→   547→        X_val_away: np.ndarray | None = None,
   548→   548→        y_val: np.ndarray | None = None,
   549→   549→    ) -> dict[str, Any]:
   550→   550→        """Train the LSTM model.
   551→   551→
   552→   552→        Parameters
   553→   553→        ----------
   554→   554→        X_home, X_away : np.ndarray, shape (N, seq_len, input_size)
   555→   555→            Training sequences for home and away teams.
   556→   556→        y : np.ndarray, shape (N,), dtype int
   557→   557→            Labels (0=home, 1=draw, 2=away).
   558→   558→        X_val_home, X_val_away, y_val : optional
   559→   559→            Validation set.  When provided, early stopping monitors
   560→   560→            validation loss.
   561→   561→
   562→   562→        Returns
   563→   563→        -------
   564→   564→        dict
   565→   565→            Training metrics: epochs_run, best_epoch, train_loss_best,
   566→   566→            val_loss_best (if validation), learning_rates, etc.
   567→   567→        """
   568→   568→        # --- Validate inputs ---
   569→   569→        X_home = np.asarray(X_home, dtype=np.float32)
   570→   570→        X_away = np.asarray(X_away, dtype=np.float32)
   571→   571→        y = np.asarray(y, dtype=np.int64)
   572→   572→
   573→   573→        # Auto-detect input_size from data if not set
   574→   574→        if X_home.ndim == 3 and X_home.shape[2] != self.config.input_size:
   575→   575→            logger.info(
   576→   576→                "input_size mismatch: config=%d, data=%d. Updating config.",
   577→   577→                self.config.input_size,
   578→   578→                X_home.shape[2],
   579→   579→            )
   580→   580→            self.config.input_size = X_home.shape[2]
   581→   581→
   582→   582→        # Set random seed for reproducibility
   583→   583→        torch.manual_seed(self.config.random_seed)
   584→   584→        if torch.cuda.is_available():
   585→   585→            torch.cuda.manual_seed_all(self.config.random_seed)
   586→   586→
   587→   587→        # --- Instantiate model ---
   588→   588→        self.model = FootballLSTM(self.config).to(self.device)
   589→   589→
   590→   590→        # --- Data loaders ---
   591→   591→        train_dataset = _MatchSequenceDataset(X_home, X_away, y)
   592→   592→        train_loader = torch.utils.data.DataLoader(
   593→   593→            train_dataset,
   594→   594→            batch_size=self.config.batch_size,
   595→   595→            shuffle=True,
   596→   596→            drop_last=False,
   597→   597→            num_workers=0,
   598→   598→            pin_memory=self.device.type == "cuda",
   599→   599→        )
   600→   600→
   601→   601→        has_val = (
   602→   602→            X_val_home is not None
   603→   603→            and X_val_away is not None
   604→   604→            and y_val is not None
   605→   605→        )
   606→   606→
   607→   607→        if has_val:
   608→   608→            X_val_home = np.asarray(X_val_home, dtype=np.float32)
   609→   609→            X_val_away = np.asarray(X_val_away, dtype=np.float32)
   610→   610→            y_val = np.asarray(y_val, dtype=np.int64)
   611→   611→
   612→   612→            val_dataset = _MatchSequenceDataset(X_val_home, X_val_away, y_val)
   613→   613→            val_loader = torch.utils.data.DataLoader(
   614→   614→                val_dataset,
   615→   615→                batch_size=self.config.batch_size,
   616→   616→                shuffle=False,
   617→   617→                drop_last=False,
   618→   618→                num_workers=0,
   619→   619→                pin_memory=self.device.type == "cuda",
   620→   620→            )
   621→   621→
   622→   622→        # --- Optimiser, loss, scheduler ---
   623→   623→        optimizer = torch.optim.Adam(
   624→   624→            self.model.parameters(),
   625→   625→            lr=self.config.learning_rate,
   626→   626→            weight_decay=self.config.weight_decay,
   627→   627→        )
   628→   628→        criterion = nn.CrossEntropyLoss()
   629→   629→        scheduler = torch.optim.lr_scheduler.ReduceLROnPlateau(
   630→   630→            optimizer,
   631→   631→            mode="min",
   632→   632→            factor=self.config.lr_scheduler_factor,
   633→   633→            patience=self.config.lr_scheduler_patience,
   634→   634→            min_lr=self.config.lr_scheduler_min_lr,
   635→   635→            verbose=False,
   636→   636→        )
   637→   637→
   638→   638→        # --- Training loop ---
   639→   639→        cfg = self.config
   640→   640→        best_val_loss = float("inf")
   641→   641→        best_train_loss = float("inf")
   642→   642→        best_epoch = 0
   643→   643→        epochs_no_improve = 0
   644→   644→        train_loss_history: list[float] = []
   645→   645→        val_loss_history: list[float] = []
   646→   646→        lr_history: list[float] = []
   647→   647→
   648→   648→        self._best_state = None
   649→   649→
   650→   650→        for epoch in range(1, cfg.epochs + 1):
   651→   651→            # --- Train ---
   652→   652→            self.model.train()
   653→   653→            epoch_loss = 0.0
   654→   654→            n_batches = 0
   655→   655→
   656→   656→            for batch_xh, batch_xa, batch_y in train_loader:
   657→   657→                batch_xh = batch_xh.to(self.device)
   658→   658→                batch_xa = batch_xa.to(self.device)
   659→   659→                batch_y = batch_y.to(self.device)
   660→   660→
   661→   661→                optimizer.zero_grad()
   662→   662→                logits = self.model(batch_xh, batch_xa)  # (B, 3)
   663→   663→                loss = criterion(logits, batch_y)
   664→   664→
   665→   665→                loss.backward()
   666→   666→                torch.nn.utils.clip_grad_norm_(
   667→   667→                    self.model.parameters(), max_norm=cfg.grad_clip_max_norm
   668→   668→                )
   669→   669→                optimizer.step()
   670→   670→
   671→   671→                epoch_loss += loss.item()
   672→   672→                n_batches += 1
   673→   673→
   674→   674→            avg_train_loss = epoch_loss / max(n_batches, 1)
   675→   675→            train_loss_history.append(avg_train_loss)
   676→   676→            lr_history.append(optimizer.param_groups[0]["lr"])
   677→   677→
   678→   678→            if avg_train_loss < best_train_loss:
   679→   679→                best_train_loss = avg_train_loss
   680→   680→
   681→   681→            # --- Validate ---
   682→   682→            if has_val:
   683→   683→                val_loss = self._eval_loss(val_loader, criterion)
   684→   684→                val_loss_history.append(val_loss)
   685→   685→                scheduler.step(val_loss)
   686→   686→
   687→   687→                if val_loss < best_val_loss:
   688→   688→                    best_val_loss = val_loss
   689→   689→                    best_epoch = epoch
   690→   690→                    epochs_no_improve = 0
   691→   691→                    # Deepcopy best model state
   692→   692→                    self._best_state = copy.deepcopy(self.model.state_dict())
   693→   693→                else:
   694→   694→                    epochs_no_improve += 1
   695→   695→
   696→   696→                if epoch % 10 == 0 or epoch == 1:
   697→   697→                    logger.info(
   698→   698→                        "Epoch %3d/%d  train_loss=%.4f  val_loss=%.4f  lr=%.2e",
   699→   699→                        epoch,
   700→   700→                        cfg.epochs,
   701→   701→                        avg_train_loss,
   702→   702→                        val_loss,
   703→   703→                        optimizer.param_groups[0]["lr"],
   704→   704→                    )
   705→   705→
   706→   706→                # Early stopping
   707→   707→                if (
   708→   708→                    cfg.early_stopping_patience > 0
   709→   709→                    and epochs_no_improve >= cfg.early_stopping_patience
   710→   710→                ):
   711→   711→                    logger.info(
   712→   712→                        "Early stopping at epoch %d (no improvement for %d epochs)",
   713→   713→                        epoch,
   714→   714→                        epochs_no_improve,
   715→   715→                    )
   716→   716→                    break
   717→   717→            else:
   718→   718→                # No validation: track training loss only
   719→   719→                scheduler.step(avg_train_loss)
   720→   720→                if avg_train_loss < best_val_loss:
   721→   721→                    best_val_loss = avg_train_loss
   722→   722→                    best_epoch = epoch
   723→   723→                    epochs_no_improve = 0
   724→   724→                    self._best_state = copy.deepcopy(self.model.state_dict())
   725→   725→                else:
   726→   726→                    epochs_no_improve += 1
   727→   727→
   728→   728→                if epoch % 10 == 0 or epoch == 1:
   729→   729→                    logger.info(
   730→   730→                        "Epoch %3d/%d  train_loss=%.4f  lr=%.2e",
   731→   731→                        epoch,
   732→   732→                        cfg.epochs,
   733→   733→                        avg_train_loss,
   734→   734→                        optimizer.param_groups[0]["lr"],
   735→   735→                    )
   736→   736→
   737→   737→                if (
   738→   738→                    cfg.early_stopping_patience > 0
   739→   739→                    and epochs_no_improve >= cfg.early_stopping_patience
   740→   740→                ):
   741→   741→                    logger.info(
   742→   742→                        "Early stopping at epoch %d (no improvement for %d epochs)",
   743→   743→                        epoch,
   744→   744→                        epochs_no_improve,
   745→   745→                    )
   746→   746→                    break
   747→   747→
   748→   748→        # --- Restore best weights ---
   749→   749→        if self._best_state is not None:
   750→   750→            self.model.load_state_dict(self._best_state)
   751→   751→            logger.info("Restored best model weights from epoch %d", best_epoch)
   752→   752→
   753→   753→        self.model.eval()
   754→   754→        self.is_trained = True
   755→   755→
   756→   756→        metrics: dict[str, Any] = {
   757→   757→            "epochs_run": epoch,
   758→   758→            "best_epoch": best_epoch,
   759→   759→            "train_loss_best": best_train_loss,
   760→   760→            "train_loss_final": train_loss_history[-1] if train_loss_history else None,
   761→   761→            "n_samples_train": int(len(y)),
   762→   762→            "n_samples_val": int(len(y_val)) if has_val else 0,
   763→   763→            "device": str(self.device),
   764→   764→            "loss_history": train_loss_history,
   765→   765→            "lr_history": lr_history,
   766→   766→        }
   767→   767→
   768→   768→        if has_val and val_loss_history:
   769→   769→            metrics["val_loss_best"] = best_val_loss
   770→   770→            metrics["val_loss_final"] = val_loss_history[-1]
   771→   771→
   772→   772→        logger.info(
   773→   773→            "LSTM training complete – epochs=%d, best_epoch=%d, best_train_loss=%.4f%s",
   774→   774→            metrics["epochs_run"],
   775→   775→            best_epoch,
   776→   776→            best_train_loss,
   777→   777→            f", best_val_loss={best_val_loss:.4f}" if has_val else "",
   778→   778→        )
   779→   779→
   780→   780→        return metrics
   781→   781→
   782→   782→    # ------------------------------------------------------------------
   783→   783→    # Validation loss helper
   784→   784→    # ------------------------------------------------------------------
   785→   785→
   786→   786→    def _eval_loss(
   787→   787→        self,
   788→   788→        loader: torch.utils.data.DataLoader,
   789→   789→        criterion: nn.Module,
   790→   790→    ) -> float:
   791→   791→        """Compute average loss over a data loader (no gradient)."""
   792→   792→        self.model.eval()
   793→   793→        total_loss = 0.0
   794→   794→        n_batches = 0
   795→   795→        with torch.no_grad():
   796→   796→            for batch_xh, batch_xa, batch_y in loader:
   797→   797→                batch_xh = batch_xh.to(self.device)
   798→   798→                batch_xa = batch_xa.to(self.device)
   799→   799→                batch_y = batch_y.to(self.device)
   800→   800→                logits = self.model(batch_xh, batch_xa)
   801→   801→                loss = criterion(logits, batch_y)
   802→   802→                total_loss += loss.item()
   803→   803→                n_batches += 1
   804→   804→        return total_loss / max(n_batches, 1)

   806→   806→    # ==================================================================
   807→   807→    # Prediction
   808→   808→    # ==================================================================
   809→   809→
   810→   810→    def predict(
   811→   811→        self,
   812→   812→        X_home: np.ndarray,
   813→   813→        X_away: np.ndarray,
   814→   814→    ) -> dict[str, Any]:
   815→   815→        """Predict match outcome probabilities.
   816→   816→
   817→   817→        Parameters
   818→   818→        ----------
   819→   819→        X_home : np.ndarray, shape (N, seq_len, input_size) or (seq_len, input_size)
   820→   820→        X_away : np.ndarray, same shape as X_home
   821→   821→
   822→   822→        Returns
   823→   823→        -------
   824→   824→        dict
   825→   825→            For a **single** sample: ``{home_win, draw, away_win,
   826→   826→            predicted_class, confidence}``.  Probabilities are softmax-
   827→   827→            normalised and sum to exactly 1.0.
   828→   828→        """
   829→   829→        if not self.is_trained or self.model is None:
   830→   830→            raise RuntimeError("Model has not been trained or loaded yet.")
   831→   831→
   832→   832→        X_home = np.asarray(X_home, dtype=np.float32)
   833→   833→        X_away = np.asarray(X_away, dtype=np.float32)
   834→   834→
   835→   835→        # Ensure 3-D
   836→   836→        if X_home.ndim == 2:
   837→   837→            X_home = X_home[np.newaxis, ...]
   838→   838→        if X_away.ndim == 2:
   839→   839→            X_away = X_away[np.newaxis, ...]
   840→   840→
   841→   841→        dataset = _SingleMatchDataset(X_home, X_away)
   842→   842→        loader = torch.utils.data.DataLoader(
   843→   843→            dataset, batch_size=len(dataset), shuffle=False
   844→   844→        )
   845→   845→
   846→   846→        self.model.eval()
   847→   847→        all_probs: list[np.ndarray] = []
   848→   848→
   849→   849→        with torch.no_grad():
   850→   850→            for batch_xh, batch_xa in loader:
   851→   851→                batch_xh = batch_xh.to(self.device)
   852→   852→                batch_xa = batch_xa.to(self.device)
   853→   853→                logits = self.model(batch_xh, batch_xa)  # (B, 3)
   854→   854→                probs = torch.softmax(logits, dim=1).cpu().numpy()
   855→   855→                all_probs.append(probs)
   856→   856→
   857→   857→        probs_matrix = np.concatenate(all_probs, axis=0)  # (N, 3)
   858→   858→
   859→   859→        # For the common single-sample case, return a flat dict
   860→   860→        if probs_matrix.shape[0] == 1:
   861→   861→            p = probs_matrix[0]
   862→   862→            # Normalise to exactly 1.0
   863→   863→            p = p / p.sum()
   864→   864→            pred_idx = int(np.argmax(p))
   865→   865→            return {
   866→   866→                "home_win": float(p[0]),
   867→   867→                "draw": float(p[1]),
   868→   868→                "away_win": float(p[2]),
   869→   869→                "predicted_class": _CLASS_IDX_TO_LABEL[pred_idx],
   870→   870→                "confidence": float(p[pred_idx]),
   871→   871→            }
   872→   872→
   873→   873→        # Batch case: return list of dicts
   874→   874→        results = []
   875→   875→        for i in range(probs_matrix.shape[0]):
   876→   876→            p = probs_matrix[i]
   877→   877→            p = p / p.sum()
   878→   878→            pred_idx = int(np.argmax(p))
   879→   879→            results.append(
   880→   880→                {
   881→   881→                    "home_win": float(p[0]),
   882→   882→                    "draw": float(p[1]),
   883→   883→                    "away_win": float(p[2]),
   884→   884→                    "predicted_class": _CLASS_IDX_TO_LABEL[pred_idx],
   885→   885→                    "confidence": float(p[pred_idx]),
   886→   886→                }
   887→   887→            )
   888→   888→        return {"batch": results}
   889→   889→
   890→   890→    # ==================================================================
   891→   891→    # Evaluation
   892→   892→    # ==================================================================
   893→   893→
   894→   894→    def evaluate(
   895→   895→        self,
   896→   896→        X_home: np.ndarray,
   897→   897→        X_away: np.ndarray,
   898→   898→        y: np.ndarray,
   899→   899→    ) -> dict[str, Any]:
   900→   900→        """Evaluate the model on a labelled dataset.
   901→   901→
   902→   902→        Computes: accuracy, per-class precision / recall / F1, macro
   903→   903→        averages, Brier score, log-loss, and confusion matrix.
   904→   904→
   905→   905→        Parameters
   906→   906→        ----------
   907→   907→        X_home, X_away : np.ndarray, shape (N, seq_len, input_size)
   908→   908→        y : np.ndarray, shape (N,), dtype int
   909→   909→
   910→   910→        Returns
   911→   911→        -------
   912→   912→        dict
   913→   913→            Comprehensive evaluation metrics.
   914→   914→        """
   915→   915→        if not self.is_trained or self.model is None:
   916→   916→            raise RuntimeError("Model has not been trained or loaded yet.")
   917→   917→
   918→   918→        X_home = np.asarray(X_home, dtype=np.float32)
   919→   919→        X_away = np.asarray(X_away, dtype=np.float32)
   920→   920→        y = np.asarray(y, dtype=np.int64)
   921→   921→
   922→   922→        if X_home.ndim == 2:
   923→   923→            X_home = X_home[np.newaxis, ...]
   924→   924→        if X_away.ndim == 2:
   925→   925→            X_away = X_away[np.newaxis, ...]
   926→   926→
   927→   927→        dataset = _MatchSequenceDataset(X_home, X_away, y)
   928→   928→        loader = torch.utils.data.DataLoader(
   929→   929→            dataset, batch_size=self.config.batch_size, shuffle=False
   930→   930→        )
   931→   931→
   932→   932→        self.model.eval()
   933→   933→        all_probs: list[np.ndarray] = []
   934→   934→        all_labels: list[int] = []
   935→   935→
   936→   936→        with torch.no_grad():
   937→   937→            for batch_xh, batch_xa, batch_y in loader:
   938→   938→                batch_xh = batch_xh.to(self.device)
   939→   939→                batch_xa = batch_xa.to(self.device)
   940→   940→                logits = self.model(batch_xh, batch_xa)
   941→   941→                probs = torch.softmax(logits, dim=1).cpu().numpy()
   942→   942→                all_probs.append(probs)
   943→   943→                all_labels.extend(batch_y.numpy().tolist())
   944→   944→
   945→   945→        proba = np.concatenate(all_probs, axis=0)  # (N, 3)
   946→   946→        y_arr = np.array(all_labels, dtype=np.int64)
   947→   947→        preds = np.argmax(proba, axis=1)
   948→   948→
   949→   949→        n = len(y_arr)
   950→   950→
   951→   951→        # --- Overall accuracy ---
   952→   952→        correct = int(np.sum(preds == y_arr))
   953→   953→        accuracy = correct / n
   954→   954→
   955→   955→        # --- Log loss ---
   956→   956→        eps = 1e-15
   957→   957→        clipped = np.clip(proba, eps, 1.0 - eps)
   958→   958→        eye = np.eye(_N_CLASSES)[y_arr]
   959→   959→        log_loss_val = float(
   960→   960→            -np.mean(np.sum(eye * np.log(clipped), axis=1))
   961→   961→        )
   962→   962→
   963→   963→        # --- Brier score (multi-class) ---
   964→   964→        brier = float(np.mean(np.sum((proba - eye) ** 2, axis=1)))
   965→   965→
   966→   966→        # --- Per-class metrics ---
   967→   967→        class_metrics: dict[str, dict[str, float]] = {}
   968→   968→        for cls_idx, cls_label in _CLASS_IDX_TO_LABEL.items():
   969→   969→            tp = int(np.sum((preds == cls_idx) & (y_arr == cls_idx)))
   970→   970→            fp = int(np.sum((preds == cls_idx) & (y_arr != cls_idx)))
   971→   971→            fn = int(np.sum((preds != cls_idx) & (y_arr == cls_idx)))
   972→   972→
   973→   973→            precision = tp / (tp + fp) if (tp + fp) > 0 else 0.0
   974→   974→            recall = tp / (tp + fn) if (tp + fn) > 0 else 0.0
   975→   975→            f1 = (
   976→   976→                2 * precision * recall / (precision + recall)
   977→   977→                if (precision + recall) > 0
   978→   978→                else 0.0
   979→   979→            )
   980→   980→
   981→   981→            class_metrics[cls_label] = {
   982→   982→                "precision": precision,
   983→   983→                "recall": recall,
   984→   984→                "f1": f1,
   985→   985→                "support": int(np.sum(y_arr == cls_idx)),
   986→   986→            }
   987→   987→
   988→   988→        # Macro averages
   989→   989→        macro_precision = float(
   990→   990→            np.mean([m["precision"] for m in class_metrics.values()])
   991→   991→        )
   992→   992→        macro_recall = float(
   993→   993→            np.mean([m["recall"] for m in class_metrics.values()])
   994→   994→        )
   995→   995→        macro_f1 = float(
   996→   996→            np.mean([m["f1"] for m in class_metrics.values()])
   997→   997→        )
   998→   998→
   999→   999→        # --- Confusion matrix ---
  1000→  1000→        cm = np.zeros((_N_CLASSES, _N_CLASSES), dtype=int)
  1001→  1001→        for true, pred in zip(y_arr, preds):
  1002→  1002→            cm[int(true)][int(pred)] += 1
  1003→  1003→
  1004→  1004→        # --- Calibration: expected calibration error (ECE) ---
  1005→  1005→        ece = self._compute_ece(proba, y_arr, n_bins=10)
  1006→  1006→
  1007→  1007→        return {
  1008→  1008→            "n_samples": n,
  1009→  1009→            "accuracy": accuracy,
  1010→  1010→            "log_loss": log_loss_val,
  1011→  1011→            "brier_score": brier,
  1012→  1012→            "ece": ece,
  1013→  1013→            "macro_precision": macro_precision,
  1014→  1014→            "macro_recall": macro_recall,
  1015→  1015→            "macro_f1": macro_f1,
  1016→  1016→            "per_class": class_metrics,
  1017→  1017→            "confusion_matrix": cm.tolist(),
  1018→  1018→        }
  1019→  1019→
  1020→  1020→    # ------------------------------------------------------------------
  1021→  1021→    # Expected Calibration Error
  1022→  1022→    # ------------------------------------------------------------------
  1023→  1023→
  1024→  1024→    @staticmethod
  1025→  1025→    def _compute_ece(
  1026→  1026→        proba: np.ndarray, y: np.ndarray, n_bins: int = 10
  1027→  1027→    ) -> float:
  1028→  1028→        """Compute Expected Calibration Error.
  1029→  1029→
  1030→  1030→        Bins predictions by confidence and measures the gap between
  1031→  1031→        average confidence and average accuracy in each bin.
  1032→  1032→        """
  1033→  1033→        confidences = np.max(proba, axis=1)
  1034→  1034→        predictions = np.argmax(proba, axis=1)
  1035→  1035→        accuracies = (predictions == y).astype(np.float32)
  1036→  1036→
  1037→  1037→        bin_boundaries = np.linspace(0.0, 1.0, n_bins + 1)
  1038→  1038→        ece = 0.0
  1039→  1039→        for i in range(n_bins):
  1040→  1040→            lo, hi = bin_boundaries[i], bin_boundaries[i + 1]
  1041→  1041→            if i == n_bins - 1:
  1042→  1042→                mask = (confidences >= lo) & (confidences <= hi)
  1043→  1043→            else:
  1044→  1044→                mask = (confidences >= lo) & (confidences < hi)
  1045→  1045→            if mask.sum() == 0:
  1046→  1046→                continue
  1047→  1047→            avg_conf = confidences[mask].mean()
  1048→  1048→            avg_acc = accuracies[mask].mean()
  1049→  1049→            ece += (mask.sum() / len(y)) * abs(avg_conf - avg_acc)
  1050→  1050→        return float(ece)
  1051→  1051→

  1101→  1101→    def load(self, name: str = "lstm_v1") -> bool:
  1102→  1102→        """Load a previously saved model.
  1103→  1103→
  1104→  1104→        Parameters
  1105→  1105→        ----------
  1106→  1106→        name : str
  1107→  1107→            Model identifier used in :meth:`save`.
  1108→  1108→
  1109→  1109→        Returns
  1110→  1110→        -------
  1111→  1111→        bool
  1112→  1112→            ``True`` if loading succeeded, ``False`` otherwise.
  1113→  1113→        """
  1114→  1114→        model_path = self.model_dir / f"{name}.pt"
  1115→  1115→        meta_path = self.model_dir / f"{name}_meta.joblib"
  1116→  1116→
  1117→  1117→        if not model_path.exists():
  1118→  1118→            logger.warning("Model file not found: %s", model_path)
  1119→  1119→            return False
  1120→  1120→
  1121→  1121→        try:
  1122→  1122→            # Load metadata first to reconstruct config
  1123→  1123→            if meta_path.exists():
  1124→  1124→                meta: dict[str, Any] = joblib.load(meta_path)
  1125→  1125→                loaded_config = meta.get("config", {})
  1126→  1126→                if loaded_config:
  1127→  1127→                    self.config = LSTMConfig(**loaded_config)
  1128→  1128→                self.feature_cols = meta.get("feature_cols", [])
  1129→  1129→
  1130→  1130→                # Schema version check — TEMPORAL SEQUENCE CONTRACT
  1131→  1131→                from src.ml.feature_schema import (
  1132→  1132→                    SCHEMA_VERSION as CURRENT_SCHEMA_VERSION,
  1133→  1133→                    SCHEMA_HASH as CURRENT_SCHEMA_HASH,
  1134→  1134→                    LSTM_SEQUENCE_FEATURES as CURRENT_LSTM_FEATS,
  1135→  1135→                )

  1137→  1137→                saved_version = meta.get("schema_version")
  1138→  1138→                saved_hash = meta.get("schema_hash")
  1139→  1139→                saved_lstm_feats = meta.get("lstm_sequence_features", [])
  1140→  1140→
  1141→  1141→                if saved_hash and saved_hash != CURRENT_SCHEMA_HASH:
  1142→  1142→                    logger.warning(
  1143→  1143→                        "LSTM model schema hash mismatch: model=%s, current=%s. "
  1144→  1144→                        "The model was trained with a different feature schema.",
  1145→  1145→                        saved_hash,
  1146→  1146→                        CURRENT_SCHEMA_HASH,
  1147→  1147→                    )
  1148→  1148→
  1149→  1149→                if saved_lstm_feats and list(saved_lstm_feats) != list(CURRENT_LSTM_FEATS):
  1150→  1150→                    logger.error(
  1151→  1151→                        "LSTM SEQUENCE CONTRACT VIOLATION: model trained with features %s, "
  1152→  1152→                        "current schema expects %s. Refusing to load — retrain the model.",
  1153→  1153→                        saved_lstm_feats,
  1154→  1154→                        CURRENT_LSTM_FEATS,
  1155→  1155→                    )
  1156→  1156→                    return False
  1157→  1157→
  1158→  1158→                if saved_version:
  1159→  1159→                    logger.info(
  1160→  1160→                        "LSTM model trained with schema version %s (current: %s)",
  1161→  1161→                        saved_version,
  1162→  1162→                        CURRENT_SCHEMA_VERSION,
  1163→  1163→                    )
  1164→  1164→
  1165→  1165→            # Rebuild model architecture and load weights
  1166→  1166→            self.model = FootballLSTM(self.config).to(self.device)
  1167→  1167→            self.model.load_state_dict(
  1168→  1168→                torch.load(str(model_path), map_location=self.device, weights_only=True)
  1169→  1169→            )
  1170→  1170→            self.model.eval()
  1171→  1171→            self.is_trained = True
  1172→  1172→
  1173→  1173→            logger.info("LSTM model loaded from %s", model_path)
  1174→  1174→            return True
  1175→  1175→
  1176→  1176→        except Exception as exc:
  1177→  1177→            logger.error("Failed to load LSTM model: %s", exc)
  1178→  1178→            return False
