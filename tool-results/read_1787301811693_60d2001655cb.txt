   138→# Super-Ensemble Engine
   139→# ──────────────────────────────────────────────────────────────────────────────

   141→
   142→class SuperEnsemble:
   143→    """Combines all prediction models into a calibrated super-ensemble.
   144→
   145→    Weighted average with:
   146→    1. Per-model calibration factors (based on recent accuracy)
   147→    2. Market signal integration (odds movement detection)
   148→    3. Kelly Criterion for bankroll management
   149→    4. Auto-weight optimisation when enough historical data
   150→
   151→    Falls back gracefully when ML models aren't trained.
   152→    """
   153→
   154→    def __init__(self, config: EnsembleConfig | None = None) -> None:
   155→        self.config = config or EnsembleConfig()
   156→        self.model_dir = Path(self.config.model_dir)
   157→
   158→        # Lazy-loaded ML models
   159→        self._xgboost_engine = None
   160→        self._lstm_engine = None
   161→        self._xgboost_loaded = False
   162→        self._lstm_loaded = False
   163→
   164→    # ==================================================================
   165→    # Public API
   166→    # ==================================================================
   167→
   168→    def predict(self, match_data: dict) -> SuperEnsembleResult:
   169→        """Run full super-ensemble prediction.
   170→
   171→        match_data should contain:
   172→        - home_team, away_team (names)
   173→        - home_elo, away_elo
   174→        - home_avg_goals, away_avg_goals
   175→        - home_avg_conceded, away_avg_conceded
   176→        - home_xg, away_xg (optional, defaults to avg_goals)
   177→        - odds_home, odds_draw, odds_away (optional, for market analysis)
   178→        - home_form, away_form (list of recent results, optional)
   179→        - league (optional)
   180→        - match_id (optional)
   181→
   182→        Returns
   183→        -------
   184→        SuperEnsembleResult
   185→            Comprehensive prediction result with probabilities,
   186→            markets, Kelly analysis, and signals.
   187→        """
   188→        match_id = match_data.get("match_id")
   189→        home_team = match_data.get("home_team", "Home")
   190→        away_team = match_data.get("away_team", "Away")
   191→
   192→        home_elo = float(match_data.get("home_elo", 1500.0))
   193→        away_elo = float(match_data.get("away_elo", 1500.0))
   194→        home_avg_goals = float(match_data.get("home_avg_goals", 1.3))
   195→        away_avg_goals = float(match_data.get("away_avg_goals", 1.1))
   196→        home_avg_conceded = float(match_data.get("home_avg_conceded", 1.0))
   197→        away_avg_conceded = float(match_data.get("away_avg_conceded", 1.2))
   198→        home_xg = float(match_data.get("home_xg", home_avg_goals))
   199→        away_xg = float(match_data.get("away_xg", away_avg_goals))
   200→        odds_home = match_data.get("odds_home")
   201→        odds_draw = match_data.get("odds_draw")
   202→        odds_away = match_data.get("odds_away")
   203→
   204→        # Compute expected goals for Poisson-based models
   205→        lambda_h = (home_avg_goals + away_avg_conceded) / 2.0
   206→        lambda_a = (away_avg_goals + home_avg_conceded) / 2.0
   207→
   208→        per_model: dict[str, dict] = {}
   209→        available_models: list[str] = []
   210→
   211→        # ── 1. ELO Model ────────────────────────────────────────────────
   212→        try:
   213→            elo_probs = compute_elo_prob(home_elo, away_elo)
   214→            per_model["elo"] = {
   215→                "home_win": elo_probs["home_win"],
   216→                "draw": elo_probs["draw"],
   217→                "away_win": elo_probs["away_win"],
   218→            }
   219→            available_models.append("elo")
   220→        except Exception as exc:
   221→            logger.warning("ELO model failed: %s", exc)
   222→
   223→        # ── 2. Poisson Model ────────────────────────────────────────────
   224→        try:
   225→            poisson_probs = compute_poisson_prob(lambda_h, lambda_a)
   226→            per_model["poisson"] = {
   227→                "home_win": poisson_probs["home_win"],
   228→                "draw": poisson_probs["draw"],
   229→                "away_win": poisson_probs["away_win"],
   230→            }
   231→            available_models.append("poisson")
   232→        except Exception as exc:
   233→            logger.warning("Poisson model failed: %s", exc)
   234→
   235→        # ── 3. Dixon-Coles Model ────────────────────────────────────────
   236→        try:
   237→            dc_probs = compute_dixon_coles_prob(lambda_h, lambda_a)
   238→            per_model["dixon_coles"] = {
   239→                "home_win": dc_probs["home_win"],
   240→                "draw": dc_probs["draw"],
   241→                "away_win": dc_probs["away_win"],
   242→            }
   243→            available_models.append("dixon_coles")
   244→        except Exception as exc:
   245→            logger.warning("Dixon-Coles model failed: %s", exc)
   246→
   247→        # ── 4. Stochastic (Merton + GARCH Monte Carlo) ──────────────────
   248→        stochastic_result: StochasticResult | None = None
   249→        try:
   250→            stochastic_result = run_stochastic_simulation(
   251→                home_xg=home_xg,
   252→                away_xg=away_xg,
   253→                home_conceded=home_avg_conceded,
   254→                away_conceded=away_avg_conceded,
   255→                home_elo=home_elo,
   256→                away_elo=away_elo,
   257→                config=EngineConfig(num_simulations=50_000),
   258→            )
   259→            per_model["stochastic"] = {
   260→                "home_win": stochastic_result.match_probabilities["home_victory"],
   261→                "draw": stochastic_result.match_probabilities["draw"],
   262→                "away_win": stochastic_result.match_probabilities["away_victory"],
   263→            }
   264→            available_models.append("stochastic")
   265→        except Exception as exc:
   266→            logger.warning("Stochastic model failed: %s", exc)
   267→
   268→        # ── 5. XGBoost ───────────────────────────────────────────────────
   269→        try:
   270→            xgb_probs = self._run_xgboost(match_data)
   271→            if xgb_probs:
   272→                per_model["xgboost"] = xgb_probs
   273→                available_models.append("xgboost")
   274→        except Exception as exc:
   275→            logger.debug("XGBoost not available: %s", exc)
   276→
   277→        # ── 6. LSTM ─────────────────────────────────────────────────────
   278→        try:
   279→            lstm_probs = self._run_lstm(match_data)
   280→            if lstm_probs:
   281→                per_model["lstm"] = lstm_probs
   282→                available_models.append("lstm")
   283→        except Exception as exc:
   284→            logger.debug("LSTM not available: %s", exc)
   285→
   286→        # ── 7. Weighted Ensemble ────────────────────────────────────────
   287→        weights = self._get_model_weights(available_models)
   288→        final_probs = self._compute_weighted_average(per_model, weights)
   289→
   290→        # ── 8. Market Signals ───────────────────────────────────────────
   291→        market_signals: list[dict] = []
   292→        if odds_home and odds_draw and odds_away:
   293→            try:
   294→                # Use current odds as both opening and current when we
   295→                # don't have opening odds — signals will be empty, but
   296→                # we can still compute Kelly
   297→                opening_odds = match_data.get("opening_odds_home")
   298→                if opening_odds and match_data.get("opening_odds_draw"):
   299→                    opening = {
   300→                        "home": float(match_data["opening_odds_home"]),
   301→                        "draw": float(match_data["opening_odds_draw"]),
   302→                        "away": float(match_data["opening_odds_away"]),
   303→                    }
   304→                    current = {
   305→                        "home": float(odds_home),
   306→                        "draw": float(odds_draw),
   307→                        "away": float(odds_away),
   308→                    }
   309→                    detected = analyze_market_signals(opening, current)
   310→                    for sig in detected:
   311→                        market_signals.append(sig.model_dump())
   312→            except Exception as exc:
   313→                logger.debug("Market signal analysis failed: %s", exc)
   314→
   315→        # ── 9. Kelly Criterion ──────────────────────────────────────────
   316→        kelly_results: dict[str, dict] = {}
   317→        if odds_home and odds_draw and odds_away:
   318→            kelly_results["home_win"] = calculate_kelly(
   319→                final_probs["home_win"],
   320→                float(odds_home),
   321→                bankroll=self.config.bankroll,
   322→                fraction=self.config.kelly_fraction,
   323→            ).model_dump()
   324→            kelly_results["draw"] = calculate_kelly(
   325→                final_probs["draw"],
   326→                float(odds_draw),
   327→                bankroll=self.config.bankroll,
   328→                fraction=self.config.kelly_fraction,
   329→            ).model_dump()
   330→            kelly_results["away_win"] = calculate_kelly(
   331→                final_probs["away_win"],
   332→                float(odds_away),
   333→                bankroll=self.config.bankroll,
   334→                fraction=self.config.kelly_fraction,
   335→            ).model_dump()
   336→
   337→        # ── 10. Signals & Risks ────────────────────────────────────────
   338→        signals, risks = self._generate_signals_and_risks(
   339→            match_data, final_probs
   340→        )
   341→
   342→        # Add market signal descriptions to signals list
   343→        for sig in market_signals:
   344→            signals.append(f"[{sig.get('signal_type', '').upper()}] {sig.get('description', '')}")
   345→
   346→        # ── 11. Markets (derived) ──────────────────────────────────────
   347→        markets: dict = {}
   348→        if stochastic_result:
   349→            markets["over_25"] = stochastic_result.totals_market.get("over25", 0.0)
   350→            markets["over_15"] = stochastic_result.totals_market.get("over15", 0.0)
   351→            markets["over_35"] = stochastic_result.totals_market.get("over35", 0.0)
   352→            markets["btts"] = stochastic_result.both_teams_to_score
   353→            markets["asian_handicap"] = stochastic_result.asian_handicap
   354→        else:
   355→            # Fallback: estimate from Poisson
   356→            from scipy.stats import poisson as poisson_dist
   357→            over_25 = 0.0
   358→            btts = 0.0
   359→            for i in range(11):
   360→                for j in range(11):
   361→                    p = poisson_dist.pmf(i, lambda_h) * poisson_dist.pmf(j, lambda_a)
   362→                    if i + j > 2:
   363→                        over_25 += p
   364→                    if i > 0 and j > 0:
   365→                        btts += p
   366→            markets["over_25"] = round(over_25, 4)
   367→            markets["over_15"] = 0.0
   368→            markets["over_35"] = 0.0
   369→            markets["btts"] = round(btts, 4)
   370→            markets["asian_handicap"] = {
   371→                "0": round(final_probs["home_win"], 4),
   372→                "-0.5": round(final_probs["home_win"], 4),
   373→                "-1.0": 0.0,
   374→                "-1.5": 0.0,
   375→            }
   376→
   377→        # ── 12. Recommended Bet ────────────────────────────────────────
   378→        recommended_bet = None
   379→        if odds_home and odds_draw and odds_away:
   380→            best_edge = -1.0
   381→            for outcome_key, odds_val in [
   382→                ("home_win", odds_home),
   383→                ("draw", odds_draw),
   384→                ("away_win", odds_away),
   385→            ]:
   386→                kr = kelly_results.get(outcome_key, {})
   387→                edge = kr.get("edge", 0.0)
   388→                kelly_frac = kr.get("optimal_fraction", 0.0)
   389→                if edge > best_edge and kelly_frac > 0.001:
   390→                    best_edge = edge
   391→                    recommended_bet = {
   392→                        "outcome": outcome_key,
   393→                        "edge": round(edge, 4),
   394→                        "kelly_fraction": round(kelly_frac, 6),
   395→                        "odds": float(odds_val),
   396→                        "recommended_stake": kr.get("recommended_stake", 0.0),
   397→                    }
   398→
   399→        # ── 13. Expected Goals ──────────────────────────────────────────
   400→        if stochastic_result:
   401→            expected_goals = {
   402→                "home": stochastic_result.expected_means["home"],
   403→                "away": stochastic_result.expected_means["away"],
   404→                "total": stochastic_result.expected_means["total"],
   405→            }
   406→            volatility_index = stochastic_result.volatility_index
   407→        else:
   408→            expected_goals = {
   409→                "home": round(lambda_h, 3),
   410→                "away": round(lambda_a, 3),
   411→                "total": round(lambda_h + lambda_a, 3),
   412→            }
   413→            volatility_index = 0.0
   414→
   415→        # ── 14. Confidence ──────────────────────────────────────────────
   416→        confidence = self._compute_confidence(final_probs)
   417→
   418→        return SuperEnsembleResult(
   419→            match_id=match_id,
   420→            home_team=home_team,
   421→            away_team=away_team,
   422→            probabilities=final_probs,
   423→            per_model=per_model,
   424→            expected_goals=expected_goals,
   425→            markets=markets,
   426→            kelly=kelly_results,
   427→            market_signals=market_signals,
   428→            confidence=confidence,
   429→            signals=signals,
   430→            risks=risks,
   431→            recommended_bet=recommended_bet,
   432→            volatility_index=volatility_index,
   433→        )
   434→

   437→
   438→        Parameters
   439→        ----------

   460→                # Insert a placeholder with error info
   461→                results.append(
   462→                    SuperEnsembleResult(
   463→                        match_id=match.get("match_id"),
   464→                        home_team=match.get("home_team"),
   465→                        away_team=match.get("away_team"),
   466→                        probabilities={"home_win": 1/3, "draw": 1/3, "away_win": 1/3},
   467→                        per_model={},
   468→                        expected_goals={"home": 0, "away": 0, "total": 0},

   476→        return results

   478→    # ==================================================================
   479→    # Weight Management
   480→    # ==================================================================
   481→
   482→    def _get_model_weights(self, available_models: list[str]) -> dict[str, float]:
   483→        """Get normalized weights for available models only.
   484→
   485→        If some models are unavailable, their weight is redistributed
   486→        proportionally among the remaining models.
   487→
   488→        Parameters
   489→        ----------
   490→        available_models : list[str]
   491→            List of model names that produced valid results.
   492→
   493→        Returns
   494→        -------
   495→        dict[str, float]
   496→            Normalized weights summing to 1.0, keyed by model name.
   497→        """
   498→        base_weights = self.config.weights
   499→
   500→        # Filter to available models
   501→        filtered = {
   502→            name: base_weights[name]
   503→            for name in available_models
   504→            if name in base_weights
   505→        }
   506→
   507→        if not filtered:
   508→            # Fallback: equal weight for all available
   509→            n = len(available_models)
   510→            return {name: 1.0 / n for name in available_models} if n > 0 else {}
   511→
   512→        # Normalize to sum to 1.0
   513→        total = sum(filtered.values())
   514→        return {name: w / total for name, w in filtered.items()}
   515→
   516→    def _compute_weighted_average(
   517→        self,
   518→        probs: dict[str, dict],
   519→        weights: dict[str, float],
   520→    ) -> dict[str, float]:
   521→        """Weighted average of model probabilities.
   522→
   523→        Parameters
   524→        ----------
   525→        probs : dict[str, dict]
   526→            Per-model probability dicts: {"elo": {"home_win": 0.5, ...}, ...}
   527→        weights : dict[str, float]
   528→            Normalized weights per model.
   529→
   530→        Returns
   531→        -------
   532→        dict[str, float]
   533→            Combined {"home_win": ..., "draw": ..., "away_win": ...}.
   534→        """
   535→        if not probs or not weights:
   536→            return {"home_win": 1/3, "draw": 1/3, "away_win": 1/3}
   537→
   538→        combined = {"home_win": 0.0, "draw": 0.0, "away_win": 0.0}
   539→
   540→        for model_name, model_probs in probs.items():
   541→            w = weights.get(model_name, 0.0)
   542→            if w <= 0:
   543→                continue
   544→            for key in ("home_win", "draw", "away_win"):
   545→                combined[key] += w * model_probs.get(key, 0.0)
   546→
   547→        # Normalize to exactly 1.0
   548→        total = sum(combined.values())
   549→        if total > 0:
   550→            combined = {k: round(v / total, 6) for k, v in combined.items()}
   551→
   552→        return combined
   553→
   554→    # ==================================================================
   555→    # Confidence & Signal Generation
   556→    # ==================================================================
   557→
   558→    def _compute_confidence(self, probs: dict) -> str:
   559→        """Classify prediction confidence based on max probability.
   560→
   561→        - high: max probability >= 0.55
   562→        - medium: max probability >= 0.42
   563→        - low: otherwise
   564→        """
   565→        max_prob = max(probs.get("home_win", 0), probs.get("draw", 0), probs.get("away_win", 0))
   566→        if max_prob >= 0.55:
   567→            return "high"
   568→        elif max_prob >= 0.42:
   569→            return "medium"
   570→        return "low"
   571→
   572→    def _generate_signals_and_risks(
   573→        self,
   574→        match_data: dict,
   575→        probs: dict,
   576→    ) -> tuple[list[str], list[str]]:
   577→        """Generate human-readable signals and risk factors.
   578→
   579→        Parameters
   580→        ----------
   581→        match_data : dict
   582→            Original match data dict.
   583→        probs : dict
   584→            Final ensemble probabilities.
   585→
   586→        Returns
   587→        -------
   588→        tuple[list[str], list[str]]
   589→            (signals, risks) lists.
   590→        """
   591→        signals: list[str] = []
   592→        risks: list[str] = []
   593→
   594→        home_elo = float(match_data.get("home_elo", 1500))
   595→        away_elo = float(match_data.get("away_elo", 1500))
   596→        elo_diff = home_elo - away_elo
   597→        home_xg = float(match_data.get("home_xg", match_data.get("home_avg_goals", 1.3)))
   598→        away_xg = float(match_data.get("away_xg", match_data.get("away_avg_goals", 1.1)))
   599→        odds_home = match_data.get("odds_home")
   600→
   601→        # ELO signals
   602→        if elo_diff > 150:
   603→            signals.append(f"Strong ELO advantage for {match_data.get('home_team', 'home')} (+{elo_diff:.0f})")
   604→        elif elo_diff < -150:
   605→            signals.append(f"Strong ELO advantage for {match_data.get('away_team', 'away')} (+{abs(elo_diff):.0f})")
   606→        elif abs(elo_diff) > 80:
   607→            team = "home" if elo_diff > 0 else "away"
   608→            signals.append(f"Moderate ELO edge for {team} team")
   609→
   610→        # xG signals
   611→        if home_xg > away_xg * 1.3:
   612→            signals.append(f"Home xG ({home_xg:.2f}) significantly higher than away ({away_xg:.2f})")
   613→        elif away_xg > home_xg * 1.3:
   614→            signals.append(f"Away xG ({away_xg:.2f}) significantly higher than home ({home_xg:.2f})")
   615→
   616→        # Form signals
   617→        home_form = match_data.get("home_form", [])
   618→        away_form = match_data.get("away_form", [])
   619→        if len(home_form) >= 3:
   620→            recent_wins = sum(1 for r in home_form[-3:] if r == "W")
   621→            if recent_wins == 3:
   622→                signals.append(f"{match_data.get('home_team', 'Home')} on a 3-match winning streak")
   623→        if len(away_form) >= 3:
   624→            recent_wins = sum(1 for r in away_form[-3:] if r == "W")
   625→            if recent_wins == 3:
   626→                signals.append(f"{match_data.get('away_team', 'Away')} on a 3-match winning streak")
   627→
   628→        # Model agreement signal
   629→        max_prob = max(probs["home_win"], probs["draw"], probs["away_win"])
   630→        if max_prob >= 0.55:
   631→            signals.append(f"High model agreement ({max_prob*100:.1f}% on most likely outcome)")
   632→
   633→        # Risks
   634→        if odds_home is None:
   635→            risks.append("No market odds available — Kelly criterion and market signals excluded")
   636→
   637→        if odds_home and float(odds_home) < 1.3:
   638→            risks.append("Heavy favorite — low value in outcome market")
   639→
   640→        if abs(elo_diff) < 30:
   641→            risks.append("Teams are very evenly matched on ELO — high draw probability")
   642→
   643→        # Check if any model is missing
   644→        expected_models = {"elo", "poisson", "dixon_coles", "stochastic"}
   645→        # We don't risk-flag missing xgboost/lstm since they're optional
   646→
   647→        # Derive risk from model disagreement
   648→        if probs["draw"] > 0.32:
   649→            risks.append("High draw probability — consider Double Chance market")
   650→
   651→        return signals, risks

   657→    def _run_xgboost(self, match_data: dict) -> dict[str, float] | None:
   658→        """Try to load and run XGBoost model.
   659→
   660→        Returns home_win/draw/away_win probabilities or None if
   661→        the model is not available.
   662→        """
   663→        if not self._xgboost_loaded:
   664→            try:
   665→                from src.ml.xgboost_engine import XGBoostEngine
   666→
   667→                self._xgboost_engine = XGBoostEngine(
   668→                    model_dir=str(self.model_dir),
   669→                )
   670→                loaded = self._xgboost_engine.load("xgboost_v1")
   671→                self._xgboost_loaded = True  # mark as attempted regardless
   672→                if not loaded:
   673→                    logger.info("XGBoost model not found — skipping")
   674→                    return None
   675→            except Exception as exc:
   676→                self._xgboost_loaded = True
   677→                logger.debug("XGBoost load failed: %s", exc)
   678→                return None
   679→
   680→        if self._xgboost_engine is None or not self._xgboost_engine.is_trained:
   681→            return None
   682→
   683→        # Build a feature vector from match_data
   684→        # The XGBoost model expects the same feature order as training
   685→        # We construct a basic feature vector; if the model was trained with
   686→        # the full feature engineer, it will need the full 50-feature vector.
   687→        # For the ensemble, we use a simplified 6-feature approximation.
   688→        from src.ml.feature_schema import FEATURE_NAMES, FEATURE_DEFAULTS
   689→
   690→        # Build feature vector using canonical feature schema
   691→        home_avg_goals = float(match_data.get("home_avg_goals", 1.3))
   692→        away_avg_goals = float(match_data.get("away_avg_goals", 1.1))
   693→        home_avg_conceded = float(match_data.get("home_avg_conceded", 1.0))
   694→        away_avg_conceded = float(match_data.get("away_avg_conceded", 1.2))
   695→        home_elo = float(match_data.get("home_elo", 1500))
   696→        away_elo = float(match_data.get("away_elo", 1500))
   697→
   698→        # Build feature vector using canonical feature schema
   699→        from src.ml.feature_schema import FEATURE_NAMES, FEATURE_DEFAULTS
   700→
   701→        feature_dict = dict(FEATURE_DEFAULTS)  # start with all defaults
   702→
   703→        # Override with available match data (using CANONICAL names)
   704→        feature_dict["elo_home"] = home_elo
   705→        feature_dict["elo_away"] = away_elo
   706→        feature_dict["elo_diff"] = home_elo - away_elo
   707→        feature_dict["elo_home_advantage_adjusted"] = home_elo + 65.0  # standard home advantage
   708→        feature_dict["avg_goals_scored_home"] = home_avg_goals
   709→        feature_dict["avg_goals_scored_away"] = away_avg_goals
   710→        feature_dict["avg_goals_conceded_home"] = home_avg_conceded
   711→        feature_dict["avg_goals_conceded_away"] = away_avg_conceded
   712→        feature_dict["avg_goals_scored"] = home_avg_goals
   713→        feature_dict["avg_goals_conceded"] = home_avg_conceded
   714→        feature_dict["poisson_lambda_home"] = (home_avg_goals + away_avg_conceded) / 2.0
   715→        feature_dict["poisson_lambda_away"] = (away_avg_goals + home_avg_conceded) / 2.0
   716→
   717→        # Handle optional xG data
   718→        home_xg = match_data.get("home_xg")
   719→        away_xg = match_data.get("away_xg")
   720→        if home_xg is not None:
   721→            feature_dict["xg_for_home"] = float(home_xg)
   722→        if away_xg is not None:
   723→            feature_dict["xg_for_away"] = float(away_xg)
   724→
   725→        # Handle optional odds data for market features
   726→        odds_home = match_data.get("odds_home")
   727→        odds_draw = match_data.get("odds_draw")
   728→        odds_away = match_data.get("odds_away")
   729→        if odds_home and odds_draw and odds_away:
   730→            total_implied = 1.0/float(odds_home) + 1.0/float(odds_draw) + 1.0/float(odds_away)
   731→            feature_dict["implied_prob_home"] = (1.0/float(odds_home)) / total_implied
   732→            feature_dict["implied_prob_draw"] = (1.0/float(odds_draw)) / total_implied
   733→            feature_dict["implied_prob_away"] = (1.0/float(odds_away)) / total_implied
   734→            feature_dict["overround"] = total_implied - 1.0
   735→
   736→        # Order features correctly
   737→        feature_vector = np.array(
   738→            [feature_dict[fname] for fname in FEATURE_NAMES],
   739→            dtype=np.float32,
   740→        )
   741→
   742→        try:
   743→            pred = self._xgboost_engine.predict(
   744→                feature_vector, feature_names=FEATURE_NAMES
   745→            )
   746→            return {
   747→                "home_win": pred.home_win_prob,
   748→                "draw": pred.draw_prob,
   749→                "away_win": pred.away_win_prob,
   750→            }
   751→        except Exception as exc:
   752→            logger.warning("XGBoost prediction failed: %s", exc)
   753→            return None

   755→    def _run_lstm(self, match_data: dict) -> dict[str, float] | None:
   756→        """Try to load and run LSTM model.
   757→
   758→        Returns home_win/draw/away_win probabilities or None.
   759→
   760→        CRITICAL: The LSTM is a SEQUENCE model.  It requires genuine temporal
   761→        match histories — not a single feature vector repeated N times.
   762→        If real sequences are not available, this method returns None
   763→        rather than producing meaningless output from padded data.
   764→        """
   765→        if not self._lstm_loaded:
   766→            try:
   767→                from src.ml.lstm_engine import LSTMEngine
   768→
   769→                self._lstm_engine = LSTMEngine(
   770→                    model_dir=str(self.model_dir),
   771→                )
   772→                loaded = self._lstm_engine.load("lstm_v1")
   773→                self._lstm_loaded = True  # mark as attempted regardless
   774→                if not loaded:
   775→                    logger.info("LSTM model not found — skipping")
   776→                    return None
   777→            except Exception as exc:
   778→                self._lstm_loaded = True
   779→                logger.debug("LSTM load failed: %s", exc)
   780→                return None
   781→
   782→        if self._lstm_engine is None or not self._lstm_engine.is_trained:
   783→            return None
   784→
   785→        # ── HONEST BEHAVIOR: require real sequences ─────────────────────
   786→        # The LSTM model requires (seq_len, input_size) tensors built from
   787→        # actual past match histories.  Repeating a single feature vector
   788→        # across timesteps is NOT a valid sequence — it destroys the
   789→        # temporal structure the model was trained on.
   790→        #
   791→        # Real sequences must be provided via match_data["home_sequence"]
   792→        # and match_data["away_sequence"] (each a list of dicts or a
   793→        # 2-D numpy array of shape (seq_len, n_features)).
   794→        #
   795→        # If these are not present, we honestly return None.
   796→        home_seq = match_data.get("home_sequence")
   797→        away_seq = match_data.get("away_sequence")
   798→
   799→        if home_seq is None or away_seq is None:
   800→            logger.debug(
   801→                "LSTM: no real sequences in match_data — honestly skipping. "
   802→                "Pass home_sequence and away_sequence (each a list of dicts "
   803→                "or np.ndarray of shape (seq_len, n_features))."
   804→            )
   805→            return None
   806→
   807→        try:
   808→            import numpy as np
   809→
   810→            X_home = np.asarray(home_seq, dtype=np.float32)
   811→            X_away = np.asarray(away_seq, dtype=np.float32)
   812→
   813→            # Ensure 3-D: (1, seq_len, n_features)
   814→            if X_home.ndim == 2:
   815→                X_home = X_home[np.newaxis, ...]
   816→            if X_away.ndim == 2:
   817→                X_away = X_away[np.newaxis, ...]
   818→
   819→            # Validate shape matches what model expects
   820→            expected_features = len(self._lstm_engine.feature_cols)
   821→            if X_home.shape[2] != expected_features:
   822→                logger.warning(
   823→                    "LSTM: feature mismatch — model expects %d features, "
   824→                    "got %d. Cannot produce valid prediction.",
   825→                    expected_features, X_home.shape[2],
   826→                )
   827→                return None
   828→
   829→            result = self._lstm_engine.predict(X_home[0], X_away[0])
   830→
   831→            if isinstance(result, dict) and "batch" not in result:
   832→                return {
   833→                    "home_win": result["home_win"],
   834→                    "draw": result["draw"],
   835→                    "away_win": result["away_win"],
   836→                }
   837→
   838→            return None
   839→        except Exception as exc:
   840→            logger.warning("LSTM prediction failed: %s", exc)
   841→            return None
