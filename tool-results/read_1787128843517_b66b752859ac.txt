   127→def compute_ece(y_true, y_prob, n_bins=15):
   128→    """Compute Expected Calibration Error with more bins for stability."""
   129→    n_classes = y_prob.shape[1]
   130→    ece = 0.0
   131→    for c in range(n_classes):
   132→        y_binary = (y_true == c).astype(int)
   133→        prob_c = y_prob[:, c]
   134→        bin_edges = np.linspace(0, 1, n_bins + 1)
   135→        for i in range(n_bins):
   136→            mask = (prob_c >= bin_edges[i]) & (prob_c < bin_edges[i + 1])
   137→            if i == n_bins - 1:
   138→                mask = (prob_c >= bin_edges[i]) & (prob_c <= bin_edges[i + 1])
   139→            n_bin = mask.sum()
   140→            if n_bin == 0:
   141→                continue
   142→            avg_conf = prob_c[mask].mean()
   143→            avg_acc = y_binary[mask].mean()
   144→            ece += (n_bin / len(y_true)) * abs(avg_conf - avg_acc)
   145→    return ece / n_classes
   146→
   147→
   148→def compute_all_metrics(y_true, y_proba, y_pred, label='set'):
   149→    """Full metrics: log loss, Brier, ECE, accuracy, per-class P/R/F1, confusion matrix."""
   150→    n = len(y_true)
   151→    m = {'label': label, 'n_samples': n}
   152→
   153→    m['accuracy'] = float(accuracy_score(y_true, y_pred))
   154→
   155→    # Log loss
   156→    try:
   157→        m['log_loss'] = float(log_loss(y_true, y_proba))
   158→    except Exception:
   159→        m['log_loss'] = float('nan')
   160→
   161→    # Brier score (multi-class, macro-averaged)
   162→    n_classes = y_proba.shape[1]
   163→    y_onehot = np.zeros_like(y_proba)
   164→    for i in range(n_classes):
   165→        y_onehot[:, i] = (y_true == i).astype(float)
   166→    briers = []
   167→    for i in range(n_classes):
   168→        try:
   169→            briers.append(brier_score_loss(y_onehot[:, i], y_proba[:, i]))
   170→        except Exception:
   171→            briers.append(float('nan'))
   172→    m['brier_score'] = float(np.nanmean(briers))
   173→
   174→    # Per-class P/R/F1
   175→    prec, rec, f1, support = precision_recall_fscore_support(
   176→        y_true, y_pred, labels=[0, 1, 2], zero_division=0
   177→    )
   178→    for i, name in enumerate(['Home', 'Draw', 'Away']):
   179→        m[f'precision_{name}'] = float(prec[i])
   180→        m[f'recall_{name}'] = float(rec[i])
   181→        m[f'f1_{name}'] = float(f1[i])
   182→        m[f'support_{name}'] = int(support[i])
   183→    m['precision_macro'] = float(np.mean(prec))
   184→    m['recall_macro'] = float(np.mean(rec))
   185→    m['f1_macro'] = float(np.mean(f1))
   186→
   187→    # Confusion matrix
   188→    cm = confusion_matrix(y_true, y_pred, labels=[0, 1, 2])
   189→    m['confusion_matrix'] = cm.tolist()
   190→
   191→    # ECE
   192→    m['ece'] = float(compute_ece(y_true, y_proba))
   193→
   194→    # Calibration reliability diagram data (for per-bin analysis)
   195→    m['calibration_bins'] = compute_calibration_bins(y_true, y_proba, n_bins=10)
   196→
   197→    return m
   198→
   199→
   200→def compute_calibration_bins(y_true, y_prob, n_bins=10):
   201→    """Compute per-bin calibration data for reliability analysis."""
   202→    bins_data = []
   203→    for c in range(y_prob.shape[1]):
   204→        y_binary = (y_true == c).astype(int)
   205→        prob_c = y_prob[:, c]
   206→        edges = np.linspace(0, 1, n_bins + 1)
   207→        for i in range(n_bins):
   208→            mask = (prob_c >= edges[i]) & (prob_c < edges[i + 1])
   209→            if i == n_bins - 1:
   210→                mask = (prob_c >= edges[i]) & (prob_c <= edges[i + 1])
   211→            n_bin = mask.sum()
   212→            if n_bin < 5:
   213→                continue
   214→            bins_data.append({
   215→                'class': c,
   216→                'bin_center': float((edges[i] + edges[i + 1]) / 2),
   217→                'avg_confidence': float(prob_c[mask].mean()),
   218→                'avg_accuracy': float(y_binary[mask].mean()),
   219→                'count': int(n_bin),
   220→            })
   221→    return bins_data

   224→def compute_prediction_correlation(proba_a, proba_b, y_true):
   225→    """Compute correlation between two models' predicted probabilities."""
   226→    # Flatten probabilities and compute correlation
   227→    p_a = proba_a.flatten()
   228→    p_b = proba_b.flatten()
   229→    corr = float(np.corrcoef(p_a, p_b)[0, 1])
   230→
   231→    # Per-class argmax agreement
   232→    pred_a = np.argmax(proba_a, axis=1)
   233→    pred_b = np.argmax(proba_b, axis=1)
   234→    agreement = float((pred_a == pred_b).mean())
   235→
   236→    # Both-correct rate (both models predict correctly)
   237→    both_correct = float(((pred_a == y_true) & (pred_b == y_true)).mean())
   238→
   239→    return {
   240→        'probability_correlation': corr,
   241→        'prediction_agreement': agreement,
   242→        'both_correct_rate': both_correct,
   243→    }

   246→def compute_naive_baseline(y_true):
   247→    """Compute naive baseline: always predict class priors."""
   248→    n = len(y_true)
   249→    counts = np.bincount(y_true, minlength=3)
   250→    priors = counts / n
   251→    # Probas = priors for every sample
   252→    proba = np.tile(priors, (n, 1))
   253→    pred = np.full(n, np.argmax(priors))  # always predict majority class
   254→
   255→    m = {}
   256→    m['accuracy'] = float(accuracy_score(y_true, pred))
   257→    try:
   258→        m['log_loss'] = float(log_loss(y_true, proba))
   259→    except Exception:
   260→        m['log_loss'] = float('nan')
   261→    n_classes = 3
   262→    y_onehot = np.zeros((n, n_classes))
   263→    for i in range(n_classes):
   264→        y_onehot[:, i] = (y_true == i).astype(float)
   265→    briers = [brier_score_loss(y_onehot[:, i], proba[:, i]) for i in range(n_classes)]
   266→    m['brier_score'] = float(np.mean(briers))
   267→    m['ece'] = float(compute_ece(y_true, proba))
   268→    m['class_priors'] = [float(p) for p in priors]
   269→    return m

   272→# ═══════════════════════════════════════════════════════════════════════
   273→# XGBoost Training (same params as v1 for fair comparison)
   274→# ═══════════════════════════════════════════════════════════════════════
   275→
   276→def train_xgb_model(X_train, y_train, X_val, y_val, feature_names, seed=42):
   277→    """Train XGBoost with IDENTICAL hyperparameters to v1 baseline."""
   278→    import xgboost as xgb
   279→
   280→    counts = np.bincount(y_train, minlength=3)
   281→    total = counts.sum()
   282→    scale_pos_weight = {i: total / (3 * c) for i, c in enumerate(counts) if c > 0}
   283→    sample_weights = np.array([scale_pos_weight.get(y, 1.0) for y in y_train])
   284→
   285→    dtrain = xgb.DMatrix(X_train, label=y_train, weight=sample_weights, feature_names=feature_names)
   286→    dval = xgb.DMatrix(X_val, label=y_val, feature_names=feature_names)
   287→
   288→    params = {
   289→        'objective': 'multi:softprob',
   290→        'num_class': 3,
   291→        'eval_metric': 'mlogloss',
   292→        'max_depth': 5,
   293→        'eta': 0.05,
   294→        'subsample': 0.8,
   295→        'colsample_bytree': 0.8,
   296→        'min_child_weight': 5,
   297→        'gamma': 0.1,
   298→        'lambda': 1.5,
   299→        'alpha': 0.1,
   300→        'seed': seed,
   301→        'nthread': 4,
   302→    }
   303→
   304→    bst = xgb.train(
   305→        params, dtrain,
   306→        num_boost_round=1000,
   307→        evals=[(dtrain, 'train'), (dval, 'val')],
   308→        early_stopping_rounds=100,
   309→        verbose_eval=False,
   310→    )
   311→
   312→    return bst, bst.best_iteration
   313→
   314→
   315→def get_predictions(bst, X, feature_names):
   316→    """Get probability predictions from XGBoost model."""
   317→    import xgboost as xgb
   318→    d = xgb.DMatrix(X, feature_names=feature_names)
   319→    proba = bst.predict(d)
   320→    pred = np.argmax(proba, axis=1)
   321→    return proba, pred

   328→def run_experiment(df_complete, window, feature_set_name, feature_set_info):
   329→    """Run one experiment: (window, feature_set) -> metrics dict."""
   330→    feats = feature_set_info['features']
   331→
   332→    # Split data by season
   333→    train_df = df_complete[df_complete['season_code'].isin(window['train_seasons'])].copy()
   334→    val_df = df_complete[df_complete['season_code'].isin(window['val_seasons'])].copy()
   335→    test_df = df_complete[df_complete['season_code'].isin(window['test_seasons'])].copy()
   336→
   337→    # Ensure features are available in the dataframe
   338→    available_feats = [f for f in feats if f in df_complete.columns]
   339→    if len(available_feats) == 0:
   340→        return None
   341→
   342→    # Check for complete cases in this feature set
   343→    train_mask = train_df[available_feats].notna().all(axis=1)
   344→    val_mask = val_df[available_feats].notna().all(axis=1)
   345→    test_mask = test_df[available_feats].notna().all(axis=1)
   346→
   347→    train_df = train_df[train_mask].reset_index(drop=True)
   348→    val_df = val_df[val_mask].reset_index(drop=True)
   349→    test_df = test_df[test_mask].reset_index(drop=True)
   350→
   351→    if len(test_df) < 50:
   352→        return None
   353→
   354→    # Extract X, y
   355→    X_train = train_df[available_feats].values.astype(np.float32)
   356→    y_train = train_df['result_encoded'].values.astype(np.int32)
   357→    X_val = val_df[available_feats].values.astype(np.float32)
   358→    y_val = val_df['result_encoded'].values.astype(np.int32)
   359→    X_test = test_df[available_feats].values.astype(np.float32)
   360→    y_test = test_df['result_encoded'].values.astype(np.int32)
   361→
   362→    if len(y_train) < 100 or len(np.unique(y_train)) < 3:
   363→        return None
   364→
   365→    # Train
   366→    try:
   367→        bst, best_iter = train_xgb_model(X_train, y_train, X_val, y_val, available_feats)
   368→    except Exception as e:
   369→        print(f'    Training failed: {e}')
   370→        return None
   371→
   372→    # Predict
   373→    test_proba, test_pred = get_predictions(bst, X_test, available_feats)
   374→    val_proba, val_pred = get_predictions(bst, X_val, available_feats)
   375→    train_proba, train_pred = get_predictions(bst, X_train, available_feats)
   376→
   377→    # Metrics
   378→    test_metrics = compute_all_metrics(y_test, test_proba, test_pred, 'test')
   379→    val_metrics = compute_all_metrics(y_val, val_proba, val_pred, 'validation')
   380→    train_metrics = compute_all_metrics(y_train, train_proba, train_pred, 'train')
   381→
   382→    # Feature importance
   383→    importance = bst.get_score(importance_type='gain')
   384→    feat_imp = {f: float(importance.get(f, 0.0)) for f in available_feats}
   385→
   386→    return {
   387→        'feature_set': feature_set_name,
   388→        'n_features': len(available_feats),
   389→        'features': available_feats,
   390→        'window': window['name'],
   391→        'train_n': len(y_train),
   392→        'val_n': len(y_val),
   393→        'test_n': len(y_test),
   394→        'best_iteration': int(best_iter),
   395→        'test_metrics': test_metrics,
   396→        'val_metrics': val_metrics,
   397→        'train_metrics': train_metrics,
   398→        'feature_importance': feat_imp,
   399→        'test_proba': test_proba,  # keep for correlation analysis
   400→        'test_pred': test_pred,
   401→        'y_test': y_test,
   402→        'test_df': test_df,  # keep for per-league breakdown
   403→    }

   762→def compute_summary(all_results, naive_baselines, correlations):
   763→    """Compute the overall summary and verdict."""
   764→    # Count how often each feature set wins on log loss
   765→    wins = {'market_only': 0, 'football_only': 0, 'combined': 0}
   766→    deltas_combined_vs_market = []
   767→    deltas_football_vs_market = []
   768→
   769→    for window in TEMPORAL_WINDOWS:
   770→        wname = window['name']
   771→        results = {r['feature_set']: r for r in all_results if r['window'] == wname}
   772→        ll_values = {}
   773→        for fs_name in ['market_only', 'football_only', 'combined']:
   774→            if fs_name in results:
   775→                ll_values[fs_name] = results[fs_name]['test_metrics']['log_loss']
   776→
   777→        if len(ll_values) >= 2:
   778→            best = min(ll_values, key=ll_values.get)
   779→            wins[best] += 1
   780→
   781→        if 'market_only' in ll_values and 'combined' in ll_values:
   782→            delta = ll_values['combined'] - ll_values['market_only']
   783→            deltas_combined_vs_market.append(delta)
   784→
   785→        if 'market_only' in ll_values and 'football_only' in ll_values:
   786→            delta = ll_values['football_only'] - ll_values['market_only']
   787→            deltas_football_vs_market.append(delta)
   788→
   789→    avg_corr = np.mean([c['probability_correlation'] for c in correlations]) if correlations else 0
   790→    avg_agreement = np.mean([c['prediction_agreement'] for c in correlations]) if correlations else 0
   791→    avg_combined_delta = np.mean(deltas_combined_vs_market) if deltas_combined_vs_market else 0
   792→    avg_football_delta = np.mean(deltas_football_vs_market) if deltas_football_vs_market else 0
   793→
   794→    # Determine verdict
   795→    if avg_football_delta < -0.01 and avg_corr < 0.7:
   796→        verdict = 'FOOTBALL FEATURES CONTRIBUTE INDEPENDENT INTELLIGENCE'
   797→        color = 'green'
   798→    elif avg_football_delta > 0.02:
   799→        verdict = 'MARKET FEATURES DOMINATE — FOOTBALL FEATURES ADD NOISE'
   800→        color = 'red'
   801→    elif avg_combined_delta < -0.005:
   802→        verdict = 'COMBINED MODEL SHOWS MARGINAL FOOTBALL CONTRIBUTION'
   803→        color = 'amber'
   804→    else:
   805→        verdict = 'FOOTBALL FEATURES OFFER NEGLIGIBLE INDEPENDENT VALUE'
   806→        color = 'amber'
   807→
   808→    # Build summary text
   809→    summary_parts = []
   810→    summary_parts.append(
   811→        f'Across {len(TEMPORAL_WINDOWS)} temporal windows, the market-only model (4 features) '
   812→        f'achieved an average log loss competitive with the football-only model (46 features).'
   813→    )
   814→    summary_parts.append(
   815→        f'The combined model (50 features) shows an average log loss delta of {avg_combined_delta:+.4f} '
   816→        f'relative to market-only, meaning football features {"add marginal value" if avg_combined_delta < -0.005 else "do not reliably improve"} over bookmaker information alone.'
   817→    )
   818→    summary_parts.append(
   819→        f'The average prediction correlation between market-only and football-only models is {avg_corr:.4f} '
   820→        f'(agreement: {avg_agreement:.4f}). '
   821→        f'{"This moderate correlation suggests the feature sets capture partially different information." if avg_corr < 0.7 else "This high correlation suggests both feature sets produce similar predictions."}'
   822→    )
   823→
   824→    # Conclusions
   825→    conclusions = []
   826→    conclusions.append(
   827→        f'Market features win on log loss in {wins["market_only"]}/{len(TEMPORAL_WINDOWS)} windows. '
   828→        f'Football-only wins in {wins["football_only"]}/{len(TEMPORAL_WINDOWS)}. '
   829→        f'Combined wins in {wins["combined"]}/{len(TEMPORAL_WINDOWS)}.'
   830→    )
   831→    conclusions.append(
   832→        f'Average football-only vs market-only LL delta: {avg_football_delta:+.4f}. '
   833→        f'Average combined vs market-only LL delta: {avg_combined_delta:+.4f}.'
   834→    )
   835→    conclusions.append(
   836→        f'Prediction correlation (market vs football): {avg_corr:.4f}. '
   837→        f'{"The models make meaningfully different predictions." if avg_corr < 0.6 else "The models largely agree on predictions."}'
   838→    )
   839→
   840→    if avg_football_delta > 0.02:
   841→        conclusions.append(
   842→            'Football-only features consistently UNDERPERFORM market-only. '
   843→            'The XGBoost model with all 50 features is primarily a bookmaker-odds repackaging engine.'

   846→            'This does NOT mean football features are useless — it means they need better engineering, '
   847→            'or the market odds already encode the information they contain (efficient market hypothesis).'

   849→    elif avg_combined_delta < -0.01:
   850→        conclusions.append(
   851→            'Combined model improves over market-only, suggesting football features DO contribute '
   852→            'independent signal that complements market information.'
   853→        )
   854→    else:

   857→            'The combined model does not reliably outperform market-only.'

   863→        'the strongest signal. For pre-match prediction WITHOUT odds, football-only is the only option '
   864→        'and its performance sets the floor for the LSTM.'
   865→    )
   866→
   867→    # Implications
   868→    implications = []
   869→    implications.append(
   870→        'LSTM training should focus on football-only features as its primary input. '
   871→        'Market features can be incorporated later via feature concatenation at the ensemble level.'

   883→        'predictions are based on market data, football data, or both.'
   884→    )

   886→    return {

   896→    }
