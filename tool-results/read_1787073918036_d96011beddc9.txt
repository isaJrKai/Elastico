   216→def extract_team_features(row, is_home):
   217→    """Extract 15 team-centric features from a match row.
   218→
   219→    Phase 4C: Home/Away Representation.
   220→    When the team was home, use home columns; when away, use away columns.
   221→    Plus opponent features (from the opposite columns) and is_home_flag.
   222→    """
   223→    feat = np.zeros(N_SEQ_FEATURES, dtype=np.float32)
   224→    for i, (home_col, away_col) in enumerate(TEAM_FEATURE_MAP):
   225→        col = home_col if is_home else away_col
   226→        feat[i] = float(row.get(col, 0.0))
   227→    feat[14] = 1.0 if is_home else 0.0
   228→    return feat

   231→def build_all_sequences(df_complete):
   232→    """Build team-centric temporal sequences for EVERY match in the dataset.
   233→
   234→    Phase 4B + 4N: For each target match at time T:
   235→      1. Find all past matches for HomeTeam where Date < T
   236→      2. Find all past matches for AwayTeam where Date < T
   237→      3. Take the last SEQ_LEN matches for each team
   238→      4. Extract 15 team-centric features per timestep
   239→      5. If either team has < SEQ_LEN past matches, SKIP (honest, no padding)
   240→
   241→    Sequences are built once using ALL available history.
   242→    Window filtering happens AFTER construction.
   243→
   244→    Returns:
   245→        X_home: np.ndarray (N_valid, SEQ_LEN, 15)
   246→        X_away: np.ndarray (N_valid, SEQ_LEN, 15)
   247→        y:      np.ndarray (N_valid,) int32
   248→        valid_indices: list[int] — row indices into df_complete
   249→        skipped: int
   250→    """
   251→    print('Building temporal sequences for all matches...')
   252→
   253→    # Build per-team match index: team -> [(date, row_idx, is_home), ...]
   254→    team_matches = {}
   255→    for idx, row in df_complete.iterrows():
   256→        ht = row['HomeTeam']
   257→        at = row['AwayTeam']
   258→        dt = row['Date']
   259→        for team, is_home in [(ht, True), (at, False)]:
   260→            if team not in team_matches:
   261→                team_matches[team] = []
   262→            team_matches[team].append((dt, idx, is_home))
   263→
   264→    # Sort each team's matches chronologically
   265→    for team in team_matches:
   266→        team_matches[team].sort(key=lambda x: x[0])
   267→
   268→    X_home_list = []
   269→    X_away_list = []
   270→    y_list = []
   271→    valid_indices = []
   272→    skipped = 0
   273→
   274→    for idx, row in df_complete.iterrows():
   275→        ht = row['HomeTeam']
   276→        at = row['AwayTeam']
   277→        target_date = row['Date']
   278→
   279→        # Get past matches for home team (Date < target_date, excluding self)
   280→        home_past = [(dt, ri, ih) for dt, ri, ih in team_matches[ht]
   281→                     if dt < target_date and ri != idx]
   282→        # Get past matches for away team
   283→        away_past = [(dt, ri, ih) for dt, ri, ih in team_matches[at]
   284→                     if dt < target_date and ri != idx]
   285→
   286→        # Phase 4N: SKIP if insufficient history. No padding.
   287→        if len(home_past) < SEQ_LEN or len(away_past) < SEQ_LEN:
   288→            skipped += 1
   289→            continue
   290→
   291→        # Take last SEQ_LEN matches for each team
   292→        home_seq_raw = home_past[-SEQ_LEN:]
   293→        away_seq_raw = away_past[-SEQ_LEN:]
   294→
   295→        # Build feature matrices
   296→        home_features = np.zeros((SEQ_LEN, N_SEQ_FEATURES), dtype=np.float32)
   297→        away_features = np.zeros((SEQ_LEN, N_SEQ_FEATURES), dtype=np.float32)
   298→
   299→        for t, (dt, ri, is_home) in enumerate(home_seq_raw):
   300→            past_row = df_complete.loc[ri]
   301→            feat = extract_team_features(past_row, is_home)
   302→            home_features[t] = feat
   303→
   304→        for t, (dt, ri, is_home) in enumerate(away_seq_raw):
   305→            past_row = df_complete.loc[ri]
   306→            feat = extract_team_features(past_row, is_home)
   307→            away_features[t] = feat
   308→
   309→        # Replace any NaN/inf with 0 (cold-start residual)
   310→        home_features = np.nan_to_num(home_features, nan=0.0, posinf=0.0, neginf=0.0)
   311→        away_features = np.nan_to_num(away_features, nan=0.0, posinf=0.0, neginf=0.0)
   312→
   313→        X_home_list.append(home_features)
   314→        X_away_list.append(away_features)
   315→        y_list.append(int(row['result_encoded']))
   316→        valid_indices.append(idx)
   317→
   318→    X_home = np.stack(X_home_list, axis=0)
   319→    X_away = np.stack(X_away_list, axis=0)
   320→    y = np.array(y_list, dtype=np.int32)
   321→
   322→    print(f'  Total matches: {len(df_complete)}')
   323→    print(f'  Valid sequences: {len(y)}')
   324→    print(f'  Skipped (insufficient history): {skipped}')
   325→    print(f'  Coverage: {len(y)/len(df_complete)*100:.1f}%')
   326→
   327→    return X_home, X_away, y, valid_indices, skipped
   328→

   331→# Phase 4D — NORMALIZATION
   332→# ═════════════════════════════════════════════════════════════════════════════════════════
   333→
   334→class Normalizer:
   335→    """StandardScaler fit on training data only."""
   336→
   337→    def __init__(self):
   338→        self.mean = None
   339→        self.std = None
   340→        self.n_features = None
   341→
   342→    def fit(self, X):
   343→        """Fit on training data. X shape: (N, seq_len, n_features)."""
   344→        X_flat = X.reshape(-1, X.shape[-1])
   345→        self.n_features = X_flat.shape[1]
   346→        self.mean = X_flat.mean(axis=0)
   347→        self.std = X_flat.std(axis=0)
   348→        self.std[self.std < 1e-8] = 1.0
   349→        return self
   350→
   351→    def transform(self, X):
   352→        """Transform using stored parameters."""
   353→        shape = X.shape
   354→        X_flat = X.reshape(-1, self.n_features)
   355→        X_norm = (X_flat - self.mean) / self.std
   356→        return X_norm.reshape(shape)
   357→
   358→    def to_dict(self):
   359→        return {'mean': self.mean.tolist(), 'std': self.std.tolist(),
   360→                'n_features': self.n_features}
   361→
   362→    @classmethod
   363→    def from_dict(cls, d):
   364→        n = cls()
   365→        n.mean = np.array(d['mean'])
   366→        n.std = np.array(d['std'])
   367→        n.n_features = d['n_features']
   368→        return n

   372→# Phase 4F — LSTM ARCHITECTURE
   373→# ═════════════════════════════════════════════════════════════════════════════════════════
   374→
   375→import torch
   376→import torch.nn as nn
   377→
   378→class FootballLSTM_v1(nn.Module):
   379→    """Two-branch BiLSTM with attention for football prediction.
   380→
   381→    Architecture:
   382→        Home seq (B, T, 15) → BiLSTM → Attention → home_ctx (B, 96)
   383→        Away seq (B, T, 15) → BiLSTM(shared) → Attention → away_ctx (B, 96)
   384→        Concat → FC(192→64) → BN → ReLU → Dropout(0.4) → FC(64→3)
   385→
   386→    Design rationale:
   387→      hidden_size=48 (reduced from 64 for ~3-4K training sequences)
   388→      FC head: 2 layers (reduced from 3 for less overfitting)
   389→      dropout=0.4 (increased from 0.3 given XGBoost overfitting evidence)
   390→      Shared BiLSTM weights: both branches use same LSTM (team-agnostic)
   391→    """
   392→
   393→    def __init__(self, input_size=15, hidden_size=48, num_layers=2,
   394→                 dropout=0.4, n_classes=3):
   395→        super().__init__()
   396→        self.input_size = input_size
   397→        self.hidden_size = hidden_size
   398→        self.num_layers = num_layers
   399→        self.dropout_rate = dropout
   400→        bidir_hidden = hidden_size * 2  # 96
   401→
   402→        # Shared BiLSTM
   403→        self.lstm = nn.LSTM(
   404→            input_size=input_size,
   405→            hidden_size=hidden_size,
   406→            num_layers=num_layers,
   407→            batch_first=True,
   408→            bidirectional=True,
   409→            dropout=dropout if num_layers > 1 else 0.0,
   410→        )
   411→
   412→        # Attention
   413→        self.attention = nn.Linear(bidir_hidden, 1)
   414→
   415→        # FC head
   416→        fc_input = bidir_hidden * 2  # 192
   417→        self.fc = nn.Sequential(
   418→            nn.Linear(fc_input, 64),
   419→            nn.BatchNorm1d(64),
   420→            nn.ReLU(),
   421→            nn.Dropout(dropout),
   422→            nn.Linear(64, n_classes),
   423→        )
   424→
   425→        self._init_weights()
   426→
   427→    def _init_weights(self):
   428→        for name, param in self.named_parameters():
   429→            if 'lstm' in name and 'weight_ih' in name:
   430→                nn.init.xavier_uniform_(param)
   431→            elif 'lstm' in name and 'weight_hh' in name:
   432→                nn.init.orthogonal_(param)
   433→            elif 'lstm' in name and 'bias' in name:
   434→                nn.init.zeros_(param)
   435→                n = param.size(0)
   436→                start, end = n // 4, n // 2
   437→                nn.init.constant_(param.data[start:end], 1.0)
   438→            elif 'weight' in name and param.ndim >= 2:
   439→                nn.init.xavier_uniform_(param)
   440→            elif 'bias' in name:
   441→                nn.init.zeros_(param)
   442→
   443→    def _attend(self, lstm_out):
   444→        """Softmax attention: (B, T, H*2) → (B, H*2)"""
   445→        attn_logits = self.attention(lstm_out)
   446→        attn_weights = torch.softmax(attn_logits, dim=1)
   447→        context = torch.sum(attn_weights * lstm_out, dim=1)
   448→        return context
   449→
   450→    def forward(self, x_home, x_away):
   451→        """
   452→        x_home: (B, T, 15)  x_away: (B, T, 15)
   453→        Returns: logits (B, 3)
   454→        """
   455→        home_out, _ = self.lstm(x_home)
   456→        home_ctx = self._attend(home_out)
   457→
   458→        away_out, _ = self.lstm(x_away)
   459→        away_ctx = self._attend(away_out)
   460→
   461→        combined = torch.cat([home_ctx, away_ctx], dim=1)
   462→        return self.fc(combined)

   469→def train_lstm(model, train_loader, val_loader, config):
   470→    """Train LSTM with early stopping, gradient clipping, LR scheduling."""
   471→    device = torch.device('cpu')
   472→    model = model.to(device)
   473→
   474→    # Set PyTorch seeds for reproducibility
   475→    torch.manual_seed(SEED)
   476→
   477→    # Class weights for imbalance
   478→    y_all = []
   479→    for _, _, yb in train_loader:
   480→        y_all.append(yb.numpy())
   481→    y_all = np.concatenate(y_all)
   482→    counts = np.bincount(y_all, minlength=3).astype(float)
   483→    total = counts.sum()
   484→    class_weights = torch.tensor(total / (3 * counts), dtype=torch.float32).to(device)
   485→    print(f'    Class weights: {class_weights.tolist()}')
   486→
   487→    optimizer = torch.optim.Adam(model.parameters(), lr=config['lr'],
   488→                                  weight_decay=config['weight_decay'])
   489→    scheduler = torch.optim.lr_scheduler.ReduceLROnPlateau(
   490→        optimizer, factor=config['lr_factor'], patience=config['lr_patience'],
   491→        min_lr=config['lr_min'], mode='min')
   492→    criterion = nn.CrossEntropyLoss(weight=class_weights)
   493→
   494→    best_val_loss = float('inf')
   495→    best_state = None
   496→    patience_counter = 0
   497→    history = {'train_loss': [], 'val_loss': [],
   498→               'train_acc': [], 'val_acc': [], 'lr': []}
   499→
   500→    for epoch in range(config['max_epochs']):
   501→        # --- Train ---
   502→        model.train()
   503→        train_loss_sum = 0.0
   504→        train_correct = 0
   505→        train_total = 0
   506→        for xh, xa, yb in train_loader:
   507→            xh, xa, yb = xh.to(device), xa.to(device), yb.to(device)
   508→            optimizer.zero_grad()
   509→            logits = model(xh, xa)
   510→            loss = criterion(logits, yb)
   511→            loss.backward()
   512→            torch.nn.utils.clip_grad_norm_(model.parameters(), config['grad_clip'])
   513→            optimizer.step()
   514→            train_loss_sum += loss.item() * yb.size(0)
   515→            train_correct += (logits.argmax(1) == yb).sum().item()
   516→            train_total += yb.size(0)
   517→
   518→        train_loss = train_loss_sum / train_total
   519→        train_acc = train_correct / train_total
   520→
   521→        # --- Validate ---
   522→        model.eval()
   523→        val_loss_sum = 0.0
   524→        val_correct = 0
   525→        val_total = 0
   526→        with torch.no_grad():
   527→            for xh, xa, yb in val_loader:
   528→                xh, xa, yb = xh.to(device), xa.to(device), yb.to(device)
   529→                logits = model(xh, xa)
   530→                loss = criterion(logits, yb)
   531→                val_loss_sum += loss.item() * yb.size(0)
   532→                val_correct += (logits.argmax(1) == yb).sum().item()
   533→                val_total += yb.size(0)
   534→
   535→        val_loss = val_loss_sum / val_total
   536→        val_acc = val_correct / val_total
   537→        current_lr = optimizer.param_groups[0]['lr']
   538→
   539→        scheduler.step(val_loss)
   540→
   541→        history['train_loss'].append(train_loss)
   542→        history['val_loss'].append(val_loss)
   543→        history['train_acc'].append(train_acc)
   544→        history['val_acc'].append(val_acc)
   545→        history['lr'].append(current_lr)
   546→
   547→        if (epoch + 1) % 10 == 0 or epoch == 0:
   548→            print(f'      Epoch {epoch+1:3d}: train_loss={train_loss:.4f} train_acc={train_acc:.4f} '
   549→                  f'val_loss={val_loss:.4f} val_acc={val_acc:.4f} lr={current_lr:.6f}')
   550→
   551→        # Early stopping
   552→        if val_loss < best_val_loss:
   553→            best_val_loss = val_loss
   554→            best_state = copy.deepcopy(model.state_dict())
   555→            patience_counter = 0
   556→        else:
   557→            patience_counter += 1
   558→            if patience_counter >= config['patience']:
   559→                print(f'      Early stopping at epoch {epoch+1}')
   560→                break
   561→
   562→    # Restore best model
   563→    if best_state is not None:
   564→        model.load_state_dict(best_state)
   565→
   566→    return model, history

   573→def compute_ece(y_true, y_prob, n_bins=15):
   574→    """Expected Calibration Error."""
   575→    n_classes = y_prob.shape[1]
   576→    ece = 0.0
   577→    for c in range(n_classes):
   578→        y_binary = (y_true == c).astype(int)
   579→        prob_c = y_prob[:, c]
   580→        edges = np.linspace(0, 1, n_bins + 1)
   581→        for i in range(n_bins):
   582→            mask = (prob_c >= edges[i]) & (prob_c < edges[i + 1])
   583→            if i == n_bins - 1:
   584→                mask = (prob_c >= edges[i]) & (prob_c <= edges[i + 1])
   585→            n_bin = mask.sum()
   586→            if n_bin == 0:
   587→                continue
   588→            ece += (n_bin / len(y_true)) * abs(prob_c[mask].mean() - y_binary[mask].mean())
   589→    return ece / n_classes

   592→def compute_metrics(y_true, y_proba, y_pred, label='set'):
   593→    """Full metrics suite."""
   594→    n = len(y_true)
   595→    m = {'label': label, 'n_samples': n}
   596→    m['accuracy'] = float(accuracy_score(y_true, y_pred))
   597→    try:
   598→        m['log_loss'] = float(log_loss(y_true, y_proba))
   599→    except Exception:
   600→        m['log_loss'] = float('nan')
   601→    n_classes = y_proba.shape[1]
   602→    y_onehot = np.zeros_like(y_proba)
   603→    for i in range(n_classes):
   604→        y_onehot[:, i] = (y_true == i).astype(float)
   605→    briers = []
   606→    for i in range(n_classes):
   607→        try:
   608→            briers.append(brier_score_loss(y_onehot[:, i], y_proba[:, i]))
   609→        except Exception:
   610→            briers.append(float('nan'))
   611→    m['brier_score'] = float(np.nanmean(briers))
   612→    prec, rec, f1, support = precision_recall_fscore_support(
   613→        y_true, y_pred, labels=[0, 1, 2], zero_division=0)
   614→    for i, name in enumerate(['Home', 'Draw', 'Away']):
   615→        m[f'precision_{name}'] = float(prec[i])
   616→        m[f'recall_{name}'] = float(rec[i])
   617→        m[f'f1_{name}'] = float(f1[i])
   618→        m[f'support_{name}'] = int(support[i])
   619→    m['precision_macro'] = float(np.mean(prec))
   620→    m['recall_macro'] = float(np.mean(rec))
   621→    m['f1_macro'] = float(np.mean(f1))
   622→    m['confusion_matrix'] = confusion_matrix(y_true, y_pred, labels=[0, 1, 2]).tolist()
   623→    m['ece'] = float(compute_ece(y_true, y_proba))
   624→    return m

   627→def predict_proba(model, X_home, X_away, batch_size=256):
   628→    """Get probability predictions from model."""
   629→    model.eval()
   630→    device = torch.device('cpu')
   631→    all_proba = []
   632→    with torch.no_grad():
   633→        for i in range(0, len(X_home), batch_size):
   634→            xh = torch.as_tensor(X_home[i:i+batch_size], dtype=torch.float32).to(device)
   635→            xa = torch.as_tensor(X_away[i:i+batch_size], dtype=torch.float32).to(device)
   636→            logits = model(xh, xa)
   637→            proba = torch.softmax(logits, dim=1).cpu().numpy()
   638→            all_proba.append(proba)
   639→    return np.concatenate(all_proba, axis=0)

   646→def compute_prediction_correlation(proba_a, proba_b, y_true):
   647→    """Correlation between two models' predictions."""
   648→    p_a = proba_a.flatten()
   649→    p_b = proba_b.flatten()
   650→    corr = float(np.corrcoef(p_a, p_b)[0, 1])
   651→    pred_a = np.argmax(proba_a, axis=1)
   652→    pred_b = np.argmax(proba_b, axis=1)
   653→    agreement = float((pred_a == pred_b).mean())
   654→    both_correct = float(((pred_a == y_true) & (pred_b == y_true)).mean())
   655→    disagree_mask = pred_a != pred_b
   656→    if disagree_mask.sum() > 0:
   657→        a_right = (pred_a[disagree_mask] == y_true[disagree_mask]).mean()
   658→        b_right = (pred_b[disagree_mask] == y_true[disagree_mask]).mean()
   659→    else:
   660→        a_right = b_right = 0.0
   661→    return {
   662→        'probability_correlation': corr,
   663→        'prediction_agreement': agreement,
   664→        'both_correct_rate': both_correct,
   665→        'disagree_rate': float(disagree_mask.mean()),
   666→        'lstm_right_on_disagree': float(a_right),
   667→        'xgb_right_on_disagree': float(b_right),
   668→    }

   671→def train_xgb_football_only(df_train, df_test, football_features):
   672→    """Train a football-only XGBoost on the given split. Same params as v1."""
   673→    import xgboost as xgb
   674→
   675→    X_train = df_train[football_features].values.astype(np.float32)

   678→    y_test = df_test['result_encoded'].values.astype(np.int32)
   679→
   680→    # Filter complete cases
   681→    train_mask = ~np.isnan(X_train).any(axis=1)
   682→    test_mask = ~np.isnan(X_test).any(axis=1)

   690→    total = counts.sum()
   691→    sw = {i: total / (3 * c) for i, c in enumerate(counts) if c > 0}
   692→    sample_weights = np.array([sw.get(y, 1.0) for y in y_train])
   693→

   701→              'min_child_weight': 5, 'gamma': 0.1,

   913→    b['name'] = 'Uniform random'
   914→    baselines.append(b)
   915→
   916→    result['baselines'] = baselines
   917→    result['model'] = model
   918→    result['normalizer_obj'] = normalizer
   919→
   920→    return result
   921→
   922→
   923→# ═════════════════════════════════════════════════════════════════════════════════════════
   924→# Phase 4K — MODEL VALUE DECISION
   925→# ═════════════════════════════════════════════════════════════════════════════════════════
   926→
   927→def compute_verdict(window_results):
   928→    """Aggregate verdict across all windows.
   929→
   930→    Classification criteria:
   931→      A. SUPERIOR:     LSTM beats XGB on LL by >=0.01 in majority of windows
   932→      B. COMPLEMENTARY: LSTM not worse by >=0.02 on LL, and disagree rate > 0.15
   933→      C. REDUNDANT:    LSTM not worse by >=0.02, but high agreement (>0.85)
   934→      D. FAILED:       LSTM worse by >=0.02 on LL in majority of windows
   935→    """
   936→    valid_windows = [w for w in window_results if not w.get('skipped', True) and w.get('xgb_test_metrics')]
   937→    n_valid = len(valid_windows)
   938→
   939→    if n_valid == 0:
   940→        return {
   941→            'classification': 'D. FAILED',
   942→            'explanation': 'No valid windows to evaluate.',
   943→            'conclusions': ['All windows had insufficient data.'],
   944→            'recommendations': ['Insufficient data for any conclusion.'],
   945→            'per_window': [],
   946→        }
   947→
   948→    per_window = []
   949→    lstm_ll_better_count = 0
   950→    lstm_acc_better_count = 0
   951→    lstm_ece_better_count = 0
   952→    sum_disagree = 0.0
   953→    sum_disagree_count = 0
   954→
   955→    for w in valid_windows:
   956→        lstm_ll = w['lstm_test_metrics']['log_loss']
   957→        xgb_ll = w['xgb_test_metrics']['log_loss']
   958→        lstm_acc = w['lstm_test_metrics']['accuracy']
   959→        xgb_acc = w['xgb_test_metrics']['accuracy']
   960→        lstm_ece = w['lstm_test_metrics']['ece']
   961→        xgb_ece = w['xgb_test_metrics']['ece']
   962→
   963→        ll_delta = lstm_ll - xgb_ll
   964→        acc_delta = lstm_acc - xgb_acc
   965→        ece_delta = lstm_ece - xgb_ece
   966→
   967→        lstm_ll_better = ll_delta < -0.005
   968→        lstm_acc_better = acc_delta > 0.005
   969→        lstm_ece_better = ece_delta < -0.005
   970→
   971→        if lstm_ll_better:
   972→            lstm_ll_better_count += 1
   973→        if lstm_acc_better:
   974→            lstm_acc_better_count += 1
   975→        if lstm_ece_better:
   976→            lstm_ece_better_count += 1
   977→
   978→        corr = w.get('correlation')
   979→        disagree_rate = corr['disagree_rate'] if corr else 0.0
   980→        if corr:
   981→            sum_disagree += disagree_rate
   982→            sum_disagree_count += 1
   983→
   984→        per_window.append({
   985→            'window': w['window'],
   986→            'lstm_ll': lstm_ll, 'xgb_ll': xgb_ll, 'll_delta': ll_delta,
   987→            'lstm_acc': lstm_acc, 'xgb_acc': xgb_acc, 'acc_delta': acc_delta,
   988→            'lstm_ece': lstm_ece, 'xgb_ece': xgb_ece, 'ece_delta': ece_delta,
   989→            'disagree_rate': disagree_rate,
   990→            'lstm_ll_better': lstm_ll_better,
   991→            'lstm_acc_better': lstm_acc_better,
   992→        })
   993→
   994→    # Average deltas
   995→    avg_ll_delta = np.mean([pw['ll_delta'] for pw in per_window])
   996→    avg_acc_delta = np.mean([pw['acc_delta'] for pw in per_window])
   997→    avg_ece_delta = np.mean([pw['ece_delta'] for pw in per_window])
   998→    avg_disagree = sum_disagree / max(sum_disagree_count, 1)
   999→
  1000→    # Classification logic
  1001→    majority_ll_better = lstm_ll_better_count > n_valid / 2
  1002→    majority_acc_better = lstm_acc_better_count > n_valid / 2
  1003→    lstm_not_worse_ll = avg_ll_delta < 0.02
  1004→    lstm_not_worse_acc = avg_acc_delta > -0.02
  1005→
  1006→    if majority_ll_better and avg_ll_delta < -0.005:
  1007→        classification = 'A. SUPERIOR'
  1008→    elif (lstm_not_worse_ll or majority_ll_better) and avg_disagree > 0.15:
  1009→        classification = 'B. COMPLEMENTARY'
  1010→    elif lstm_not_worse_ll and avg_disagree <= 0.15:
  1011→        classification = 'C. REDUNDANT'
  1012→    else:
  1013→        classification = 'D. FAILED'
  1014→
  1015→    # Build conclusions
  1016→    conclusions = []
  1017→    conclusions.append(
  1018→        f'Evaluated across {n_valid} temporal windows. '
  1019→        f'LSTM beat XGBoost on log loss in {lstm_ll_better_count}/{n_valid} windows '
  1020→        f'(avg delta: {avg_ll_delta:+.4f}).')
  1021→    conclusions.append(
  1022→        f'LSTM beat XGBoost on accuracy in {lstm_acc_better_count}/{n_valid} windows '
  1023→        f'(avg delta: {avg_acc_delta:+.4f}).')
  1024→    conclusions.append(
  1025→        f'LSTM beat XGBoost on ECE in {lstm_ece_better_count}/{n_valid} windows '
  1026→        f'(avg delta: {avg_ece_delta:+.4f}).')
  1027→    if sum_disagree_count > 0:
  1028→        conclusions.append(
  1029→            f'Average prediction disagreement rate: {avg_disagree:.1%}. '
  1030→            f'This {"suggests complementary signal" if avg_disagree > 0.15 else "suggests redundant predictions"}.')
  1031→    else:
  1032→        conclusions.append('Prediction correlation could not be computed for any window.')
  1033→
  1034→    if classification == 'A. SUPERIOR':
  1035→        conclusions.append(
  1036→            'The LSTM extracts genuinely useful temporal patterns from football history '
  1037→            'that static XGBoost cannot capture. This justifies its inclusion in the ensemble.')
  1038→    elif classification == 'B. COMPLEMENTARY':
  1039→        conclusions.append(
  1040→            'The LSTM does not consistently outperform XGBoost, but the models disagree '
  1041→            'frequently enough that combining them may yield improvement.')
  1042→    elif classification == 'C. REDUNDANT':
  1043→        conclusions.append(
  1044→            'The LSTM performs similarly to XGBoost with high agreement. '
  1045→            'Temporal structure provides no meaningful additional signal.')
  1046→    else:
  1047→        conclusions.append(
  1048→            'The LSTM does not outperform a static football-only XGBoost on any meaningful metric. '
  1049→            'Temporal structure in football features does not provide extractable predictive value.')
  1050→
  1051→    # Recommendations
  1052→    recommendations = []
  1053→    if classification == 'A. SUPERIOR':
  1054→        recommendations.append('Promote LSTM as a primary football-intelligence model in the ensemble.')
  1055→        recommendations.append('Investigate stacking or blending with market-based XGBoost.')
  1056→        recommendations.append('The combination of market XGBoost + temporal LSTM may approach genuine predictive value.')
  1057→    elif classification == 'B. COMPLEMENTARY':
  1058→        recommendations.append('Preserve LSTM for ensemble experimentation in Phase 5.')
  1059→        recommendations.append('Investigate whether the LSTM\'s disagreements with XGBoost are calibrated.')
  1060→        recommendations.append('Consider a weighted ensemble that downweights the LSTM relative to XGBoost.')
  1061→        recommendations.append('The LSTM may add value as a diversity source rather than a primary predictor.')
  1062→    elif classification == 'C. REDUNDANT':
  1063→        recommendations.append('LSTM is redundant. Reserve ensemble slots for models with independent signal.')
  1064→        recommendations.append('Consider whether the LSTM architecture needs fundamental redesign.')
  1065→        recommendations.append('Focus ensemble work on combining market-based XGBoost with other diverse models.')
  1066→    else:
  1067→        recommendations.append('Do NOT include LSTM in the final ensemble.')
  1068→        recommendations.append('Focus ensemble work on combining market-based XGBoost with football-only XGBoost.')
  1069→        recommendations.append('If ensemble diversity is needed, consider simpler statistical models instead.')
  1070→
  1071→    return {
  1072→        'classification': classification,
  1073→        'explanation': (f'Based on {n_valid} windows: avg LL delta={avg_ll_delta:+.4f}, '
  1074→                       f'avg Acc delta={avg_acc_delta:+.4f}, avg ECE delta={avg_ece_delta:+.4f}. '
  1075→                       f'Avg disagreement={avg_disagree:.1%}.'),
  1076→        'conclusions': conclusions,
  1077→        'recommendations': recommendations,
  1078→        'per_window': per_window,
  1079→        'n_windows_evaluated': n_valid,
  1080→        'avg_ll_delta': float(avg_ll_delta),
  1081→        'avg_acc_delta': float(avg_acc_delta),
  1082→        'avg_ece_delta': float(avg_ece_delta),
  1083→        'avg_disagree_rate': float(avg_disagree),
  1084→    }
