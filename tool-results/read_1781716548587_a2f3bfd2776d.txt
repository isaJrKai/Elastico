   100→   100→function formatMatchDate(dateStr: string | null): string {
   101→   101→  if (!dateStr) return 'TBD'
   102→   102→  const d = new Date(dateStr)
   103→   103→  return d.toLocaleDateString('en-US', {
   104→   104→    month: 'short',
   105→   105→    day: 'numeric',
   106→   106→    hour: '2-digit',
   107→   107→    minute: '2-digit',
   108→   108→  })
   109→   109→}
   110→   110→
   111→   111→function formatMatchTime(dateStr: string | null): string {
   112→   112→  if (!dateStr) return ''
   113→   113→  const d = new Date(dateStr)
   114→   114→  return d.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' })
   115→   115→}

   117→   117→// ── Helper: Prediction Result Badge ──────────────────────────────────────────
   118→   118→
   119→   119→function getResultBadge(
   120→   120→  match: Match,
   121→   121→  userPredictionOutcome: string | null,
   122→   122→): { label: string; className: string } | null {
   123→   123→  if (match.status !== 'finished' || !userPredictionOutcome) return null
   124→   124→
   125→   125→  const actual =
   126→   126→    match.homeScore > match.awayScore
   127→   127→      ? 'home'
   128→   128→      : match.homeScore === match.awayScore
   129→   129→        ? 'draw'
   130→   130→        : 'away'
   131→   131→
   132→   132→  if (userPredictionOutcome === actual) {
   133→   133→    return { label: '✓ Won', className: 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30' }
   134→   134→  }
   135→   135→  return { label: '✗ Lost', className: 'bg-red-500/15 text-red-400 border-red-500/30' }

   345→   345→// ── Main MatchesView Component ─────────────────────────────────────────────────
   346→   346→
   347→   347→export function MatchesView() {
   348→   348→  const { token, selectMatch } = useElasticoStore()
   349→   349→
   350→   350→  const [activeTab, setActiveTab] = useState<MatchTab>('live')
   351→   351→  const [searchQuery, setSearchQuery] = useState('')
   352→   352→  const [stageFilter, setStageFilter] = useState<string>('all')
   353→   353→  const [groupFilter, setGroupFilter] = useState<string>('all')
   354→   354→  const [matchesMap, setMatchesMap] = useState<Record<MatchTab, Match[]>>({
   355→   355→    live: [],
   356→   356→    upcoming: [],
   357→   357→    finished: [],
   358→   358→    all: [],
   359→   359→  })
   360→   360→  const [loadingMap, setLoadingMap] = useState<Record<MatchTab, boolean>>({
   361→   361→    live: false,
   362→   362→    upcoming: false,
   363→   363→    finished: false,
   364→   364→    all: false,
   365→   365→  })
   366→   366→  const [lastRefresh, setLastRefresh] = useState<Date>(new Date())
   367→   367→  const liveRefreshRef = useRef<ReturnType<typeof setInterval> | null>(null)
   368→   368→
   369→   369→  // ── Fetch matches for a given tab ──────────────────────────────────────
   370→   370→  const fetchMatches = useCallback(
   371→   371→    async (tab: MatchTab, force = false) => {
   372→   372→      // Don't re-fetch if already loading (unless forced)
   373→   373→      if (!force && loadingMap[tab]) return
   374→   374→
   375→   375→      setLoadingMap((prev) => ({ ...prev, [tab]: true }))
   376→   376→
   377→   377→      try {
   378→   378→        const params = new URLSearchParams()
   379→   379→        if (tab !== 'all') params.set('status', tab)
   380→   380→        if (searchQuery) params.set('search', searchQuery)
   381→   381→        if (stageFilter !== 'all') params.set('stage', stageFilter)
   382→   382→        if (groupFilter !== 'all') params.set('group', groupFilter)
   383→   383→
   384→   384→        const headers: Record<string, string> = {}
   385→   385→        if (token) headers['Authorization'] = `Bearer ${token}`
   386→   386→
   387→   387→        const res = await fetch(`/api/matches?${params.toString()}`, {
   388→   388→          headers,
   389→   389→        })
   390→   390→        const data: MatchesApiResponse = await res.json()
   391→   391→
   392→   392→        if (data.matches) {
   393→   393→          setMatchesMap((prev) => ({ ...prev, [tab]: data.matches }))
   394→   394→        }
   395→   395→      } catch (err) {
   396→   396→        console.error(`Failed to fetch matches (${tab}):`, err)
   397→   397→      } finally {
   398→   398→        setLoadingMap((prev) => ({ ...prev, [tab]: false }))
   399→   399→        setLastRefresh(new Date())
   400→   400→      }
   401→   401→    },
   402→   402→    [searchQuery, stageFilter, groupFilter, token, loadingMap],
   403→   403→  )
   404→   404→
   405→   405→  // ── Initial fetch on mount and when tab changes ─────────────────────────
   406→   406→  useEffect(() => {
   407→   407→    fetchMatches(activeTab)
   408→   408→  }, [activeTab, fetchMatches])
   409→   409→
   410→   410→  // ── Re-fetch when filters change ───────────────────────────────────────
   411→   411→  useEffect(() => {
   412→   412→    fetchMatches(activeTab, true)
   413→   413→  }, [searchQuery, stageFilter, groupFilter, activeTab, fetchMatches])
   414→   414→
   415→   415→  // ── Auto-refresh live tab every 15 seconds ──────────────────────────────
   416→   416→  useEffect(() => {
   417→   417→    if (activeTab === 'live') {
   418→   418→      liveRefreshRef.current = setInterval(() => {
   419→   419→        fetchMatches('live', true)
   420→   420→      }, 15_000)
   421→   421→    }
   422→   422→
   423→   423→    return () => {
   424→   424→      if (liveRefreshRef.current) {
   425→   425→        clearInterval(liveRefreshRef.current)
   426→   426→        liveRefreshRef.current = null
   427→   427→      }
   428→   428→    }
   429→   429→  }, [activeTab, fetchMatches])
   430→   430→
   431→   431→  // ── Also keep live data fresh even on other tabs ───────────────────────
   432→   432→  useEffect(() => {
   433→   433→    const interval = setInterval(() => {
   434→   434→      fetchMatches('live', true)
   435→   435→    }, 30_000)
   436→   436→    return () => clearInterval(interval)
   437→   437→  }, [fetchMatches])
   438→   438→
   439→   439→  // ── Debounced search ──────────────────────────────────────────────────
   440→   440→  const searchTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   441→   441→  const handleSearchChange = useCallback((value: string) => {
   442→   442→    setSearchQuery(value)
   443→   443→    if (searchTimeoutRef.current) clearTimeout(searchTimeoutRef.current)
   444→   444→    searchTimeoutRef.current = setTimeout(() => {
   445→   445→      // The effect above will trigger refetch
   446→   446→    }, 300)
   447→   447→  }, [])
   448→   448→
   449→   449→  // ── Current matches ───────────────────────────────────────────────────
   450→   450→  const currentMatches = matchesMap[activeTab] || []
   451→   451→  const isLoading = loadingMap[activeTab]
   452→   452→
   453→   453→  // ── Live match count for tab badges ────────────────────────────────────
   454→   454→  const liveCount = (matchesMap['live'] || []).length
   455→   455→
   456→   456→  // ── Count matches per tab (from the "all" set for total reference) ──────
   457→   457→  const allMatches = matchesMap['all'] || []
   458→   458→  const upcomingCount = allMatches.filter((m) => m.status === 'upcoming').length
   459→   459→  const finishedCount = allMatches.filter((m) => m.status === 'finished').length
   460→   460→
   461→   461→  // ── Extract unique stages / groups from fetched data ─────────────────────
   462→   462→  const availableStages = useMemo(() => {
   463→   463→    const stages = new Set(allMatches.map((m) => m.stage))
   464→   464→    return STAGES.filter((s) => stages.has(s))
   465→   465→  }, [allMatches])
   466→   466→
   467→   467→  const availableGroups = useMemo(() => {
   468→   468→    const groups = new Set(allMatches.map((m) => m.group).filter(Boolean) as string[])
   469→   469→    return GROUPS.filter((g) => groups.has(g))
   470→   470→  }, [allMatches])
   471→   471→
   472→   472→  return (
   473→   473→    <div className="flex flex-col gap-4 animate-fade-in-up">
   474→   474→      {/* ── Filter Bar ────────────────────────────────────────────────────── */}
   475→   475→      <div className="glass-card rounded-xl p-4">
   476→   476→        <div className="flex flex-col sm:flex-row gap-3">
   477→   477→          {/* Search */}
   478→   478→          <div className="relative flex-1">
   479→   479→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   480→   480→            <Input
   481→   481→              placeholder="Search by team name..."
   482→   482→              value={searchQuery}
   483→   483→              onChange={(e) => handleSearchChange(e.target.value)}
   484→   484→              className="pl-9 h-9 bg-muted/50 border-border text-sm"
   485→   485→            />
   486→   486→          </div>
   487→   487→
   488→   488→          {/* Stage Filter */}
   489→   489→          <Select value={stageFilter} onValueChange={setStageFilter}>

   494→   494→            <SelectContent className="glass-card border-border">
   495→   495→              <SelectItem value="all">All Stages</SelectItem>
   496→   496→              {availableStages.map((stage) => (
   497→   497→                <SelectItem key={stage} value={stage}>
   498→   498→                  {stage}
   499→   499→                </SelectItem>
   500→   500→              ))}
   501→   501→            </SelectContent>
   502→   502→          </Select>
   503→   503→
   504→   504→          {/* Group Filter */}
   505→   505→          <Select value={groupFilter} onValueChange={setGroupFilter}>
   506→   506→            <SelectTrigger className="h-9 w-full sm:w-[120px] bg-muted/50 border-border text-sm">
   507→   507→              <SelectValue placeholder="Group" />
   508→   508→            </SelectTrigger>
   509→   509→            <SelectContent className="glass-card border-border">
   510→   510→              <SelectItem value="all">All Groups</SelectItem>
   511→   511→              {availableGroups.map((group) => (
   512→   512→                <SelectItem key={group} value={group}>
   513→   513→                  Group {group}
   514→   514→                </SelectItem>
   515→   515→              ))}
   516→   516→            </SelectContent>
   517→   517→          </Select>
   518→   518→
   519→   519→          {/* Refresh */}
   520→   520→          <Button
   521→   521→            variant="outline"
   522→   522→            size="icon"
   523→   523→            className="h-9 w-9 shrink-0 border-border bg-muted/50 hover:bg-accent"
   524→   524→            onClick={() => fetchMatches(activeTab, true)}
   525→   525→            disabled={isLoading}
   526→   526→            aria-label="Refresh matches"
   527→   527→          >
   528→   528→            <RefreshCw className={cn('size-4', isLoading && 'animate-spin')} />
   529→   529→          </Button>
   530→   530→        </div>
   531→   531→      </div>
   532→   532→
   533→   533→      {/* ── Tabs ───────────────────────────────────────────────────────────── */}
   534→   534→      <Tabs
   535→   535→        value={activeTab}
   536→   536→        onValueChange={(v) => setActiveTab(v as MatchTab)}
   537→   537→        className="w-full"
   538→   538→      >
   539→   539→        <TabsList className="glass-card w-full h-10 bg-muted/30 p-1 rounded-lg">
   540→   540→          {STATUS_TABS.map((tab) => {
   541→   541→            const count =
   542→   542→              tab.value === 'live'
   543→   543→                ? liveCount
   544→   544→                : tab.value === 'upcoming'
   545→   545→                  ? upcomingCount
   546→   546→                  : tab.value === 'finished'
   547→   547→                    ? finishedCount
   548→   548→                    : allMatches.length
   549→   549→
   550→   550→            return (
   551→   551→              <TabsTrigger
   552→   552→                key={tab.value}
   553→   553→                value={tab.value}
   554→   554→                className={cn(
   555→   555→                  'flex-1 h-8 text-xs font-semibold rounded-md transition-all gap-1.5 data-[state=active]:bg-primary/15 data-[state=active]:text-primary data-[state=active]:shadow-sm',
   556→   556→                )}
   557→   557→              >
   558→   558→                {tab.value === 'live' && liveCount > 0 && (
   559→   559→                  <span className="relative flex size-1.5">
   560→   560→                    <span className="absolute inline-flex size-full animate-ping rounded-full bg-red-400 opacity-75" />
   561→   561→                    <span className="relative inline-flex size-1.5 rounded-full bg-red-500" />
   562→   562→                  </span>
   563→   563→                )}
   564→   564→                {tab.label}
   565→   565→                {count > 0 && (
   566→   566→                  <span className="ml-0.5 text-[10px] opacity-60">
   567→   567→                    ({count})
   568→   568→                  </span>
   569→   569→                )}
   570→   570→              </TabsTrigger>
   571→   571→            )
   572→   572→          })}
   573→   573→        </TabsList>
   574→   574→
   575→   575→        {/* ── Tab Content ────────────────────────────────────────────────── */}
   576→   576→        {STATUS_TABS.map((tab) => (

   578→   578→            {/* ── Loading State ──────────────────────────────────────────── */}
   579→   579→            {isLoading && currentMatches.length === 0 && (
   580→   580→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   581→   581→                {Array.from({ length: 6 }).map((_, i) => (

   584→   584→              </div>
   585→   585→            )}
   586→   586→
   587→   587→            {/* ── Empty State ────────────────────────────────────────────── */}
   588→   588→            {!isLoading && currentMatches.length === 0 && (
   589→   589→              <div className="flex flex-col items-center justify-center py-16 text-center">
   590→   590→                <div className="size-16 rounded-full bg-muted/50 flex items-center justify-center mb-4">

   592→   592→                    <Zap className="size-7 text-muted-foreground" />

   594→   594→                    <Eye className="size-7 text-muted-foreground" />
   595→   595→                  )}
   596→   596→                </div>
   597→   597→                <h3 className="text-sm font-semibold text-foreground mb-1">
   598→   598→                  No {tab.value === 'all' ? '' : tab.value} matches found
   599→   599→                </h3>
   600→   600→                <p className="text-xs text-muted-foreground max-w-[260px]">
   601→   601→                  {tab.value === 'live'
   602→   602→                    ? 'There are no live matches at the moment. Check back later!'
   603→   603→                    : tab.value === 'upcoming'
   604→   604→                      ? 'No upcoming matches scheduled. Stay tuned for updates!'
   605→   605→                      : tab.value === 'finished'
   606→   606→                        ? 'No finished matches yet. Matches will appear here after completion.'
   607→   607→                        : searchQuery
   608→   608→                          ? 'Try adjusting your search or filters to find matches.'
   609→   609→                          : 'No matches available. Check back later!'}
   610→   610→                </p>
   611→   611→                {(searchQuery || stageFilter !== 'all' || groupFilter !== 'all') && (
   612→   612→                  <Button
   613→   613→                    variant="outline"
   614→   614→                    size="sm"
   615→   615→                    className="mt-4 h-8 text-xs border-border"
   616→   616→                    onClick={() => {
   617→   617→                      setSearchQuery('')
   618→   618→                      setStageFilter('all')
   619→   619→                      setGroupFilter('all')
   620→   620→                    }}
   621→   621→                  >
   622→   622→                    Clear Filters
   623→   623→                  </Button>
   624→   624→                )}
   625→   625→              </div>
   626→   626→            )}
   627→   627→
   628→   628→            {/* ── Match Grid ─────────────────────────────────────────────── */}
   629→   629→            {currentMatches.length > 0 && (
   630→   630→              <>
   631→   631→                <div className="flex items-center justify-between mb-2 px-1">
   632→   632→                  <span className="text-[11px] text-muted-foreground">
   633→   633→                    {currentMatches.length} match{currentMatches.length !== 1 ? 'es' : ''}
   634→   634→                    {isLoading && ' (refreshing...)'}
   635→   635→                  </span>
   636→   636→                  <span className="text-[10px] text-muted-foreground/50">
   637→   637→                    Updated {lastRefresh.toLocaleTimeString()}
   638→   638→                  </span>
   639→   639→                </div>
   640→   640→                <div
   641→   641→                  className={cn(
   642→   642→                    'grid gap-4',
   643→   643→                    // Responsive columns: 1 on mobile, 2 on sm, 3 on lg
   644→   644→                    'grid-cols-1 sm:grid-cols-2 lg:grid-cols-3',
   645→   645→                  )}
   646→   646→                >
   647→   647→                  {currentMatches.map((match, idx) => (
   648→   648→                    <div
   649→   649→                      key={match.id}
   650→   650→                      style={{ animationDelay: `${Math.min(idx * 50, 300)}ms` }}
   651→   651→                    >
   652→   652→                      <MatchCard match={match} />
   653→   653→                    </div>
   654→   654→                  ))}
   655→   655→                </div>
   656→   656→              </>

   658→   658→          </TabsContent>
   659→   659→        ))}
   660→   660→      </Tabs>
