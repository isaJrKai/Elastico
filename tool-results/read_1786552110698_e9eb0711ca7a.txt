    69→function getStatusConfig(status: string) {
    70→  switch (status) {
    71→    case 'live': return { label: 'LIVE', cls: 'bg-red-500/15 text-red-400 border-red-500/30', pulse: true }
    72→    case 'halftime': return { label: 'HT', cls: 'bg-yellow-500/15 text-yellow-400 border-yellow-500/30', pulse: false }
    73→    case 'finished': return { label: 'FT', cls: 'bg-muted text-muted-foreground border-border', pulse: false }
    74→    case 'postponed': return { label: 'PPD', cls: 'bg-orange-500/15 text-orange-400 border-orange-500/30', pulse: false }
    75→    default: return { label: 'Upcoming', cls: 'bg-primary/15 text-primary border-primary/30', pulse: false }
    76→  }
    77→}
    78→
    79→function formatMatchDate(dateStr: string | null): string {
    80→  if (!dateStr) return 'TBD'
    81→  return new Date(dateStr).toLocaleDateString('en-US', { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' })
    82→}

   344→export function MatchesView() {
   345→  const token = useElasticoStore(s => s.token)
   346→  const [activeTab, setActiveTab] = useState<MatchTab>('all')
   347→  const [leagueFilter, setLeagueFilter] = useState('PL')
   348→  const [searchQuery, setSearchQuery] = useState('')
   349→  const [stageFilter, setStageFilter] = useState('all')
   350→  const [groupFilter, setGroupFilter] = useState('all')
   351→  const [sortBy, setSortBy] = useState<SortKey>('date')
   352→  const [matchesData, setMatchesData] = useState<Match[]>([])
   353→  const [loading, setLoading] = useState(true)
   354→  const [lastRefresh, setLastRefresh] = useState(new Date())
   355→
   356→  const fetchMatches = useCallback(async () => {
   357→    setLoading(true)
   358→    try {
   359→      const headers: Record<string, string> = {}
   360→      if (token) headers['Authorization'] = `Bearer ${token}`
   361→
   362→      // Map activeTab to football-data.org status
   363→      let statusParam = 'FINISHED'
   364→      if (activeTab === 'upcoming') statusParam = 'SCHEDULED,TIMED'
   365→      else if (activeTab === 'live') statusParam = 'IN_PLAY,PAUSED'
   366→      else if (activeTab === 'all') statusParam = 'FINISHED'
   367→
   368→      const leagueParam = leagueFilter === 'all' ? 'PL' : leagueFilter
   369→      const res = await fetch(`/api/football-data?action=matches&competition=${leagueParam}&status=${statusParam}`, { headers })
   370→      const data: any = await res.json()
   371→      const matches = (data.data || []).map((m: any) => ({
   372→        ...m,
   373→        id: m.id || `fd-${Math.random().toString(36).slice(2)}`,
   374→        competition: m.competition,
   375→        homeTeam: m.homeTeam,
   376→        awayTeam: m.awayTeam,
   377→        homeScore: m.homeScore,
   378→        awayScore: m.awayScore,
   379→        status: m.status,
   380→        date: m.date,
   381→        matchday: m.matchday,
   382→      }))
   383→      setMatchesData(matches)
   384→    } catch (err) {
   385→      console.error('Failed to fetch matches:', err)
   386→    } finally {
   387→      setLoading(false)
   388→      setLastRefresh(new Date())
   389→    }
   390→  }, [activeTab, leagueFilter, searchQuery, stageFilter, groupFilter, token])
   391→
   392→  useEffect(() => { fetchMatches() }, [fetchMatches])
   393→
   394→  // Auto-refresh live
   395→  useEffect(() => {
   396→    const iv = setInterval(() => { if (activeTab === 'live') fetchMatches() }, 15000)
   397→    return () => clearInterval(iv)
   398→  }, [activeTab, fetchMatches])
   399→
   400→  // Sort
   401→  const sortedMatches = useMemo(() => {
   402→    const arr = [...matchesData]
   403→    switch (sortBy) {
   404→      case 'elo': return arr.sort((a, b) => Math.abs((b.homeEloBefore ?? 0) - (b.awayEloBefore ?? 0)) - Math.abs((a.homeEloBefore ?? 0) - (a.awayEloBefore ?? 0)))
   405→      case 'xg': return arr.sort((a, b) => (b.homeXg + b.awayXg) - (a.homeXg + a.awayXg))
   406→      default: return arr
   407→    }
   408→  }, [matchesData, sortBy])
   409→
   410→  const allMatches = matchesData
   411→  const liveCount = allMatches.filter((m) => m.status === 'live' || m.status === 'halftime').length
   412→  const availableStages = useMemo(() => STAGES.filter((s) => allMatches.some((m) => m.stage === s)), [allMatches])
   413→  const availableGroups = useMemo(() => GROUPS.filter((g) => allMatches.some((m) => m.group === g)), [allMatches])
   414→
   415→  return (
   416→    <div className="flex flex-col gap-4 animate-fade-in-up">
   417→      {/* Header */}
   418→      <div className="flex items-center gap-3">
