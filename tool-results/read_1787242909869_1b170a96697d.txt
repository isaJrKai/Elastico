   361→   361→    } catch (err) {
   362→   362→      console.error('Failed to fetch matches:', err)
   363→   363→    } finally {
   364→   364→      setLoading(false)
   365→   365→      setLastRefresh(new Date())
   366→   366→    }
   367→   367→  }, [activeTab, token])
   368→   368→
   369→   369→  useEffect(() => { fetchMatches() }, [fetchMatches])
   370→   370→
   371→   371→  // No auto-refresh — user can manually refresh
   372→   372→
   373→   373→  // Sort
   374→   374→  const sortedMatches = useMemo(() => {
   375→   375→    const arr = [...matchesData]
   376→   376→    switch (sortBy) {
   377→   377→      case 'elo': return arr.sort((a, b) => Math.abs((b.homeEloBefore ?? 0) - (b.awayEloBefore ?? 0)) - Math.abs((a.homeEloBefore ?? 0) - (a.awayEloBefore ?? 0)))
   378→   378→      case 'xg': return arr.sort((a, b) => (b.homeXg + b.awayXg) - (a.homeXg + a.awayXg))
   379→   379→      default: return arr
   380→   380→    }
   381→   381→  }, [matchesData, sortBy])
   382→   382→
   383→   383→  const allMatches = matchesData
   384→   384→  const liveCount = allMatches.filter((m) => m.status === 'live' || m.status === 'halftime').length
   385→   385→  const availableStages = useMemo(() => STAGES.filter((s) => allMatches.some((m) => m.stage === s)), [allMatches])
   386→   386→  const availableGroups = useMemo(() => GROUPS.filter((g) => allMatches.some((m) => m.group === g)), [allMatches])
   387→   387→
   388→   388→  return (
   389→   389→    <div className="flex flex-col gap-4 animate-fade-in-up">
   390→   390→      {/* Header */}
   391→   391→      <div className="flex items-center gap-3">
   392→   392→        <div className="flex size-10 items-center justify-center rounded-xl bg-primary/15"><Zap className="size-5 text-primary" /></div>
   393→   393→        <div>
   394→   394→          <h1 className="text-2xl font-bold tracking-tight">Matches</h1>
   395→   395→          <p className="text-sm text-muted-foreground">Browse, filter, and predict match outcomes</p>
   396→   396→        </div>
   397→   397→      </div>
   398→   398→
   399→   399→      {/* Filter Bar */}
   400→   400→      <div className="glass-card-premium rounded-xl p-4 space-y-3">
   401→   401→        <div className="flex flex-col sm:flex-row gap-3">
   402→   402→          <div className="relative flex-1">
   403→   403→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   404→   404→            <Input placeholder="Search by team name..." value={searchQuery} onChange={(e) => setSearchQuery(e.target.value)} className="pl-9 h-9 bg-muted/50 border-border text-sm" />
   405→   405→          </div>
   406→   406→          <Select value={stageFilter} onValueChange={setStageFilter}>
   407→   407→            <SelectTrigger className="h-9 w-full sm:w-[160px] bg-muted/50 border-border text-sm">
   408→   408→              <Filter className="size-3.5 mr-1.5 text-muted-foreground" /><SelectValue placeholder="Stage" />
   409→   409→            </SelectTrigger>
   410→   410→            <SelectContent className="glass-card border-border">
   411→   411→              <SelectItem value="all">All Stages</SelectItem>
   412→   412→              {availableStages.map((s) => <SelectItem key={s} value={s}>{s}</SelectItem>)}
   413→   413→            </SelectContent>
   414→   414→          </Select>
   415→   415→          <Select value={groupFilter} onValueChange={setGroupFilter}>
   416→   416→            <SelectTrigger className="h-9 w-full sm:w-[120px] bg-muted/50 border-border text-sm"><SelectValue placeholder="Group" /></SelectTrigger>
   417→   417→            <SelectContent className="glass-card border-border">
   418→   418→              <SelectItem value="all">All Groups</SelectItem>
   419→   419→              {availableGroups.map((g) => <SelectItem key={g} value={g}>Group {g}</SelectItem>)}
   420→   420→            </SelectContent>
   421→   421→          </Select>
   422→   422→          <Select value={sortBy} onValueChange={(v) => setSortBy(v as SortKey)}>
   423→   423→            <SelectTrigger className="h-9 w-full sm:w-[130px] bg-muted/50 border-border text-sm">
   424→   424→              <ArrowUpDown className="size-3.5 mr-1.5 text-muted-foreground" /><SelectValue placeholder="Sort" />
   425→   425→            </SelectTrigger>
   426→   426→            <SelectContent className="glass-card border-border">
   427→   427→              <SelectItem value="date">By Date</SelectItem>
   428→   428→              <SelectItem value="elo">By ELO Diff</SelectItem>
   429→   429→              <SelectItem value="xg">By xG Total</SelectItem>
   430→   430→            </SelectContent>
   431→   431→          </Select>
   432→   432→          <Button variant="outline" size="icon" className="h-9 w-9 shrink-0 border-border bg-muted/50 hover:bg-accent" onClick={() => fetchMatches()} disabled={loading} aria-label="Refresh">
   433→   433→            <RefreshCw className={cn('size-4', loading && 'animate-spin')} />
   434→   434→          </Button>
   435→   435→        </div>
   436→   436→        {/* Group quick-buttons */}
   437→   437→        <div className="flex items-center gap-1.5 flex-wrap">
   438→   438→          <span className="text-[10px] text-muted-foreground mr-1">Groups:</span>
   439→   439→          {GROUPS.map((g) => (
   440→   440→            <button key={g} onClick={() => setGroupFilter(groupFilter === g ? 'all' : g)}
   441→   441→              className={cn('flex size-7 items-center justify-center rounded-md text-[10px] font-bold transition-all',
   442→   442→                groupFilter === g ? 'bg-primary/20 text-primary ring-glow-emerald' : 'bg-muted/30 text-muted-foreground hover:bg-muted/50 hover:text-foreground')}>
   443→   443→              {g}
   444→   444→            </button>
   445→   445→          ))}
   446→   446→        </div>
   447→   447→      </div>
   448→   448→
   449→   449→      {/* Tabs */}
   450→   450→      <Tabs value={activeTab} onValueChange={(v) => setActiveTab(v as MatchTab)} className="w-full">
   451→   451→        <TabsList className="glass-card w-full h-10 bg-muted/30 p-1 rounded-lg">
   452→   452→          {STATUS_TABS.map((tab) => {
   453→   453→            const count = tab.value === 'live' ? liveCount : allMatches.length
   454→   454→            return (
   455→   455→              <TabsTrigger key={tab.value} value={tab.value} className="flex-1 h-8 text-xs font-semibold rounded-md transition-all gap-1.5 data-[state=active]:bg-primary/15 data-[state=active]:text-primary data-[state=active]:shadow-sm">
   456→   456→                {tab.value === 'live' && liveCount > 0 && (
   457→   457→                  <span className="relative flex size-1.5"><span className="absolute inline-flex size-full animate-ping rounded-full bg-red-400 opacity-75" /><span className="relative inline-flex size-1.5 rounded-full bg-red-500" /></span>
   458→   458→                )}
   459→   459→                {tab.label}
   460→   460→                {count > 0 && <span className="ml-0.5 text-[10px] opacity-60">({count})</span>}
   461→   461→              </TabsTrigger>
   462→   462→            )
   463→   463→          })}
   464→   464→        </TabsList>
   465→   465→
   466→   466→        {STATUS_TABS.map((tab) => (
   467→   467→          <TabsContent key={tab.value} value={tab.value} className="mt-4">
   468→   468→            {loading && sortedMatches.length === 0 && (
   469→   469→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   470→   470→                {Array.from({ length: 6 }).map((_, i) => <MatchCardSkeleton key={i} />)}
   471→   471→              </div>
   472→   472→            )}
   473→   473→
   474→   474→            {!loading && sortedMatches.length === 0 && (
   475→   475→              <div className="flex flex-col items-center justify-center py-16 text-center">
   476→   476→                <div className="size-16 rounded-full bg-muted/50 flex items-center justify-center mb-4"><Eye className="size-7 text-muted-foreground" /></div>
   477→   477→                <h3 className="text-sm font-semibold mb-1">No {tab.value === 'all' ? '' : tab.value} matches found</h3>
   478→   478→                <p className="text-xs text-muted-foreground max-w-[260px]">
   479→   479→                  {searchQuery || stageFilter !== 'all' || groupFilter !== 'all' ? 'Try adjusting your filters.' : 'No matches available. Check back later!'}
   480→   480→                </p>
   481→   481→                {(searchQuery || stageFilter !== 'all' || groupFilter !== 'all') && (
   482→   482→                  <Button variant="outline" size="sm" className="mt-4 h-8 text-xs border-border" onClick={() => { setSearchQuery(''); setStageFilter('all'); setGroupFilter('all') }}>Clear Filters</Button>
   483→   483→                )}
   484→   484→              </div>
   485→   485→            )}
   486→   486→
   487→   487→            {sortedMatches.length > 0 && (
   488→   488→              <>
   489→   489→                <div className="flex items-center justify-between mb-2 px-1">
   490→   490→                  <span className="text-[11px] text-muted-foreground">{sortedMatches.length} match{sortedMatches.length !== 1 ? 'es' : ''}{loading && ' (refreshing...)'}</span>
   491→   491→                  <span className="text-[10px] text-muted-foreground/50">Updated {lastRefresh.toLocaleTimeString()}</span>
   492→   492→                </div>
   493→   493→                <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   494→   494→                  {sortedMatches.map((match, idx) => (
   495→   495→                    <div key={match.id} style={{ animationDelay: `${Math.min(idx * 50, 300)}ms` }}><MatchCard match={match} /></div>
   496→   496→                  ))}
   497→   497→                </div>
   498→   498→              </>
   499→   499→            )}
   500→   500→          </TabsContent>
   501→   501→        ))}
   502→   502→      </Tabs>
   503→   503→    </div>
   504→   504→  )
