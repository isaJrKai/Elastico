    21→// ── Helper functions ──
    22→function h1(text) {
    23→  return new Paragraph({
    24→    heading: HeadingLevel.HEADING_1,
    25→    spacing: { before: 480, after: 200 },
    26→    children: [new TextRun({ text, bold: true, size: 32, color: "0A1628", font: { ascii: "Times New Roman" } })],
    27→  });
    28→}

    30→  return new Paragraph({
    31→    heading: HeadingLevel.HEADING_2,
    32→    spacing: { before: 360, after: 160 },
    33→    children: [new TextRun({ text, bold: true, size: 28, color: "0A1628", font: { ascii: "Times New Roman" } })],

    43→function body(text) {

    48→}
    49→function bodyBold(label, text) {
    50→  return new Paragraph({
    51→    spacing: { after: 100, line: 312 },
    52→    children: [

    56→  });
    57→}

    59→  return new Paragraph({
    60→    spacing: { after: 60, line: 312 },

    67→}

    73→function specTable(rows) {
    74→  return new Table({
    75→    width: { size: 100, type: WidthType.PERCENTAGE },
    76→    layout: TableLayoutType.FIXED,
    77→    borders: {
    78→      top: { style: BorderStyle.SINGLE, size: 2, color: c(PAL.tblLine) },
    79→      bottom: { style: BorderStyle.SINGLE, size: 2, color: c(PAL.tblLine) },
    80→      left: { style: BorderStyle.NONE }, right: { style: BorderStyle.NONE },
    81→      insideHorizontal: { style: BorderStyle.SINGLE, size: 1, color: c(PAL.tblInner) },
    82→      insideVertical: { style: BorderStyle.NONE },
    83→    },
    84→    rows: rows.map((r, i) =>
    85→      new TableRow({
    86→        children: [
    87→          new TableCell({
    88→            width: { size: 30, type: WidthType.PERCENTAGE },
    89→            shading: { type: ShadingType.CLEAR, fill: i === 0 ? c(PAL.tblHeaderBg) : (i % 2 === 0 ? c(PAL.tblSurface) : "FFFFFF") },
    90→            borders: noBorders,
    91→            margins: { top: 60, bottom: 60, left: 120, right: 120 },
    92→            children: [new Paragraph({ children: [new TextRun({ text: r[0], bold: i === 0, size: 21, color: i === 0 ? c(PAL.tblHeaderText) : "0A1628", font: { ascii: "Calibri" } })] })],
    93→          }),
    94→          new TableCell({

    96→            shading: { type: ShadingType.CLEAR, fill: i === 0 ? c(PAL.tblHeaderBg) : (i % 2 === 0 ? c(PAL.tblSurface) : "FFFFFF") },
    97→            borders: noBorders,
    98→            margins: { top: 60, bottom: 60, left: 120, right: 120 },

   100→          }),
   101→        ],
   102→      })
   103→    ),
   104→  });

   107→// ── Screen section generator ──
   108→function screenSection(num, title, refFile, data) {
   109→  const rows = [
   110→    ["Specification", "Detail"],
   111→    ["Reference Image", `Reference ${String(num).padStart(2, "0")} (${refFile})`],
   112→    ["ELASTICO Screen", data.screen],
   113→    ["Existing Component", data.component],
   114→    ["Data Source (API/Store)", data.dataSource],
   115→    ["Assets to Source", data.assets],
   116→  ];
   117→  return [
   118→    h1(`Reference ${String(num).padStart(2, "0")}: ${title}`),
   119→    bodyBold("Purpose: ", data.purpose),
   120→    bodyBold("Primary Question: ", data.primaryQuestion),
   121→    bodyBold("Primary Information: ", data.primaryInfo),
   122→    bodyBold("Primary Action: ", data.primaryAction),
   123→    spacer(80),
   124→    specTable(rows),
   125→    spacer(60),
   126→    h2("What to Preserve from Reference"),
   127→    ...data.preserve.map((t) => bullet(t)),
   128→    h2("What to Change"),
   129→    ...data.change.map((t) => bullet(t)),
   130→    h2("What Information Belongs Here"),
   131→    ...data.infoBelongs.map((t) => bullet(t)),
   132→    h2("Which Existing ELASTICO Component Supplies It"),
   133→    body(data.componentDetail),
   134→    h2("Which API/Store Supplies the Data"),
   135→    body(data.dataDetail),
   136→    h2("Which Assets Should Be Sourced"),
   137→    ...data.assetsDetail.map((t) => bullet(t)),
   138→    h2("What Must NOT Be Copied"),
   139→    ...data.mustNotCopy.map((t) => bullet(t)),
   140→    h2("1366 x 768 Composition"),
   141→    body(data.composition),
   142→    h2("Responsive Behavior"),
   143→    body(data.responsive),
   144→    h2("Acceptance Criteria"),
   145→    ...data.acceptance.map((t) => bullet(t)),
   146→  ];
   147→}

   721→function calcTitleLayout(title, maxWidth, preferredPt = 40, minPt = 24) {
   722→  const CHARS_PER_PT = 0.52;
   723→  let pt = preferredPt;
   724→  while (pt > minPt) {
   725→    const charsPerLine = Math.floor(maxWidth / (pt * 14.6));
   726→    if (charsPerLine <= 0) { pt--; continue; }
   727→    const lines = Math.ceil(title.length / charsPerLine);
   728→    if (lines <= 3) return { titlePt: pt, titleLines: splitTitleLines(title, charsPerLine) };
   729→    pt--;
   730→  }
   731→  return { titlePt: minPt, titleLines: [title] };
   732→}
   733→function splitTitleLines(title, charsPerLine) {
   734→  const breakAfter = new Set(" ,.!?;:-/()[]{}".split(""));
   735→  const lines = [];
   736→  let remaining = title;
   737→  while (remaining.length > 0) {
   738→    if (remaining.length <= charsPerLine) { lines.push(remaining.trim()); break; }
   739→    let breakAt = -1;
   740→    for (let i = Math.min(charsPerLine, remaining.length); i > charsPerLine * 0.7; i--) {
   741→      if (breakAfter.has(remaining[i - 1])) { breakAt = i; break; }
   742→    }
   743→    if (breakAt === -1) breakAt = charsPerLine;
   744→    lines.push(remaining.slice(0, breakAt).trim());
   745→    remaining = remaining.slice(breakAt).trim();
   746→  }
   747→  return lines;
   748→}
   749→function calcCoverSpacing(p) {
   750→  const SAFETY = 1200;
   751→  const usable = 16838 - (p.marginTop || 0) - (p.marginBottom || 0) - SAFETY;
   752→  const contentH =
   753→    (p.titleLineCount || 1) * ((p.titlePt || 36) * 23 + 200) +
   754→    (p.hasSubtitle ? (12 * 23 + 600) : 0) +
   755→    (p.metaLineCount || 0) * (10 * 23 + 100) +
   756→    (p.fixedHeight || 800) + 900;
   757→  const remaining = Math.max(usable - contentH, 400);
   758→  const rawTop = Math.floor(remaining * 0.45);
   759→  const rawBottom = Math.floor(remaining * 0.45);
   760→  return { topSpacing: Math.max(rawTop, 400), bottomSpacing: Math.max(rawBottom, 800) };
   761→}
