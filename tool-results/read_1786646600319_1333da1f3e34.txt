    43→function computeEloProb(homeElo: number, awayElo: number, outcome: string): number {
    44→  const expectedHome = 1 / (1 + Math.pow(10, (awayElo - homeElo) / 400))
    45→  if (outcome === 'home') return Math.round(expectedHome * 100)
    46→  if (outcome === 'away') return Math.round((1 - expectedHome) * 100 * 0.75)
    47→  return Math.round(100 - expectedHome * 100 - (1 - expectedHome) * 100 * 0.75)
    48→}

    50→function StatBarRow({ label, homeValue, awayValue, homeColor, awayColor, suffix = '', decimals = 0 }: { label: string; homeValue: number | null; awayValue: number | null; homeColor?: string; awayColor?: string; suffix?: string; decimals?: number }) {
    51→  if (homeValue === null || awayValue === null) {
    52→    return (
    53→      <div className="space-y-1.5">
    54→        <div className="flex items-center justify-between text-xs">
    55→          <span className="font-semibold tabular-nums text-muted-foreground">N/A</span>
    56→          <span className="text-[10px] text-muted-foreground font-medium uppercase tracking-wider">{label}</span>
    57→          <span className="font-semibold tabular-nums text-muted-foreground">N/A</span>
    58→        </div>
    59→        <div className="flex h-2 rounded-full overflow-hidden bg-muted">
    60→          <div className="h-full w-1/2 rounded-full bg-muted-foreground/20" />
    61→          <div className="h-full w-1/2 rounded-full bg-muted-foreground/20 ml-auto" />
    62→        </div>
    63→      </div>
    64→    )
    65→  }
    66→  const total = homeValue + awayValue || 1
    67→  const hp = (homeValue / total) * 100
    68→  const fmt = (v: number) => decimals > 0 ? v.toFixed(decimals) : String(Math.round(v))
    69→  return (
    70→    <div className="space-y-1.5">
    71→      <div className="flex items-center justify-between text-xs">
    72→        <span className={cn('font-semibold tabular-nums', hp >= 50 ? 'text-foreground' : 'text-muted-foreground')}>{fmt(homeValue)}{suffix}</span>
    73→        <span className="text-[10px] text-muted-foreground font-medium uppercase tracking-wider">{label}</span>
    74→        <span className={cn('font-semibold tabular-nums', (100 - hp) > hp ? 'text-foreground' : 'text-muted-foreground')}>{fmt(awayValue)}{suffix}</span>
    75→      </div>
    76→      <div className="flex h-2 gap-0.5 rounded-full overflow-hidden bg-muted">
    77→        <div className="h-full rounded-full transition-all duration-700 ease-out" style={{ width: `${hp}%`, backgroundColor: homeColor || 'oklch(0.72 0.19 163)' }} />
    78→        <div className="h-full rounded-full transition-all duration-700 ease-out ml-auto" style={{ width: `${100 - hp}%`, backgroundColor: awayColor || 'oklch(0.68 0.15 230)' }} />
    79→      </div>
    80→    </div>
    81→  )
    82→}

   111→// Fallback: generate shot positions from match events when no deep data source
   112→function generateShotsFromEvents(events: MatchEvent[], homeXg: number, awayXg: number): ShotMapPoint[] {
   113→  const shots: ShotMapPoint[] = []
   114→  const homeGoals = events.filter(e => e.type === 'goal' && e.team === 'home')
   115→  const awayGoals = events.filter(e => e.type === 'goal' && e.team === 'away')
   116→  for (let i = 0; i < Math.ceil(homeXg * 3); i++) {
   117→    shots.push({ x: 15 + (i * 17 % 30), y: 10 + (i * 23 % 80), team: 'home', goal: i < homeGoals.length })
   118→  }
   119→  for (let i = 0; i < Math.ceil(awayXg * 3); i++) {
   120→    shots.push({ x: 55 + (i * 17 % 30), y: 10 + (i * 23 % 80), team: 'away', goal: i < awayGoals.length })
   121→  }
   122→  return shots
   123→}
