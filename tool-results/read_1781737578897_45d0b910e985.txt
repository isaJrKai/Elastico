    46→function getStatusConfig(status: string) {
    47→  switch (status) {
    48→    case 'live': return { label: 'LIVE', cls: 'bg-red-500/15 text-red-400 border-red-500/30', pulse: true }
    49→    case 'halftime': return { label: 'HT', cls: 'bg-yellow-500/15 text-yellow-400 border-yellow-500/30', pulse: false }
    50→    case 'finished': return { label: 'FT', cls: 'bg-muted text-muted-foreground border-border', pulse: false }
    51→    case 'postponed': return { label: 'PPD', cls: 'bg-orange-500/15 text-orange-400 border-orange-500/30', pulse: false }
    52→    default: return { label: 'Upcoming', cls: 'bg-primary/15 text-primary border-primary/30', pulse: false }
    53→  }
    54→}
    55→
    56→function formatMatchDate(dateStr: string | null): string {
    57→  if (!dateStr) return 'TBD'
    58→  return new Date(dateStr).toLocaleDateString('en-US', { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' })
    59→}

   318→// MAIN MATCHES VIEW
   319→// ═══════════════════════════════════════════════════════════════════════════════
   320→
   321→export function MatchesView() {
   322→  const token = useElasticoStore(s => s.token)
   323→  const [activeTab, setActiveTab] = useState<MatchTab>('all')
   324→  const [searchQuery, setSearchQuery] = useState('')
   325→  const [stageFilter, setStageFilter] = useState('all')
   326→  const [groupFilter, setGroupFilter] = useState('all')
   327→  const [sortBy, setSortBy] = useState<SortKey>('date')
   328→  const [matchesData, setMatchesData] = useState<Match[]>([])
   329→  const [loading, setLoading] = useState(true)
   330→  const [lastRefresh, setLastRefresh] = useState(new Date())
   331→
   332→  const fetchMatches = useCallback(async () => {
   333→    setLoading(true)
   334→    try {
   335→      const params = new URLSearchParams()
   336→      if (activeTab !== 'all') params.set('status', activeTab)
   337→      if (searchQuery) params.set('search', searchQuery)
   338→      if (stageFilter !== 'all') params.set('stage', stageFilter)
   339→      if (groupFilter !== 'all') params.set('group', groupFilter)
   340→      const headers: Record<string, string> = {}
   341→      if (token) headers['Authorization'] = `Bearer ${token}`
   342→      const res = await fetch(`/api/matches?${params.toString()}`, { headers })
   343→      const data: MatchesApiResponse = await res.json()
   344→      setMatchesData(data.matches || [])
   345→    } catch (err) {
   346→      console.error('Failed to fetch matches:', err)
   347→    } finally {
   348→      setLoading(false)
   349→      setLastRefresh(new Date())
   350→    }
   351→  }, [activeTab, searchQuery, stageFilter, groupFilter, token])
   352→
   353→  useEffect(() => { fetchMatches() }, [fetchMatches])
   354→
   355→  // Auto-refresh live
   356→  useEffect(() => {
   357→    const iv = setInterval(() => { if (activeTab === 'live') fetchMatches() }, 15000)
   358→    return () => clearInterval(iv)
   359→  }, [activeTab, fetchMatches])
   360→
   361→  // Sort
   362→  const sortedMatches = useMemo(() => {
   363→    const arr = [...matchesData]
   364→    switch (sortBy) {
   365→      case 'elo': return arr.sort((a, b) => Math.abs((b.homeEloBefore ?? 0) - (b.awayEloBefore ?? 0)) - Math.abs((a.homeEloBefore ?? 0) - (a.awayEloBefore ?? 0)))
   366→      case 'xg': return arr.sort((a, b) => (b.homeXg + b.awayXg) - (a.homeXg + a.awayXg))
   367→      default: return arr
   368→    }
   369→  }, [matchesData, sortBy])
   370→
   371→  const allMatches = matchesData
   372→  const liveCount = allMatches.filter((m) => m.status === 'live' || m.status === 'halftime').length
   373→  const availableStages = useMemo(() => STAGES.filter((s) => allMatches.some((m) => m.stage === s)), [allMatches])
   374→  const availableGroups = useMemo(() => GROUPS.filter((g) => allMatches.some((m) => m.group === g)), [allMatches])
   375→
   376→  return (
   377→    <div className="flex flex-col gap-4 animate-fade-in-up">
   378→      {/* Header */}
   379→      <div className="flex items-center gap-3">
   380→        <div className="flex size-10 items-center justify-center rounded-xl bg-primary/15"><Zap className="size-5 text-primary" /></div>
   381→        <div>
   382→          <h1 className="text-2xl font-bold tracking-tight">Matches</h1>
   383→          <p className="text-sm text-muted-foreground">Browse, filter, and predict match outcomes</p>
   384→        </div>
   385→      </div>
   386→
   387→      {/* Filter Bar */}
   388→      <div className="glass-card-premium rounded-xl p-4 space-y-3">
   389→        <div className="flex flex-col sm:flex-row gap-3">
   390→          <div className="relative flex-1">
   391→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   392→            <Input placeholder="Search by team name..." value={searchQuery} onChange={(e) => setSearchQuery(e.target.value)} className="pl-9 h-9 bg-muted/50 border-border text-sm" />
   393→          </div>
   394→          <Select value={stageFilter} onValueChange={setStageFilter}>
   395→            <SelectTrigger className="h-9 w-full sm:w-[160px] bg-muted/50 border-border text-sm">
   396→              <Filter className="size-3.5 mr-1.5 text-muted-foreground" /><SelectValue placeholder="Stage" />
   397→            </SelectTrigger>
   398→            <SelectContent className="glass-card border-border">
   399→              <SelectItem value="all">All Stages</SelectItem>
   400→              {availableStages.map((s) => <SelectItem key={s} value={s}>{s}</SelectItem>)}
   401→            </SelectContent>
   402→          </Select>
   403→          <Select value={groupFilter} onValueChange={setGroupFilter}>
   404→            <SelectTrigger className="h-9 w-full sm:w-[120px] bg-muted/50 border-border text-sm"><SelectValue placeholder="Group" /></SelectTrigger>
   405→            <SelectContent className="glass-card border-border">
   406→              <SelectItem value="all">All Groups</SelectItem>
   407→              {availableGroups.map((g) => <SelectItem key={g} value={g}>Group {g}</SelectItem>)}
   408→            </SelectContent>
   409→          </Select>
   410→          <Select value={sortBy} onValueChange={(v) => setSortBy(v as SortKey)}>
