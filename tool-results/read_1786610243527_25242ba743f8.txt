    43→function computeEloProb(homeElo: number, awayElo: number, outcome: string): number {
    44→  const expectedHome = 1 / (1 + Math.pow(10, (awayElo - homeElo) / 400))
    45→  if (outcome === 'home') return Math.round(expectedHome * 100)
    46→  if (outcome === 'away') return Math.round((1 - expectedHome) * 100 * 0.75)
    47→  return Math.round(100 - expectedHome * 100 - (1 - expectedHome) * 100 * 0.75)
    48→}

    50→function StatBarRow({ label, homeValue, awayValue, homeColor, awayColor, suffix = '', decimals = 0 }: { label: string; homeValue: number; awayValue: number; homeColor?: string; awayColor?: string; suffix?: string; decimals?: number }) {
    51→  const total = homeValue + awayValue || 1
    52→  const hp = (homeValue / total) * 100
    53→  const fmt = (v: number) => decimals > 0 ? v.toFixed(decimals) : String(Math.round(v))
    54→  return (
    55→    <div className="space-y-1.5">
    56→      <div className="flex items-center justify-between text-xs">
    57→        <span className={cn('font-semibold tabular-nums', hp >= 50 ? 'text-foreground' : 'text-muted-foreground')}>{fmt(homeValue)}{suffix}</span>
    58→        <span className="text-[10px] text-muted-foreground font-medium uppercase tracking-wider">{label}</span>
    59→        <span className={cn('font-semibold tabular-nums', (100 - hp) > hp ? 'text-foreground' : 'text-muted-foreground')}>{fmt(awayValue)}{suffix}</span>
    60→      </div>
    61→      <div className="flex h-2 gap-0.5 rounded-full overflow-hidden bg-muted">
    62→        <div className="h-full rounded-full transition-all duration-700 ease-out" style={{ width: `${hp}%`, backgroundColor: homeColor || 'oklch(0.72 0.19 163)' }} />
    63→        <div className="h-full rounded-full transition-all duration-700 ease-out ml-auto" style={{ width: `${100 - hp}%`, backgroundColor: awayColor || 'oklch(0.68 0.15 230)' }} />
    64→      </div>
    65→    </div>
    66→  )
    67→}

    95→// Fallback: generate shot positions from match events when no deep data source
    96→function generateShotsFromEvents(events: MatchEvent[], homeXg: number, awayXg: number): ShotMapPoint[] {
    97→  const shots: ShotMapPoint[] = []
    98→  const homeGoals = events.filter(e => e.type === 'goal' && e.team === 'home')
    99→  const awayGoals = events.filter(e => e.type === 'goal' && e.team === 'away')
   100→  for (let i = 0; i < Math.ceil(homeXg * 3); i++) {
   101→    shots.push({ x: 15 + (i * 17 % 30), y: 10 + (i * 23 % 80), team: 'home', goal: i < homeGoals.length })
   102→  }
   103→  for (let i = 0; i < Math.ceil(awayXg * 3); i++) {
   104→    shots.push({ x: 55 + (i * 17 % 30), y: 10 + (i * 23 % 80), team: 'away', goal: i < awayGoals.length })
   105→  }
   106→  return shots
   107→}
