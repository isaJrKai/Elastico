    22→    22→// ── Security Authentication Middleware ───────────────────────────────────────
    23→    23→export function authMiddleware(req: AuthenticatedRequest, res: Response, next: NextFunction) {
    24→    24→  const header = req.headers.authorization;
    25→    25→  const qToken = req.query.token as string | undefined;
    26→    26→  const raw = header?.startsWith("Bearer ") ? header.slice(7) : qToken;
    27→    27→
    28→    28→  if (!raw) {
    29→    29→    return res.status(401).json({ error: "Authorization credentials required" });
    30→    30→  }
    31→    31→
    32→    32→  try {
    33→    33→    const decoded = jwt.verify(raw, JWT_SECRET) as { id: number; sessionId?: string };
    34→    34→    const user = db.users.find((u) => u.id === decoded.id && u.is_active === 1);
    35→    35→
    36→    36→    if (!user) {
    37→    37→      return res.status(401).json({ error: "Authenticated user not found or suspended" });
    38→    38→    }
    39→    39→
    40→    40→    // Strict single active login session security check
    41→    41→    if (decoded.sessionId && user.session_id && user.session_id !== decoded.sessionId) {
    42→    42→      return res.status(401).json({
    43→    43→        error: "Multiple logins detected. This active session has been terminated because the account was logged in from another device or browser tab.",
    44→    44→        code: "CONCURRENT_LOGIN_TERMINATION"
    45→    45→      });
    46→    46→    }
    47→    47→
    48→    48→    // Force owner email to be ROOT ADMIN with elite billing override
    49→    49→    const activeAdminEmail = "kaisoisaac@gmail.com";
    50→    50→    if (user.email?.toLowerCase() === activeAdminEmail) {
    51→    51→      user.role = "admin";
    52→    52→      user.plan = "elite";
    53→    53→    } else {
    54→    54→      // Strictly downgrade any unauthorized admin role to user
    55→    55→      if (user.role === "admin") {
    56→    56→        user.role = "user";
    57→    57→        if (user.plan === "elite" && !db.whitelisted_free_emails?.includes(user.email?.toLowerCase() || "")) {
    58→    58→          user.plan = "free";
    59→    59→        }
    60→    60→      }
    61→    61→    }
    62→    62→
    63→    63→    // Direct Whitelisted Manual free promotional bypass
    64→    64→    if (user.email && db.whitelisted_free_emails?.includes(user.email.toLowerCase())) {
    65→    65→      user.plan = "elite";
    66→    66→    }
    67→    67→
    68→    68→    req.user = user;
    69→    69→    next();
    70→    70→  } catch (err) {
    71→    71→    return res.status(401).json({
    72→    72→      error: err instanceof jwt.TokenExpiredError ? "Token expired" : "Invalid token",
    73→    73→    });
    74→    74→  }
    75→    75→}
