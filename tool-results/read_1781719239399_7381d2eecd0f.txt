    60→function StatCard({ title, value, icon: Icon, change, color = 'emerald', suffix = '' }: {
    61→  title: string; value: string | number; icon: any; change?: number; color?: string; suffix?: string
    62→}) {
    63→  const isPositive = change !== undefined && change >= 0
    64→  const colorMap: Record<string, string> = {

    69→    purple: 'bg-purple-500/10 text-purple-400',

    71→  }
    72→  return (
    73→    <Card className="glass-card glass-card-hover animate-fade-in-up">
    74→      <CardContent className="p-4">
    75→        <div className="flex items-center justify-between">
    76→          <div className="space-y-1 min-w-0">
    77→            <p className="text-xs text-muted-foreground uppercase tracking-wider flex items-center gap-1.5">
    78→              <Icon className="h-3 w-3" />

    99→function GaugeBar({ value, max = 100, label, color = '#00e676' }: { value: number; max?: number; label: string; color?: string }) {
   100→  const pct = Math.min((value / max) * 100, 100)
   101→  return (
   102→    <div className="space-y-1.5">
   103→      <div className="flex items-center justify-between text-xs">
   104→        <span className="text-muted-foreground">{label}</span>
   105→        <span className="font-mono font-bold" style={{ color }}>{value}{max === 100 ? '%' : ''}</span>
   106→      </div>
   107→      <div className="h-2 bg-muted/50 rounded-full overflow-hidden">
   108→        <div className="h-full rounded-full transition-all duration-1000" style={{ width: `${pct}%`, backgroundColor: color }} />
   109→      </div>
   110→    </div>
   111→  )
