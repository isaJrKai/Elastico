    63→    63→interface EnhancedPlayer extends Player {
    64→    64→  teamName?: string
    65→    65→  teamCode?: string

    71→    71→// ── Helpers ────────────────────────────────────────────────────────────────────
    72→    72→
    73→    73→function generateRadarStats(p: EnhancedPlayer) {
    74→    74→  return {
    75→    75→    Pace: p.position === 'FWD' || p.position === 'MID' ? 75 : 55,
    76→    76→    Shooting: p.position === 'FWD' ? 75 : 50,
    77→    77→    Passing: p.position === 'MID' ? 75 : 55,
    78→    78→    Defending: p.position === 'DEF' ? 75 : 45,
    79→    79→    Physical: 65,
    80→    80→    Dribbling: p.position === 'FWD' || p.position === 'MID' ? 72 : 50,
    81→    81→  }
    82→    82→}

   101→   101→    async function fetchPlayers() {

   103→   103→        const headers: Record<string, string> = {}

   108→   108→        if (search) params.set('search', search)

   111→   111→        if (res.ok) {
   112→   112→          const data = await res.json()

   114→   114→            setPlayers(data.players.map((p: Record<string, unknown>) => ({
   115→   115→              ...p,
   116→   116→              teamName: (p.team as Record<string, unknown>)?.name as string || p.teamName,
   117→   117→              teamCode: (p.team as Record<string, unknown>)?.code as string || p.teamCode,
   118→   118→              teamColor: (p.team as Record<string, unknown>)?.primaryColor as string || '#00e676',

   120→   120→            return
   121→   121→          }
   122→   122→        }
   123→   123→      } catch {
   124→   124→        // DB empty — fallback to ESPN
   125→   125→      }
   126→   126→
   127→   127→      // ESPN fallback: fetch teams for selected league, then fetch rosters
   128→   128→      try {
   129→   129→        const leagueCode = teamFilter !== 'all' ? undefined : 'PL'
   130→   130→        const teamsRes = await fetch(`/api/live?action=teams&league=${teamFilter !== 'all' ? teamFilter : 'PL'}`)
   131→   131→        const teamsData = await teamsRes.json()
   132→   132→        const espnTeams = teamsData.data || teamsData.teams || []
   133→   133→
   134→   134→        if (espnTeams.length === 0) return
   135→   135→
   136→   136→        // Fetch rosters for up to 6 teams (to avoid too many requests)
   137→   137→        const allPlayers: EnhancedPlayer[] = []
   138→   138→        const teamsToFetch = espnTeams.slice(0, 6)
   139→   139→
   140→   140→        await Promise.allSettled(teamsToFetch.map(async (t: any) => {
   141→   141→          try {
   142→   142→            const rosterRes = await fetch(`/api/live?action=roster&league=PL&team=${t.id}`)
   143→   143→            if (!rosterRes.ok) return
   144→   144→            const rosterData = await rosterRes.json()

   146→   146→            for (const a of athletes) {
   147→   147→              allPlayers.push({
   148→   148→                id: String(a.id || ''),
   149→   149→                name: a.name || a.displayName || 'Unknown',
   150→   150→                position: a.position || 'MID',
   151→   151→                age: a.age || 25,
   152→   152→                nationality: a.nationality || '',
   153→   153→                rating: 0, // ESPN roster API does not provide match ratings
   154→   154→                goals: a.goals || 0,
   155→   155→                assists: a.assists || 0,
   156→   156→                teamName: t.name,
   157→   157→                teamCode: t.abbreviation || t.code,
   158→   158→                teamColor: t.color || '#00e676',
   159→   159→                teamId: String(t.id || ''),
   160→   160→                appearances: a.appearances || 0,
   161→   161→                minutesPlayed: 0,
   162→   162→                shirtNumber: a.shirtNumber || a.jersey || 0,
   163→   163→              } as unknown as EnhancedPlayer)
   164→   164→            }
   165→   165→          } catch { /* skip this team */ }
   166→   166→        }))
   167→   167→
   168→   168→        if (allPlayers.length > 0) setPlayers(allPlayers)
   169→   169→      } catch {
   170→   170→        // ESPN fallback also failed
   171→   171→      }
