    49→   144→    arr = np.asarray(values, dtype=float)

    55→   150→def _linear_trend(values: list[float] | np.ndarray, default: float = 0.0) -> float:
    56→   151→    """Slope of a simple linear fit (most recent → oldest = positive trend)."""
    57→   152→    arr = np.asarray(values, dtype=float)
    58→   153→    if arr.size < 2:
    59→   154→        return default
    60→   155→    x = np.arange(len(arr), dtype=float)
    61→   156→    valid = np.isfinite(arr)
    62→   157→    if valid.sum() < 2:
    63→   158→        return default
    64→   159→    x = x[valid]
    65→   160→    y = arr[valid]
    66→   161→    slope = np.polyfit(x, y, 1)[0]
    67→   162→    return float(slope) if math.isfinite(slope) else default

    70→   165→def _compute_streaks(
    71→   166→    results: list[str] | list[int] | None,
    72→   167→) -> tuple[int, int, int]:
    73→   168→    """Return (win_streak, unbeaten_streak, loss_streak) from recent results.
    74→   169→
    75→   170→    *results* is ordered **most recent first**.  Each element is one of:
    76→   171→    - 'W' / 1  → win
    77→   172→    - 'D' / 0  → draw
    78→   173→    - 'L' / -1 → loss
    79→   174→    """
    80→   175→    if not results:
    81→   176→        return 0, 0, 0
    82→   177→    win_s = unbeaten_s = loss_s = 0
    83→   178→    for r in results:
    84→   179→        if r in ('W', 1):
    85→   180→            win_s += 1
    86→   181→            unbeaten_s += 1
    87→   182→        elif r in ('D', 0):
    88→   183→            unbeaten_s += 1
    89→   184→            break
    90→   185→        elif r in ('L', -1):
    91→   186→            break
    92→   187→        else:
    93→   188→            break
    94→   189→    for r in results:
    95→   190→        if r in ('L', -1):
    96→   191→            loss_s += 1
    97→   192→        else:
    98→   193→            break
    99→   194→    return win_s, unbeaten_s, loss_s

   102→   197→def _form_points(results: list[str] | list[int] | None) -> float:
   103→   198→    """Sum of 3/1/0 for W/D/L over recent results (max 15 for 5 matches)."""
   104→   199→    if not results:
   105→   200→        return 0.0
   106→   201→    mapping = {'W': 3, 'D': 1, 'L': 0, 1: 3, 0: 1, -1: 0}
   107→   202→    return sum(mapping.get(r, 0) for r in results)

   111→   206→    """Coerce *value* to a ``datetime``.  Returns ``None`` on failure."""
   112→   207→    if isinstance(value, datetime):
   113→   208→        return value
   114→   209→    if isinstance(value, str):
   115→   210→        for fmt in ("%Y-%m-%d", "%Y-%m-%dT%H:%M:%S", "%Y-%m-%d %H:%M:%S"):
   116→   211→            try:
   117→   212→                return datetime.strptime(value, fmt)

   121→   216→        try:

   229→   324→    def build_training_dataset(
   230→   325→        self,
   231→   326→        matches: list[dict[str, Any]],
   232→   327→        window: int = 10,
   233→   328→    ) -> pd.DataFrame:
   234→   329→        """Build a training dataset from historical matches.
   235→   330→
   236→   331→        Each match in *matches* must contain:
   237→   332→
   238→   333→        - ``home_team``, ``away_team``
   239→   334→        - ``home_goals``, ``away_goals``  (for target + stats)
   240→   335→        - ``date``
   241→   336→        - ``home_odds``, ``draw_odds``, ``away_odds``  (optional)
   242→   337→
   243→   338→        The method computes a sliding-window of *window* previous matches
   244→   339→        per team to derive form, streak, and trend features.  Missing
   245→   340→        historical data is back-filled with league-average defaults.
   246→   341→
   247→   342→        Parameters
   248→   343→        ----------
   249→   344→        matches : list[dict]
   250→   345→            Historical match records.  Must be **chronologically sorted**
   251→   346→            (oldest first) for correct sliding-window behaviour.
   252→   347→        window : int
   253→   348→            Number of recent matches to use for rolling form stats.
   254→   349→
   255→   350→        Returns
   256→   351→        -------
   257→   352→        pd.DataFrame
   258→   353→            One row per match.  Columns are the 50 features plus
   259→   354→            ``target`` (0=away, 1=draw, 2=home).
   260→   355→        """
   261→   356→        if not matches:
   262→   357→            return pd.DataFrame(columns=[*FEATURE_NAMES, "target"])
   263→   358→
   264→   359→        # Index: team_id → list of (date, goals_scored, goals_conceded, result)
   265→   360→        team_history: dict[str, list[tuple]] = defaultdict(list)
   266→   361→        # team_id → list of home-only / away-only records
   267→   362→        team_home_history: dict[str, list[tuple]] = defaultdict(list)
   268→   363→        team_away_history: dict[str, list[tuple]] = defaultdict(list)
   269→   364→
   270→   365→        rows: list[dict[str, float]] = []
   271→   366→
   272→   367→        for match in matches:
   273→   368→            home_team = str(match.get("home_team", ""))
   274→   369→            away_team = str(match.get("away_team", ""))
   275→   370→            hg = int(match.get("home_goals", 0))
   276→   371→            ag = int(match.get("away_goals", 0))
   277→   372→
   278→   373→            # Derive result codes (from home perspective)
   279→   374→            if hg > ag:
   280→   375→                home_result, away_result = "W", "L"
   281→   376→            elif hg < ag:
   282→   377→                home_result, away_result = "L", "W"
   283→   378→            else:
   284→   379→                home_result, away_result = "D", "D"
   285→   380→
   286→   381→            match_date = _parse_date(match.get("date")) or datetime.now()
   287→   382→
   288→   383→            # ── Build per-team stats dicts from rolling history ───────
   289→   384→            home_stats = self._stats_from_history(
   290→   385→                team_history[home_team],
   291→   386→                team_home_history[home_team],
   292→   387→                team_away_history[home_team],
   293→   388→                window=window,
   294→   389→            )
   295→   390→            away_stats = self._stats_from_history(
   296→   391→                team_history[away_team],
   297→   392→                team_home_history[away_team],
   298→   393→                team_away_history[away_team],
   299→   394→                window=window,
   300→   395→            )
   301→   396→
   302→   397→            # Set last-match date for rest-day calculation
   303→   398→            if team_history[home_team]:
   304→   399→                home_stats["last_match_date"] = team_history[home_team][-1][0].isoformat()
   305→   400→            if team_history[away_team]:
   306→   401→                away_stats["last_match_date"] = team_history[away_team][-1][0].isoformat()
   307→   402→
   308→   403→            # ── Head-to-head (use previous matches between these teams) ─
   309→   404→            h2h: list[dict[str, Any]] = []
   310→   405→            for prev in matches:
   311→   406→                if prev is match:
   312→   407→                    continue
   313→   408→                if (
   314→   409→                    str(prev.get("home_team")) == home_team
   315→   410→                    and str(prev.get("away_team")) == away_team
   316→   411→                ) or (
   317→   412→                    str(prev.get("home_team")) == away_team
   318→   413→                    and str(prev.get("away_team")) == home_team
   319→   414→                ):
   320→   415→                    phg = int(prev.get("home_goals", 0))
   321→   416→                    pag = int(prev.get("away_goals", 0))
   322→   417→                    # Normalise to current home perspective
   323→   418→                    if str(prev.get("home_team")) == home_team:
   324→   419→                        h2h.append({"home_goals": phg, "away_goals": pag})
   325→   420→                    else:
   326→   421→                        h2h.append({"home_goals": pag, "away_goals": phg})
   327→   422→
   328→   423→            # ── Odds ──────────────────────────────────────────────────
   329→   424→            odds: dict[str, Any] | None = None
   330→   425→            if match.get("home_odds") and match.get("draw_odds") and match.get("away_odds"):
   331→   426→                odds = {
   332→   427→                    "home_odds": float(match["home_odds"]),
   333→   428→                    "draw_odds": float(match["draw_odds"]),
   334→   429→                    "away_odds": float(match["away_odds"]),
   335→   430→                }
   336→   431→
   337→   432→            # ── Build features ────────────────────────────────────────
   338→   433→            feats = self.build_match_features(
   339→   434→                match_data={"date": match_date},
   340→   435→                home_stats=home_stats,
   341→   436→                away_stats=away_stats,
   342→   437→                h2h_data=h2h if h2h else None,
   343→   438→                odds=odds,
   344→   439→            )
   345→   440→
   346→   441→            # Target: 0=away, 1=draw, 2=home
   347→   442→            if hg > ag:
   348→   443→                target = 2
   349→   444→            elif hg < ag:
   350→   445→                target = 0
   351→   446→            else:
   352→   447→                target = 1
   353→   448→
   354→   449→            feats["target"] = float(target)
   355→   450→            rows.append(feats)
   356→   451→
   357→   452→            # ── Update histories for future windows ───────────────────
   358→   453→            team_history[home_team].append(
   359→   454→                (match_date, hg, ag, home_result)
   360→   455→            )
   361→   456→            team_history[away_team].append(
   362→   457→                (match_date, ag, hg, away_result)
   363→   458→            )
   364→   459→            team_home_history[home_team].append(
   365→   460→                (match_date, hg, ag, home_result)
   366→   461→            )
   367→   462→            team_away_history[away_team].append(
   368→   463→                (match_date, ag, hg, away_result)
   369→   464→            )
   370→   465→
   371→   466→        df = pd.DataFrame(rows)
   372→   467→        # Ensure column order
   373→   468→        cols = [*FEATURE_NAMES, "target"]
   374→   469→        for c in cols:
   375→   470→            if c not in df.columns:
   376→   471→                df[c] = 0.0
   377→   472→        return df[cols]

   416→   511→    def _goal_features(
   417→   512→        self, home_stats: dict[str, Any], away_stats: dict[str, Any]

   420→   515→

   422→   517→        h_gs_away = float(home_stats.get("goals_scored_away", lag))

   424→   519→        h_gc_home = float(home_stats.get("goals_conceded_home", lag))
   425→   520→        h_gc_away = float(home_stats.get("goals_conceded_away", lag))
   426→   521→        h_gc = float(home_stats.get("goals_conceded", lag))
   427→   522→
   428→   523→        a_gs_home = float(away_stats.get("goals_scored_home", lag))
   429→   524→        a_gs_away = float(away_stats.get("goals_scored_away", lag))
   430→   525→        a_gs = float(away_stats.get("goals_scored", lag))
   431→   526→        a_gc_home = float(away_stats.get("goals_conceded_home", lag))
   432→   527→        a_gc_away = float(away_stats.get("goals_conceded_away", lag))
   433→   528→        a_gc = float(away_stats.get("goals_conceded", lag))
   434→   529→
   435→   530→        gd = h_gs - h_gc
   436→   531→
   437→   532→        xg_h = float(home_stats.get("xg_for_home", lag))

   483→   578→
   484→   579→    def _h2h_features(
   485→   580→        self, h2h_data: list[dict[str, Any]] | None
   486→   581→    ) -> dict[str, float]:
   487→   582→        if not h2h_data:
   488→   583→            return {
   489→   584→                "h2h_home_win_pct": 0.0,
   490→   585→                "h2h_away_win_pct": 0.0,
   491→   586→                "h2h_draw_pct": 0.0,
   492→   587→                "h2h_avg_goals": self.league_avg_goals * 2,
   493→   588→            }
   494→   589→
   495→   590→        n = len(h2h_data)
   496→   591→        home_wins = sum(1 for m in h2h_data if m.get("home_goals", 0) > m.get("away_goals", 0))
   497→   592→        away_wins = sum(1 for m in h2h_data if m.get("home_goals", 0) < m.get("away_goals", 0))
   498→   593→        draws = n - home_wins - away_wins
   499→   594→
   500→   595→        total_goals_list = [m.get("home_goals", 0) + m.get("away_goals", 0) for m in h2h_data]
   501→   596→        avg_goals = _safe_avg(total_goals_list, default=self.league_avg_goals * 2)
   502→   597→
   503→   598→        return {
   504→   599→            "h2h_home_win_pct": _safe_div(home_wins, n),
   505→   600→            "h2h_away_win_pct": _safe_div(away_wins, n),
   506→   601→            "h2h_draw_pct": _safe_div(draws, n),
   507→   602→            "h2h_avg_goals": avg_goals,
   508→   603→        }

   513→   608→        self, home_stats: dict[str, Any], away_stats: dict[str, Any]
   514→   609→    ) -> dict[str, float]:
   515→   610→        lag = self.league_avg_goals
   516→   611→
   517→   612→        # Home team's home record
   518→   613→        hw = int(home_stats.get("home_wins", 0))
   519→   614→        hd = int(home_stats.get("home_draws", 0))
   520→   615→        hl = int(home_stats.get("home_losses", 0))
   521→   616→        h_home_total = hw + hd + hl
   522→   617→        h_home_goals = float(home_stats.get("home_goals", lag))
   523→   618→
   524→   619→        # Away team's away record
   525→   620→        aw = int(away_stats.get("away_wins", 0))
   526→   621→        ad = int(away_stats.get("away_draws", 0))
   527→   622→        al = int(away_stats.get("away_losses", 0))
   528→   623→        a_away_total = aw + ad + al
   529→   624→        a_away_goals = float(away_stats.get("away_goals", lag))
   530→   625→
   531→   626→        return {
   532→   627→            "home_win_pct": _safe_div(hw, h_home_total),
   533→   628→            "away_win_pct": _safe_div(aw, a_away_total),
   534→   629→            "home_goals_avg": h_home_goals,
   535→   630→            "away_goals_avg": a_away_goals,
   536→   631→        }

   538→   633→    # ---------- 6. Market / odds features ------------------------------
   539→   634→
   540→   635→    def _market_features(
   541→   636→        self, odds: dict[str, Any] | None
   542→   637→    ) -> dict[str, float]:
   543→   638→        default_odds = 1.0
   544→   639→        if odds is None:
   545→   640→            return {
   546→   641→                "implied_prob_home": 1.0 / 3.0,
   547→   642→                "implied_prob_draw": 1.0 / 3.0,
   548→   643→                "implied_prob_away": 1.0 / 3.0,
   549→   644→                "overround": 1.0,
   550→   645→            }
   551→   646→
   552→   647→        ho = float(odds.get("home_odds", default_odds))
   553→   648→        do = float(odds.get("draw_odds", default_odds))
   554→   649→        ao = float(odds.get("away_odds", default_odds))
   555→   650→
   556→   651→        # Guard against zero
   557→   652→        ho = max(ho, 1.01)
   558→   653→        do = max(do, 1.01)
   559→   654→        ao = max(ao, 1.01)
   560→   655→
   561→   656→        ip_h = _safe_div(1.0, ho, 1.0 / 3.0)
   562→   657→        ip_d = _safe_div(1.0, do, 1.0 / 3.0)
   563→   658→        ip_a = _safe_div(1.0, ao, 1.0 / 3.0)
   564→   659→        overround = ip_h + ip_d + ip_a
   565→   660→
   566→   661→        return {
   567→   662→            "implied_prob_home": ip_h,
   568→   663→            "implied_prob_draw": ip_d,
   569→   664→            "implied_prob_away": ip_a,
   570→   665→            "overround": overround,
   571→   666→        }

   573→   668→    # ---------- 7. Temporal features -----------------------------------
   574→   669→
   575→   670→    def _temporal_features(
   576→   671→        self,
   577→   672→        match_data: dict[str, Any],
   578→   673→        home_stats: dict[str, Any],
   579→   674→        away_stats: dict[str, Any],
   580→   675→    ) -> dict[str, float]:
   581→   676→        match_date = _parse_date(match_data.get("date")) or datetime.now()
   582→   677→        dow = float(match_date.weekday())
   583→   678→        month = float(match_date.month)
   584→   679→        is_weekend = 1.0 if dow >= 5 else 0.0
   585→   680→
   586→   681→        # Rest days
   587→   682→        h_last = _parse_date(home_stats.get("last_match_date"))
   588→   683→        a_last = _parse_date(away_stats.get("last_match_date"))
   589→   684→
   590→   685→        default_rest = 7.0
   591→   686→        h_rest = default_rest
   592→   687→        a_rest = default_rest
   593→   688→        if h_last:
   594→   689→            delta = (match_date - h_last).days
   595→   690→            h_rest = float(max(delta, 0)) if delta is not None else default_rest
   596→   691→        if a_last:
   597→   692→            delta = (match_date - a_last).days
   598→   693→            a_rest = float(max(delta, 0)) if delta is not None else default_rest
   599→   694→
   600→   695→        return {
   601→   696→            "day_of_week": dow,
   602→   697→            "month": month,
   603→   698→            "is_weekend": is_weekend,
   604→   699→            "days_since_last_match_home": h_rest,
   605→   700→            "days_since_last_match_away": a_rest,
   606→   701→        }

   608→   703→    # ---------- 8. Advanced features -----------------------------------
   609→   704→
   610→   705→    def _advanced_features(
   611→   706→        self, home_stats: dict[str, Any], away_stats: dict[str, Any]
   612→   707→    ) -> dict[str, float]:
   613→   708→        lag = self.league_avg_goals
   614→   709→
   615→   710→        # Attacking / defensive strength (relative to league average)
   616→   711→        h_gs = float(home_stats.get("goals_scored", lag))
   617→   712→        h_gc = float(home_stats.get("goals_conceded", lag))
   618→   713→        a_gs = float(away_stats.get("goals_scored", lag))
   619→   714→        a_gc = float(away_stats.get("goals_conceded", lag))
   620→   715→
   621→   716→        atk_h = _safe_div(h_gs, lag, 1.0)
   622→   717→        atk_a = _safe_div(a_gs, lag, 1.0)
   623→   718→        def_h = _safe_div(h_gc, lag, 1.0)
   624→   719→        def_a = _safe_div(a_gc, lag, 1.0)
   625→   720→
   626→   721→        # Poisson lambda: attack * opponent defence * league average
   627→   722→        poisson_h = atk_h * def_a * lag
   628→   723→        poisson_a = atk_a * def_h * lag
   629→   724→
   630→   725→        # Shot conversion rates
   631→   726→        h_shots = float(home_stats.get("shots_total", 0.0))
   632→   727→        h_sot = float(home_stats.get("shots_on_target", 0.0))
   633→   728→        h_matches = max(float(home_stats.get("matches_played", 1)), 1.0)
   634→   729→        a_shots = float(away_stats.get("shots_total", 0.0))
   635→   730→        a_sot = float(away_stats.get("shots_on_target", 0.0))
   636→   731→        a_matches = max(float(away_stats.get("matches_played", 1)), 1.0)
   637→   732→
   638→   733→        h_conv = _safe_div(h_sot, h_shots, 0.33)
   639→   734→        a_conv = _safe_div(a_sot, a_shots, 0.33)
   640→   735→
   641→   736→        # Clean-sheet percentage
   642→   737→        h_cs = float(home_stats.get("clean_sheets_home", 0.0))
   643→   738→        h_home_m = max(float(home_stats.get("matches_played_home", 1)), 1.0)
   644→   739→        a_cs = float(away_stats.get("clean_sheets_away", 0.0))
   645→   740→        a_away_m = max(float(away_stats.get("matches_played_away", 1)), 1.0)
   646→   741→
   647→   742→        return {
   648→   743→            "attacking_strength_home": atk_h,
   649→   744→            "attacking_strength_away": atk_a,
   650→   745→            "defensive_strength_home": def_h,
   651→   746→            "defensive_strength_away": def_a,
   652→   747→            "poisson_lambda_home": poisson_h,
   653→   748→            "poisson_lambda_away": poisson_a,
   654→   749→            "shot_conversion_rate_home": h_conv,
   655→   750→            "shot_conversion_rate_away": a_conv,
   656→   751→            "clean_sheet_pct_home": _safe_div(h_cs, h_home_m),
   657→   752→            "clean_sheet_pct_away": _safe_div(a_cs, a_away_m),
   658→   753→        }

   663→   758→
   664→   759→    @staticmethod
   665→   760→    def _stats_from_history(
   666→   761→        overall: list[tuple],
   667→   762→        home_only: list[tuple],
   668→   763→        away_only: list[tuple],
   669→   764→        window: int = 10,
   670→   765→    ) -> dict[str, Any]:
   671→   766→        """Derive a stats dict from rolling match history.
   672→   767→
   673→   768→        Each tuple is ``(date, goals_scored, goals_conceded, result_char)``.
   674→   769→        Uses the last *window* entries for form/streak, and full history
   675→   770→        for season averages.
   676→   771→        """
   677→   772→        stats: dict[str, Any] = {}
   678→   773→
   679→   774→        if not overall:
   680→   775→            return stats
   681→   776→
   682→   777→        # Full-season averages (all matches)
   683→   778→        gs_all = [m[1] for m in overall]
   684→   779→        gc_all = [m[2] for m in overall]
   685→   780→        stats["goals_scored"] = _safe_avg(gs_all)
   686→   781→        stats["goals_conceded"] = _safe_avg(gc_all)
   687→   782→        stats["matches_played"] = float(len(overall))
   688→   783→
   689→   784→        # Venue-split averages
   690→   785→        if home_only:
   691→   786→            stats["goals_scored_home"] = _safe_avg([m[1] for m in home_only])
   692→   787→            stats["goals_conceded_home"] = _safe_avg([m[2] for m in home_only])
   693→   788→            stats["home_wins"] = sum(1 for m in home_only if m[3] == "W")
   694→   789→            stats["home_draws"] = sum(1 for m in home_only if m[3] == "D")
   695→   790→            stats["home_losses"] = sum(1 for m in home_only if m[3] == "L")
   696→   791→            stats["home_goals"] = _safe_avg([m[1] for m in home_only])
   697→   792→            stats["matches_played_home"] = float(len(home_only))
   698→   793→            stats["clean_sheets_home"] = float(
   699→   794→                sum(1 for m in home_only if m[2] == 0)
   700→   795→            )
   701→   796→
   702→   797→        if away_only:
   703→   798→            stats["goals_scored_away"] = _safe_avg([m[1] for m in away_only])
   704→   799→            stats["goals_conceded_away"] = _safe_avg([m[2] for m in away_only])
   705→   800→            stats["away_wins"] = sum(1 for m in away_only if m[3] == "W")
   706→   801→            stats["away_draws"] = sum(1 for m in away_only if m[3] == "D")
   707→   802→            stats["away_losses"] = sum(1 for m in away_only if m[3] == "L")
   708→   803→            stats["away_goals"] = _safe_avg([m[1] for m in away_only])
   709→   804→            stats["matches_played_away"] = float(len(away_only))
   710→   805→            stats["clean_sheets_away"] = float(
   711→   806→                sum(1 for m in away_only if m[2] == 0)
   712→   807→            )
   713→   808→
   714→   809→        # Recent form (sliding window)
   715→   810→        recent = overall[-window:] if len(overall) >= window else overall
   716→   811→        stats["recent_results"] = [m[3] for m in recent]
   717→   812→        stats["recent_goals_scored"] = [m[1] for m in recent]
   718→   813→
   719→   814→        return stats
