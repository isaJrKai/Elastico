   258→   258→def compute_elo_ratings(df: pd.DataFrame) -> pd.DataFrame:
   259→   259→    print(f"\n  Computing ELO ratings (temporal, no leakage)...")
   260→   260→    elo_ratings: dict[str, float] = defaultdict(lambda: 1500.0)
   261→   261→    match_elo_home = []
   262→   262→    match_elo_away = []
   263→   263→
   264→   264→    for idx, row in df.iterrows():
   265→   265→        home, away, ftr = row["HomeTeam"], row["AwayTeam"], row["FTR"]
   266→   266→        k = LEAGUE_ELO_K.get(row["league_code"], 30.0)
   267→   267→        home_adv = LEAGUE_HOME_ADVANTAGE.get(row["league_code"], 60.0)
   268→   268→        elo_h = elo_ratings[home]
   269→   269→        elo_a = elo_ratings[away]
   270→   270→        match_elo_home.append(elo_h)
   271→   271→        match_elo_away.append(elo_a)
   272→   272→        expected_h = 1.0 / (1.0 + 10 ** ((elo_a - elo_h - home_adv) / 400.0))
   273→   273→        if ftr == "H":
   274→   274→            ah, aa = 1.0, 0.0
   275→   275→        elif ftr == "A":
   276→   276→            ah, aa = 0.0, 1.0
   277→   277→        else:
   278→   278→            ah, aa = 0.5, 0.5
   279→   279→        elo_ratings[home] = elo_h + k * (ah - expected_h)
   280→   280→        elo_ratings[away] = elo_a + k * (aa - (1.0 - expected_h))
   281→   281→
   282→   282→    df = df.copy()
   283→   283→    df["elo_home"] = match_elo_home
   284→   284→    df["elo_away"] = match_elo_away
   285→   285→    df["elo_diff"] = df["elo_home"] - df["elo_away"]
   286→   286→    df["elo_home_advantage_adjusted"] = df["elo_home"] + df["league_code"].map(LEAGUE_HOME_ADVANTAGE).fillna(60.0)
   287→   287→    print(f"  ELO computed for {len(df)} matches.")
   288→   288→    return df

   311→   311→    # Feature arrays

   313→   313→        "avg_goals_scored_home", "avg_goals_scored_away",
   314→   314→        "avg_goals_conceded_home", "avg_goals_conceded_away",
   315→   315→        "avg_goals_scored", "avg_goals_conceded", "goal_difference",
   316→   316→        "xg_for_home", "xg_for_away", "xg_diff",
   317→   317→        "last_5_form_points_home", "last_5_form_points_away",
   318→   318→        "form_points_diff", "win_streak_home", "win_streak_away",
   319→   319→        "unbeaten_streak_home", "unbeaten_streak_away", "goals_per_game_trend",
   320→   320→        "h2h_home_win_pct", "h2h_away_win_pct", "h2h_draw_pct", "h2h_avg_goals",
   321→   321→        "home_win_pct", "away_win_pct", "home_goals_avg", "away_goals_avg",
   322→   322→        "implied_prob_home", "implied_prob_draw", "implied_prob_away", "overround",
   323→   323→        "day_of_week", "month", "is_weekend",
   324→   324→        "days_since_last_match_home", "days_since_last_match_away",
   325→   325→        "attacking_strength_home", "attacking_strength_away",
   326→   326→        "defensive_strength_home", "defensive_strength_away",
   327→   327→        "poisson_lambda_home", "poisson_lambda_away",
   328→   328→        "shot_conversion_rate_home", "shot_conversion_rate_away",
   329→   329→        "clean_sheet_pct_home", "clean_sheet_pct_away",
   330→   330→        "elo_form_trend",
   331→   331→    ]
   332→   332→    features = {name: np.full(n, np.nan) for name in feat_names}
   333→   333→
   334→   334→    for i in range(n):
   335→   335→        row = df.iloc[i]
   336→   336→        home = row["HomeTeam"]
   337→   337→        away = row["AwayTeam"]
   338→   338→        date = row["Date"]
   339→   339→        league = row["league_code"]
   340→   340→        home_hist = team_history[home]
   341→   341→        away_hist = team_history[away]
   342→   342→
   343→   343→        # --- Goal features (all-venue averages from PAST matches only) ---
   344→   344→        if home_hist:
   345→   345→            gf_h = [m["goals_for"] for m in home_hist]
   346→   346→            ga_h = [m["goals_against"] for m in home_hist]
   347→   347→            features["avg_goals_scored"][i] = np.mean(gf_h)
   348→   348→            features["avg_goals_conceded"][i] = np.mean(ga_h)
   349→   349→            features["goal_difference"][i] = np.mean(gf_h) - np.mean(ga_h)
   350→   350→            home_home = [m for m in home_hist if m["venue"] == "home"]
   351→   351→            home_away = [m for m in home_hist if m["venue"] == "away"]
   352→   352→            if home_home:
   353→   353→                features["avg_goals_scored_home"][i] = np.mean([m["goals_for"] for m in home_home])
   354→   354→                features["avg_goals_conceded_home"][i] = np.mean([m["goals_against"] for m in home_home])
   355→   355→                features["home_win_pct"][i] = np.mean([m["points"] == 3 for m in home_home]) * 100
   356→   356→                features["home_goals_avg"][i] = np.mean([m["goals_for"] for m in home_home])
   357→   357→            if home_away:
   358→   358→                features["avg_goals_scored_away"][i] = np.mean([m["goals_for"] for m in home_away])
   359→   359→                features["avg_goals_conceded_away"][i] = np.mean([m["goals_against"] for m in home_away])
   360→   360→            # xG proxy: recent SOT * 0.1
   361→   361→            if "shots_on_target" in home_hist[-1]:
   362→   362→                recent_sot = [m["shots_on_target"] for m in home_hist[-min(5, len(home_hist)):]]
   363→   363→                features["xg_for_home"][i] = np.mean(recent_sot) * 0.1
   364→   364→            # Shot conversion
   365→   365→            if "shots" in home_hist[-1]:
   366→   366→                recent = home_hist[-min(window, len(home_hist)):]
   367→   367→                sots = sum(m["shots_on_target"] for m in recent)
   368→   368→                shots = sum(m["shots"] for m in recent)
   369→   369→                if shots > 0:
   370→   370→                    features["shot_conversion_rate_home"][i] = sots / shots * 100
   371→   371→            # Clean sheets
   372→   372→            features["clean_sheet_pct_home"][i] = np.mean([m["clean_sheet"] for m in home_hist]) * 100
   373→   373→
   374→   374→        if away_hist:
   375→   375→            away_away = [m for m in away_hist if m["venue"] == "away"]
   376→   376→            if away_away:
   377→   377→                features["away_win_pct"][i] = np.mean([m["points"] == 3 for m in away_away]) * 100
   378→   378→                features["away_goals_avg"][i] = np.mean([m["goals_for"] for m in away_away])
   379→   379→            if "shots_on_target" in away_hist[-1]:
   380→   380→                recent_sot = [m["shots_on_target"] for m in away_hist[-min(5, len(away_hist)):]]
   381→   381→                features["xg_for_away"][i] = np.mean(recent_sot) * 0.1
   382→   382→            if "shots" in away_hist[-1]:
   383→   383→                recent = away_hist[-min(window, len(away_hist)):]
   384→   384→                sots = sum(m["shots_on_target"] for m in recent)
   385→   385→                shots = sum(m["shots"] for m in recent)
   386→   386→                if shots > 0:
   387→   387→                    features["shot_conversion_rate_away"][i] = sots / shots * 100
   388→   388→            features["clean_sheet_pct_away"][i] = np.mean([m["clean_sheet"] for m in away_hist]) * 100
   389→   389→
   390→   390→        # xG diff
   391→   391→        if not np.isnan(features["xg_for_home"][i]) and not np.isnan(features["xg_for_away"][i]):
   392→   392→            features["xg_diff"][i] = features["xg_for_home"][i] - features["xg_for_away"][i]
   393→   393→
   394→   394→        # --- Form features ---
   395→   395→        if home_hist:
   396→   396→            recent_home = home_hist[-min(5, len(home_hist)):]
   397→   397→            features["last_5_form_points_home"][i] = sum(m["points"] for m in recent_home)
   398→   398→            streak = 0
   399→   399→            for m in reversed(home_hist):
   400→   400→                if m["points"] == 3: streak += 1
   401→   401→                else: break
   402→   402→            features["win_streak_home"][i] = streak
   403→   403→            streak = 0
   404→   404→            for m in reversed(home_hist):
   405→   405→                if m["points"] >= 1: streak += 1
   406→   406→                else: break
   407→   407→            features["unbeaten_streak_home"][i] = streak
   408→   408→            if len(home_hist) >= 6:
   409→   409→                r3 = np.mean([m["goals_for"] for m in home_hist[-3:]])
   410→   410→                o3 = np.mean([m["goals_for"] for m in home_hist[-6:-3]])
   411→   411→                features["goals_per_game_trend"][i] = r3 - o3
   412→   412→            elif len(home_hist) >= 2:
   413→   413→                features["goals_per_game_trend"][i] = np.mean([m["goals_for"] for m in home_hist])
   414→   414→
   415→   415→        if away_hist:
   416→   416→            recent_away = away_hist[-min(5, len(away_hist)):]
   417→   417→            features["last_5_form_points_away"][i] = sum(m["points"] for m in recent_away)
   418→   418→            streak = 0
   419→   419→            for m in reversed(away_hist):
   420→   420→                if m["points"] == 3: streak += 1
   421→   421→                else: break
   422→   422→            features["win_streak_away"][i] = streak
   423→   423→            streak = 0
   424→   424→            for m in reversed(away_hist):
   425→   425→                if m["points"] >= 1: streak += 1
   426→   426→                else: break
   427→   427→            features["unbeaten_streak_away"][i] = streak
   428→   428→
   429→   429→        if (not np.isnan(features["last_5_form_points_home"][i])
   430→   430→                and not np.isnan(features["last_5_form_points_away"][i])):
   431→   431→            features["form_points_diff"][i] = (features["last_5_form_points_home"][i]
   432→   432→                                             - features["last_5_form_points_away"][i])
   433→   433→
   434→   434→        # --- H2H features ---
   435→   435→        h2h_matches = h2h_history[(home, away)]
   436→   436→        if h2h_matches:
   437→   437→            hw = sum(1 for m in h2h_matches if m["home_winner"])
   438→   438→            aw = sum(1 for m in h2h_matches if m["away_winner"])
   439→   439→            dr = sum(1 for m in h2h_matches if m["draw"])
   440→   440→            t = len(h2h_matches)
   441→   441→            features["h2h_home_win_pct"][i] = hw / t * 100
   442→   442→            features["h2h_away_win_pct"][i] = aw / t * 100
   443→   443→            features["h2h_draw_pct"][i] = dr / t * 100
   444→   444→            features["h2h_avg_goals"][i] = np.mean([m["total_goals"] for m in h2h_matches])
   445→   445→
   446→   446→        # --- Market features ---
   447→   447→        oh = row.get("PSH", np.nan) or row.get("B365H", np.nan)
   448→   448→        od = row.get("PSD", np.nan) or row.get("B365D", np.nan)
   449→   449→        oa = row.get("PSA", np.nan) or row.get("B365A", np.nan)
   450→   450→        try:
   451→   451→            oh_f = float(oh)
   452→   452→            od_f = float(od)
   453→   453→            oa_f = float(oa)
   454→   454→            if np.isfinite(oh_f) and np.isfinite(od_f) and np.isfinite(oa_f) and oh_f > 0:
   455→   455→                si = 1.0/oh_f + 1.0/od_f + 1.0/oa_f
   456→   456→                features["implied_prob_home"][i] = 1.0/oh_f / si * 100
   457→   457→                features["implied_prob_draw"][i] = 1.0/od_f / si * 100
   458→   458→                features["implied_prob_away"][i] = 1.0/oa_f / si * 100
   459→   459→                features["overround"][i] = si * 100
   460→   460→        except (TypeError, ValueError, ZeroDivisionError):
   461→   461→            pass
   462→   462→
   463→   463→        # --- Temporal features ---
   464→   464→        features["day_of_week"][i] = date.dayofweek
   465→   465→        features["month"][i] = date.month
   466→   466→        features["is_weekend"][i] = 1.0 if date.dayofweek >= 5 else 0.0
   467→   467→        if home in last_match_date:
   468→   468→            features["days_since_last_match_home"][i] = (date - last_match_date[home]).days
   469→   469→        if away in last_match_date:
   470→   470→            features["days_since_last_match_away"][i] = (date - last_match_date[away]).days
   471→   471→
   472→   472→        # --- Advanced features ---
   473→   473→        if league_goals[league]:
   474→   474→            league_avg = np.mean(league_goals[league])
   475→   475→            league_avg_per_team = league_avg / 2.0
   476→   476→            if home_hist:
   477→   477→                ha_s = np.mean([m["goals_for"] for m in home_hist])
   478→   478→                ha_c = np.mean([m["goals_against"] for m in home_hist])
   479→   479→                features["attacking_strength_home"][i] = ha_s / max(league_avg_per_team, 0.5)
   480→   480→                features["defensive_strength_home"][i] = ha_c / max(league_avg_per_team, 0.5)
   481→   481→            if away_hist:
   482→   482→                aa_s = np.mean([m["goals_for"] for m in away_hist])
   483→   483→                aa_c = np.mean([m["goals_against"] for m in away_hist])
   484→   484→                features["attacking_strength_away"][i] = aa_s / max(league_avg_per_team, 0.5)
   485→   485→                features["defensive_strength_away"][i] = aa_c / max(league_avg_per_team, 0.5)
   486→   486→            # Poisson lambda
   487→   487→            if (not np.isnan(features["attacking_strength_home"][i])
   488→   488→                    and not np.isnan(features["defensive_strength_away"][i])):
   489→   489→                features["poisson_lambda_home"][i] = (features["attacking_strength_home"][i]
   490→   490→                                                     * features["defensive_strength_away"][i]
   491→   491→                                                     * league_avg_per_team)
   492→   492→            if (not np.isnan(features["attacking_strength_away"][i])
   493→   493→                    and not np.isnan(features["defensive_strength_home"][i])):
   494→   494→                features["poisson_lambda_away"][i] = (features["attacking_strength_away"][i]
   495→   495→                                                     * features["defensive_strength_home"][i]
   496→   496→                                                     * league_avg_per_team)
   497→   497→
   498→   498→        # ELO form trend
   499→   499→        if home in elo_history_per_team and len(elo_history_per_team[home]) >= 5:
   500→   500→            recent_elos = elo_history_per_team[home][-10:]
   501→   501→            if len(recent_elos) >= 2:
   502→   502→                x = np.arange(len(recent_elos))
   503→   503→                features["elo_form_trend"][i] = np.polyfit(x, recent_elos, 1)[0]
   504→   504→
   505→   505→        # ══════════════════════════════════════════════════════════════════
   506→   506→        # UPDATE HISTORIES — AFTER feature computation for match i
   507→   507→        # This is the KEY to preventing temporal leakage.
   508→   508→        # ══════════════════════════════════════════════════════════════════
   509→   509→
   510→   510→        fthg = int(row["FTHG"])
   511→   511→        ftag = int(row["FTAG"])
   512→   512→        ftr = row["FTR"]
   513→   513→        hs = int(row.get("HS", 0) or 0)
   514→   514→        as_ = int(row.get("AS", 0) or 0)
   515→   515→        hst = int(row.get("HST", 0) or 0)
   516→   516→        ast = int(row.get("AST", 0) or 0)
   517→   517→
   518→   518→        team_history[home].append({
   519→   519→            "date": date, "venue": "home",
   520→   520→            "goals_for": fthg, "goals_against": ftag,
   521→   521→            "ftr": ftr, "points": 3 if ftr == "H" else (1 if ftr == "D" else 0),
   522→   522→            "shots": hs, "shots_on_target": hst,
   523→   523→            "clean_sheet": 1 if ftag == 0 else 0,
   524→   524→        })
   525→   525→        team_history[away].append({
   526→   526→            "date": date, "venue": "away",
   527→   527→            "goals_for": ftag, "goals_against": fthg,
   528→   528→            "ftr": ftr, "points": 3 if ftr == "A" else (1 if ftr == "D" else 0),
   529→   529→            "shots": as_, "shots_on_target": ast,
   530→   530→            "clean_sheet": 1 if fthg == 0 else 0,
   531→   531→        })
   532→   532→
   533→   533→        h2h_history[(home, away)].append({
   534→   534→            "home_winner": ftr == "H", "away_winner": ftr == "A",
   535→   535→            "draw": ftr == "D", "total_goals": fthg + ftag,
   536→   536→        })
   537→   537→        h2h_history[(away, home)].append({
   538→   538→            "home_winner": ftr == "A", "away_winner": ftr == "H",
   539→   539→            "draw": ftr == "D", "total_goals": fthg + ftag,
   540→   540→        })
   541→   541→
   542→   542→        league_goals[league].append(fthg + ftag)
   543→   543→        last_match_date[home] = date
   544→   544→        last_match_date[away] = date
   545→   545→        elo_history_per_team[home].append(row["elo_home"])
   546→   546→        elo_history_per_team[away].append(row["elo_away"])
   547→   547→
   548→   548→    # Attach features to dataframe
   549→   549→    for name, arr in features.items():
   550→   550→        df[name] = arr
   551→   551→
   552→   552→    print(f"  Rolling features computed for {n} matches.")
   553→   553→    return df

   557→   557→# Step 5: Temporal Leakage Audit
   558→   558→# ═══════════════════════════════════════════════════════════════════════════
   559→   559→
   560→   560→def audit_temporal_leakage(df: pd.DataFrame) -> dict:
   561→   561→    print(f"\n{'='*80}")
   562→   562→    print(f"STEP 5: TEMPORAL LEAKAGE AUDIT")
   563→   563→    print(f"{'='*80}")
   564→   564→
   565→   565→    n = len(df)
   566→   566→    violations = 0
   567→   567→    details = []
   568→   568→    sample_indices = list(range(0, n, max(1, n // 100)))
   569→   569→
   570→   570→    for i in sample_indices:
   571→   571→        row = df.iloc[i]
   572→   572→        home = row["HomeTeam"]
   573→   573→        away = row["AwayTeam"]
   574→   574→        match_date = row["Date"]
   575→   575→
   576→   576→        # Recompute avg_goals_scored from strictly prior matches
   577→   577→        prev_home = df[(df["Date"] < match_date) &
   578→   578→                       ((df["HomeTeam"] == home) | (df["AwayTeam"] == home))]
   579→   579→        if len(prev_home) > 0:
   580→   580→            past_gf = []
   581→   581→            for _, pr in prev_home.iterrows():
   582→   582→                if pr["HomeTeam"] == home:
   583→   583→                    past_gf.append(pr["FTHG"])
   584→   584→                else:
   585→   585→                    past_gf.append(pr["FTAG"])
   586→   586→            expected = np.mean(past_gf)
   587→   587→            actual = row["avg_goals_scored"]
   588→   588→            if not np.isnan(actual):
   589→   589→                diff = abs(actual - expected)
   590→   590→                if diff > 0.15:
   591→   591→                    violations += 1
   592→   592→                    details.append({"idx": i, "date": str(match_date.date()),
   593→   593→                                   "feature": "avg_goals_scored",
   594→   594→                                   "expected": round(expected, 4),
   595→   595→                                   "actual": round(actual, 4),
   596→   596→                                   "diff": round(diff, 4), "team": home})
   597→   597→
   598→   598→        # Verify form points
   599→   599→        if len(prev_home) > 0:
   600→   600→            past_pts = []
   601→   601→            for _, pr in prev_home.iterrows():
   602→   602→                if pr["HomeTeam"] == home:
   603→   603→                    past_pts.append(3 if pr["FTR"] == "H" else (1 if pr["FTR"] == "D" else 0))
   604→   604→                else:
   605→   605→                    past_pts.append(3 if pr["FTR"] == "A" else (1 if pr["FTR"] == "D" else 0))
   606→   606→            expected_form = sum(past_pts[-5:])
   607→   607→            actual_form = row["last_5_form_points_home"]
   608→   608→            if not np.isnan(actual_form) and actual_form != expected_form:
   609→   609→                violations += 1
   610→   610→                details.append({"idx": i, "date": str(match_date.date()),
   611→   611→                               "feature": "last_5_form_points_home",
   612→   612→                               "expected": expected_form, "actual": actual_form,
   613→   613→                               "diff": abs(actual_form - expected_form), "team": home})
   614→   614→
   615→   615→        # Verify rest days
   616→   616→        if home in df["HomeTeam"].values or home in df["AwayTeam"].values:
   617→   617→            prev_dates = df[(df["Date"] < match_date) &
   618→   618→                           ((df["HomeTeam"] == home) | (df["AwayTeam"] == home))]["Date"]
   619→   619→            if len(prev_dates) > 0:
   620→   620→                expected_days = (match_date - prev_dates.max()).days
   621→   621→                actual_days = row["days_since_last_match_home"]
   622→   622→                if not np.isnan(actual_days) and actual_days != expected_days:
   623→   623→                    violations += 1
   624→   624→                    details.append({"idx": i, "feature": "days_since_last_match_home",
   625→   625→                                   "expected": expected_days, "actual": actual_days})
   626→   626→
   627→   627→    # Verify first-match NaN
   628→   628→    first_teams = set()
   629→   629→    for i in range(n):
   630→   630→        row = df.iloc[i]
   631→   631→        for team in [row["HomeTeam"], row["AwayTeam"]]:
   632→   632→            if team not in first_teams:
   633→   633→                first_teams.add(team)
   634→   634→                if row["HomeTeam"] == team:
   635→   635→                    for feat in ["avg_goals_scored", "last_5_form_points_home",
   636→   636→                                "win_streak_home", "home_win_pct", "clean_sheet_pct_home"]:
   637→   637→                        if not np.isnan(row.get(feat, np.nan)):
   638→   638→                            violations += 1
   639→   639→                            details.append({"idx": i, "feature": feat,
   640→   640→                                           "violation": f"First match for {team} but not NaN",
   641→   641→                                           "value": row[feat]})
   642→   642→
   643→   643→    print(f"  Sampled {len(sample_indices)} matches")
   644→   644→    print(f"  Checked {len(first_teams)} teams for first-match NaN")
   645→   645→    print(f"  Violations: {violations}")
   646→   646→    if details:
   647→   647→        print(f"  First 5:")
   648→   648→        for v in details[:5]:
   649→   649→            print(f"    {v}")
   650→   650→    return {"samples_checked": len(sample_indices),
   651→   651→            "teams_checked": len(first_teams),
   652→   652→            "violations": violations,
   653→   653→            "details": details[:50],
   654→   654→            "passed": violations == 0}
   655→   655→

   658→   658→# Step 6: Training Dataset

   660→   660→
   661→   661→def produce_training_dataset(df: pd.DataFrame) -> tuple[pd.DataFrame, dict]:
   662→   662→    print(f"\n{'='*80}")

   703→   703→    nan_pct = (nan_counts / len(train_df) * 100).round(1)

   706→   706→    for feat, pct in nan_pct.sort_values(ascending=False).head(15).items():
   707→   707→        print(f"    {feat:40s}: {nan_counts[feat]:6.0f} NaN ({pct:5.1f}%)")
   708→   708→
   709→   709→    complete_mask = train_df[available].notna().all(axis=1)
   710→   710→    n_complete = int(complete_mask.sum())
   711→   711→    n_incomplete = int((~complete_mask).sum())
   712→   712→    print(f"\n  Complete rows: {n_complete}/{len(train_df)} ({n_complete/len(train_df)*100:.1f}%)")
   713→   713→    print(f"  Incomplete rows: {n_incomplete}")
   714→   714→
   715→   715→    return train_df, {
   716→   716→        "total_rows": len(train_df),
   717→   717→        "complete_rows": n_complete,
   718→   718→        "incomplete_rows": n_incomplete,
   719→   719→        "features_available": len(available),
   720→   720→        "features_missing": missing,
   721→   721→        "nan_per_feature": {f: int(nan_counts[f]) for f in available if nan_counts[f] > 0},
   722→   722→    }

   725→   725→# ═══════════════════════════════════════════════════════════════════════════
   726→   726→# Step 7: Report
   727→   727→# ═══════════════════════════════════════════════════════════════════════════
   728→   728→
   729→   729→def generate_report(file_info, df_final, dataset_info, leakage_audit, dupes1, dupes2):
   730→   730→    n = len(df_final)
