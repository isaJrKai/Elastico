    38→
    39→function useEspnStandings(league: string = 'PL') {
    40→  const [rows, setRows] = useState<Array<{
    41→    rank: number; name: string; code: string; logo: string; color: string
    42→    played: number; wins: number; draws: number; losses: number; points: number
    43→  }>>([])
    44→  const [loading, setLoading] = useState(true)
    45→
    46→  useEffect(() => {
    47→    setLoading(true)
    48→    fetch(`/api/live?action=standings&league=${league}`)
    49→      .then(r => r.ok ? r.json() : [])
    50→      .then(data => {
    51→        const items = (data.data || []).slice(0, 10).map((t: any, i: number) => ({
    52→          rank: i + 1,
    53→          name: t.name || t.team?.displayName || '?',
    54→          code: t.code || t.abbreviation || '',
    55→          logo: t.logo || t.team?.logo || '',
    56→          color: t.color || '',
    57→          played: t.gamesPlayed ?? t.played ?? 0,
    58→          wins: t.wins ?? 0,
    59→          draws: t.draws ?? 0,
    60→          losses: t.losses ?? 0,
    61→          points: t.points ?? 0,
    62→        }))
    63→        setRows(items)
    64→      })
    65→      .catch(() => setRows([]))
    66→      .finally(() => setLoading(false))
    67→  }, [league])
    68→
    69→  return { rows, loading }
    70→}
