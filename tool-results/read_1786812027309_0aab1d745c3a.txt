    47→function getStatusConfig(status: string) {
    48→  switch (status) {
    49→    case 'live': return { label: 'LIVE', cls: 'bg-red-500/15 text-red-400 border-red-500/30', pulse: true }
    50→    case 'halftime': return { label: 'HT', cls: 'bg-yellow-500/15 text-yellow-400 border-yellow-500/30', pulse: false }
    51→    case 'finished': return { label: 'FT', cls: 'bg-muted text-muted-foreground border-border', pulse: false }
    52→    case 'postponed': return { label: 'PPD', cls: 'bg-orange-500/15 text-orange-400 border-orange-500/30', pulse: false }
    53→    default: return { label: 'Upcoming', cls: 'bg-primary/15 text-primary border-primary/30', pulse: false }
    54→  }
    55→}

    57→function formatMatchDate(dateStr: string | null): string {
    58→  if (!dateStr) return 'TBD'
    59→  return new Date(dateStr).toLocaleDateString('en-US', { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' })
    60→}

   322→export function MatchesView() {
   323→  const token = useElasticoStore(s => s.token)
   324→  const [activeTab, setActiveTab] = useState<MatchTab>('all')
   325→  const [searchQuery, setSearchQuery] = useState('')
   326→  const [stageFilter, setStageFilter] = useState('all')
   327→  const [groupFilter, setGroupFilter] = useState('all')
   328→  const [sortBy, setSortBy] = useState<SortKey>('date')
   329→  const [matchesData, setMatchesData] = useState<Match[]>([])
   330→  const [loading, setLoading] = useState(true)
   331→  const [lastRefresh, setLastRefresh] = useState(new Date())
   332→
   333→  const fetchMatches = useCallback(async () => {
   334→    setLoading(true)
   335→    try {
   336→      const headers: Record<string, string> = {}
   337→      if (token) headers['Authorization'] = `Bearer ${token}`
   338→
   339→      // Map activeTab to football-data.org status
   340→      let statusParam = 'FINISHED'
   341→      if (activeTab === 'upcoming') statusParam = 'SCHEDULED,TIMED'
   342→      else if (activeTab === 'live') statusParam = 'IN_PLAY,PAUSED'
   343→      else if (activeTab === 'all') statusParam = 'FINISHED'
   344→
   345→      const res = await fetch(`/api/football-data?action=matches&competition=PL&status=${statusParam}`, { headers })
   346→      const data: any = await res.json()
   347→      const matches = (data.data || []).map((m: any) => ({
   348→        ...m,
   349→        id: m.id || `fd-${Math.random().toString(36).slice(2)}`,
   350→        competition: m.competition,
   351→        homeTeam: m.homeTeam,
   352→        awayTeam: m.awayTeam,
   353→        homeScore: m.homeScore,
   354→        awayScore: m.awayScore,
   355→        status: m.status,
   356→        date: m.date,
   357→        matchday: m.matchday,
   358→      }))
   359→      setMatchesData(matches)
   360→    } catch (err) {
   361→      console.error('Failed to fetch matches:', err)
   362→    } finally {
   363→      setLoading(false)
   364→      setLastRefresh(new Date())
   365→    }
   366→  }, [activeTab, searchQuery, stageFilter, groupFilter, token])
   367→
   368→  useEffect(() => { fetchMatches() }, [fetchMatches])
   369→
   370→  // Auto-refresh live
   371→  useEffect(() => {
   372→    const iv = setInterval(() => { if (activeTab === 'live') fetchMatches() }, 15000)
   373→    return () => clearInterval(iv)
   374→  }, [activeTab, fetchMatches])
   375→
   376→  // Sort
   377→  const sortedMatches = useMemo(() => {
   378→    const arr = [...matchesData]
   379→    switch (sortBy) {
   380→      case 'elo': return arr.sort((a, b) => Math.abs((b.homeEloBefore ?? 0) - (b.awayEloBefore ?? 0)) - Math.abs((a.homeEloBefore ?? 0) - (a.awayEloBefore ?? 0)))
   381→      case 'xg': return arr.sort((a, b) => (b.homeXg + b.awayXg) - (a.homeXg + a.awayXg))
   382→      default: return arr
   383→    }
   384→  }, [matchesData, sortBy])
   385→
   386→  const allMatches = matchesData
   387→  const liveCount = allMatches.filter((m) => m.status === 'live' || m.status === 'halftime').length
   388→  const availableStages = useMemo(() => STAGES.filter((s) => allMatches.some((m) => m.stage === s)), [allMatches])
   389→  const availableGroups = useMemo(() => GROUPS.filter((g) => allMatches.some((m) => m.group === g)), [allMatches])
   390→
   391→  return (
   392→    <div className="flex flex-col gap-4 animate-fade-in-up">
   393→      {/* Header */}
   394→      <div className="flex items-center gap-3">
   395→        <div className="flex size-10 items-center justify-center rounded-xl bg-primary/15"><Zap className="size-5 text-primary" /></div>
   396→        <div>
   397→          <h1 className="text-2xl font-bold tracking-tight">Matches</h1>
   398→          <p className="text-sm text-muted-foreground">Browse, filter, and predict match outcomes</p>
   399→        </div>
   400→      </div>
   401→
   402→      {/* Filter Bar */}
   403→      <div className="glass-card-premium rounded-xl p-4 space-y-3">
   404→        <div className="flex flex-col sm:flex-row gap-3">
   405→          <div className="relative flex-1">
   406→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   407→            <Input placeholder="Search by team name..." value={searchQuery} onChange={(e) => setSearchQuery(e.target.value)} className="pl-9 h-9 bg-muted/50 border-border text-sm" />
   408→          </div>
   409→          <Select value={stageFilter} onValueChange={setStageFilter}>
   410→            <SelectTrigger className="h-9 w-full sm:w-[160px] bg-muted/50 border-border text-sm">
   411→              <Filter className="size-3.5 mr-1.5 text-muted-foreground" /><SelectValue placeholder="Stage" />
   412→            </SelectTrigger>
   413→            <SelectContent className="glass-card border-border">
   414→              <SelectItem value="all">All Stages</SelectItem>
   415→              {availableStages.map((s) => <SelectItem key={s} value={s}>{s}</SelectItem>)}
   416→            </SelectContent>
   417→          </Select>
   418→          <Select value={groupFilter} onValueChange={setGroupFilter}>
