   127→   127→def compute_ece(y_true, y_prob, n_bins=15):
   128→   128→    """Compute Expected Calibration Error with more bins for stability."""
   129→   129→    n_classes = y_prob.shape[1]
   130→   130→    ece = 0.0
   131→   131→    for c in range(n_classes):
   132→   132→        y_binary = (y_true == c).astype(int)
   133→   133→        prob_c = y_prob[:, c]
   134→   134→        bin_edges = np.linspace(0, 1, n_bins + 1)
   135→   135→        for i in range(n_bins):
   136→   136→            mask = (prob_c >= bin_edges[i]) & (prob_c < bin_edges[i + 1])
   137→   137→            if i == n_bins - 1:
   138→   138→                mask = (prob_c >= bin_edges[i]) & (prob_c <= bin_edges[i + 1])
   139→   139→            n_bin = mask.sum()
   140→   140→            if n_bin == 0:
   141→   141→                continue
   142→   142→            avg_conf = prob_c[mask].mean()
   143→   143→            avg_acc = y_binary[mask].mean()
   144→   144→            ece += (n_bin / len(y_true)) * abs(avg_conf - avg_acc)
   145→   145→    return ece / n_classes

   148→   148→def compute_all_metrics(y_true, y_proba, y_pred, label='set'):
   149→   149→    """Full metrics: log loss, Brier, ECE, accuracy, per-class P/R/F1, confusion matrix."""
   150→   150→    n = len(y_true)
   151→   151→    m = {'label': label, 'n_samples': n}
   152→   152→
   153→   153→    m['accuracy'] = float(accuracy_score(y_true, y_pred))
   154→   154→
   155→   155→    # Log loss
   156→   156→    try:
   157→   157→        m['log_loss'] = float(log_loss(y_true, y_proba))
   158→   158→    except Exception:
   159→   159→        m['log_loss'] = float('nan')
   160→   160→
   161→   161→    # Brier score (multi-class, macro-averaged)
   162→   162→    n_classes = y_proba.shape[1]
   163→   163→    y_onehot = np.zeros_like(y_proba)
   164→   164→    for i in range(n_classes):
   165→   165→        y_onehot[:, i] = (y_true == i).astype(float)
   166→   166→    briers = []
   167→   167→    for i in range(n_classes):
   168→   168→        try:
   169→   169→            briers.append(brier_score_loss(y_onehot[:, i], y_proba[:, i]))
   170→   170→        except Exception:
   171→   171→            briers.append(float('nan'))
   172→   172→    m['brier_score'] = float(np.nanmean(briers))
   173→   173→
   174→   174→    # Per-class P/R/F1
   175→   175→    prec, rec, f1, support = precision_recall_fscore_support(
   176→   176→        y_true, y_pred, labels=[0, 1, 2], zero_division=0
   177→   177→    )
   178→   178→    for i, name in enumerate(['Home', 'Draw', 'Away']):
   179→   179→        m[f'precision_{name}'] = float(prec[i])
   180→   180→        m[f'recall_{name}'] = float(rec[i])
   181→   181→        m[f'f1_{name}'] = float(f1[i])
   182→   182→        m[f'support_{name}'] = int(support[i])
   183→   183→    m['precision_macro'] = float(np.mean(prec))
   184→   184→    m['recall_macro'] = float(np.mean(rec))
   185→   185→    m['f1_macro'] = float(np.mean(f1))
   186→   186→
   187→   187→    # Confusion matrix
   188→   188→    cm = confusion_matrix(y_true, y_pred, labels=[0, 1, 2])
   189→   189→    m['confusion_matrix'] = cm.tolist()
   190→   190→
   191→   191→    # ECE
   192→   192→    m['ece'] = float(compute_ece(y_true, y_proba))
   193→   193→
   194→   194→    # Calibration reliability diagram data (for per-bin analysis)
   195→   195→    m['calibration_bins'] = compute_calibration_bins(y_true, y_proba, n_bins=10)
   196→   196→
   197→   197→    return m
   198→   198→

   200→   200→def compute_calibration_bins(y_true, y_prob, n_bins=10):
   201→   201→    """Compute per-bin calibration data for reliability analysis."""
   202→   202→    bins_data = []
   203→   203→    for c in range(y_prob.shape[1]):
   204→   204→        y_binary = (y_true == c).astype(int)
   205→   205→        prob_c = y_prob[:, c]
   206→   206→        edges = np.linspace(0, 1, n_bins + 1)
   207→   207→        for i in range(n_bins):
   208→   208→            mask = (prob_c >= edges[i]) & (prob_c < edges[i + 1])
   209→   209→            if i == n_bins - 1:
   210→   210→                mask = (prob_c >= edges[i]) & (prob_c <= edges[i + 1])
   211→   211→            n_bin = mask.sum()
   212→   212→            if n_bin < 5:
   213→   213→                continue
   214→   214→            bins_data.append({
   215→   215→                'class': c,
   216→   216→                'bin_center': float((edges[i] + edges[i + 1]) / 2),
   217→   217→                'avg_confidence': float(prob_c[mask].mean()),
   218→   218→                'avg_accuracy': float(y_binary[mask].mean()),
   219→   219→                'count': int(n_bin),
   220→   220→            })
   221→   221→    return bins_data

   224→   224→def compute_prediction_correlation(proba_a, proba_b, y_true):
   225→   225→    """Compute correlation between two models' predicted probabilities."""
   226→   226→    # Flatten probabilities and compute correlation
   227→   227→    p_a = proba_a.flatten()
   228→   228→    p_b = proba_b.flatten()
   229→   229→    corr = float(np.corrcoef(p_a, p_b)[0, 1])
   230→   230→
   231→   231→    # Per-class argmax agreement
   232→   232→    pred_a = np.argmax(proba_a, axis=1)
   233→   233→    pred_b = np.argmax(proba_b, axis=1)
   234→   234→    agreement = float((pred_a == pred_b).mean())
   235→   235→
   236→   236→    # Both-correct rate (both models predict correctly)
   237→   237→    both_correct = float(((pred_a == y_true) & (pred_b == y_true)).mean())
   238→   238→
   239→   239→    return {
   240→   240→        'probability_correlation': corr,
   241→   241→        'prediction_agreement': agreement,
   242→   242→        'both_correct_rate': both_correct,
   243→   243→    }

   246→   246→def compute_naive_baseline(y_true):
   247→   247→    """Compute naive baseline: always predict class priors."""
   248→   248→    n = len(y_true)
   249→   249→    counts = np.bincount(y_true, minlength=3)
   250→   250→    priors = counts / n
   251→   251→    # Probas = priors for every sample
   252→   252→    proba = np.tile(priors, (n, 1))
   253→   253→    pred = np.full(n, np.argmax(priors))  # always predict majority class
   254→   254→
   255→   255→    m = {}
   256→   256→    m['accuracy'] = float(accuracy_score(y_true, pred))
   257→   257→    try:
   258→   258→        m['log_loss'] = float(log_loss(y_true, proba))
   259→   259→    except Exception:
   260→   260→        m['log_loss'] = float('nan')
   261→   261→    n_classes = 3
   262→   262→    y_onehot = np.zeros((n, n_classes))
   263→   263→    for i in range(n_classes):
   264→   264→        y_onehot[:, i] = (y_true == i).astype(float)
   265→   265→    briers = [brier_score_loss(y_onehot[:, i], proba[:, i]) for i in range(n_classes)]
   266→   266→    m['brier_score'] = float(np.mean(briers))
   267→   267→    m['ece'] = float(compute_ece(y_true, proba))
   268→   268→    m['class_priors'] = [float(p) for p in priors]
   269→   269→    return m

   273→   273→# XGBoost Training (same params as v1 for fair comparison)
   274→   274→# ═══════════════════════════════════════════════════════════════════════
   275→   275→
   276→   276→def train_xgb_model(X_train, y_train, X_val, y_val, feature_names, seed=42):
   277→   277→    """Train XGBoost with IDENTICAL hyperparameters to v1 baseline."""
   278→   278→    import xgboost as xgb
   279→   279→
   280→   280→    counts = np.bincount(y_train, minlength=3)
   281→   281→    total = counts.sum()
   282→   282→    scale_pos_weight = {i: total / (3 * c) for i, c in enumerate(counts) if c > 0}
   283→   283→    sample_weights = np.array([scale_pos_weight.get(y, 1.0) for y in y_train])
   284→   284→
   285→   285→    dtrain = xgb.DMatrix(X_train, label=y_train, weight=sample_weights, feature_names=feature_names)
   286→   286→    dval = xgb.DMatrix(X_val, label=y_val, feature_names=feature_names)
   287→   287→
   288→   288→    params = {
   289→   289→        'objective': 'multi:softprob',
   290→   290→        'num_class': 3,
   291→   291→        'eval_metric': 'mlogloss',
   292→   292→        'max_depth': 5,
   293→   293→        'eta': 0.05,
   294→   294→        'subsample': 0.8,
   295→   295→        'colsample_bytree': 0.8,
   296→   296→        'min_child_weight': 5,
   297→   297→        'gamma': 0.1,
   298→   298→        'lambda': 1.5,
   299→   299→        'alpha': 0.1,
   300→   300→        'seed': seed,
   301→   301→        'nthread': 4,
   302→   302→    }
   303→   303→
   304→   304→    bst = xgb.train(
   305→   305→        params, dtrain,
   306→   306→        num_boost_round=1000,
   307→   307→        evals=[(dtrain, 'train'), (dval, 'val')],
   308→   308→        early_stopping_rounds=100,
   309→   309→        verbose_eval=False,
   310→   310→    )
   311→   311→
   312→   312→    return bst, bst.best_iteration

   314→   314→
   315→   315→def get_predictions(bst, X, feature_names):
   316→   316→    """Get probability predictions from XGBoost model."""
   317→   317→    import xgboost as xgb
   318→   318→    d = xgb.DMatrix(X, feature_names=feature_names)
   319→   319→    proba = bst.predict(d)
   320→   320→    pred = np.argmax(proba, axis=1)
   321→   321→    return proba, pred

   328→   328→def run_experiment(df_complete, window, feature_set_name, feature_set_info):
   329→   329→    """Run one experiment: (window, feature_set) -> metrics dict."""
   330→   330→    feats = feature_set_info['features']
   331→   331→
   332→   332→    # Split data by season
   333→   333→    train_df = df_complete[df_complete['season_code'].isin(window['train_seasons'])].copy()
   334→   334→    val_df = df_complete[df_complete['season_code'].isin(window['val_seasons'])].copy()
   335→   335→    test_df = df_complete[df_complete['season_code'].isin(window['test_seasons'])].copy()
   336→   336→
   337→   337→    # Ensure features are available in the dataframe
   338→   338→    available_feats = [f for f in feats if f in df_complete.columns]
   339→   339→    if len(available_feats) == 0:
   340→   340→        return None
   341→   341→
   342→   342→    # Check for complete cases in this feature set
   343→   343→    train_mask = train_df[available_feats].notna().all(axis=1)
   344→   344→    val_mask = val_df[available_feats].notna().all(axis=1)
   345→   345→    test_mask = test_df[available_feats].notna().all(axis=1)
   346→   346→
   347→   347→    train_df = train_df[train_mask].reset_index(drop=True)
   348→   348→    val_df = val_df[val_mask].reset_index(drop=True)
   349→   349→    test_df = test_df[test_mask].reset_index(drop=True)
   350→   350→
   351→   351→    if len(test_df) < 50:
   352→   352→        return None
   353→   353→
   354→   354→    # Extract X, y
   355→   355→    X_train = train_df[available_feats].values.astype(np.float32)
   356→   356→    y_train = train_df['result_encoded'].values.astype(np.int32)
   357→   357→    X_val = val_df[available_feats].values.astype(np.float32)
   358→   358→    y_val = val_df['result_encoded'].values.astype(np.int32)
   359→   359→    X_test = test_df[available_feats].values.astype(np.float32)
   360→   360→    y_test = test_df['result_encoded'].values.astype(np.int32)
   361→   361→
   362→   362→    if len(y_train) < 100 or len(np.unique(y_train)) < 3:
   363→   363→        return None
   364→   364→
   365→   365→    # Train
   366→   366→    try:
   367→   367→        bst, best_iter = train_xgb_model(X_train, y_train, X_val, y_val, available_feats)
   368→   368→    except Exception as e:
   369→   369→        print(f'    Training failed: {e}')
   370→   370→        return None
   371→   371→
   372→   372→    # Predict
   373→   373→    test_proba, test_pred = get_predictions(bst, X_test, available_feats)
   374→   374→    val_proba, val_pred = get_predictions(bst, X_val, available_feats)
   375→   375→    train_proba, train_pred = get_predictions(bst, X_train, available_feats)
   376→   376→
   377→   377→    # Metrics
   378→   378→    test_metrics = compute_all_metrics(y_test, test_proba, test_pred, 'test')
   379→   379→    val_metrics = compute_all_metrics(y_val, val_proba, val_pred, 'validation')
   380→   380→    train_metrics = compute_all_metrics(y_train, train_proba, train_pred, 'train')
   381→   381→
   382→   382→    # Feature importance
   383→   383→    importance = bst.get_score(importance_type='gain')
   384→   384→    feat_imp = {f: float(importance.get(f, 0.0)) for f in available_feats}
   385→   385→
   386→   386→    return {
   387→   387→        'feature_set': feature_set_name,
   388→   388→        'n_features': len(available_feats),
   389→   389→        'features': available_feats,
   390→   390→        'window': window['name'],
   391→   391→        'train_n': len(y_train),
   392→   392→        'val_n': len(y_val),
   393→   393→        'test_n': len(y_test),
   394→   394→        'best_iteration': int(best_iter),
   395→   395→        'test_metrics': test_metrics,
   396→   396→        'val_metrics': val_metrics,
   397→   397→        'train_metrics': train_metrics,
   398→   398→        'feature_importance': feat_imp,
   399→   399→        'test_proba': test_proba,  # keep for correlation analysis
   400→   400→        'test_pred': test_pred,
   401→   401→        'y_test': y_test,
   402→   402→        'test_df': test_df,  # keep for per-league breakdown
   403→   403→    }
