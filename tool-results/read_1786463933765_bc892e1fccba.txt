   137→// ── Helpers ──────────────────────────────────────────────────────────────────
   138→
   139→function statusColor(status: string) {
   140→  if (status === 'PASS' || status === 'HEALED' || status === 'INTACT' || status === 'HEALTHY' || status === 'ONLINE' || status === 'STABLE' || status === 'ok') return '#00e676'
   141→  if (status === 'WARNING' || status === 'DRIFT_DETECTED' || status === 'QUARANTINED' || status === 'MODIFIED' || status === 'DEGRADED' || status === 'UNSTABLE' || status === 'warning') return '#eab308'
   142→  return '#ef4444'
   143→}
   144→
   145→function statusBgClass(status: string) {
   146→  const c = statusColor(status)
   147→  if (c === '#00e676') return 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30'
   148→  if (c === '#eab308') return 'bg-yellow-500/15 text-yellow-400 border-yellow-500/30'
   149→  return 'bg-red-500/15 text-red-400 border-red-500/30'
   150→}
   151→
   152→function CircGauge({ value, max = 100, size = 120, label, strokeColor }: {
   153→  value: number; max?: number; size?: number; label: string; strokeColor?: string
   154→}) {
   155→  const pct = Math.min(value / max, 1)
   156→  const r = (size - 16) / 2
   157→  const circ = 2 * Math.PI * r
   158→  const offset = circ * (1 - pct)
   159→  const color = strokeColor || (pct > 0.7 ? '#00e676' : pct > 0.4 ? '#eab308' : '#ef4444')
   160→
   161→  return (
   162→    <div className="flex flex-col items-center gap-2">
   163→      <svg width={size} height={size} className="drop-shadow-lg">
   164→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="oklch(0.15 0.02 260)" strokeWidth={8} />
   165→        <circle
   166→          cx={size / 2} cy={size / 2} r={r} fill="none"
   167→          stroke={color} strokeWidth={8} strokeLinecap="round"
   168→          strokeDasharray={circ} strokeDashoffset={offset}
   169→          transform={`rotate(-90 ${size / 2} ${size / 2})`}
   170→          className="transition-all duration-1000 ease-out"
   171→        />
   172→        <text x={size / 2} y={size / 2 - 4} textAnchor="middle" className="fill-foreground text-2xl font-bold" dominantBaseline="middle">
   173→          {Math.round(value)}
   174→        </text>
   175→        <text x={size / 2} y={size / 2 + 18} textAnchor="middle" className="fill-muted-foreground text-xs" dominantBaseline="middle">
   176→          / {max}
   177→        </text>
   178→      </svg>
   179→      <span className="text-xs text-muted-foreground font-medium">{label}</span>
   180→    </div>
   181→  )
   182→}
