   100→   135→    """Divide with safe fallback for zero denominators."""
   101→   136→    if denominator == 0 or not math.isfinite(denominator):
   102→   137→        return default
   103→   138→    result = numerator / denominator
   104→   139→    return result if math.isfinite(result) else default
   105→   140→
   106→   141→
   107→   142→def _safe_avg(values: list[float] | np.ndarray, default: float = 0.0) -> float:
   108→   143→    """Mean of a numeric sequence, returning *default* for empty input."""
   109→   144→    arr = np.asarray(values, dtype=float)
   110→   145→    if arr.size == 0:
   111→   146→        return default
   112→   147→    return float(np.nanmean(arr))

   115→   150→def _linear_trend(values: list[float] | np.ndarray, default: float = 0.0) -> float:
   116→   151→    """Slope of a simple linear fit (most recent → oldest = positive trend)."""
   117→   152→    arr = np.asarray(values, dtype=float)
   118→   153→    if arr.size < 2:
   119→   154→        return default
   120→   155→    x = np.arange(len(arr), dtype=float)
   121→   156→    valid = np.isfinite(arr)
   122→   157→    if valid.sum() < 2:
   123→   158→        return default
   124→   159→    x = x[valid]
   125→   160→    y = arr[valid]
   126→   161→    slope = np.polyfit(x, y, 1)[0]
   127→   162→    return float(slope) if math.isfinite(slope) else default

   129→   164→
   130→   165→def _compute_streaks(
   131→   166→    results: list[str] | list[int] | None,
   132→   167→) -> tuple[int, int, int]:
   133→   168→    """Return (win_streak, unbeaten_streak, loss_streak) from recent results.
   134→   169→
   135→   170→    *results* is ordered **most recent first**.  Each element is one of:
   136→   171→    - 'W' / 1  → win
   137→   172→    - 'D' / 0  → draw
   138→   173→    - 'L' / -1 → loss
   139→   174→    """
   140→   175→    if not results:
   141→   176→        return 0, 0, 0
   142→   177→    win_s = unbeaten_s = loss_s = 0
   143→   178→    for r in results:
   144→   179→        if r in ('W', 1):
   145→   180→            win_s += 1
   146→   181→            unbeaten_s += 1
   147→   182→        elif r in ('D', 0):
   148→   183→            unbeaten_s += 1
   149→   184→            break
   150→   185→        elif r in ('L', -1):
   151→   186→            break
   152→   187→        else:
   153→   188→            break
   154→   189→    for r in results:
   155→   190→        if r in ('L', -1):
   156→   191→            loss_s += 1
   157→   192→        else:
   158→   193→            break
   159→   194→    return win_s, unbeaten_s, loss_s
   160→   195→
   161→   196→
   162→   197→def _form_points(results: list[str] | list[int] | None) -> float:
   163→   198→    """Sum of 3/1/0 for W/D/L over recent results (max 15 for 5 matches)."""
   164→   199→    if not results:
   165→   200→        return 0.0
   166→   201→    mapping = {'W': 3, 'D': 1, 'L': 0, 1: 3, 0: 1, -1: 0}
   167→   202→    return sum(mapping.get(r, 0) for r in results)

   170→   205→def _parse_date(value: Any) -> Optional[datetime]:
   171→   206→    """Coerce *value* to a ``datetime``.  Returns ``None`` on failure."""
   172→   207→    if isinstance(value, datetime):
   173→   208→        return value
   174→   209→    if isinstance(value, str):
   175→   210→        for fmt in ("%Y-%m-%d", "%Y-%m-%dT%H:%M:%S", "%Y-%m-%d %H:%M:%S"):
   176→   211→            try:
   177→   212→                return datetime.strptime(value, fmt)
   178→   213→            except ValueError:
   179→   214→                continue
   180→   215→    if isinstance(value, (int, float)):
   181→   216→        try:
   182→   217→            return datetime.fromtimestamp(value)
   183→   218→        except (OSError, ValueError):
   184→   219→            pass
   185→   220→    return None

   218→   253→    def build_match_features(
   219→   254→        self,
   220→   255→        match_data: dict[str, Any],
   221→   256→        home_stats: dict[str, Any],
   222→   257→        away_stats: dict[str, Any],
   223→   258→        h2h_data: list[dict[str, Any]] | None = None,
   224→   259→        odds: dict[str, Any] | None = None,
   225→   260→    ) -> dict[str, float]:
   226→   261→        """Build a feature vector (50 values) for a **single** match prediction.
   227→   262→
   228→   263→        Parameters
   229→   264→        ----------
   230→   265→        match_data : dict
   231→   266→            Must contain ``date`` (str / datetime / timestamp).
   232→   267→        home_stats, away_stats : dict
   233→   268→            Per-team season statistics.  Expected keys (all optional):
   234→   269→
   235→   270→            ELO: ``elo``, ``elo_history`` (list[float])
   236→   271→            Goals: ``goals_scored``, ``goals_conceded``,
   237→   272→                   ``goals_scored_home``, ``goals_conceded_home`` (or ``_away``)
   238→   273→            xG: ``xg_for``, ``xg_for_home`` (or ``_away``)
   239→   274→            Form: ``recent_results`` (list[str] 'W'/'D'/'L', most-recent first),
   240→   275→                  ``recent_goals_scored`` (list[float]),
   241→   276→                  ``last_match_date`` (str/datetime)
   242→   277→            Shots: ``shots_total``, ``shots_on_target``
   243→   278→            Venue: ``home_wins``, ``home_draws``, ``home_losses``,
   244→   279→                   ``away_wins``, ``away_draws``, ``away_losses``,
   245→   280→                   ``home_goals``, ``away_goals``,
   246→   281→                   ``clean_sheets_home``, ``clean_sheets_away``,
   247→   282→                   ``matches_played_home``, ``matches_played_away``
   248→   283→        h2h_data : list[dict], optional

   250→   285→            ``home_goals`` and ``away_goals`` (or ``score`` like '2-1').
   251→   286→        odds : dict, optional
   252→   287→            ``home_odds``, ``draw_odds``, ``away_odds`` (decimal).
   253→   288→
   254→   289→        Returns
   255→   290→        -------

   257→   292→            Mapping of all 50 feature names to float values.
   258→   293→        """
   259→   294→        feats: dict[str, float] = {}
   260→   295→
   261→   296→        # 1. ELO features ─────────────────────────────────────────────
   262→   297→        feats.update(self._elo_features(home_stats, away_stats))
   263→   298→
   264→   299→        # 2. Goal features ────────────────────────────────────────────
   265→   300→        feats.update(self._goal_features(home_stats, away_stats))
   266→   301→
   267→   302→        # 3. Form features ────────────────────────────────────────────
   268→   303→        feats.update(self._form_features(home_stats, away_stats))
   269→   304→
   270→   305→        # 4. Head-to-head features ─────────────────────────────────────
   271→   306→        feats.update(self._h2h_features(h2h_data))
   272→   307→
   273→   308→        # 5. Home/away split features ──────────────────────────────────
   274→   309→        feats.update(self._venue_features(home_stats, away_stats))
   275→   310→
   276→   311→        # 6. Market / odds features ────────────────────────────────────
   277→   312→        feats.update(self._market_features(odds))
   278→   313→
   279→   314→        # 7. Temporal features ─────────────────────────────────────────
   280→   315→        feats.update(self._temporal_features(match_data, home_stats, away_stats))
   281→   316→
   282→   317→        # 8. Advanced features ─────────────────────────────────────────
   283→   318→        feats.update(self._advanced_features(home_stats, away_stats))

   288→   323→    # ------------------------------------------------------------------
   289→   324→    def build_training_dataset(
   290→   325→        self,
   291→   326→        matches: list[dict[str, Any]],
   292→   327→        window: int = 10,
   293→   328→    ) -> pd.DataFrame:
   294→   329→        """Build a training dataset from historical matches.
   295→   330→
   296→   331→        Each match in *matches* must contain:
   297→   332→
   298→   333→        - ``home_team``, ``away_team``
   299→   334→        - ``home_goals``, ``away_goals``  (for target + stats)
   300→   335→        - ``date``
   301→   336→        - ``home_odds``, ``draw_odds``, ``away_odds``  (optional)
   302→   337→
   303→   338→        The method computes a sliding-window of *window* previous matches
   304→   339→        per team to derive form, streak, and trend features.  Missing
   305→   340→        historical data is back-filled with league-average defaults.
   306→   341→
   307→   342→        Parameters
   308→   343→        ----------
   309→   344→        matches : list[dict]
   310→   345→            Historical match records.  Must be **chronologically sorted**
   311→   346→            (oldest first) for correct sliding-window behaviour.
   312→   347→        window : int
   313→   348→            Number of recent matches to use for rolling form stats.
   314→   349→
   315→   350→        Returns
   316→   351→        -------
   317→   352→        pd.DataFrame
   318→   353→            One row per match.  Columns are the 50 features plus
   319→   354→            ``target`` (0=away, 1=draw, 2=home).
   320→   355→        """
   321→   356→        if not matches:
   322→   357→            return pd.DataFrame(columns=[*FEATURE_NAMES, "target"])
   323→   358→
   324→   359→        # Index: team_id → list of (date, goals_scored, goals_conceded, result)
   325→   360→        team_history: dict[str, list[tuple]] = defaultdict(list)
   326→   361→        # team_id → list of home-only / away-only records
   327→   362→        team_home_history: dict[str, list[tuple]] = defaultdict(list)
   328→   363→        team_away_history: dict[str, list[tuple]] = defaultdict(list)
   329→   364→
   330→   365→        rows: list[dict[str, float]] = []
   331→   366→
   332→   367→        for match in matches:
   333→   368→            home_team = str(match.get("home_team", ""))
   334→   369→            away_team = str(match.get("away_team", ""))
   335→   370→            hg = int(match.get("home_goals", 0))
   336→   371→            ag = int(match.get("away_goals", 0))
   337→   372→
   338→   373→            # Derive result codes (from home perspective)
   339→   374→            if hg > ag:
   340→   375→                home_result, away_result = "W", "L"
   341→   376→            elif hg < ag:
   342→   377→                home_result, away_result = "L", "W"
   343→   378→            else:
   344→   379→                home_result, away_result = "D", "D"
   345→   380→
   346→   381→            match_date = _parse_date(match.get("date")) or datetime.now()
   347→   382→
   348→   383→            # ── Build per-team stats dicts from rolling history ───────
   349→   384→            home_stats = self._stats_from_history(
   350→   385→                team_history[home_team],
   351→   386→                team_home_history[home_team],
   352→   387→                team_away_history[home_team],
   353→   388→                window=window,
   354→   389→            )
   355→   390→            away_stats = self._stats_from_history(
   356→   391→                team_history[away_team],
   357→   392→                team_home_history[away_team],
   358→   393→                team_away_history[away_team],
   359→   394→                window=window,
   360→   395→            )
   361→   396→
   362→   397→            # Set last-match date for rest-day calculation
   363→   398→            if team_history[home_team]:
   364→   399→                home_stats["last_match_date"] = team_history[home_team][-1][0].isoformat()
   365→   400→            if team_history[away_team]:
   366→   401→                away_stats["last_match_date"] = team_history[away_team][-1][0].isoformat()
   367→   402→
   368→   403→            # ── Head-to-head (use previous matches between these teams) ─
   369→   404→            h2h: list[dict[str, Any]] = []
   370→   405→            for prev in matches:
   371→   406→                if prev is match:
   372→   407→                    continue
   373→   408→                if (
   374→   409→                    str(prev.get("home_team")) == home_team
   375→   410→                    and str(prev.get("away_team")) == away_team
   376→   411→                ) or (
   377→   412→                    str(prev.get("home_team")) == away_team
   378→   413→                    and str(prev.get("away_team")) == home_team
   379→   414→                ):
   380→   415→                    phg = int(prev.get("home_goals", 0))
   381→   416→                    pag = int(prev.get("away_goals", 0))
   382→   417→                    # Normalise to current home perspective
   383→   418→                    if str(prev.get("home_team")) == home_team:
   384→   419→                        h2h.append({"home_goals": phg, "away_goals": pag})
   385→   420→                    else:
   386→   421→                        h2h.append({"home_goals": pag, "away_goals": phg})
   387→   422→
   388→   423→            # ── Odds ──────────────────────────────────────────────────
   389→   424→            odds: dict[str, Any] | None = None
   390→   425→            if match.get("home_odds") and match.get("draw_odds") and match.get("away_odds"):
   391→   426→                odds = {
   392→   427→                    "home_odds": float(match["home_odds"]),
   393→   428→                    "draw_odds": float(match["draw_odds"]),
   394→   429→                    "away_odds": float(match["away_odds"]),
   395→   430→                }
   396→   431→
   397→   432→            # ── Build features ────────────────────────────────────────
   398→   433→            feats = self.build_match_features(
   399→   434→                match_data={"date": match_date},
   400→   435→                home_stats=home_stats,
   401→   436→                away_stats=away_stats,
   402→   437→                h2h_data=h2h if h2h else None,
   403→   438→                odds=odds,
   404→   439→            )
   405→   440→
   406→   441→            # Target: 0=away, 1=draw, 2=home
   407→   442→            if hg > ag:
   408→   443→                target = 2
   409→   444→            elif hg < ag:
   410→   445→                target = 0
   411→   446→            else:
   412→   447→                target = 1
   413→   448→
   414→   449→            feats["target"] = float(target)
   415→   450→            rows.append(feats)
   416→   451→
   417→   452→            # ── Update histories for future windows ───────────────────
   418→   453→            team_history[home_team].append(
   419→   454→                (match_date, hg, ag, home_result)
   420→   455→            )
   421→   456→            team_history[away_team].append(
   422→   457→                (match_date, ag, hg, away_result)
   423→   458→            )
   424→   459→            team_home_history[home_team].append(
   425→   460→                (match_date, hg, ag, home_result)
   426→   461→            )
   427→   462→            team_away_history[away_team].append(
   428→   463→                (match_date, ag, hg, away_result)
   429→   464→            )
   430→   465→
   431→   466→        df = pd.DataFrame(rows)
   432→   467→        # Ensure column order
   433→   468→        cols = [*FEATURE_NAMES, "target"]
   434→   469→        for c in cols:
   435→   470→            if c not in df.columns:
   436→   471→                df[c] = 0.0
   437→   472→        return df[cols]

   543→   578→
   544→   579→    def _h2h_features(
   545→   580→        self, h2h_data: list[dict[str, Any]] | None
   546→   581→    ) -> dict[str, float]:
   547→   582→        if not h2h_data:
   548→   583→            return {
   549→   584→                "h2h_home_win_pct": 0.0,
   550→   585→                "h2h_away_win_pct": 0.0,
   551→   586→                "h2h_draw_pct": 0.0,
   552→   587→                "h2h_avg_goals": self.league_avg_goals * 2,
   553→   588→            }
   554→   589→
   555→   590→        n = len(h2h_data)
   556→   591→        home_wins = sum(1 for m in h2h_data if m.get("home_goals", 0) > m.get("away_goals", 0))
   557→   592→        away_wins = sum(1 for m in h2h_data if m.get("home_goals", 0) < m.get("away_goals", 0))
   558→   593→        draws = n - home_wins - away_wins
   559→   594→
   560→   595→        total_goals_list = [m.get("home_goals", 0) + m.get("away_goals", 0) for m in h2h_data]
   561→   596→        avg_goals = _safe_avg(total_goals_list, default=self.league_avg_goals * 2)
   562→   597→
   563→   598→        return {
   564→   599→            "h2h_home_win_pct": _safe_div(home_wins, n),
   565→   600→            "h2h_away_win_pct": _safe_div(away_wins, n),
   566→   601→            "h2h_draw_pct": _safe_div(draws, n),
   567→   602→            "h2h_avg_goals": avg_goals,
   568→   603→        }
   569→   604→
   570→   605→    # ---------- 5. Venue splits ----------------------------------------
   571→   606→
   572→   607→    def _venue_features(
   573→   608→        self, home_stats: dict[str, Any], away_stats: dict[str, Any]
   574→   609→    ) -> dict[str, float]:
   575→   610→        lag = self.league_avg_goals
   576→   611→
   577→   612→        # Home team's home record
   578→   613→        hw = int(home_stats.get("home_wins", 0))
   579→   614→        hd = int(home_stats.get("home_draws", 0))
   580→   615→        hl = int(home_stats.get("home_losses", 0))
   581→   616→        h_home_total = hw + hd + hl
   582→   617→        h_home_goals = float(home_stats.get("home_goals", lag))
   583→   618→
   584→   619→        # Away team's away record
   585→   620→        aw = int(away_stats.get("away_wins", 0))
   586→   621→        ad = int(away_stats.get("away_draws", 0))
   587→   622→        al = int(away_stats.get("away_losses", 0))
   588→   623→        a_away_total = aw + ad + al
   589→   624→        a_away_goals = float(away_stats.get("away_goals", lag))
   590→   625→
   591→   626→        return {
   592→   627→            "home_win_pct": _safe_div(hw, h_home_total),
   593→   628→            "away_win_pct": _safe_div(aw, a_away_total),
   594→   629→            "home_goals_avg": h_home_goals,
   595→   630→            "away_goals_avg": a_away_goals,
   596→   631→        }
   597→   632→
   598→   633→    # ---------- 6. Market / odds features ------------------------------
   599→   634→
   600→   635→    def _market_features(
   601→   636→        self, odds: dict[str, Any] | None
   602→   637→    ) -> dict[str, float]:
   603→   638→        default_odds = 1.0
   604→   639→        if odds is None:
   605→   640→            return {
   606→   641→                "implied_prob_home": 1.0 / 3.0,
   607→   642→                "implied_prob_draw": 1.0 / 3.0,
   608→   643→                "implied_prob_away": 1.0 / 3.0,
   609→   644→                "overround": 1.0,
   610→   645→            }
   611→   646→
   612→   647→        ho = float(odds.get("home_odds", default_odds))
   613→   648→        do = float(odds.get("draw_odds", default_odds))
   614→   649→        ao = float(odds.get("away_odds", default_odds))
   615→   650→
   616→   651→        # Guard against zero
   617→   652→        ho = max(ho, 1.01)
   618→   653→        do = max(do, 1.01)
   619→   654→        ao = max(ao, 1.01)
   620→   655→
   621→   656→        ip_h = _safe_div(1.0, ho, 1.0 / 3.0)
   622→   657→        ip_d = _safe_div(1.0, do, 1.0 / 3.0)
   623→   658→        ip_a = _safe_div(1.0, ao, 1.0 / 3.0)
   624→   659→        overround = ip_h + ip_d + ip_a
   625→   660→
   626→   661→        return {
   627→   662→            "implied_prob_home": ip_h,
   628→   663→            "implied_prob_draw": ip_d,
   629→   664→            "implied_prob_away": ip_a,
   630→   665→            "overround": overround,
   631→   666→        }
   632→   667→
   633→   668→    # ---------- 7. Temporal features -----------------------------------
   634→   669→
   635→   670→    def _temporal_features(
   636→   671→        self,
   637→   672→        match_data: dict[str, Any],
   638→   673→        home_stats: dict[str, Any],
   639→   674→        away_stats: dict[str, Any],
   640→   675→    ) -> dict[str, float]:
   641→   676→        match_date = _parse_date(match_data.get("date")) or datetime.now()
   642→   677→        dow = float(match_date.weekday())
   643→   678→        month = float(match_date.month)
   644→   679→        is_weekend = 1.0 if dow >= 5 else 0.0
   645→   680→
   646→   681→        # Rest days
   647→   682→        h_last = _parse_date(home_stats.get("last_match_date"))
   648→   683→        a_last = _parse_date(away_stats.get("last_match_date"))
   649→   684→
   650→   685→        default_rest = 7.0
   651→   686→        h_rest = default_rest
   652→   687→        a_rest = default_rest
   653→   688→        if h_last:
   654→   689→            delta = (match_date - h_last).days
   655→   690→            h_rest = float(max(delta, 0)) if delta is not None else default_rest
   656→   691→        if a_last:
   657→   692→            delta = (match_date - a_last).days
   658→   693→            a_rest = float(max(delta, 0)) if delta is not None else default_rest
   659→   694→
   660→   695→        return {
   661→   696→            "day_of_week": dow,
   662→   697→            "month": month,
   663→   698→            "is_weekend": is_weekend,
   664→   699→            "days_since_last_match_home": h_rest,
   665→   700→            "days_since_last_match_away": a_rest,
   666→   701→        }

   668→   703→    # ---------- 8. Advanced features -----------------------------------
   669→   704→
   670→   705→    def _advanced_features(
   671→   706→        self, home_stats: dict[str, Any], away_stats: dict[str, Any]
   672→   707→    ) -> dict[str, float]:
   673→   708→        lag = self.league_avg_goals
   674→   709→
   675→   710→        # Attacking / defensive strength (relative to league average)
   676→   711→        h_gs = float(home_stats.get("goals_scored", lag))
   677→   712→        h_gc = float(home_stats.get("goals_conceded", lag))
   678→   713→        a_gs = float(away_stats.get("goals_scored", lag))
   679→   714→        a_gc = float(away_stats.get("goals_conceded", lag))
   680→   715→
   681→   716→        atk_h = _safe_div(h_gs, lag, 1.0)
   682→   717→        atk_a = _safe_div(a_gs, lag, 1.0)
   683→   718→        def_h = _safe_div(h_gc, lag, 1.0)
   684→   719→        def_a = _safe_div(a_gc, lag, 1.0)
   685→   720→
   686→   721→        # Poisson lambda: attack * opponent defence * league average
   687→   722→        poisson_h = atk_h * def_a * lag
   688→   723→        poisson_a = atk_a * def_h * lag
   689→   724→
   690→   725→        # Shot conversion rates
   691→   726→        h_shots = float(home_stats.get("shots_total", 0.0))
   692→   727→        h_sot = float(home_stats.get("shots_on_target", 0.0))
   693→   728→        h_matches = max(float(home_stats.get("matches_played", 1)), 1.0)
   694→   729→        a_shots = float(away_stats.get("shots_total", 0.0))
   695→   730→        a_sot = float(away_stats.get("shots_on_target", 0.0))
   696→   731→        a_matches = max(float(away_stats.get("matches_played", 1)), 1.0)
   697→   732→
   698→   733→        h_conv = _safe_div(h_sot, h_shots, 0.33)
   699→   734→        a_conv = _safe_div(a_sot, a_shots, 0.33)
   700→   735→
   701→   736→        # Clean-sheet percentage
   702→   737→        h_cs = float(home_stats.get("clean_sheets_home", 0.0))
   703→   738→        h_home_m = max(float(home_stats.get("matches_played_home", 1)), 1.0)
   704→   739→        a_cs = float(away_stats.get("clean_sheets_away", 0.0))
   705→   740→        a_away_m = max(float(away_stats.get("matches_played_away", 1)), 1.0)
   706→   741→
   707→   742→        return {
   708→   743→            "attacking_strength_home": atk_h,
   709→   744→            "attacking_strength_away": atk_a,
   710→   745→            "defensive_strength_home": def_h,
   711→   746→            "defensive_strength_away": def_a,
   712→   747→            "poisson_lambda_home": poisson_h,
   713→   748→            "poisson_lambda_away": poisson_a,
   714→   749→            "shot_conversion_rate_home": h_conv,
   715→   750→            "shot_conversion_rate_away": a_conv,
   716→   751→            "clean_sheet_pct_home": _safe_div(h_cs, h_home_m),
   717→   752→            "clean_sheet_pct_away": _safe_div(a_cs, a_away_m),
   718→   753→        }
   719→   754→

   723→   758→
   724→   759→    @staticmethod
   725→   760→    def _stats_from_history(
   726→   761→        overall: list[tuple],
   727→   762→        home_only: list[tuple],
   728→   763→        away_only: list[tuple],
   729→   764→        window: int = 10,
   730→   765→    ) -> dict[str, Any]:
   731→   766→        """Derive a stats dict from rolling match history.
   732→   767→
   733→   768→        Each tuple is ``(date, goals_scored, goals_conceded, result_char)``.
   734→   769→        Uses the last *window* entries for form/streak, and full history
   735→   770→        for season averages.
   736→   771→        """
   737→   772→        stats: dict[str, Any] = {}
   738→   773→
   739→   774→        if not overall:
   740→   775→            return stats
   741→   776→
   742→   777→        # Full-season averages (all matches)
   743→   778→        gs_all = [m[1] for m in overall]
   744→   779→        gc_all = [m[2] for m in overall]
   745→   780→        stats["goals_scored"] = _safe_avg(gs_all)
   746→   781→        stats["goals_conceded"] = _safe_avg(gc_all)
   747→   782→        stats["matches_played"] = float(len(overall))
   748→   783→
   749→   784→        # Venue-split averages
   750→   785→        if home_only:
   751→   786→            stats["goals_scored_home"] = _safe_avg([m[1] for m in home_only])
   752→   787→            stats["goals_conceded_home"] = _safe_avg([m[2] for m in home_only])
   753→   788→            stats["home_wins"] = sum(1 for m in home_only if m[3] == "W")
   754→   789→            stats["home_draws"] = sum(1 for m in home_only if m[3] == "D")
   755→   790→            stats["home_losses"] = sum(1 for m in home_only if m[3] == "L")
   756→   791→            stats["home_goals"] = _safe_avg([m[1] for m in home_only])
   757→   792→            stats["matches_played_home"] = float(len(home_only))
   758→   793→            stats["clean_sheets_home"] = float(
   759→   794→                sum(1 for m in home_only if m[2] == 0)
   760→   795→            )
   761→   796→
   762→   797→        if away_only:
   763→   798→            stats["goals_scored_away"] = _safe_avg([m[1] for m in away_only])
   764→   799→            stats["goals_conceded_away"] = _safe_avg([m[2] for m in away_only])
   765→   800→            stats["away_wins"] = sum(1 for m in away_only if m[3] == "W")
   766→   801→            stats["away_draws"] = sum(1 for m in away_only if m[3] == "D")
   767→   802→            stats["away_losses"] = sum(1 for m in away_only if m[3] == "L")
   768→   803→            stats["away_goals"] = _safe_avg([m[1] for m in away_only])
   769→   804→            stats["matches_played_away"] = float(len(away_only))
   770→   805→            stats["clean_sheets_away"] = float(
   771→   806→                sum(1 for m in away_only if m[2] == 0)
   772→   807→            )
   773→   808→
   774→   809→        # Recent form (sliding window)
   775→   810→        recent = overall[-window:] if len(overall) >= window else overall
   776→   811→        stats["recent_results"] = [m[3] for m in recent]
   777→   812→        stats["recent_goals_scored"] = [m[1] for m in recent]
   778→   813→
   779→   814→        return stats
