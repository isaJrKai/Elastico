    55→
    56→interface EnhancedPlayer extends Player {
    57→  teamName?: string
    58→  teamCode?: string
    59→  teamColor?: string
    60→  nationality?: string
    61→  appearances?: number
    62→  minutesPlayed?: number
    63→}

    65→// ── Data honesty note ─────────────────────────────────────────────────────────
    66→// This view fetches players from /api/players (DB-backed).
    67→// FIFA-style radar attributes (Pace/Shooting/Passing/Defending/Physical/Dribbling)
    68→// and per-match form ratings are NOT computed by ELASTICO's engine and have no
    69→// real data source. These sections show honest empty states instead of
    70→// fabricating plausible-looking numbers.
    71→// ─────────────────────────────────────────────────────────────────────────────
    72→
    73→// ── Component ──────────────────────────────────────────────────────────────────
    74→
    75→export function PlayerView() {
    76→  const teams = useElasticoStore(s => s.teams)
    77→  const token = useElasticoStore(s => s.token)
    78→  const [players, setPlayers] = useState<EnhancedPlayer[]>([])
    79→  const [isLoading, setIsLoading] = useState(true)
    80→  const [search, setSearch] = useState('')
    81→  const [positionFilter, setPositionFilter] = useState<string>('all')
    82→  const [teamFilter, setTeamFilter] = useState<string>('all')
    83→  const [sortBy, setSortBy] = useState<string>('rating')
    84→  const [selectedPlayer, setSelectedPlayer] = useState<EnhancedPlayer | null>(null)
    85→  const [comparePlayer, setComparePlayer] = useState<EnhancedPlayer | null>(null)
    86→  const [page, setPage] = useState(0)
    87→  const perPage = 12
    88→
    89→  // Fetch players from API
    90→  useEffect(() => {
    91→    async function fetchPlayers() {
    92→      setIsLoading(true)
    93→      try {
    94→        const headers: Record<string, string> = {}
    95→        if (token) headers['Authorization'] = `Bearer ${token}`
    96→        const params = new URLSearchParams()
    97→        if (sortBy) params.set('sortBy', sortBy)
    98→        if (positionFilter !== 'all') params.set('position', positionFilter)
    99→        if (search) params.set('search', search)
   100→        params.set('limit', '200')
   101→        const res = await fetch(`/api/players?${params}`, { headers })
   102→        if (res.ok) {
   103→          const data = await res.json()
   104→          if (data.players && data.players.length > 0) {
   105→            setPlayers(data.players.map((p: Record<string, unknown>) => ({
   106→              ...p,
   107→              teamName: (p.team as Record<string, unknown>)?.name as string || p.teamName,
   108→              teamCode: (p.team as Record<string, unknown>)?.code as string || p.teamCode,
   109→              teamColor: (p.team as Record<string, unknown>)?.primaryColor as string || '#00e676',
   110→              nationality: (p as Record<string, unknown>).nationality as string || undefined,
   111→              appearances: (p as Record<string, unknown>).appearances as number || undefined,
   112→              minutesPlayed: (p as Record<string, unknown>).minutesPlayed as number || undefined,
   113→            } as EnhancedPlayer)))
   114→          }
   115→        }
   116→      } catch {
   117→        // API unavailable — players remains empty, empty state will show
   118→      } finally {
   119→        setIsLoading(false)
   120→      }
   121→    }
   122→    fetchPlayers()
   123→  }, [token, sortBy, positionFilter, search])
   124→
   125→  // Derived data
   126→  const filteredPlayers = useMemo(() => {
   127→    let result = [...players]
   128→    if (search) result = result.filter(p => p.name.toLowerCase().includes(search.toLowerCase()))
   129→    if (positionFilter !== 'all') result = result.filter(p => p.position === positionFilter)
   130→    if (teamFilter !== 'all') result = result.filter(p => p.teamName === teamFilter)
   131→    result.sort((a, b) => {
   132→      if (sortBy === 'rating') return b.rating - a.rating
   133→      if (sortBy === 'goals') return b.goals - a.goals
   134→      if (sortBy === 'assists') return b.assists - a.assists
   135→      if (sortBy === 'name') return a.name.localeCompare(b.name)
   136→      if (sortBy === 'age') return (a.age || 0) - (b.age || 0)
   137→      if (sortBy === 'marketValue') return (b.marketValue || 0) - (a.marketValue || 0)
   138→      return 0
   139→    })
   140→    return result
   141→  }, [players, search, positionFilter, teamFilter, sortBy])
   142→
   143→  const pagedPlayers = filteredPlayers.slice(page * perPage, (page + 1) * perPage)
   144→  const totalPages = Math.ceil(filteredPlayers.length / perPage)
   145→
   146→  const topScorers = useMemo(() => [...players].sort((a, b) => b.goals - a.goals).slice(0, 10), [players])
   147→  const topByValue = useMemo(() => [...players].filter(p => p.marketValue).sort((a, b) => (b.marketValue || 0) - (a.marketValue || 0)).slice(0, 10), [players])
   148→
   149→  const positionalGroups = useMemo(() => {
   150→    const groups: Record<string, EnhancedPlayer[]> = {}
   151→    for (const p of players) {
   152→      if (!groups[p.position]) groups[p.position] = []
   153→      groups[p.position].push(p)
   154→    }
   155→    return groups
   156→  }, [players])
   157→
   158→  const ageData = useMemo(() => {
   159→    const buckets: Record<string, number> = { '18-21': 0, '22-25': 0, '26-29': 0, '30-33': 0, '34+': 0 }
   160→    for (const p of players) {
   161→      const age = p.age || 25
   162→      if (age <= 21) buckets['18-21']++
   163→      else if (age <= 25) buckets['22-25']++
   164→      else if (age <= 29) buckets['26-29']++
   165→      else if (age <= 33) buckets['30-33']++
   166→      else buckets['34+']++
   167→    }
   168→    return Object.entries(buckets).map(([range, count]) => ({ range, count }))
   169→  }, [players])
   170→
   171→  const nationalities = useMemo(() => {
   172→    const map: Record<string, number> = {}
   173→    for (const p of players) {
   174→      const nat = p.nationality || 'Unknown'
   175→      map[nat] = (map[nat] || 0) + 1
   176→    }
   177→    return Object.entries(map).sort((a, b) => b[1] - a[1]).slice(0, 10)
   178→  }, [players])
   179→
   180→  const uniqueTeams = useMemo(() => [...new Set(players.map(p => p.teamName).filter(Boolean))], [players])
   181→
   182→  // Similar players — based on real stat proximity (goals, assists, rating, age)
   183→  // Not based on fabricated radar attributes
   184→  const similarPlayers = useMemo(() => {
   185→    if (!selectedPlayer || players.length === 0) return []
   186→    return players
   187→      .filter(p => p.id !== selectedPlayer.id && p.position === selectedPlayer.position)
   188→      .map(p => {
   189→        // Normalized difference across real metrics
   190→        const maxGoals = Math.max(selectedPlayer.goals, p.goals, 1)
   191→        const maxAssists = Math.max(selectedPlayer.assists, p.assists, 1)
   192→        const goalDiff = Math.abs(selectedPlayer.goals - p.goals) / maxGoals
   193→        const assistDiff = Math.abs(selectedPlayer.assists - p.assists) / maxAssists
   194→        const ratingDiff = Math.abs(selectedPlayer.rating - p.rating) / 10
   195→        const ageDiff = Math.abs((selectedPlayer.age || 25) - (p.age || 25)) / 20
   196→        const totalDiff = goalDiff + assistDiff + ratingDiff + ageDiff
   197→        return { player: p, similarity: Math.max(0, Math.round(100 - (totalDiff / 4) * 100)) }
   198→      })
   199→      .sort((a, b) => b.similarity - a.similarity)
   200→      .slice(0, 5)
   201→  }, [selectedPlayer, players])
   202→
   203→  // CSV Export
   204→  const exportCSV = useCallback(() => {
   205→    const headers = ['Name', 'Position', 'Team', 'Age', 'Goals', 'Assists', 'Rating', 'Market Value (€M)', 'Yellow Cards', 'Red Cards', 'Nationality']
   206→    const rows = filteredPlayers.map(p => [
   207→      p.name, p.position, p.teamName || '', p.age || '', p.goals, p.assists, p.rating, p.marketValue || '', p.yellowCards, p.redCards, p.nationality || ''
   208→    ])
   209→    const csv = [headers.join(','), ...rows.map(r => r.map(v => `"${v}"`).join(','))].join('\n')
   210→    const blob = new Blob([csv], { type: 'text/csv' })
   211→    const url = URL.createObjectURL(blob)
   212→    const a = document.createElement('a')
   213→    a.href = url
   214→    a.download = 'elastico-players.csv'
   215→    a.click()
   216→    URL.revokeObjectURL(url)
   217→    toast.success('Players exported to CSV')
   218→  }, [filteredPlayers])
   219→
   220→  const getPositionColor = (pos: string) => {
   221→    switch (pos) {
   222→      case 'GK': return 'text-yellow-400 bg-yellow-400/10 border-yellow-400/30'
   223→      case 'DEF': return 'text-blue-400 bg-blue-400/10 border-blue-400/30'
   224→      case 'MID': return 'text-emerald-400 bg-emerald-400/10 border-emerald-400/30'
   225→      case 'FWD': return 'text-red-400 bg-red-400/10 border-red-400/30'
   226→      default: return ''
   227→    }
   228→  }
   229→
   230→  const getRatingColor = (r: number) => {
   231→    if (r >= 8.0) return 'text-emerald-400'
   232→    if (r >= 7.0) return 'text-emerald-300'
   233→    if (r >= 6.0) return 'text-yellow-400'
   234→    return 'text-red-400'
   235→  }
   236→
   237→  return (
   238→    <motion.div
   239→      initial={{ opacity: 0, y: 20 }}
   240→      animate={{ opacity: 1, y: 0 }}
   241→      className="space-y-6"
   242→    >
   243→      {/* Header */}
   244→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   245→        <div>
   246→          <h1 className="text-2xl font-bold flex items-center gap-2">
   247→            <Users className="text-primary" /> Player Analytics
   248→          </h1>
   249→          <p className="text-muted-foreground text-sm mt-1">Search, analyze, and compare football players</p>
   250→        </div>
   251→        <Button variant="outline" size="sm" onClick={exportCSV} className="gap-2">
   252→          <Download className="size-4" /> Export CSV
   253→        </Button>
   254→      </div>
   255→
   256→      {/* Search & Filters */}
   257→      <div className="flex flex-col sm:flex-row gap-3">
   258→        <div className="relative flex-1">
   259→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   260→          <Input
   261→            placeholder="Search players..."
   262→            value={search}
   263→            onChange={(e) => { setSearch(e.target.value); setPage(0) }}
   264→            className="pl-9"
   265→          />
   266→        </div>
   267→        <Select value={positionFilter} onValueChange={(v) => { setPositionFilter(v); setPage(0) }}>
   268→          <SelectTrigger className="w-full sm:w-36">
   269→            <SelectValue placeholder="Position" />
   270→          </SelectTrigger>
   271→          <SelectContent>
   272→            <SelectItem value="all">All Positions</SelectItem>
   273→            <SelectItem value="GK">GK</SelectItem>
   274→            <SelectItem value="DEF">DEF</SelectItem>
   275→            <SelectItem value="MID">MID</SelectItem>
   276→            <SelectItem value="FWD">FWD</SelectItem>
   277→          </SelectContent>
   278→        </Select>
   279→        <Select value={teamFilter} onValueChange={(v) => { setTeamFilter(v); setPage(0) }}>
   280→          <SelectTrigger className="w-full sm:w-40">
   281→            <SelectValue placeholder="Team" />
   282→          </SelectTrigger>
   283→          <SelectContent>
   284→            <SelectItem value="all">All Teams</SelectItem>
   285→            {uniqueTeams.map(t => (
   286→              <SelectItem key={t} value={t!}>{t}</SelectItem>
   287→            ))}
   288→          </SelectContent>
   289→        </Select>
   290→        <Select value={sortBy} onValueChange={setSortBy}>
   291→          <SelectTrigger className="w-full sm:w-36">
   292→            <SelectValue placeholder="Sort by" />
   293→          </SelectTrigger>
   294→          <SelectContent>
   295→            <SelectItem value="rating">Rating</SelectItem>
   296→            <SelectItem value="goals">Goals</SelectItem>
   297→            <SelectItem value="assists">Assists</SelectItem>
   298→            <SelectItem value="marketValue">Market Value</SelectItem>
   299→            <SelectItem value="age">Age</SelectItem>
   300→            <SelectItem value="name">Name</SelectItem>
   301→          </SelectContent>
   302→        </Select>
   303→      </div>
   304→
   305→      <Tabs defaultValue="grid" className="space-y-4">
   306→        <TabsList className="flex flex-wrap gap-1 h-auto bg-card/50 p-1 rounded-lg">
   307→          <TabsTrigger value="grid" className="text-xs">Player Grid</TabsTrigger>
   308→          <TabsTrigger value="leaderboard" className="text-xs">Top Scorers</TabsTrigger>
   309→          <TabsTrigger value="radar" className="text-xs">Radar Charts</TabsTrigger>
   310→          <TabsTrigger value="compare" className="text-xs">Compare</TabsTrigger>
   311→          <TabsTrigger value="value" className="text-xs">Market Value</TabsTrigger>
   312→          <TabsTrigger value="positions" className="text-xs">By Position</TabsTrigger>
   313→          <TabsTrigger value="age" className="text-xs">Age Dist.</TabsTrigger>
   314→          <TabsTrigger value="nationality" className="text-xs">Nationality</TabsTrigger>
   315→          <TabsTrigger value="cards" className="text-xs">Cards</TabsTrigger>
   316→          <TabsTrigger value="subs" className="text-xs">Substitutions</TabsTrigger>
   317→          <TabsTrigger value="similar" className="text-xs">Similar</TabsTrigger>
   318→          <TabsTrigger value="form" className="text-xs">Form</TabsTrigger>
   319→        </TabsList>
   320→
   321→        {/* Player Cards Grid */}
   322→        <TabsContent value="grid">
   323→          {isLoading ? (
   324→            <Card className="glass-card"><CardContent className="p-12 text-center text-muted-foreground">Loading players...</CardContent></Card>
   325→          ) : pagedPlayers.length === 0 ? (
   326→            <Card className="glass-card"><CardContent className="p-12 text-center">
   327→              <Users className="size-8 mx-auto text-muted-foreground/50 mb-3" />
   328→              <p className="text-sm text-muted-foreground">No players found in the database.</p>
   329→              <p className="text-xs text-muted-foreground mt-1">Player data is populated when teams and matches are synced. Use the admin panel or /api/sync to load data.</p>
   330→            </CardContent></Card>
   331→          ) : (
   332→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4">
   333→            {pagedPlayers.map((player) => (
   334→              <motion.div
   335→                key={player.id}
   336→                whileHover={{ scale: 1.02, y: -4 }}
   337→                whileTap={{ scale: 0.98 }}
   338→                className="cursor-pointer"
   339→                onClick={() => setSelectedPlayer(player)}
   340→              >
   341→                <Card className="glass-card h-full overflow-hidden">
   342→                  <div className="h-2" style={{ backgroundColor: player.teamColor || '#00e676' }} />
   343→                  <CardContent className="p-4">
   344→                    <div className="flex items-start gap-3">
   345→                      <div className="w-14 h-14 rounded-full bg-muted flex items-center justify-center text-lg font-bold shrink-0 border-2 border-border">
   346→                        {player.number}
   347→                      </div>
   348→                      <div className="min-w-0 flex-1">
   349→                        <h3 className="text-sm font-semibold truncate">{player.name}</h3>
   350→                        <div className="flex items-center gap-2 mt-1">
   351→                          <Badge variant="outline" className={cn('text-[10px]', getPositionColor(player.position))}>
   352→                            {player.position}
   353→                          </Badge>
   354→                          <span className="text-[10px] text-muted-foreground truncate">{player.teamName}</span>
   355→                        </div>
   356→                      </div>
   357→                      <div className={cn('text-xl font-bold', getRatingColor(player.rating))}>
   358→                        {player.rating.toFixed(1)}
   359→                      </div>
   360→                    </div>
   361→                    <div className="grid grid-cols-3 gap-2 mt-4">
   362→                      <div className="text-center">
   363→                        <div className="text-lg font-bold text-primary">{player.goals}</div>
   364→                        <div className="text-[10px] text-muted-foreground">Goals</div>
   365→                      </div>
   366→                      <div className="text-center">
   367→                        <div className="text-lg font-bold text-primary">{player.assists}</div>
   368→                        <div className="text-[10px] text-muted-foreground">Assists</div>
   369→                      </div>
   370→                      <div className="text-center">
   371→                        <div className="text-lg font-bold text-primary">{player.yellowCards + player.redCards}</div>
   372→                        <div className="text-[10px] text-muted-foreground">Cards</div>
   373→                      </div>
   374→                    </div>
   375→                    {player.marketValue && (
   376→                      <div className="mt-3 pt-3 border-t border-border/50 flex items-center gap-1">
   377→                        <TrendingUp className="size-3 text-muted-foreground" />
   378→                        <span className="text-xs text-muted-foreground">€{player.marketValue}M</span>
   379→                      </div>
   380→                    )}
   381→                  </CardContent>
   382→                </Card>
   383→              </motion.div>
   384→            ))}
   385→          </div>
   386→          {/* Pagination */}
   387→          {totalPages > 1 && (
   388→            <div className="flex items-center justify-center gap-2 mt-4">
   389→              <Button variant="outline" size="icon" className="h-8 w-8" onClick={() => setPage(Math.max(0, page - 1))} disabled={page === 0}>
   390→                <ChevronLeft className="size-4" />
   391→              </Button>
   392→              <span className="text-sm text-muted-foreground">{page + 1} / {totalPages}</span>
   393→              <Button variant="outline" size="icon" className="h-8 w-8" onClick={() => setPage(Math.min(totalPages - 1, page + 1))} disabled={page >= totalPages - 1}>
   394→                <ChevronRight className="size-4" />
   395→              </Button>
   396→            </div>
   397→          )}
   398→          )}
   399→        </TabsContent>
   400→
   401→        {/* Top Scorers Leaderboard */}
   402→        <TabsContent value="leaderboard">
   403→          <Card className="glass-card">
   404→            <CardHeader className="pb-2">
   405→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   406→                <Star className="size-4 text-yellow-400" /> Top Scorers Leaderboard
   407→              </CardTitle>
   408→            </CardHeader>
   409→            <CardContent>
   410→              <div className="max-h-96 overflow-y-auto">
   411→                <Table>
   412→                  <TableHeader>
   413→                    <TableRow>
   414→                      <TableHead className="w-10">#</TableHead>
   415→                      <TableHead>Player</TableHead>
   416→                      <TableHead className="text-center">Goals</TableHead>
   417→                      <TableHead className="text-center">Assists</TableHead>
   418→                      <TableHead className="text-center">Mins</TableHead>
   419→                      <TableHead className="text-center">G/90</TableHead>
   420→                      <TableHead className="text-center">Rating</TableHead>
   421→                    </TableRow>
   422→                  </TableHeader>
   423→                  <TableBody>
   424→                    {topScorers.map((p, i) => (
   425→                      <TableRow key={p.id} className="cursor-pointer hover:bg-accent/50" onClick={() => setSelectedPlayer(p)}>
   426→                        <TableCell className="font-bold text-muted-foreground">{i + 1}</TableCell>
   427→                        <TableCell>
   428→                          <div className="flex items-center gap-2">
   429→                            <span className="text-sm font-medium">{p.name}</span>
   430→                            <Badge variant="outline" className={cn('text-[9px]', getPositionColor(p.position))}>{p.position}</Badge>
   431→                          </div>
   432→                          <span className="text-[10px] text-muted-foreground">{p.teamName}</span>
   433→                        </TableCell>
   434→                        <TableCell className="text-center font-bold text-primary">{p.goals}</TableCell>
   435→                        <TableCell className="text-center">{p.assists}</TableCell>
   436→                        <TableCell className="text-center text-muted-foreground">{p.minutesPlayed || (p.appearances || 0) * 90}</TableCell>
   437→                        <TableCell className="text-center font-medium">
   438→                          {((p.goals / ((p.minutesPlayed || (p.appearances || 0) * 90) / 90))).toFixed(2)}
   439→                        </TableCell>
   440→                        <TableCell className={cn('text-center font-bold', getRatingColor(p.rating))}>{p.rating.toFixed(1)}</TableCell>
   441→                      </TableRow>
   442→                    ))}
   443→                  </TableBody>
   444→                </Table>
   445→              </div>
   446→            </CardContent>
   447→          </Card>
   448→        </TabsContent>
   449→
   450→        {/* Player Radar Chart */}
   451→        <TabsContent value="radar">
   452→          <Card className="glass-card">
   453→            <CardHeader className="pb-2">
   454→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   455→                <Award className="size-4 text-primary" /> Player Radar Charts
   456→              </CardTitle>
   457→            </CardHeader>
   458→            <CardContent>
   459→              <div className="p-8 text-center">
   460→                <Award className="size-8 mx-auto text-muted-foreground/50 mb-3" />
   461→                <p className="text-sm text-muted-foreground">FIFA-style attribute radars (Pace, Shooting, Passing, Defending, Physical, Dribbling) are not a metric ELASTICO computes.</p>
   462→                <p className="text-xs text-muted-foreground mt-1">This feature requires an external attribute data source. Showing real stats (goals, assists, rating) instead — see the Player Grid and Leaderboard tabs.</p>
   463→              </div>
   464→            </CardContent>
   465→          </Card>
   466→        </TabsContent>
