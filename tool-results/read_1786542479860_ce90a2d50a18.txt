    43→    43→function getEloProb(eloA: number, eloB: number) {
    44→    44→  const eA = 1 / (1 + Math.pow(10, (eloB - eloA) / 400))
    45→    45→  return { home: +(eA * 100).toFixed(1), away: +((1 - eA) * 100).toFixed(1), draw: +(Math.min(eA, 1 - eA) * 30).toFixed(1) }
    46→    46→}
