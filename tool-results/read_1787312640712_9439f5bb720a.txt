   164→   164→function NvidiaApiStatusBadge({ token }: { token?: string }) {
   165→   165→  const [status, setStatus] = React.useState<'loading' | 'connected' | 'disconnected'>('loading')
   166→   166→
   167→   167→  React.useEffect(() => {
   168→   168→    fetch('/api/chat', {
   169→   169→      method: 'POST',
   170→   170→      headers: {
   171→   171→        'Content-Type': 'application/json',
   172→   172→        ...(token ? { Authorization: `Bearer ${token}` } : {}),
   173→   173→      },
   174→   174→      body: JSON.stringify({ message: '__status_check__', model: 'pro', stream: false }),
   175→   175→    })
   176→   176→      .then((res) => res.json())
   177→   177→      .then((data) => {
   178→   178→        if (data.model === 'pro') {
   179→   179→          setStatus('connected')
   180→   180→        } else if (data.model === 'mock-fallback') {
   181→   181→          setStatus('disconnected')
   182→   182→        } else {
   183→   183→          setStatus('disconnected')
   184→   184→        }
   185→   185→      })
   186→   186→      .catch(() => setStatus('disconnected'))
   187→   187→  }, [])
   188→   188→
   189→   189→  if (status === 'loading') {
   190→   190→    return (
   191→   191→      <Badge variant="outline" className="gap-1.5 text-xs border-border">
   192→   192→        <Loader2 className="size-3 animate-spin text-muted-foreground" />
   193→   193→        Checking...
   194→   194→      </Badge>
   195→   195→    )
   196→   196→  }
   197→   197→
   198→   198→  if (status === 'connected') {
   199→   199→    return (
   200→   200→      <Badge variant="outline" className="gap-1.5 text-xs border-emerald-500/30 text-emerald-400 bg-emerald-500/5">
   201→   201→        <CheckCircle2 className="size-3" />
   202→   202→        Connected
   203→   203→      </Badge>
   204→   204→    )
   205→   205→  }
   206→   206→
   207→   207→  return (
   208→   208→    <Badge variant="outline" className="gap-1.5 text-xs border-amber-500/30 text-amber-400 bg-amber-500/5">
   209→   209→      <XCircle className="size-3" />
   210→   210→      Not Configured
   211→   211→    </Badge>
   212→   212→  )
   213→   213→}
