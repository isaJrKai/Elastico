   122→function statusColor(status: string) {
   123→  if (status === 'PASS' || status === 'HEALED' || status === 'INTACT' || status === 'HEALTHY' || status === 'ONLINE') return '#00e676'
   124→  if (status === 'WARNING' || status === 'DRIFT_DETECTED' || status === 'QUARANTINED' || status === 'MODIFIED') return '#eab308'
   125→  return '#ef4444'
   126→}
   127→
   128→function statusBgClass(status: string) {
   129→  const c = statusColor(status)
   130→  if (c === '#00e676') return 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30'
   131→  if (c === '#eab308') return 'bg-yellow-500/15 text-yellow-400 border-yellow-500/30'
   132→  return 'bg-red-500/15 text-red-400 border-red-500/30'
   133→}
   134→
   135→function CircGauge({ value, max = 100, size = 120, label, strokeColor }: {
   136→  value: number; max?: number; size?: number; label: string; strokeColor?: string
   137→}) {
   138→  const pct = Math.min(value / max, 1)
   139→  const r = (size - 16) / 2
   140→  const circ = 2 * Math.PI * r
   141→  const offset = circ * (1 - pct)
   142→  const color = strokeColor || (pct > 0.7 ? '#00e676' : pct > 0.4 ? '#eab308' : '#ef4444')
   143→
   144→  return (
   145→    <div className="flex flex-col items-center gap-2">
   146→      <svg width={size} height={size} className="drop-shadow-lg">
   147→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="oklch(0.15 0.02 260)" strokeWidth={8} />
   148→        <circle
   149→          cx={size / 2} cy={size / 2} r={r} fill="none"
   150→          stroke={color} strokeWidth={8} strokeLinecap="round"
   151→          strokeDasharray={circ} strokeDashoffset={offset}
   152→          transform={`rotate(-90 ${size / 2} ${size / 2})`}
   153→          className="transition-all duration-1000 ease-out"
   154→        />
   155→        <text x={size / 2} y={size / 2 - 4} textAnchor="middle" className="fill-foreground text-2xl font-bold" dominantBaseline="middle">
   156→          {Math.round(value)}
   157→        </text>
   158→        <text x={size / 2} y={size / 2 + 18} textAnchor="middle" className="fill-muted-foreground text-xs" dominantBaseline="middle">
   159→          / {max}
   160→        </text>
   161→      </svg>
   162→      <span className="text-xs text-muted-foreground font-medium">{label}</span>
   163→    </div>
   164→  )
   165→}
