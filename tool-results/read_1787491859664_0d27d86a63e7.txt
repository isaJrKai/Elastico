    54→async function syncFixturesFromApiSports(): Promise<{
    55→  created: number
    56→  updated: number
    57→  processed: number
    58→}> {
    59→  const result = { created: 0, updated: 0, processed: 0 }
    60→
    61→  try {
    62→    // 1. Live fixtures (1 API call — covers all leagues)
    63→    const liveFixtures = await fetchLiveFixtures()
    64→    console.log(`[SYNC/API-Sports] Got ${liveFixtures.length} live fixtures`)
    65→
    66→    for (const f of liveFixtures) {
    67→      checkTimeout()
    68→      await upsertApiSportsFixture(f, result)
    69→    }
    70→
    71→    // 2. Today's fixtures from top 5 leagues (5 API calls)
    72→    const top5 = AS_LEAGUES.slice(0, 5) // PL, LIGA, SA, BL, L1
    73→    for (const league of top5) {
    74→      checkTimeout()
    75→      try {
    76→        const fixtures = await fetchFixtures(league.id)
    77→        for (const f of fixtures) {
    78→          checkTimeout()
    79→          await upsertApiSportsFixture(f, result)
    80→        }
    81→      } catch (err: any) {
    82→        if (err.message?.includes('timeout')) throw err
    83→        console.warn(`[SYNC/API-Sports] Failed fixtures for ${league.code}:`, err.message)
    84→      }
    85→    }
    86→  } catch (err: any) {
    87→    if (err.message?.includes('timeout')) throw err
    88→    console.warn('[SYNC/API-Sports] Fixtures failed:', err.message || err)
    89→  }
    90→
    91→  return result
    92→}

   103→    'UNKNOWN'

   109→      name: f.teams.home.name,
   110→      logo: f.teams.home.logo,
   111→      league: f.league?.name || '',
   112→      leagueCode,
   113→      source: 'api-sports',

   115→    },
   116→    create: {
   117→      externalId: String(f.teams.home.id),
   118→      name: f.teams.home.name,
   119→      code: f.teams.home.name.substring(0, 3).toUpperCase(),
   120→      logo: f.teams.home.logo,

   176→    await db.match.upsert({
   177→      where: { source_sourceId: { source: 'api-sports', sourceId: String(f.id) } },
   178→      update: matchData,
   179→      create: matchData,
   180→    })

   184→  }

   267→// UNDERSTAT ANALYTICAL ENRICHMENT (Phase 1: team xG/xGA only)
   268→// Runs AFTER API-Sports sync so that canonical teams exist in DB.
   269→// ═════════════════════════════════════════════════════════════════════════
   270→
   271→const UNDERSTAT_LEAGUES = ['PL', 'LIGA', 'SA', 'BL', 'L1'] as const
   272→const UNDERSTAT_SEASON = '2024' // Understat uses calendar year for 24/25 season
   273→
   274→async function syncUnderstatAnalytics(): Promise<{
   275→  resolved: number
   276→  unresolved: number
   277→  updated: number
   278→}> {
   279→  const result = { resolved: 0, unresolved: 0, updated: 0 }
   280→
   281→  for (const leagueCode of UNDERSTAT_LEAGUES) {
   282→    checkTimeout()
   283→    try {
   284→      const understatTeams = await fetchUnderstatLeagueTeams(leagueCode, parseInt(UNDERSTAT_SEASON))
   285→      if (understatTeams.length === 0) continue
   286→
   287→      const { resolved, unresolved } = await resolveUnderstatTeams(
   288→        db, understatTeams, leagueCode, UNDERSTAT_SEASON,
   289→      )
   290→
   291→      result.resolved += resolved.length
   292→      result.unresolved += unresolved.length
   293→
   294→      // Upsert resolved teams' analytics
   295→      for (const r of resolved) {
   296→        checkTimeout()
   297→        try {
   298→          const ut = understatTeams.find(t => t.id === r.understatTeamId)
   299→          if (!ut) continue
   300→
   301→          // Validate: xG must be non-negative
   302→          const xg = parseFloat(String(ut.xG)) || 0
   303→          const xga = parseFloat(String(ut.xGA)) || 0
   304→          const npxg = parseFloat(String(ut.npxG)) || 0
   305→          const npxga = parseFloat(String(ut.npxGA)) || 0
   306→          if (xg < 0 || xga < 0 || npxg < 0 || npxga < 0) {
   307→            console.warn(`[SYNC/Understat] Invalid xG values for ${r.matchedDbTeamName}: xG=${xg}, xGA=${xga}`)
   308→            continue
   309→          }
   310→
   311→          // Calculate per-game values (Understat xG is season total)
   312→          const matchesPlayed = (ut.wins || 0) + (ut.draws || 0) + (ut.losses || 0)
   313→          const xgPerGame = matchesPlayed > 0 ? Math.round((xg / matchesPlayed) * 100) / 100 : null
   314→          const xgaPerGame = matchesPlayed > 0 ? Math.round((xga / matchesPlayed) * 100) / 100 : null
   315→          const ppda = ut.ppda?.def > 0 ? Math.round((ut.ppda.att / ut.ppda.def) * 100) / 100 : null
   316→          const deep = matchesPlayed > 0 ? Math.round((ut.deep / matchesPlayed) * 100) / 100 : null
   317→          const ppdaAllowed = ut.ppda_allowed?.def > 0
   318→            ? Math.round((ut.ppda_allowed.att / ut.ppda_allowed.def) * 100) / 100
   319→            : null
   320→          const deepAllowed = matchesPlayed > 0
   321→            ? Math.round((ut.deep_allowed / matchesPlayed) * 100) / 100
   322→            : null
   323→
   324→          await db.teamAnalytic.upsert({
   325→            where: {
   326→              teamId_source_season_leagueCode: {
   327→                teamId: r.dbTeamId,
   328→                source: 'understat',
   329→                season: UNDERSTAT_SEASON,
   330→                leagueCode,
   331→              },
   332→            },
   333→            update: {
   334→              xgPerGame,
   335→              xgaPerGame,
   336→              npxGPerGame: matchesPlayed > 0 ? Math.round((npxg / matchesPlayed) * 100) / 100 : null,
   337→              npxgaPerGame: matchesPlayed > 0 ? Math.round((npxga / matchesPlayed) * 100) / 100 : null,
   338→              ppda,
   339→              ppdaAllowed,
   340→              deep,
   341→              deepAllowed,
   342→              syncedAt: new Date(),
   343→            },
   344→            create: {
   345→              teamId: r.dbTeamId,
   346→              source: 'understat',
   347→              season: UNDERSTAT_SEASON,
   348→              leagueCode,
   349→              xgPerGame,
   350→              xgaPerGame,
   351→              npxGPerGame: matchesPlayed > 0 ? Math.round((npxg / matchesPlayed) * 100) / 100 : null,
   352→              npxgaPerGame: matchesPlayed > 0 ? Math.round((npxga / matchesPlayed) * 100) / 100 : null,
   353→              ppda,
   354→              ppdaAllowed,
   355→              deep,
   356→              deepAllowed,
   357→              sourceTeamId: String(ut.id),
   358→              sourceTeamName: r.understatTeamName,
   359→              syncedAt: new Date(),
   360→            },
   361→          })
   362→          result.updated++
   363→        } catch (err) {
   364→          console.warn(`[SYNC/Understat] Failed to upsert for ${r.matchedDbTeamName}:`, err)
   365→        }
   366→      }
   367→
   368→      // Log unresolved teams
   369→      for (const u of unresolved) {
   370→        console.warn(
   371→          `[SYNC/Understat] UNRESOLVED: ${u.understatTeamName} ` +
   372→          `(id=${u.understatTeamId}, league=${leagueCode})`,
   373→        )
   374→      }
   375→    } catch (err: any) {
   376→      if (err.message?.includes('timeout')) throw err
   377→      console.warn(`[SYNC/Understat] Failed for ${leagueCode}:`, err.message)
   378→    }
   379→  }
   380→
   381→  await logSync('understat', 'team_analytics', 'success',
   382→    result.resolved + result.unresolved, result.resolved, result.updated)
   383→
   384→  return result
   385→}

   548→async function fetchScoreboardForLeague(league: { espnId: string; name: string; code: string }) {
   549→  const SITE_V2 = 'https://site.api.espn.com/apis/v2/sports/soccer'
   550→  const url = `${SITE_V2}/${league.espnId}/scoreboard`
   551→  try {
   552→    const res = await fetch(url, { cache: 'no-store' })
   553→    if (!res.ok) return []
   554→    const data = await res.json()
   555→    const events: any[] = []
   556→    for (const event of data.events || []) {
   557→      const comp = event.competitions?.[0]
   558→      if (!comp) continue
   559→      const competitors = comp.competitors || []
   560→      if (competitors.length < 2) continue
   561→      const home = competitors.find((c: any) => c.homeAway === 'home') || competitors[0]
   562→      const away = competitors.find((c: any) => c.homeAway === 'away') || competitors[1]
   563→      const statusDetail = comp.status?.type?.detail || ''
   564→      const minuteMatch = statusDetail.match(/(\d+)\s*'/)
   565→      events.push({
   566→        id: event.id?.toString() || '',
   567→        homeTeamId: home.team?.id?.toString() || '',
   568→        homeTeamName: home.team?.displayName || 'Unknown',
   569→        homeTeamAbbr: home.team?.abbreviation || '???',
   570→        homeTeamLogo: home.team?.logo || '',

   573→        awayTeamName: away.team?.displayName || 'Unknown',

   575→        awayTeamLogo: away.team?.logo || '',
   576→        awayTeamColor: away.team?.color || '#ffffff',
   577→        homeScore: parseInt(home.score) || 0,
   578→        awayScore: parseInt(away.score) || 0,
   579→        status: comp.status?.type?.name || 'STATUS_SCHEDULED',

   586→    }

   591→  }

   595→  created: number
   596→  updated: number
   597→  processed: number
   598→}> {
   599→  const result = { created: 0, updated: 0, processed: 0 }
   600→
   601→  const results = await Promise.allSettled(
   602→    ESPN_LEAGUES.map(league => fetchScoreboardForLeague(league))
   603→  )
   604→
   605→  for (const r of results) {
   606→    if (r.status !== 'fulfilled') continue
   607→    const matches = r.value
   608→    for (const m of matches) {
   609→      checkTimeout()
   610→      result.processed++
   611→
   612→      const homeTeam = await db.team.upsert({
   613→        where: { source_sourceId: { source: 'espn', sourceId: m.homeTeamId } },

   619→          leagueCode: m.competitionCode,
   620→          source: 'espn',
   621→          lastSyncedAt: new Date(),
   622→        },
   623→        create: {

   625→          name: m.homeTeamName,
   626→          code: m.homeTeamAbbr,
   627→          logo: m.homeTeamLogo,
   628→          primaryColor: m.homeTeamColor,
   629→          leagueCode: m.competitionCode,
   630→          source: 'espn',
   631→          sourceId: m.homeTeamId,
   632→        },
   633→      })
   634→
   635→      const awayTeam = await db.team.upsert({
   636→        where: { source_sourceId: { source: 'espn', sourceId: m.awayTeamId } },
   637→        update: {
   638→          name: m.awayTeamName,
   639→          code: m.awayTeamAbbr,
   640→          logo: m.awayTeamLogo,
   641→          primaryColor: m.awayTeamColor,
   642→          leagueCode: m.competitionCode,
   643→          source: 'espn',

   646→        create: {
   647→          externalId: m.awayTeamId,
   648→          name: m.awayTeamName,
   649→          code: m.awayTeamAbbr,
   650→          logo: m.awayTeamLogo,
   651→          primaryColor: m.awayTeamColor,
   652→          leagueCode: m.competitionCode,
   653→          source: 'espn',
   654→          sourceId: m.awayTeamId,
   655→        },
   656→      })
   657→
   658→      const matchData = {
   659→        externalId: m.id,
   660→        homeTeamId: homeTeam.id,
   661→        awayTeamId: awayTeam.id,
   662→        competition: m.competition,
   663→        competitionCode: m.competitionCode,
   664→        date: m.date ? new Date(m.date) : null,
   665→        status: mapStatus(m.status),
   666→        minute: m.minute ?? null,
   667→        venue: m.venue,
   668→        homeScore: m.homeScore,
   669→        awayScore: m.awayScore,
   670→        source: 'espn',
   671→        sourceId: m.id,
   672→        lastSyncedAt: new Date(),
   673→      }
   674→
   675→      try {
   676→        await db.match.upsert({
   677→          where: { source_sourceId: { source: 'espn', sourceId: m.id } },
   678→          update: matchData,
   679→          create: matchData,
   680→        })
   681→        result.created++
   682→      } catch {
   683→        result.updated++
   684→      }
   685→    }
   686→  }
   687→
   688→  return result
   689→}

   779→async function logSync(
   780→  source: string,
   781→  action: string,
   782→  status: string,
   783→  processed: number,
   784→  created: number,
   785→  updated: number,
   786→  errorMessage?: string
   787→) {
   788→  try {
   789→    await db.syncLog.create({
   790→      data: {
   791→        source,
   792→        action,
   793→        status,
   794→        recordsProcessed: processed,
   795→        recordsCreated: created,
   796→        recordsUpdated: updated,
   797→        durationMs: elapsed(),
   798→        errorMessage: errorMessage || null,
   799→      },
   800→    })
   801→  } catch (err) {
   802→    console.error('[SYNC] Failed to write SyncLog:', err)
   803→  }
   804→}

   806→// ═══════════════════════════════════════════════════════════════════════════════
   807→// MAIN HANDLER
   808→// ═══════════════════════════════════════════════════════════════════════════════
   809→
   810→export async function GET(req: NextRequest) {
   811→  // 1. Validate cron secret
   812→  if (!validateCron(req)) {
   813→    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
   814→  }
   815→
   816→  const summary: any = {
   817→    success: true,
   818→    syncs: {
   819→      fixtures: { source: 'none', created: 0, updated: 0 },
   820→      standings: { source: 'none', created: 0, updated: 0 },
   821→      teams: { source: 'none', created: 0, updated: 0 },
   822→      players: { source: 'none', created: 0, updated: 0 },
   823→      understat: { source: 'none', resolved: 0, unresolved: 0, updated: 0 },
   824→    },
   825→    durationMs: 0,
   826→  }
   827→
   828→  try {
   829→    // ═══ 2. API-Sports: Fixtures + Standings + Teams + Players ═══
   830→    let apiSportsAvailable = false
   831→
   832→    // 2a. Fixtures (live + today's top 5 leagues)
   833→    try {
   834→      const fixResult = await syncFixturesFromApiSports()
   835→      if (fixResult.processed > 0) {
   836→        apiSportsAvailable = true
   837→        summary.syncs.fixtures = {
   838→          source: 'api-sports',
   839→          created: fixResult.created,
   840→          updated: fixResult.updated,
   841→        }
   842→        await logSync('api-sports', 'fixtures', 'success', fixResult.processed, fixResult.created, fixResult.updated)
   843→      }
   844→    } catch (err: any) {
   845→      if (err.message?.includes('timeout')) {
   846→        console.warn('[SYNC] API-Sports fixtures timeout')
   847→        await logSync('api-sports', 'fixtures', 'partial', 0, 0, 0, err.message)
   848→      } else {
   849→        console.warn('[SYNC] API-Sports fixtures failed:', err.message)
   850→      }
   851→    }
   852→
   853→    // 2b. Standings (top 10 leagues — only if API-Sports is working)
   854→    if (apiSportsAvailable) {
   855→      try {
   856→        const stResult = await syncStandingsFromApiSports()
   857→        summary.syncs.standings = {
   858→          source: 'api-sports',
   859→          created: stResult.created,
   860→          updated: stResult.updated,
   861→        }
   862→        await logSync('api-sports', 'standings', 'success', stResult.created, stResult.created, stResult.updated)
   863→      } catch (err: any) {
   864→        if (err.message?.includes('timeout')) {
   865→          console.warn('[SYNC] API-Sports standings timeout')
   866→          await logSync('api-sports', 'standings', 'partial', 0, 0, 0, err.message)
   867→        } else {
   868→          console.warn('[SYNC] API-Sports standings failed:', err.message)
   869→        }
   870→      }
   871→    }
   872→
   873→    // 2c. Teams (top 5 leagues — venue info, country, etc.)
   874→    if (apiSportsAvailable) {
   875→      try {
   876→        const tmResult = await syncTeamsFromApiSports()
   877→        summary.syncs.teams = {
   878→          source: 'api-sports',
   879→          created: tmResult.created,
   880→          updated: tmResult.updated,
   881→        }
   882→        await logSync('api-sports', 'teams', 'success', tmResult.created + tmResult.updated, tmResult.created, tmResult.updated)
   883→      } catch (err: any) {
   884→        if (err.message?.includes('timeout')) {
   885→          console.warn('[SYNC] API-Sports teams timeout')
   886→        } else {
   887→          console.warn('[SYNC] API-Sports teams failed:', err.message)
   888→        }
   889→      }
   890→    }
   891→
   892→    // 2d. Player stats (top scorers, top 5 leagues)
   893→    if (apiSportsAvailable) {
   894→      try {
   895→        const plResult = await syncPlayersFromApiSports()
   896→        summary.syncs.players = {
   897→          source: 'api-sports',
   898→          created: plResult.created,
   899→          updated: plResult.updated,
   900→        }
   901→        await logSync('api-sports', 'players', 'success', plResult.created + plResult.updated, plResult.created, plResult.updated)
   902→      } catch (err: any) {
   903→        if (err.message?.includes('timeout')) {
   904→          console.warn('[SYNC] API-Sports players timeout')
   905→        } else {
   906→          console.warn('[SYNC] API-Sports players failed:', err.message)
   907→        }
   908→      }
   909→    }
   910→
   911→    // ═══ 3. Understat Analytical Enrichment ═══
   912→    try {
   913→      checkTimeout()
   914→      const usResult = await syncUnderstatAnalytics()
   915→      summary.syncs.understat = {
   916→        source: 'understat',
   917→        resolved: usResult.resolved,
   918→        unresolved: usResult.unresolved,
   919→        updated: usResult.updated,
   920→      }
   921→    } catch (err: any) {
   922→      if (err.message?.includes('timeout')) {
   923→        console.warn('[SYNC] Understat timeout')
   924→        await logSync('understat', 'team_analytics', 'partial', 0, 0, 0, err.message)
   925→      } else {
   926→        console.warn('[SYNC] Understat failed:', err.message)
   927→      }
   928→    }
   929→
   930→    // ═══ 4. ESPN Fallback — only if API-Sports didn't return fixtures ═══
   931→    if (!apiSportsAvailable) {
   932→      try {
   933→        const espnResult = await syncFixturesFromESPN()
   934→        summary.syncs.fixtures = {
   935→          source: 'espn',
   936→          created: espnResult.created,
   937→          updated: espnResult.updated,
   938→        }
   939→        await logSync('espn', 'fixtures', 'success', espnResult.processed, espnResult.created, espnResult.updated)
   940→      } catch (err: any) {
   941→        console.error('[SYNC] ESPN fixtures failed:', err)
   942→        await logSync('espn', 'fixtures', 'error', 0, 0, 0, String(err))
   943→        summary.syncs.fixtures.error = String(err)
   944→      }
   945→
   946→      // ESPN Standings fallback
   947→      try {
   948→        const standingsResult = await syncStandingsFromESPN()
   949→        summary.syncs.standings = {
   950→          source: 'espn',
   951→          created: standingsResult.created,
   952→          updated: 0,
   953→        }
   954→        await logSync('espn', 'standings', 'success', standingsResult.created, standingsResult.created, 0)
   955→      } catch (err: any) {
   956→        if (err.message?.includes('timeout')) {
   957→          console.warn('[SYNC] ESPN standings timeout')
   958→          await logSync('espn', 'standings', 'partial', 0, 0, 0, err.message)
   959→        } else {
   960→          console.error('[SYNC] ESPN standings failed:', err)
   961→          await logSync('espn', 'standings', 'error', 0, 0, 0, String(err))
   962→        }
   963→      }
   964→    }
   965→
   966→    summary.durationMs = elapsed()
   967→    console.log(`[SYNC] Complete in ${summary.durationMs}ms`, JSON.stringify(summary.syncs))
   968→
   969→    return NextResponse.json(summary)
   970→  } catch (error) {
   971→    const duration = elapsed()
   972→    console.error(`[SYNC] Fatal error after ${duration}ms:`, error)
   973→    summary.success = false
   974→    summary.durationMs = duration
   975→    summary.error = error instanceof Error ? error.message : String(error)
   976→    return NextResponse.json(summary, { status: 500 })
   977→  }
   978→}
