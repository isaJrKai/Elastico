    55→async function syncFixturesFromApiSports(): Promise<{
    56→  created: number
    57→  updated: number
    58→  processed: number
    59→}> {
    60→  const result = { created: 0, updated: 0, processed: 0 }
    61→
    62→  try {
    63→    // 1. Live fixtures (1 API call — covers all leagues)
    64→    const liveFixtures = await fetchLiveFixtures()
    65→    console.log(`[SYNC/API-Sports] Got ${liveFixtures.length} live fixtures`)
    66→
    67→    for (const f of liveFixtures) {
    68→      checkTimeout()
    69→      await upsertApiSportsFixture(f, result)
    70→    }
    71→
    72→    // 2. Today's fixtures from top 5 leagues (5 API calls)
    73→    const top5 = AS_LEAGUES.slice(0, 5) // PL, LIGA, SA, BL, L1
    74→    for (const league of top5) {
    75→      checkTimeout()
    76→      try {
    77→        const fixtures = await fetchFixtures(league.id)
    78→        for (const f of fixtures) {
    79→          checkTimeout()
    80→          await upsertApiSportsFixture(f, result)
    81→        }
    82→      } catch (err: any) {
    83→        if (err.message?.includes('timeout')) throw err
    84→        console.warn(`[SYNC/API-Sports] Failed fixtures for ${league.code}:`, err.message)
    85→      }
    86→    }
    87→  } catch (err: any) {
    88→    if (err.message?.includes('timeout')) throw err
    89→    console.warn('[SYNC/API-Sports] Fixtures failed:', err.message || err)
    90→  }
    91→
    92→  return result
    93→}

    97→  result: { created: number; updated: number; processed: number }
    98→) {
    99→  result.processed++
   100→  const season = String(f.league?.season || new Date().getFullYear())
   101→  const leagueCode =
   102→    AS_LEAGUES.find(l => l.id === f.league?.id)?.code ||
   103→    f.league?.country?.substring(0, 4).toUpperCase() ||
   104→    'UNKNOWN'
   105→
   106→  // Upsert home team
   107→  const homeTeam = await db.team.upsert({
   108→    where: { source_sourceId: { source: 'api-sports', sourceId: String(f.teams.home.id) } },
   109→    update: {
   110→      name: f.teams.home.name,
   111→      logo: f.teams.home.logo,
   112→      league: f.league?.name || '',
   113→      leagueCode,
   114→      source: 'api-sports',
   115→      lastSyncedAt: new Date(),
   116→    },
   117→    create: {
   118→      externalId: String(f.teams.home.id),
   119→      name: f.teams.home.name,
   120→      code: f.teams.home.name.substring(0, 3).toUpperCase(),
   121→      logo: f.teams.home.logo,
   122→      league: f.league?.name || '',
   123→      leagueCode,
   124→      country: f.league?.country || '',
   125→      source: 'api-sports',
   126→      sourceId: String(f.teams.home.id),
   127→    },
   128→  })
   129→
   130→  // Upsert away team
   131→  const awayTeam = await db.team.upsert({
   132→    where: { source_sourceId: { source: 'api-sports', sourceId: String(f.teams.away.id) } },
   133→    update: {
   134→      name: f.teams.away.name,
   135→      logo: f.teams.away.logo,
   136→      league: f.league?.name || '',
   137→      leagueCode,
   138→      source: 'api-sports',
   139→      lastSyncedAt: new Date(),
   140→    },
   141→    create: {
   142→      externalId: String(f.teams.away.id),
   143→      name: f.teams.away.name,
   144→      code: f.teams.away.name.substring(0, 3).toUpperCase(),
   145→      logo: f.teams.away.logo,
   146→      league: f.league?.name || '',
   147→      leagueCode,
   148→      country: f.league?.country || '',
   149→      source: 'api-sports',
   150→      sourceId: String(f.teams.away.id),

   153→
   154→  // Upsert match
   155→  const matchData = {
   156→    externalId: String(f.id),
   157→    homeTeamId: homeTeam.id,
   158→    awayTeamId: awayTeam.id,
   159→    competition: f.league?.name || '',
   160→    competitionCode: leagueCode,

   169→    halfTimeHome: f.score?.halftime?.home ?? null,
   170→    halfTimeAway: f.score?.halftime?.away ?? null,
   171→    source: 'api-sports',
   172→    sourceId: String(f.id),
   173→    lastSyncedAt: new Date(),
   174→  }
   175→
   176→  try {
   177→    await db.match.upsert({
   178→      where: { source_sourceId: { source: 'api-sports', sourceId: String(f.id) } },
   179→      update: matchData,
   180→      create: matchData,
   181→    })
   182→    result.created++
   183→  } catch {
   184→    result.updated++
   185→  }
   186→}

   268→// UNDERSTAT ANALYTICAL ENRICHMENT (Phase 1: team xG/xGA only)
   269→// Runs AFTER API-Sports sync so that canonical teams exist in DB.
   270→// ═════════════════════════════════════════════════════════════════════════
   271→
   272→const UNDERSTAT_LEAGUES = ['PL', 'LIGA', 'SA', 'BL', 'L1'] as const
   273→const UNDERSTAT_SEASON = '2024' // Understat uses calendar year for 24/25 season
   274→
   275→async function syncUnderstatAnalytics(): Promise<{
   276→  resolved: number
   277→  unresolved: number
   278→  updated: number
   279→}> {
   280→  const result = { resolved: 0, unresolved: 0, updated: 0 }
   281→
   282→  for (const leagueCode of UNDERSTAT_LEAGUES) {
   283→    checkTimeout()
   284→    try {
   285→      const understatTeams = await fetchUnderstatLeagueTeams(leagueCode, parseInt(UNDERSTAT_SEASON))
   286→      if (understatTeams.length === 0) continue
   287→
   288→      const { resolved, unresolved } = await resolveUnderstatTeams(
   289→        db, understatTeams.map(t => ({ id: t.id, title: t.title, team_name: t.team_name || t.title })), leagueCode, UNDERSTAT_SEASON,
   290→      )
   291→
   292→      result.resolved += resolved.length
   293→      result.unresolved += unresolved.length
   294→
   295→      // Upsert resolved teams' analytics
   296→      // NOTE: The new Understat API does NOT return xG/xGA at team level.
   297→      // We compute per-game xG from match-level data via computeTeamXgFromMatches.
   298→      // This makes one additional fetch per resolved team (rate-limited, so we
   299→      // process only the first 3 resolved teams per league per sync cycle).
   300→      const teamsToProcess = resolved.slice(0, 3)
   301→
   302→      for (const r of teamsToProcess) {
   303→        checkTimeout()
   304→        try {
   305→          // Compute real xG from match history
   306→          const xgData = await computeTeamXgFromMatches(
   307→            r.understatTeamId, parseInt(UNDERSTAT_SEASON),
   308→          )
   309→
   310→          if (!xgData || xgData.matchesPlayed === 0) {
   311→            console.warn(`[SYNC/Understat] No xG data for ${r.matchedDbTeamName} (${r.understatTeamId})`)
   312→            continue
   313→          }
   314→
   315→          const xgPerGame = Math.round((xgData.totalXg / xgData.matchesPlayed) * 100) / 100
   316→          const xgaPerGame = Math.round((xgData.totalXga / xgData.matchesPlayed) * 100) / 100
   317→
   318→          if (xgPerGame < 0 || xgaPerGame < 0) {
   319→            console.warn(`[SYNC/Understat] Invalid xG for ${r.matchedDbTeamName}: xG=${xgPerGame}, xGA=${xgaPerGame}`)
   320→            continue
   321→          }
   322→
   323→          // Link understat identity to canonical team
   324→          const canonicalId = await linkUnderstatToCanonical(
   325→            db, r.dbTeamId, r.understatTeamId,
   326→            r.understatTeamName, r.confidence, r.method,
   327→          )
   328→
   329→          const now = new Date()
   330→          const freshness = classifyFreshness(now)
   331→
   332→          await db.teamAnalytic.upsert({
   333→            where: {
   334→              teamId_source_season_leagueCode: {
   335→                teamId: r.dbTeamId,
   336→                source: 'understat',
   337→                season: UNDERSTAT_SEASON,
   338→                leagueCode,
   339→              },
   340→            },
   341→            update: {
   342→              xgPerGame,
   343→              xgaPerGame,
   344→              npxGPerGame: xgData.matchesPlayed > 0
   345→                ? Math.round((xgData.npxG / xgData.matchesPlayed) * 100) / 100
   346→                : null,
   347→              truthClass: 'REAL',
   348→              dataFreshness: freshness,
   349→              canonicalTeamId: canonicalId,
   350→              syncedAt: now,
   351→            },
   352→            create: {
   353→              teamId: r.dbTeamId,
   354→              source: 'understat',
   355→              season: UNDERSTAT_SEASON,
   356→              leagueCode,
   357→              xgPerGame,
   358→              xgaPerGame,
   359→              npxGPerGame: xgData.matchesPlayed > 0
   360→                ? Math.round((xgData.npxG / xgData.matchesPlayed) * 100) / 100
   361→                : null,
   362→              truthClass: 'REAL',
   363→              dataFreshness: freshness,
   364→              sourceTeamId: String(r.understatTeamId),
   365→              sourceTeamName: r.understatTeamName,
   366→              canonicalTeamId: canonicalId,
   367→              syncedAt: now,
   368→            },
   369→          })
   370→          result.updated++
   371→          console.log(`[SYNC/Understat] ${r.matchedDbTeamName}: xG/g=${xgPerGame}, xGA/g=${xgaPerGame} (${xgData.matchesPlayed} matches, ${r.confidence})`)
   372→        } catch (err) {
   373→          console.warn(`[SYNC/Understat] Failed to upsert for ${r.matchedDbTeamName}:`, err)
   374→        }
   375→      }
   376→
   377→      if (resolved.length > 3) {
   378→        console.log(`[SYNC/Understat] ${leagueCode}: ${resolved.length - 3} teams deferred to next sync cycle (rate limit)`)
   379→      }
   380→
   381→      // Log unresolved teams
   382→      for (const u of unresolved) {
   383→        console.warn(
   384→          `[SYNC/Understat] UNRESOLVED: ${u.understatTeamName} ` +
   385→          `(id=${u.understatTeamId}, league=${leagueCode})`,
   386→        )
   387→      }
   388→    } catch (err: any) {
   389→      if (err.message?.includes('timeout')) throw err
   390→      console.warn(`[SYNC/Understat] Failed for ${leagueCode}:`, err.message)
   391→    }
   392→  }
   393→
   394→  await logSync('understat', 'team_analytics', 'success',
   395→    result.resolved + result.unresolved, result.resolved, result.updated)
   396→
   397→  return result
   398→}

   558→// ESPN SYNC — Fallback (no API key needed, but lower quality)
   559→// ═══════════════════════════════════════════════════════════════════════════════
   560→
   561→async function fetchScoreboardForLeague(league: { espnId: string; name: string; code: string }) {
   562→  const SITE_V2 = 'https://site.api.espn.com/apis/v2/sports/soccer'
   563→  const url = `${SITE_V2}/${league.espnId}/scoreboard`
   564→  try {
   565→    const res = await fetch(url, { cache: 'no-store' })
   566→    if (!res.ok) return []
   567→    const data = await res.json()
   568→    const events: any[] = []
   569→    for (const event of data.events || []) {
   570→      const comp = event.competitions?.[0]
   571→      if (!comp) continue
   572→      const competitors = comp.competitors || []
   573→      if (competitors.length < 2) continue
   574→      const home = competitors.find((c: any) => c.homeAway === 'home') || competitors[0]
   575→      const away = competitors.find((c: any) => c.homeAway === 'away') || competitors[1]
   576→      const statusDetail = comp.status?.type?.detail || ''
   577→      const minuteMatch = statusDetail.match(/(\d+)\s*'/)
   578→      events.push({
   579→        id: event.id?.toString() || '',
   580→        homeTeamId: home.team?.id?.toString() || '',
   581→        homeTeamName: home.team?.displayName || 'Unknown',
   582→        homeTeamAbbr: home.team?.abbreviation || '???',
   583→        homeTeamLogo: home.team?.logo || '',
   584→        homeTeamColor: home.team?.color || '#00e676',
   585→        awayTeamId: away.team?.id?.toString() || '',
   586→        awayTeamName: away.team?.displayName || 'Unknown',
   587→        awayTeamAbbr: away.team?.abbreviation || '???',
   588→        awayTeamLogo: away.team?.logo || '',
   589→        awayTeamColor: away.team?.color || '#ffffff',
   590→        homeScore: parseInt(home.score) || 0,
   591→        awayScore: parseInt(away.score) || 0,
   592→        status: comp.status?.type?.name || 'STATUS_SCHEDULED',
   593→        date: event.date || '',
   594→        venue: comp.venue?.fullName || '',
   595→        competition: event.name || league.name,
   596→        competitionCode: league.code,
   597→        minute: minuteMatch ? parseInt(minuteMatch[1]) : undefined,
   598→      })
   599→    }
   600→    return events
   601→  } catch (err) {
   602→    console.error(`[SYNC/ESPN] Failed to fetch ${league.espnId}:`, err)
   603→    return []
   604→  }
   605→}

   695→      } catch {
   696→        result.updated++
   697→      }
   698→    }
   699→  }
   700→
   701→  return result

   792→async function logSync(
   793→  source: string,
   794→  action: string,
   795→  status: string,
   796→  processed: number,
   797→  created: number,
   798→  updated: number,
   799→  errorMessage?: string
   800→) {
   801→  try {
   802→    await db.syncLog.create({
   803→      data: {
   804→        source,
   805→        action,
   806→        status,
   807→        recordsProcessed: processed,
   808→        recordsCreated: created,
   809→        recordsUpdated: updated,
   810→        durationMs: elapsed(),
   811→        errorMessage: errorMessage || null,
   812→      },
   813→    })
   814→  } catch (err) {
   815→    console.error('[SYNC] Failed to write SyncLog:', err)
   816→  }
   817→}
   818→
   819→// ═══════════════════════════════════════════════════════════════════════════════
   820→// MAIN HANDLER
   821→// ═══════════════════════════════════════════════════════════════════════════════
   822→
   823→export async function GET(req: NextRequest) {
   824→  // 1. Validate cron secret
   825→  if (!validateCron(req)) {
   826→    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
   827→  }
   828→
   829→  const summary: any = {
   830→    success: true,
   831→    syncs: {
   832→      fixtures: { source: 'none', created: 0, updated: 0 },
   833→      standings: { source: 'none', created: 0, updated: 0 },
   834→      teams: { source: 'none', created: 0, updated: 0 },
   835→      players: { source: 'none', created: 0, updated: 0 },
   836→      understat: { source: 'none', resolved: 0, unresolved: 0, updated: 0 },
   837→    },
   838→    durationMs: 0,
   839→  }
   840→
   841→  try {
   842→    // ═══ 2. API-Sports: Fixtures + Standings + Teams + Players ═══
   843→    let apiSportsAvailable = false
   844→
   845→    // 2a. Fixtures (live + today's top 5 leagues)
   846→    try {
   847→      const fixResult = await syncFixturesFromApiSports()
   848→      if (fixResult.processed > 0) {
   849→        apiSportsAvailable = true
   850→        summary.syncs.fixtures = {
   851→          source: 'api-sports',
   852→          created: fixResult.created,
   853→          updated: fixResult.updated,
   854→        }
   855→        await logSync('api-sports', 'fixtures', 'success', fixResult.processed, fixResult.created, fixResult.updated)
   856→      }
   857→    } catch (err: any) {
   858→      if (err.message?.includes('timeout')) {
   859→        console.warn('[SYNC] API-Sports fixtures timeout')
   860→        await logSync('api-sports', 'fixtures', 'partial', 0, 0, 0, err.message)
   861→      } else {
   862→        console.warn('[SYNC] API-Sports fixtures failed:', err.message)
   863→      }
   864→    }
   865→
   866→    // 2b. Standings (top 10 leagues — only if API-Sports is working)
   867→    if (apiSportsAvailable) {
   868→      try {
   869→        const stResult = await syncStandingsFromApiSports()
   870→        summary.syncs.standings = {
   871→          source: 'api-sports',
   872→          created: stResult.created,
   873→          updated: stResult.updated,
   874→        }
   875→        await logSync('api-sports', 'standings', 'success', stResult.created, stResult.created, stResult.updated)
   876→      } catch (err: any) {
   877→        if (err.message?.includes('timeout')) {
   878→          console.warn('[SYNC] API-Sports standings timeout')
   879→          await logSync('api-sports', 'standings', 'partial', 0, 0, 0, err.message)
   880→        } else {
   881→          console.warn('[SYNC] API-Sports standings failed:', err.message)
   882→        }
   883→      }
   884→    }
   885→
   886→    // 2c. Teams (top 5 leagues — venue info, country, etc.)
   887→    if (apiSportsAvailable) {
   888→      try {
   889→        const tmResult = await syncTeamsFromApiSports()
   890→        summary.syncs.teams = {
   891→          source: 'api-sports',
   892→          created: tmResult.created,
   893→          updated: tmResult.updated,
   894→        }
   895→        await logSync('api-sports', 'teams', 'success', tmResult.created + tmResult.updated, tmResult.created, tmResult.updated)
   896→      } catch (err: any) {
   897→        if (err.message?.includes('timeout')) {
   898→          console.warn('[SYNC] API-Sports teams timeout')
   899→        } else {
   900→          console.warn('[SYNC] API-Sports teams failed:', err.message)
   901→        }
   902→      }
   903→    }
   904→
   905→    // 2d. Player stats (top scorers, top 5 leagues)
   906→    if (apiSportsAvailable) {
   907→      try {
   908→        const plResult = await syncPlayersFromApiSports()
   909→        summary.syncs.players = {
   910→          source: 'api-sports',
   911→          created: plResult.created,
   912→          updated: plResult.updated,
   913→        }
   914→        await logSync('api-sports', 'players', 'success', plResult.created + plResult.updated, plResult.created, plResult.updated)
   915→      } catch (err: any) {
   916→        if (err.message?.includes('timeout')) {
   917→          console.warn('[SYNC] API-Sports players timeout')
   918→        } else {
   919→          console.warn('[SYNC] API-Sports players failed:', err.message)
   920→        }
   921→      }
   922→    }
   923→
   924→    // ═══ 2.5 Build Canonical Entities (after teams exist) ═══
   925→    try {
   926→      checkTimeout()
   927→      const canonResult = await buildCanonicalEntities(db)
   928→      summary.syncs.canonical = {
   929→        source: 'canonical',
   930→        canonicalTeams: canonResult.canonicalTeamsCreated,
   931→        identities: canonResult.identitiesCreated,
   932→      }
   933→      console.log(`[SYNC] Canonical: ${canonResult.canonicalTeamsCreated} teams, ${canonResult.identitiesCreated} identities`)
   934→    } catch (err: any) {
   935→      if (err.message?.includes('timeout')) throw err
   936→      console.warn('[SYNC] Canonical entity build failed:', err.message)
   937→    }
   938→
   939→    // ═══ 3. Understat Analytical Enrichment ═══
   940→    try {
   941→      checkTimeout()
   942→      const usResult = await syncUnderstatAnalytics()
   943→      summary.syncs.understat = {
   944→        source: 'understat',
   945→        resolved: usResult.resolved,
   946→        unresolved: usResult.unresolved,
   947→        updated: usResult.updated,
   948→      }
   949→    } catch (err: any) {
   950→      if (err.message?.includes('timeout')) {
   951→        console.warn('[SYNC] Understat timeout')
   952→        await logSync('understat', 'team_analytics', 'partial', 0, 0, 0, err.message)
   953→      } else {
   954→        console.warn('[SYNC] Understat failed:', err.message)
   955→      }
   956→    }
   957→
   958→    // ═══ 4. ESPN Fallback — only if API-Sports didn't return fixtures ═══
   959→    if (!apiSportsAvailable) {
   960→      try {
   961→        const espnResult = await syncFixturesFromESPN()
   962→        summary.syncs.fixtures = {
   963→          source: 'espn',
   964→          created: espnResult.created,
   965→          updated: espnResult.updated,
   966→        }
   967→        await logSync('espn', 'fixtures', 'success', espnResult.processed, espnResult.created, espnResult.updated)
   968→      } catch (err: any) {
   969→        console.error('[SYNC] ESPN fixtures failed:', err)
   970→        await logSync('espn', 'fixtures', 'error', 0, 0, 0, String(err))
   971→        summary.syncs.fixtures.error = String(err)
   972→      }
   973→
   974→      // ESPN Standings fallback
   975→      try {
   976→        const standingsResult = await syncStandingsFromESPN()
   977→        summary.syncs.standings = {
   978→          source: 'espn',
   979→          created: standingsResult.created,
   980→          updated: 0,
   981→        }
   982→        await logSync('espn', 'standings', 'success', standingsResult.created, standingsResult.created, 0)
   983→      } catch (err: any) {
   984→        if (err.message?.includes('timeout')) {
   985→          console.warn('[SYNC] ESPN standings timeout')
   986→          await logSync('espn', 'standings', 'partial', 0, 0, 0, err.message)
   987→        } else {
   988→          console.error('[SYNC] ESPN standings failed:', err)
   989→          await logSync('espn', 'standings', 'error', 0, 0, 0, String(err))
   990→        }
   991→      }
   992→    }
   993→
   994→    summary.durationMs = elapsed()
   995→    console.log(`[SYNC] Complete in ${summary.durationMs}ms`, JSON.stringify(summary.syncs))
   996→
   997→    return NextResponse.json(summary)
   998→  } catch (error) {
   999→    const duration = elapsed()
  1000→    console.error(`[SYNC] Fatal error after ${duration}ms:`, error)
  1001→    summary.success = false
  1002→    summary.durationMs = duration
  1003→    summary.error = error instanceof Error ? error.message : String(error)
  1004→    return NextResponse.json(summary, { status: 500 })
  1005→  }
  1006→}
