   100→   169→   169→   318→            kelly_results["home_win"] = calculate_kelly(
   101→   170→   170→   319→                final_probs["home_win"],
   102→   171→   171→   320→                float(odds_home),
   103→   172→   172→   321→                bankroll=self.config.bankroll,

   118→   187→   187→   336→
   119→   188→   188→   337→        # ── 10. Signals & Risks ────────────────────────────────────────

   127→   196→   196→   345→

   130→   199→   199→   348→        if stochastic_result:

   133→   202→   202→   351→            markets["over_35"] = stochastic_result.totals_market.get("over35", 0.0)

   135→   204→   204→   353→            markets["asian_handicap"] = stochastic_result.asian_handicap
   136→   205→   205→   354→        else:
   137→   206→   206→   355→            # Fallback: estimate from Poisson
   138→   207→   207→   356→            from scipy.stats import poisson as poisson_dist
   139→   208→   208→   357→            over_25 = 0.0
   140→   209→   209→   358→            btts = 0.0
   141→   210→   210→   359→            for i in range(11):

   145→   214→   214→   363→                        over_25 += p
   146→   215→   215→   364→                    if i > 0 and j > 0:
   147→   216→   216→   365→                        btts += p
   148→   217→   217→   366→            markets["over_25"] = round(over_25, 4)

   154→   223→   223→   372→                "-0.5": round(final_probs["home_win"], 4),
   155→   224→   224→   373→                "-1.0": 0.0,
   156→   225→   225→   374→                "-1.5": 0.0,

   158→   227→   227→   376→

   161→   230→   230→   379→        if odds_home and odds_draw and odds_away:
   162→   231→   231→   380→            best_edge = -1.0

   179→   248→   248→   397→                    }
   180→   249→   249→   398→

   189→   258→   258→   407→        else:

   191→   260→   260→   409→                "home": round(lambda_h, 3),
   192→   261→   261→   410→                "away": round(lambda_a, 3),
   193→   262→   262→   411→                "total": round(lambda_h + lambda_a, 3),
   194→   263→   263→   412→            }
   195→   264→   264→   413→            volatility_index = 0.0
   196→   265→   265→   414→

   264→   333→   333→   482→    def _get_model_weights(self, available_models: list[str]) -> dict[str, float]:
   265→   334→   334→   483→        """Get normalized weights for available models only.
   266→   335→   335→   484→
   267→   336→   336→   485→        If some models are unavailable, their weight is redistributed
   268→   337→   337→   486→        proportionally among the remaining models.
   269→   338→   338→   487→
   270→   339→   339→   488→        Parameters
   271→   340→   340→   489→        ----------
   272→   341→   341→   490→        available_models : list[str]
   273→   342→   342→   491→            List of model names that produced valid results.
   274→   343→   343→   492→
   275→   344→   344→   493→        Returns
   276→   345→   345→   494→        -------
   277→   346→   346→   495→        dict[str, float]
   278→   347→   347→   496→            Normalized weights summing to 1.0, keyed by model name.
   279→   348→   348→   497→        """
   280→   349→   349→   498→        base_weights = self.config.weights
   281→   350→   350→   499→
   282→   351→   351→   500→        # Filter to available models
   283→   352→   352→   501→        filtered = {
   284→   353→   353→   502→            name: base_weights[name]
   285→   354→   354→   503→            for name in available_models
   286→   355→   355→   504→            if name in base_weights
   287→   356→   356→   505→        }
   288→   357→   357→   506→
   289→   358→   358→   507→        if not filtered:
   290→   359→   359→   508→            # Fallback: equal weight for all available
   291→   360→   360→   509→            n = len(available_models)
   292→   361→   361→   510→            return {name: 1.0 / n for name in available_models} if n > 0 else {}
   293→   362→   362→   511→
   294→   363→   363→   512→        # Normalize to sum to 1.0
   295→   364→   364→   513→        total = sum(filtered.values())
   296→   365→   365→   514→        return {name: w / total for name, w in filtered.items()}

   298→   367→   367→   516→    def _compute_weighted_average(
   299→   368→   368→   517→        self,
   300→   369→   369→   518→        probs: dict[str, dict],
   301→   370→   370→   519→        weights: dict[str, float],
   302→   371→   371→   520→    ) -> dict[str, float]:
   303→   372→   372→   521→        """Weighted average of model probabilities.
   304→   373→   373→   522→
   305→   374→   374→   523→        Parameters
   306→   375→   375→   524→        ----------
   307→   376→   376→   525→        probs : dict[str, dict]
   308→   377→   377→   526→            Per-model probability dicts: {"elo": {"home_win": 0.5, ...}, ...}
   309→   378→   378→   527→        weights : dict[str, float]
   310→   379→   379→   528→            Normalized weights per model.
   311→   380→   380→   529→
   312→   381→   381→   530→        Returns
   313→   382→   382→   531→        -------
   314→   383→   383→   532→        dict[str, float]
   315→   384→   384→   533→            Combined {"home_win": ..., "draw": ..., "away_win": ...}.
   316→   385→   385→   534→        """
   317→   386→   386→   535→        if not probs or not weights:
   318→   387→   387→   536→            return {"home_win": 1/3, "draw": 1/3, "away_win": 1/3}
   319→   388→   388→   537→
   320→   389→   389→   538→        combined = {"home_win": 0.0, "draw": 0.0, "away_win": 0.0}
   321→   390→   390→   539→
   322→   391→   391→   540→        for model_name, model_probs in probs.items():
   323→   392→   392→   541→            w = weights.get(model_name, 0.0)
   324→   393→   393→   542→            if w <= 0:
   325→   394→   394→   543→                continue
   326→   395→   395→   544→            for key in ("home_win", "draw", "away_win"):
   327→   396→   396→   545→                combined[key] += w * model_probs.get(key, 0.0)
   328→   397→   397→   546→
   329→   398→   398→   547→        # Normalize to exactly 1.0
   330→   399→   399→   548→        total = sum(combined.values())
   331→   400→   400→   549→        if total > 0:
   332→   401→   401→   550→            combined = {k: round(v / total, 6) for k, v in combined.items()}
   333→   402→   402→   551→
   334→   403→   403→   552→        return combined
   335→   404→   404→   553→

   340→   409→   409→   558→    def _compute_confidence(self, probs: dict) -> str:
   341→   410→   410→   559→        """Classify prediction confidence based on max probability.
   342→   411→   411→   560→
   343→   412→   412→   561→        - high: max probability >= 0.55
   344→   413→   413→   562→        - medium: max probability >= 0.42
   345→   414→   414→   563→        - low: otherwise
   346→   415→   415→   564→        """
   347→   416→   416→   565→        max_prob = max(probs.get("home_win", 0), probs.get("draw", 0), probs.get("away_win", 0))
   348→   417→   417→   566→        if max_prob >= 0.55:
   349→   418→   418→   567→            return "high"
   350→   419→   419→   568→        elif max_prob >= 0.42:
   351→   420→   420→   569→            return "medium"
   352→   421→   421→   570→        return "low"

   354→   423→   423→   572→    def _generate_signals_and_risks(
   355→   424→   424→   573→        self,
   356→   425→   425→   574→        match_data: dict,
   357→   426→   426→   575→        probs: dict,
   358→   427→   427→   576→    ) -> tuple[list[str], list[str]]:
   359→   428→   428→   577→        """Generate human-readable signals and risk factors.
   360→   429→   429→   578→
   361→   430→   430→   579→        Parameters
   362→   431→   431→   580→        ----------
   363→   432→   432→   581→        match_data : dict
   364→   433→   433→   582→            Original match data dict.
   365→   434→   434→   583→        probs : dict
   366→   435→   435→   584→            Final ensemble probabilities.
   367→   436→   436→   585→
   368→   437→   437→   586→        Returns
   369→   438→   438→   587→        -------
   370→   439→   439→   588→        tuple[list[str], list[str]]
   371→   440→   440→   589→            (signals, risks) lists.
   372→   441→   441→   590→        """
   373→   442→   442→   591→        signals: list[str] = []
   374→   443→   443→   592→        risks: list[str] = []
   375→   444→   444→   593→
   376→   445→   445→   594→        home_elo = float(match_data.get("home_elo", 1500))
   377→   446→   446→   595→        away_elo = float(match_data.get("away_elo", 1500))
   378→   447→   447→   596→        elo_diff = home_elo - away_elo
   379→   448→   448→   597→        home_xg = float(match_data.get("home_xg", match_data.get("home_avg_goals", 1.3)))
   380→   449→   449→   598→        away_xg = float(match_data.get("away_xg", match_data.get("away_avg_goals", 1.1)))
   381→   450→   450→   599→        odds_home = match_data.get("odds_home")
   382→   451→   451→   600→
   383→   452→   452→   601→        # ELO signals
   384→   453→   453→   602→        if elo_diff > 150:
   385→   454→   454→   603→            signals.append(f"Strong ELO advantage for {match_data.get('home_team', 'home')} (+{elo_diff:.0f})")
   386→   455→   455→   604→        elif elo_diff < -150:
   387→   456→   456→   605→            signals.append(f"Strong ELO advantage for {match_data.get('away_team', 'away')} (+{abs(elo_diff):.0f})")
   388→   457→   457→   606→        elif abs(elo_diff) > 80:
   389→   458→   458→   607→            team = "home" if elo_diff > 0 else "away"
   390→   459→   459→   608→            signals.append(f"Moderate ELO edge for {team} team")
   391→   460→   460→   609→
   392→   461→   461→   610→        # xG signals
   393→   462→   462→   611→        if home_xg > away_xg * 1.3:
   394→   463→   463→   612→            signals.append(f"Home xG ({home_xg:.2f}) significantly higher than away ({away_xg:.2f})")
   395→   464→   464→   613→        elif away_xg > home_xg * 1.3:
   396→   465→   465→   614→            signals.append(f"Away xG ({away_xg:.2f}) significantly higher than home ({home_xg:.2f})")
   397→   466→   466→   615→
   398→   467→   467→   616→        # Form signals
   399→   468→   468→   617→        home_form = match_data.get("home_form", [])
   400→   469→   469→   618→        away_form = match_data.get("away_form", [])
   401→   470→   470→   619→        if len(home_form) >= 3:
   402→   471→   471→   620→            recent_wins = sum(1 for r in home_form[-3:] if r == "W")
   403→   472→   472→   621→            if recent_wins == 3:
   404→   473→   473→   622→                signals.append(f"{match_data.get('home_team', 'Home')} on a 3-match winning streak")
   405→   474→   474→   623→        if len(away_form) >= 3:
   406→   475→   475→   624→            recent_wins = sum(1 for r in away_form[-3:] if r == "W")
   407→   476→   476→   625→            if recent_wins == 3:
   408→   477→   477→   626→                signals.append(f"{match_data.get('away_team', 'Away')} on a 3-match winning streak")
   409→   478→   478→   627→
   410→   479→   479→   628→        # Model agreement signal
   411→   480→   480→   629→        max_prob = max(probs["home_win"], probs["draw"], probs["away_win"])
   412→   481→   481→   630→        if max_prob >= 0.55:
   413→   482→   482→   631→            signals.append(f"High model agreement ({max_prob*100:.1f}% on most likely outcome)")
   414→   483→   483→   632→
   415→   484→   484→   633→        # Risks
   416→   485→   485→   634→        if odds_home is None:
   417→   486→   486→   635→            risks.append("No market odds available — Kelly criterion and market signals excluded")
   418→   487→   487→   636→
   419→   488→   488→   637→        if odds_home and float(odds_home) < 1.3:
   420→   489→   489→   638→            risks.append("Heavy favorite — low value in outcome market")
   421→   490→   490→   639→
   422→   491→   491→   640→        if abs(elo_diff) < 30:
   423→   492→   492→   641→            risks.append("Teams are very evenly matched on ELO — high draw probability")
   424→   493→   493→   642→
   425→   494→   494→   643→        # Check if any model is missing
   426→   495→   495→   644→        expected_models = {"elo", "poisson", "dixon_coles", "stochastic"}
   427→   496→   496→   645→        # We don't risk-flag missing xgboost/lstm since they're optional
   428→   497→   497→   646→
   429→   498→   498→   647→        # Derive risk from model disagreement
   430→   499→   499→   648→        if probs["draw"] > 0.32:
   431→   500→   500→   649→            risks.append("High draw probability — consider Double Chance market")
   432→   501→   501→   650→
   433→   502→   502→   651→        return signals, risks

   439→   508→   508→   657→    def _run_xgboost(self, match_data: dict) -> dict[str, float] | None:
   440→   509→   509→   658→        """Try to load and run XGBoost model.
   441→   510→   510→   659→
   442→   511→   511→   660→        Returns home_win/draw/away_win probabilities or None if
