    87→// ── Demo Tactical Profiles ─────────────────────────────────────────────
    88→
    89→function generateDemoProfile(team: { style: string | null; xgPerGame: number; xgaPerGame: number; possession: number; pressIntensity: number; passAccuracy: number }, seed: number) {
    90→  const hash = (s: string) => s.split('').reduce((a, c) => a + c.charCodeAt(0), 0)
    91→  const base = hash(team.style || 'balanced') + seed
    92→  const jitter = (i: number, range: number) => ((base * (i + 1) * 7) % range)
    93→
    94→  return {
    95→    attack: Math.min(95, Math.max(30, team.xgPerGame * 25 + jitter(0, 20) - 10)),
    96→    midfield: Math.min(95, Math.max(30, team.possession * 0.7 + team.passAccuracy * 0.2 + jitter(1, 15) - 7)),
    97→    defense: Math.min(95, Math.max(30, 100 - team.xgaPerGame * 30 + jitter(2, 15) - 7)),
    98→    pressing: Math.min(95, Math.max(20, team.pressIntensity + jitter(3, 20) - 10)),
    99→    possession: Math.min(95, Math.max(20, team.possession + jitter(4, 10) - 5)),
   100→    setPiece: Math.min(95, Math.max(25, 40 + jitter(5, 40))),
   101→  }
   102→}

   104→// ── Pitch Component ─────────────────────────────────────────────────────
   105→
   106→function FormationPitch({
   107→  formation,
   108→  teamColor,
   109→  mirror,
   110→  players,
   111→}: {
   112→  formation: string
   113→  teamColor: string
   114→  mirror?: boolean
   115→  players?: { name: string; number: number; position: string }[]
   116→}) {
   117→  const fmt = FORMATIONS[formation]
   118→  if (!fmt) return null
   119→
   120→  return (
   121→    <div className="relative w-full aspect-[3/4] rounded-lg border border-border/60 overflow-hidden bg-emerald-950/30">
   122→      {/* Pitch markings */}
   123→      <div className="absolute inset-2 border border-white/10 rounded-sm" />
   124→      <div className="absolute top-1/2 left-2 right-2 border-t border-white/10" />
   125→      <div className="absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 size-24 rounded-full border border-white/10" />
   126→      <div className="absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 size-1 rounded-full bg-white/20" />
   127→      {/* Penalty areas */}
   128→      <div className="absolute top-2 left-1/2 -translate-x-1/2 w-[40%] h-[16%] border border-white/10 border-t-0 rounded-b-md" />
   129→      <div className="absolute bottom-2 left-1/2 -translate-x-1/2 w-[40%] h-[16%] border border-white/10 border-b-0 rounded-t-md" />
   130→      {/* Goal areas */}
   131→      <div className="absolute top-2 left-1/2 -translate-x-1/2 w-[20%] h-[8%] border border-white/10 border-t-0" />
   132→      <div className="absolute bottom-2 left-1/2 -translate-x-1/2 w-[20%] h-[8%] border border-white/10 border-b-0" />
   133→
   134→      {/* Player dots */}
   135→      {fmt.positions.map((pos, i) => {
   136→        const x = mirror ? 100 - pos.x : pos.x
   137→        const y = pos.y
   138→        const player = players?.[i]
   139→        return (
   140→          <div
   141→            key={i}
   142→            className="absolute flex flex-col items-center -translate-x-1/2 -translate-y-1/2 group"
   143→            style={{ left: `${x}%`, top: `${y}%` }}
   144→          >
   145→            <div
   146→              className="size-5 rounded-full border-2 border-white/40 shadow-md flex items-center justify-center transition-transform hover:scale-125"
   147→              style={{ backgroundColor: teamColor }}
   148→            >
   149→              <span className="text-[8px] font-bold text-white leading-none">
   150→                {player?.number || (i === 0 ? '1' : '')}
   151→              </span>
   152→            </div>
   153→            {player && (
   154→              <span className="text-[8px] text-white/70 mt-0.5 whitespace-nowrap font-medium opacity-0 group-hover:opacity-100 transition-opacity">
   155→                {player.name}
   156→              </span>
   157→            )}
   158→            {!player && (
   159→              <span className="text-[8px] text-white/40 mt-0.5 font-mono">
   160→                {pos.role}
   161→              </span>
   162→            )}
   163→          </div>
   164→        )
   165→      })}
   166→    </div>
   167→  )
   168→}

   172→export function TacticalView() {

   182→  const awayTeam = useMemo(() => teams.find((t) => t.id === awayTeamId), [teams, awayTeamId])
   183→
   184→  // Pre-select from a selected match
   185→  React.useEffect(() => {
   186→    if (selectedMatchId && matches.length > 0) {
   187→      const m = matches.find((match) => match.id === selectedMatchId)
   188→      if (m) {
   189→        setHomeTeamId(m.homeTeamId)
   190→        setAwayTeamId(m.awayTeamId)
   191→      }
   192→    }
   193→  }, [selectedMatchId, matches])
   194→
   195→  // Auto-select first two teams if none selected
   196→  React.useEffect(() => {
   197→    if (!homeTeamId && teams.length >= 2) {
   198→      setHomeTeamId(teams[0].id)
   199→      setAwayTeamId(teams[1].id)
   200→    }
   201→  }, [teams, homeTeamId])
   202→
   203→  // Generate tactical profiles
   204→  const homeProfile = useMemo(
   205→    () => homeTeam ? generateDemoProfile(homeTeam, 1) : null,
