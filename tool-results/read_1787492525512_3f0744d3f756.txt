    55→async function syncFixturesFromApiSports(): Promise<{
    56→  created: number
    57→  updated: number
    58→  processed: number
    59→}> {
    60→  const result = { created: 0, updated: 0, processed: 0 }
    61→
    62→  try {
    63→    // 1. Live fixtures (1 API call — covers all leagues)
    64→    const liveFixtures = await fetchLiveFixtures()
    65→    console.log(`[SYNC/API-Sports] Got ${liveFixtures.length} live fixtures`)
    66→
    67→    for (const f of liveFixtures) {
    68→      checkTimeout()
    69→      await upsertApiSportsFixture(f, result)
    70→    }
    71→
    72→    // 2. Today's fixtures from top 5 leagues (5 API calls)
    73→    const top5 = AS_LEAGUES.slice(0, 5) // PL, LIGA, SA, BL, L1
    74→    for (const league of top5) {
    75→      checkTimeout()
    76→      try {
    77→        const fixtures = await fetchFixtures(league.id)
    78→        for (const f of fixtures) {
    79→          checkTimeout()
    80→          await upsertApiSportsFixture(f, result)
    81→        }
    82→      } catch (err: any) {
    83→        if (err.message?.includes('timeout')) throw err
    84→        console.warn(`[SYNC/API-Sports] Failed fixtures for ${league.code}:`, err.message)
    85→      }
    86→    }
    87→  } catch (err: any) {
    88→    if (err.message?.includes('timeout')) throw err
    89→    console.warn('[SYNC/API-Sports] Fixtures failed:', err.message || err)
    90→  }
    91→
    92→  return result

   121→      logo: f.teams.home.logo,

   272→const UNDERSTAT_LEAGUES = ['PL', 'LIGA', 'SA', 'BL', 'L1'] as const
   273→const UNDERSTAT_SEASON = '2024' // Understat uses calendar year for 24/25 season
   274→
   275→async function syncUnderstatAnalytics(): Promise<{
   276→  resolved: number
   277→  unresolved: number
   278→  updated: number
   279→}> {
   280→  const result = { resolved: 0, unresolved: 0, updated: 0 }
   281→
   282→  for (const leagueCode of UNDERSTAT_LEAGUES) {
   283→    checkTimeout()
   284→    try {
   285→      const understatTeams = await fetchUnderstatLeagueTeams(leagueCode, parseInt(UNDERSTAT_SEASON))
   286→      if (understatTeams.length === 0) continue
   287→
   288→      const { resolved, unresolved } = await resolveUnderstatTeams(
   289→        db, understatTeams, leagueCode, UNDERSTAT_SEASON,
   290→      )
   291→
   292→      result.resolved += resolved.length
   293→      result.unresolved += unresolved.length
   294→
   295→      // Upsert resolved teams' analytics
   296→      // NOTE: The new Understat API does NOT return xG/xGA at team level.
   297→      // We compute per-game xG from match-level data via computeTeamXgFromMatches.
   298→      // This makes one additional fetch per resolved team (rate-limited, so we
   299→      // process only the first 3 resolved teams per league per sync cycle).
   300→      const teamsToProcess = resolved.slice(0, 3)
   301→
   302→      for (const r of teamsToProcess) {
   303→        checkTimeout()
   304→        try {
   305→          // Compute real xG from match history
   306→          const xgData = await computeTeamXgFromMatches(
   307→            r.understatTeamId, parseInt(UNDERSTAT_SEASON),
   308→          )
   309→
   310→          if (!xgData || xgData.matchesPlayed === 0) {
   311→            console.warn(`[SYNC/Understat] No xG data for ${r.matchedDbTeamName} (${r.understatTeamId})`)
   312→            continue
   313→          }
   314→
   315→          const xgPerGame = Math.round((xgData.totalXg / xgData.matchesPlayed) * 100) / 100
   316→          const xgaPerGame = Math.round((xgData.totalXga / xgData.matchesPlayed) * 100) / 100
   317→
   318→          if (xgPerGame < 0 || xgaPerGame < 0) {
   319→            console.warn(`[SYNC/Understat] Invalid xG for ${r.matchedDbTeamName}: xG=${xgPerGame}, xGA=${xgaPerGame}`)
   320→            continue
   321→          }
   322→
   323→          await db.teamAnalytic.upsert({
   324→            where: {
   325→              teamId_source_season_leagueCode: {
   326→                teamId: r.dbTeamId,
   327→                source: 'understat',
   328→                season: UNDERSTAT_SEASON,
   329→                leagueCode,
   330→              },
   331→            },
   332→            update: {
   333→              xgPerGame,
   334→              xgaPerGame,
   335→              npxGPerGame: xgData.matchesPlayed > 0
   336→                ? Math.round((xgData.npxG / xgData.matchesPlayed) * 100) / 100
   337→                : null,
   338→              syncedAt: new Date(),
   339→            },
   340→            create: {
   341→              teamId: r.dbTeamId,
   342→              source: 'understat',
   343→              season: UNDERSTAT_SEASON,
   344→              leagueCode,
   345→              xgPerGame,
   346→              xgaPerGame,
   347→              npxGPerGame: xgData.matchesPlayed > 0
   348→                ? Math.round((xgData.npxG / xgData.matchesPlayed) * 100) / 100
   349→                : null,
   350→              sourceTeamId: String(r.understatTeamId),
   351→              sourceTeamName: r.understatTeamName,
   352→              syncedAt: new Date(),
   353→            },
   354→          })
   355→          result.updated++
   356→          console.log(`[SYNC/Understat] ${r.matchedDbTeamName}: xG/g=${xgPerGame}, xGA/g=${xgaPerGame} (${xgData.matchesPlayed} matches, ${r.confidence})`)
   357→        } catch (err) {
   358→          console.warn(`[SYNC/Understat] Failed to upsert for ${r.matchedDbTeamName}:`, err)
   359→        }
   360→      }
   361→
   362→      if (resolved.length > 3) {
   363→        console.log(`[SYNC/Understat] ${leagueCode}: ${resolved.length - 3} teams deferred to next sync cycle (rate limit)`)
   364→      }
   365→
   366→      // Log unresolved teams
   367→      for (const u of unresolved) {
   368→        console.warn(
   369→          `[SYNC/Understat] UNRESOLVED: ${u.understatTeamName} ` +
   370→          `(id=${u.understatTeamId}, league=${leagueCode})`,
   371→        )
   372→      }
   373→    } catch (err: any) {
   374→      if (err.message?.includes('timeout')) throw err
   375→      console.warn(`[SYNC/Understat] Failed for ${leagueCode}:`, err.message)
   376→    }
   377→  }
   378→
   379→  await logSync('understat', 'team_analytics', 'success',
   380→    result.resolved + result.unresolved, result.resolved, result.updated)
   381→
   382→  return result
   383→}

   546→async function fetchScoreboardForLeague(league: { espnId: string; name: string; code: string }) {
   547→  const SITE_V2 = 'https://site.api.espn.com/apis/v2/sports/soccer'
   548→  const url = `${SITE_V2}/${league.espnId}/scoreboard`
   549→  try {
   550→    const res = await fetch(url, { cache: 'no-store' })
   551→    if (!res.ok) return []
   552→    const data = await res.json()
   553→    const events: any[] = []
   554→    for (const event of data.events || []) {
   555→      const comp = event.competitions?.[0]
   556→      if (!comp) continue
   557→      const competitors = comp.competitors || []
   558→      if (competitors.length < 2) continue
   559→      const home = competitors.find((c: any) => c.homeAway === 'home') || competitors[0]
   560→      const away = competitors.find((c: any) => c.homeAway === 'away') || competitors[1]
   561→      const statusDetail = comp.status?.type?.detail || ''
   562→      const minuteMatch = statusDetail.match(/(\d+)\s*'/)
   563→      events.push({
   564→        id: event.id?.toString() || '',
   565→        homeTeamId: home.team?.id?.toString() || '',
   566→        homeTeamName: home.team?.displayName || 'Unknown',
   567→        homeTeamAbbr: home.team?.abbreviation || '???',
   568→        homeTeamLogo: home.team?.logo || '',
   569→        homeTeamColor: home.team?.color || '#00e676',
   570→        awayTeamId: away.team?.id?.toString() || '',
   571→        awayTeamName: away.team?.displayName || 'Unknown',
   572→        awayTeamAbbr: away.team?.abbreviation || '???',
   573→        awayTeamLogo: away.team?.logo || '',
   574→        awayTeamColor: away.team?.color || '#ffffff',
   575→        homeScore: parseInt(home.score) || 0,
   576→        awayScore: parseInt(away.score) || 0,
   577→        status: comp.status?.type?.name || 'STATUS_SCHEDULED',
   578→        date: event.date || '',
   579→        venue: comp.venue?.fullName || '',
   580→        competition: event.name || league.name,
   581→        competitionCode: league.code,
   582→        minute: minuteMatch ? parseInt(minuteMatch[1]) : undefined,
   583→      })
   584→    }
   585→    return events
   586→  } catch (err) {
   587→    console.error(`[SYNC/ESPN] Failed to fetch ${league.espnId}:`, err)
   588→    return []
   589→  }
   590→}

   593→  created: number
   594→  updated: number
   595→  processed: number
   596→}> {
   597→  const result = { created: 0, updated: 0, processed: 0 }
   598→
   599→  const results = await Promise.allSettled(
   600→    ESPN_LEAGUES.map(league => fetchScoreboardForLeague(league))
   601→  )
   602→
   603→  for (const r of results) {
   604→    if (r.status !== 'fulfilled') continue
   605→    const matches = r.value
   606→    for (const m of matches) {
   607→      checkTimeout()
   608→      result.processed++
   609→
   610→      const homeTeam = await db.team.upsert({
   611→        where: { source_sourceId: { source: 'espn', sourceId: m.homeTeamId } },

   614→          code: m.homeTeamAbbr,
   615→          logo: m.homeTeamLogo,
   616→          primaryColor: m.homeTeamColor,
   617→          leagueCode: m.competitionCode,
   618→          source: 'espn',
   619→          lastSyncedAt: new Date(),
   620→        },
   621→        create: {
   622→          externalId: m.homeTeamId,
   623→          name: m.homeTeamName,
   624→          code: m.homeTeamAbbr,
   625→          logo: m.homeTeamLogo,
   626→          primaryColor: m.homeTeamColor,
   627→          leagueCode: m.competitionCode,
   628→          source: 'espn',
   629→          sourceId: m.homeTeamId,
   630→        },
   631→      })
   632→
   633→      const awayTeam = await db.team.upsert({
   634→        where: { source_sourceId: { source: 'espn', sourceId: m.awayTeamId } },
   635→        update: {
   636→          name: m.awayTeamName,
   637→          code: m.awayTeamAbbr,
   638→          logo: m.awayTeamLogo,
   639→          primaryColor: m.awayTeamColor,
   640→          leagueCode: m.competitionCode,
   641→          source: 'espn',
   642→          lastSyncedAt: new Date(),
   643→        },
   644→        create: {
   645→          externalId: m.awayTeamId,
   646→          name: m.awayTeamName,
   647→          code: m.awayTeamAbbr,
   648→          logo: m.awayTeamLogo,
   649→          primaryColor: m.awayTeamColor,
   650→          leagueCode: m.competitionCode,
   651→          source: 'espn',
   652→          sourceId: m.awayTeamId,
   653→        },
   654→      })
   655→
   656→      const matchData = {
   657→        externalId: m.id,
   658→        homeTeamId: homeTeam.id,
   659→        awayTeamId: awayTeam.id,
   660→        competition: m.competition,
   661→        competitionCode: m.competitionCode,
   662→        date: m.date ? new Date(m.date) : null,
   663→        status: mapStatus(m.status),
   664→        minute: m.minute ?? null,
   665→        venue: m.venue,
   666→        homeScore: m.homeScore,
   667→        awayScore: m.awayScore,
   668→        source: 'espn',
   669→        sourceId: m.id,
   670→        lastSyncedAt: new Date(),
   671→      }
   672→
   673→      try {
   674→        await db.match.upsert({
   675→          where: { source_sourceId: { source: 'espn', sourceId: m.id } },
   676→          update: matchData,
   677→          create: matchData,
   678→        })
   679→        result.created++
   680→      } catch {
   681→        result.updated++
   682→      }
   683→    }
   684→  }
   685→
   686→  return result
   687→}

   777→async function logSync(
   778→  source: string,
   779→  action: string,
   780→  status: string,
   781→  processed: number,
   782→  created: number,
   783→  updated: number,
   784→  errorMessage?: string
   785→) {
   786→  try {
   787→    await db.syncLog.create({
   788→      data: {
   789→        source,
   790→        action,
   791→        status,
   792→        recordsProcessed: processed,
   793→        recordsCreated: created,
   794→        recordsUpdated: updated,
   795→        durationMs: elapsed(),
   796→        errorMessage: errorMessage || null,
   797→      },
   798→    })
   799→  } catch (err) {
   800→    console.error('[SYNC] Failed to write SyncLog:', err)
   801→  }
   802→}

   804→// ═══════════════════════════════════════════════════════════════════════════════
   805→// MAIN HANDLER
   806→// ═══════════════════════════════════════════════════════════════════════════════
   807→
   808→export async function GET(req: NextRequest) {
   809→  // 1. Validate cron secret
   810→  if (!validateCron(req)) {
   811→    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
   812→  }
   813→
   814→  const summary: any = {
   815→    success: true,
   816→    syncs: {
   817→      fixtures: { source: 'none', created: 0, updated: 0 },
   818→      standings: { source: 'none', created: 0, updated: 0 },
   819→      teams: { source: 'none', created: 0, updated: 0 },
   820→      players: { source: 'none', created: 0, updated: 0 },
   821→      understat: { source: 'none', resolved: 0, unresolved: 0, updated: 0 },
   822→    },
   823→    durationMs: 0,
   824→  }
   825→
   826→  try {
   827→    // ═══ 2. API-Sports: Fixtures + Standings + Teams + Players ═══
   828→    let apiSportsAvailable = false
   829→
   830→    // 2a. Fixtures (live + today's top 5 leagues)
   831→    try {
   832→      const fixResult = await syncFixturesFromApiSports()
   833→      if (fixResult.processed > 0) {
   834→        apiSportsAvailable = true
   835→        summary.syncs.fixtures = {
   836→          source: 'api-sports',
   837→          created: fixResult.created,
   838→          updated: fixResult.updated,
   839→        }
   840→        await logSync('api-sports', 'fixtures', 'success', fixResult.processed, fixResult.created, fixResult.updated)
   841→      }
   842→    } catch (err: any) {
   843→      if (err.message?.includes('timeout')) {
   844→        console.warn('[SYNC] API-Sports fixtures timeout')
   845→        await logSync('api-sports', 'fixtures', 'partial', 0, 0, 0, err.message)
   846→      } else {
   847→        console.warn('[SYNC] API-Sports fixtures failed:', err.message)
   848→      }
   849→    }
   850→
   851→    // 2b. Standings (top 10 leagues — only if API-Sports is working)
   852→    if (apiSportsAvailable) {
   853→      try {
   854→        const stResult = await syncStandingsFromApiSports()
   855→        summary.syncs.standings = {
   856→          source: 'api-sports',
   857→          created: stResult.created,
   858→          updated: stResult.updated,
   859→        }
   860→        await logSync('api-sports', 'standings', 'success', stResult.created, stResult.created, stResult.updated)
   861→      } catch (err: any) {
   862→        if (err.message?.includes('timeout')) {
   863→          console.warn('[SYNC] API-Sports standings timeout')
   864→          await logSync('api-sports', 'standings', 'partial', 0, 0, 0, err.message)
   865→        } else {
   866→          console.warn('[SYNC] API-Sports standings failed:', err.message)
   867→        }
   868→      }
   869→    }
   870→
   871→    // 2c. Teams (top 5 leagues — venue info, country, etc.)
   872→    if (apiSportsAvailable) {
   873→      try {
   874→        const tmResult = await syncTeamsFromApiSports()
   875→        summary.syncs.teams = {
   876→          source: 'api-sports',
   877→          created: tmResult.created,
   878→          updated: tmResult.updated,
   879→        }
   880→        await logSync('api-sports', 'teams', 'success', tmResult.created + tmResult.updated, tmResult.created, tmResult.updated)
   881→      } catch (err: any) {
   882→        if (err.message?.includes('timeout')) {
   883→          console.warn('[SYNC] API-Sports teams timeout')
   884→        } else {
   885→          console.warn('[SYNC] API-Sports teams failed:', err.message)
   886→        }
   887→      }
   888→    }
   889→
   890→    // 2d. Player stats (top scorers, top 5 leagues)
   891→    if (apiSportsAvailable) {
   892→      try {
   893→        const plResult = await syncPlayersFromApiSports()
   894→        summary.syncs.players = {
   895→          source: 'api-sports',
   896→          created: plResult.created,
   897→          updated: plResult.updated,
   898→        }
   899→        await logSync('api-sports', 'players', 'success', plResult.created + plResult.updated, plResult.created, plResult.updated)
   900→      } catch (err: any) {
   901→        if (err.message?.includes('timeout')) {
   902→          console.warn('[SYNC] API-Sports players timeout')
   903→        } else {
   904→          console.warn('[SYNC] API-Sports players failed:', err.message)
   905→        }
   906→      }
   907→    }
   908→
   909→    // ═══ 3. Understat Analytical Enrichment ═══
   910→    try {
   911→      checkTimeout()
   912→      const usResult = await syncUnderstatAnalytics()
   913→      summary.syncs.understat = {
   914→        source: 'understat',
   915→        resolved: usResult.resolved,
   916→        unresolved: usResult.unresolved,
   917→        updated: usResult.updated,
   918→      }
   919→    } catch (err: any) {
   920→      if (err.message?.includes('timeout')) {
   921→        console.warn('[SYNC] Understat timeout')
   922→        await logSync('understat', 'team_analytics', 'partial', 0, 0, 0, err.message)
   923→      } else {
   924→        console.warn('[SYNC] Understat failed:', err.message)
   925→      }
   926→    }
   927→
   928→    // ═══ 4. ESPN Fallback — only if API-Sports didn't return fixtures ═══
   929→    if (!apiSportsAvailable) {
   930→      try {
   931→        const espnResult = await syncFixturesFromESPN()
   932→        summary.syncs.fixtures = {
   933→          source: 'espn',
   934→          created: espnResult.created,
   935→          updated: espnResult.updated,
   936→        }
   937→        await logSync('espn', 'fixtures', 'success', espnResult.processed, espnResult.created, espnResult.updated)
   938→      } catch (err: any) {
   939→        console.error('[SYNC] ESPN fixtures failed:', err)
   940→        await logSync('espn', 'fixtures', 'error', 0, 0, 0, String(err))
   941→        summary.syncs.fixtures.error = String(err)
   942→      }
   943→
   944→      // ESPN Standings fallback
   945→      try {
   946→        const standingsResult = await syncStandingsFromESPN()
   947→        summary.syncs.standings = {
   948→          source: 'espn',
   949→          created: standingsResult.created,
   950→          updated: 0,
   951→        }
   952→        await logSync('espn', 'standings', 'success', standingsResult.created, standingsResult.created, 0)
   953→      } catch (err: any) {
   954→        if (err.message?.includes('timeout')) {
   955→          console.warn('[SYNC] ESPN standings timeout')
   956→          await logSync('espn', 'standings', 'partial', 0, 0, 0, err.message)
   957→        } else {
   958→          console.error('[SYNC] ESPN standings failed:', err)
   959→          await logSync('espn', 'standings', 'error', 0, 0, 0, String(err))
   960→        }
   961→      }
   962→    }
   963→
   964→    summary.durationMs = elapsed()
   965→    console.log(`[SYNC] Complete in ${summary.durationMs}ms`, JSON.stringify(summary.syncs))
   966→
   967→    return NextResponse.json(summary)
   968→  } catch (error) {
   969→    const duration = elapsed()
   970→    console.error(`[SYNC] Fatal error after ${duration}ms:`, error)
   971→    summary.success = false
   972→    summary.durationMs = duration
   973→    summary.error = error instanceof Error ? error.message : String(error)
   974→    return NextResponse.json(summary, { status: 500 })
   975→  }
   976→}
