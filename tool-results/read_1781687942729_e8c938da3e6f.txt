    15→    15→// Update team ELO Ratings based on match outcome
    16→    16→function updateTeamEloRatings(match: Match) {
    17→    17→  const homeTeam = db.teams.find((t) => t.id === match.home_team_id);
    18→    18→  const awayTeam = db.teams.find((t) => t.id === match.away_team_id);
    19→    19→
    20→    20→  if (!homeTeam || !awayTeam) return;
    21→    21→
    22→    22→  const R_home = homeTeam.elo_rating;
    23→    23→  const R_away = awayTeam.elo_rating;
    24→    24→
    25→    25→  // Expected outcome E(A) = 1 / (1 + 10^((Elo_B - Elo_A) / 400))
    26→    26→  const E_home = 1 / (1 + Math.pow(10, (R_away - R_home) / 400));
    27→    27→  const E_away = 1 - E_home;
    28→    28→
    29→    29→  // S_home / S_away values (1 = win, 0.5 = draw, 0 = loss)
    30→    30→  let S_home = 0.5;
    31→    31→  let S_away = 0.5;
    32→    32→
    33→    33→  if (match.home_score > match.away_score) {
    34→    34→    S_home = 1;
    35→    35→    S_away = 0;
    36→    36→  } else if (match.home_score < match.away_score) {
    37→    37→    S_home = 0;
    38→    38→    S_away = 1;
    39→    39→  }
    40→    40→
    41→    41→  // Calculate new ELOs
    42→    42→  homeTeam.elo_rating = Math.round((R_home + ELO_K_FACTOR * (S_home - E_home)) * 10) / 10;
    43→    43→  awayTeam.elo_rating = Math.round((R_away + ELO_K_FACTOR * (S_away - E_away)) * 10) / 10;
    44→    44→
    45→    45→  console.log(`[ELO SYSTEM] ${homeTeam.name} ELO updated ${R_home} -> ${homeTeam.elo_rating}`);
    46→    46→  console.log(`[ELO SYSTEM] ${awayTeam.name} ELO updated ${R_away} -> ${awayTeam.elo_rating}`);
    47→    47→}

    49→    49→// Generate continuous past head to head meetings deterministic indexes
    50→    50→export function getPastMeetingsTrend(homeTeamName: string, awayTeamName: string, homeElo: number = 1500, awayElo: number = 1500) {
    51→    51→  const dates = ["2021-06-12", "2022-11-20", "2023-06-25", "2024-10-18", "2025-03-31", "2026-01-14"];
    52→    52→  const stages = ["World Cup QF", "Friendly Cup", "International League", "Euro-Am Cup", "Continental Challenge", "Global Playoff"];
    53→    53→  const sumVal = (homeTeamName.charCodeAt(0) || 65) + (awayTeamName.charCodeAt(0) || 66);
    54→    54→
    55→    55→  return Array.from({ length: 6 }).map((_, idx) => {
    56→    56→    const factor = (idx + 1) * sumVal;
    57→    57→    const homeScore = (factor % 3);
    58→    58→    let awayScore = ((factor + idx + 1) % 4);
    59→    59→    if (homeScore === 0 && awayScore === 0 && idx % 2 === 0) {
    60→    60→      awayScore = 1; // avoid excessive boring draws
    61→    61→    }
    62→    62→    const date = dates[idx];
    63→    63→    const stage = stages[idx];
    64→    64→    const scoreText = `${homeScore}-${awayScore}`;
    65→    65→
    66→    66→    // Gracefully simulate past ELO Ratings progressing over meetings up to current actual ratings
    67→    67→    const homeEloAtClash = Math.round(homeElo - (6 - idx) * 16 + (homeScore - awayScore) * 12);
    68→    68→    const awayEloAtClash = Math.round(awayElo - (6 - idx) * 14 + (awayScore - homeScore) * 12);
    69→    69→
    70→    70→    return {
    71→    71→      meetingNumber: idx + 1,
    72→    72→      date,
    73→    73→      stage,
    74→    74→      score: scoreText,
    75→    75→      homeScore,
    76→    76→      awayScore,
    77→    77→      goalDifference: homeScore - awayScore,
    78→    78→      homePerformanceIndex: 50 + (homeScore - awayScore) * 15 + (idx % 2 === 0 ? 5 : -4),
    79→    79→      awayPerformanceIndex: 50 + (awayScore - homeScore) * 15 + (idx % 2 === 0 ? -5 : 4),
    80→    80→      homeElo: homeEloAtClash,
    81→    81→      awayElo: awayEloAtClash,
    82→    82→    };
    83→    83→  });
    84→    84→}

    86→    86→function generateTeamForm(teamName: string, seed: number): ("W" | "D" | "L")[] {
    87→    87→  const formOptions: ("W" | "D" | "L")[][] = [
    88→    88→    ["W", "D", "L", "W", "W"],
    89→    89→    ["L", "W", "D", "W", "L"],
    90→    90→    ["W", "W", "W", "D", "W"],
    91→    91→    ["L", "L", "D", "W", "D"],
    92→    92→    ["D", "W", "L", "W", "W"],
    93→    93→    ["W", "D", "W", "L", "W"],
    94→    94→    ["L", "W", "L", "W", "D"],
    95→    95→    ["D", "D", "W", "W", "L"],
    96→    96→  ];
    97→    97→  const charCodeSum = Array.from(teamName).reduce((acc, char) => acc + char.charCodeAt(0), 0) + seed;
    98→    98→  return formOptions[charCodeSum % formOptions.length];
    99→    99→}

   126→   126→// Helper to pre-calculate team profiles for matches
   127→   127→function getMatchWithMetadata(match: Match) {
   128→   128→  const homeTeam = db.teams.find((t) => t.id === match.home_team_id);
   129→   129→  const awayTeam = db.teams.find((t) => t.id === match.away_team_id);
   130→   130→
   131→   131→  const homeElo = homeTeam?.elo_rating || 1500;
   132→   132→  const awayElo = awayTeam?.elo_rating || 1500;
   133→   133→  const prediction = predictMatch(homeElo, awayElo, 2000);
   134→   134→
   135→   135→  const homeName = homeTeam?.name || "Home Team";
   136→   136→  const awayName = awayTeam?.name || "Away Team";
   137→   137→
   138→   138→  // Calculate percentage divisions for analyst public votes
   139→   139→  const votesHome = match.votes_home || 0;
   140→   140→  const votesDraw = match.votes_draw || 0;
   141→   141→  const votesAway = match.votes_away || 0;
   142→   142→  const totalVotes = votesHome + votesDraw + votesAway;
   143→   143→
   144→   144→  const votePercentages = {
   145→   145→    home: totalVotes > 0 ? (votesHome / totalVotes) : 0.33,
   146→   146→    draw: totalVotes > 0 ? (votesDraw / totalVotes) : 0.33,
   147→   147→    away: totalVotes > 0 ? (votesAway / totalVotes) : 0.34,
   148→   148→  };
   149→   149→
   150→   150→  const homeColors = getTeamJerseyColors(homeName);
   151→   151→  const awayColors = getTeamJerseyColors(awayName);
   152→   152→
   153→   153→  return {
   154→   154→    ...match,
   155→   155→    home_team: homeName,
   156→   156→    away_team: awayName,
   157→   157→    home_elo: homeElo,
   158→   158→    away_elo: awayElo,
   159→   159→    home_crest: homeTeam?.crest_url || `https://crests.football-data.org/${homeName.substring(0, 3).toUpperCase()}.png`,
   160→   160→    away_crest: awayTeam?.crest_url || `https://crests.football-data.org/${awayName.substring(0, 3).toUpperCase()}.png`,
   161→   161→    home_jersey_color: homeColors.primary,
   162→   162→    away_jersey_color: awayColors.primary,
   163→   163→    win_probability: prediction.win_probability,
   164→   164→    mu_home: prediction.mu_home,
   165→   165→    mu_away: prediction.mu_away,
   166→   166→    confidence_intervals: prediction.confidence_intervals,
   167→   167→    score_grid: prediction.score_grid,
   168→   168→    votes_distribution: {
   169→   169→      home: votesHome,
   170→   170→      draw: votesDraw,
   171→   171→      away: votesAway,
   172→   172→      total: totalVotes,
   173→   173→      percentages: votePercentages,
   174→   174→    },
   175→   175→    home_form: generateTeamForm(homeName, match.id * 3),
   176→   176→    away_form: generateTeamForm(awayName, match.id * 7),
   177→   177→    historical_clashes: getPastMeetingsTrend(homeName, awayName, homeElo, awayElo),
   178→   178→  };
   179→   179→}

   260→   260→// ── CORE TICKING SIMULATOR INJECTED WITH HALFTIME PAUSE ──────────────────────
   261→   261→function runTickingSimulator(mId: number, startMin: number) {
   262→   262→  const match = db.matches.find((m) => m.id === mId);
   263→   263→  if (!match) return;
   264→   264→
   265→   265→  // Clear existing intervals completely
   266→   266→  if (activeIntervals[mId]) {
   267→   267→    clearInterval(activeIntervals[mId]);
   268→   268→  }
   269→   269→
   270→   270→  const homeTeam = db.teams.find((t) => t.id === match.home_team_id);
   271→   271→  const awayTeam = db.teams.find((t) => t.id === match.away_team_id);
   272→   272→  const homePlayers = db.players.filter((p) => p.team_id === match.home_team_id).map((p) => p.name);
   273→   273→  const awayPlayers = db.players.filter((p) => p.team_id === match.away_team_id).map((p) => p.name);
   274→   274→
   275→   275→  const matchMeta = getMatchWithMetadata(match);
   276→   276→  const mu_home = matchMeta.mu_home;
   277→   277→  const mu_away = matchMeta.mu_away;
   278→   278→
   279→   279→  // Generate full potential timeline matching actual intensity
   280→   280→  const fullTimeline = generateTimelineEvents(
   281→   281→    mId,
   282→   282→    match.home_team_id,
   283→   283→    match.away_team_id,
   284→   284→    homePlayers.length > 0 ? homePlayers : ["Striker A", "Midfielder A"],
   285→   285→    awayPlayers.length > 0 ? awayPlayers : ["Striker B", "Midfielder B"],
   286→   286→    mu_home,
   287→   287→    mu_away
   288→   288→  );
   289→   289→
   290→   290→  let currentMin = startMin;
   291→   291→
   292→   292→  activeIntervals[mId] = setInterval(() => {
   293→   293→    currentMin += 5;
   294→   294→
   295→   295→    // Check Halftime Boundary
   296→   296→    if (currentMin === 45 && !match.halftime_prediction) {
   297→   297→      match.simulation_minute = 45;
   298→   298→      match.status = "live";
   299→   299→      
   300→   300→      // Seed a Halftime Break commentary log
   301→   301→      const evId = db.match_events.length > 0 ? Math.max(...db.match_events.map((o) => o.id)) + 1 : 1;
   302→   302→      db.match_events.push({
   303→   303→        id: evId,
   304→   304→        match_id: mId,
   305→   305→        minute: 45,
   306→   306→        event_type: "substitution",
   307→   307→        description: `HALFTIME WHISTLE! Teams march into the locker rooms with scores lock at ${match.home_score} - ${match.away_score}. Match timeline paused for analyst halftime forecasts.`,
   308→   308→        created_at: new Date().toISOString()
   309→   309→      });
   310→   310→
   311→   311→      db.save();
   312→   312→      
   313→   313→      // Stop ticking until prediction is saved (which will restart ticking at 45')
   314→   314→      clearInterval(activeIntervals[mId]);
   315→   315→      delete activeIntervals[mId];
   316→   316→      console.log(`[SIMULATOR PAUSED] Match #${mId} paused at Halftime for predictions.`);
   317→   317→      return;
   318→   318→    }
   319→   319→
   320→   320→    if (currentMin >= 90) {
   321→   321→      currentMin = 90;
   322→   322→      match.simulation_minute = 90;
   323→   323→      match.status = "finished";
   324→   324→
   325→   325→      clearInterval(activeIntervals[mId]);
   326→   326→      delete activeIntervals[mId];
   327→   327→
   328→   328→      // Evaluate Halftime prediction if exists
   329→   329→      if (match.halftime_prediction) {
   330→   330→        const hGoals = db.match_events.filter(
   331→   331→          (e) => e.match_id === mId && e.minute > 45 && e.event_type === "goal"
   332→   332→        );
   333→   333→        const homeGoals = hGoals.filter((e) => e.team_id === match.home_team_id).length;
   334→   334→        const awayGoals = hGoals.filter((e) => e.team_id === match.away_team_id).length;
   335→   335→        const totalGoals = homeGoals + awayGoals;
   336→   336→
   337→   337→        let isCorrect = false;
   338→   338→        if (match.halftime_prediction === "over15" && totalGoals >= 2) isCorrect = true;
   339→   339→        if (match.halftime_prediction === "under15" && totalGoals < 2) isCorrect = true;
   340→   340→        
   341→   341→        if (match.halftime_prediction === "homeScoreNext" && homeGoals > 0) {
   342→   342→          // Check if home scored before away in second half
   343→   343→          const homeFirst = hGoals.filter(e => e.team_id === match.home_team_id)[0];
   344→   344→          const awayFirst = hGoals.filter(e => e.team_id === match.away_team_id)[0];
   345→   345→          if (!awayFirst || (homeFirst && homeFirst.minute <= awayFirst.minute)) {
   346→   346→            isCorrect = true;
   347→   347→          }
   348→   348→        }
   349→   349→        if (match.halftime_prediction === "awayScoreNext" && awayGoals > 0) {
   350→   350→          const homeFirst = hGoals.filter(e => e.team_id === match.home_team_id)[0];
   351→   351→          const awayFirst = hGoals.filter(e => e.team_id === match.away_team_id)[0];
   352→   352→          if (!homeFirst || (awayFirst && awayFirst.minute <= homeFirst.minute)) {
   353→   353→            isCorrect = true;
   354→   354→          }
   355→   355→        }
   356→   356→
   357→   357→        match.halftime_prediction_status = isCorrect ? "correct" : "incorrect";
   358→   358→      }
   359→   359→
   360→   360→      // Record final timeline commentary
   361→   361→      const finalEvId = db.match_events.length > 0 ? Math.max(...db.match_events.map((o) => o.id)) + 1 : 1;
   362→   362→      db.match_events.push({
   363→   363→        id: finalEvId,
   364→   364→        match_id: mId,
   365→   365→        minute: 90,
   366→   366→        event_type: "substitution",
   367→   367→        description: `FULL TIME CONCLUDED! The referee has blown the final whistle. Final tactical audit logged.`,
   368→   368→        created_at: new Date().toISOString()
   369→   369→      });
   370→   370→
   371→   371→      updateTeamEloRatings(match);
   372→   372→      db.save();
   373→   373→      return;
   374→   374→    }
   375→   375→
   376→   376→    match.simulation_minute = currentMin;
   377→   377→
   378→   378→    // Filter and insert events up to current minute
   379→   379→    const currentEvents = fullTimeline.filter((e) => e.minute > currentMin - 5 && e.minute <= currentMin);
   380→   380→    currentEvents.forEach((ev) => {
   381→   381→      const evId = db.match_events.length > 0 ? Math.max(...db.match_events.map((o) => o.id)) + 1 : 1;
   382→   382→      const newEv: MatchEvent = {
   383→   383→        id: evId,
   384→   384→        match_id: mId,
   385→   385→        minute: ev.minute,
   386→   386→        event_type: ev.event_type,
   387→   387→        team_id: ev.team_id,
   388→   388→        player_name: ev.player_name,
   389→   389→        description: ev.description,
   390→   390→        created_at: new Date().toISOString(),
   391→   391→      };
   392→   392→
   393→   393→      db.match_events.push(newEv);
   394→   394→
   395→   395→      // Update score
   396→   396→      if (ev.event_type === "goal") {
   397→   397→        if (ev.team_id === match.home_team_id) {
   398→   398→          match.home_score++;
   399→   399→        } else if (ev.team_id === match.away_team_id) {
   400→   400→          match.away_score++;
   401→   401→        }
   402→   402→      }
   403→   403→    });
   404→   404→
   405→   405→    db.save();
   406→   406→  }, 1200); // 5 minute segments every 1.2 seconds
   407→   407→}
