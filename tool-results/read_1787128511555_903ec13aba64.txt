   140→def _safe_div(numerator: float, denominator: float, default: float = 0.0) -> float:
   141→    """Divide with safe fallback for zero denominators."""
   142→    if denominator == 0 or not math.isfinite(denominator):
   143→        return default
   144→    result = numerator / denominator
   145→    return result if math.isfinite(result) else default
   146→
   147→
   148→def _safe_avg(values: list[float] | np.ndarray, default: float = 0.0) -> float:
   149→    """Mean of a numeric sequence, returning *default* for empty input."""
   150→    arr = np.asarray(values, dtype=float)
   151→    if arr.size == 0:
   152→        return default
   153→    return float(np.nanmean(arr))
   154→
   155→
   156→def _linear_trend(values: list[float] | np.ndarray, default: float = 0.0) -> float:
   157→    """Slope of a simple linear fit (most recent → oldest = positive trend)."""
   158→    arr = np.asarray(values, dtype=float)
   159→    if arr.size < 2:
   160→        return default
   161→    x = np.arange(len(arr), dtype=float)
   162→    valid = np.isfinite(arr)
   163→    if valid.sum() < 2:
   164→        return default
   165→    x = x[valid]
   166→    y = arr[valid]
   167→    slope = np.polyfit(x, y, 1)[0]
   168→    return float(slope) if math.isfinite(slope) else default

   170→
   171→def _compute_streaks(
   172→    results: list[str] | list[int] | None,
   173→) -> tuple[int, int, int]:
   174→    """Return (win_streak, unbeaten_streak, loss_streak) from recent results.
   175→
   176→    *results* is ordered **most recent first**.  Each element is one of:
   177→    - 'W' / 1  → win
   178→    - 'D' / 0  → draw
   179→    - 'L' / -1 → loss
   180→    """
   181→    if not results:
   182→        return 0, 0, 0
   183→    win_s = unbeaten_s = loss_s = 0
   184→    for r in results:
   185→        if r in ('W', 1):
   186→            win_s += 1
   187→            unbeaten_s += 1
   188→        elif r in ('D', 0):
   189→            unbeaten_s += 1
   190→            break
   191→        elif r in ('L', -1):
   192→            break
   193→        else:
   194→            break
   195→    for r in results:
   196→        if r in ('L', -1):
   197→            loss_s += 1
   198→        else:
   199→            break
   200→    return win_s, unbeaten_s, loss_s
   201→
   202→
   203→def _form_points(results: list[str] | list[int] | None) -> float:
   204→    """Sum of 3/1/0 for W/D/L over recent results (max 15 for 5 matches)."""
   205→    if not results:
   206→        return 0.0
   207→    mapping = {'W': 3, 'D': 1, 'L': 0, 1: 3, 0: 1, -1: 0}
   208→    return sum(mapping.get(r, 0) for r in results)

   211→def _parse_date(value: Any) -> Optional[datetime]:
   212→    """Coerce *value* to a ``datetime``.  Returns ``None`` on failure."""
   213→    if isinstance(value, datetime):
   214→        return value
   215→    if isinstance(value, str):
   216→        for fmt in ("%Y-%m-%d", "%Y-%m-%dT%H:%M:%S", "%Y-%m-%d %H:%M:%S"):
   217→            try:
   218→                return datetime.strptime(value, fmt)
   219→            except ValueError:
   220→                continue
   221→    if isinstance(value, (int, float)):
   222→        try:
   223→            return datetime.fromtimestamp(value)
   224→        except (OSError, ValueError):
   225→            pass
   226→    return None

   236→
   237→    Parameters
   238→    ----------
   239→    league_avg_goals : float

   242→        default fallback when team data is missing.
   243→    home_advantage_elo : float
   244→        ELO points added to home team (standard ≈ 65).
   245→    """
   246→
   247→    def __init__(
   248→        self,
   249→        league_avg_goals: float = 1.35,
   250→        home_advantage_elo: float = 65.0,
   251→    ) -> None:
   252→        self.league_avg_goals = league_avg_goals
   253→        self.home_advantage_elo = home_advantage_elo
   254→
   255→    # ------------------------------------------------------------------
   256→    # Public API
   257→    # ------------------------------------------------------------------
   258→
   259→    def build_match_features(
   260→        self,
   261→        match_data: dict[str, Any],
   262→        home_stats: dict[str, Any],
   263→        away_stats: dict[str, Any],
   264→        h2h_data: list[dict[str, Any]] | None = None,
   265→        odds: dict[str, Any] | None = None,
   266→    ) -> dict[str, float]:
   267→        """Build a feature vector (50 values) for a **single** match prediction.
   268→
   269→        Parameters
   270→        ----------
   271→        match_data : dict
   272→            Must contain ``date`` (str / datetime / timestamp).
   273→        home_stats, away_stats : dict
   274→            Per-team season statistics.  Expected keys (all optional):
   275→
   276→            ELO: ``elo``, ``elo_history`` (list[float])
   277→            Goals: ``goals_scored``, ``goals_conceded``,
   278→                   ``goals_scored_home``, ``goals_conceded_home`` (or ``_away``)
   279→            xG: ``xg_for``, ``xg_for_home`` (or ``_away``)
   280→            Form: ``recent_results`` (list[str] 'W'/'D'/'L', most-recent first),
   281→                  ``recent_goals_scored`` (list[float]),
   282→                  ``last_match_date`` (str/datetime)
   283→            Shots: ``shots_total``, ``shots_on_target``
   284→            Venue: ``home_wins``, ``home_draws``, ``home_losses``,
   285→                   ``away_wins``, ``away_draws``, ``away_losses``,
   286→                   ``home_goals``, ``away_goals``,
   287→                   ``clean_sheets_home``, ``clean_sheets_away``,
   288→                   ``matches_played_home``, ``matches_played_away``
   289→        h2h_data : list[dict], optional
   290→            List of previous head-to-head match dicts.  Each needs
   291→            ``home_goals`` and ``away_goals`` (or ``score`` like '2-1').
   292→        odds : dict, optional
   293→            ``home_odds``, ``draw_odds``, ``away_odds`` (decimal).
   294→
   295→        Returns
   296→        -------
   297→        dict[str, float]
   298→            Mapping of all 50 feature names to float values.
   299→        """
   300→        feats: dict[str, float] = {}
   301→
   302→        # 1. ELO features ─────────────────────────────────────────────
   303→        feats.update(self._elo_features(home_stats, away_stats))
   304→
   305→        # 2. Goal features ────────────────────────────────────────────
   306→        feats.update(self._goal_features(home_stats, away_stats))
   307→
   308→        # 3. Form features ────────────────────────────────────────────
   309→        feats.update(self._form_features(home_stats, away_stats))
   310→
   311→        # 4. Head-to-head features ─────────────────────────────────────
   312→        feats.update(self._h2h_features(h2h_data))
   313→
   314→        # 5. Home/away split features ──────────────────────────────────
   315→        feats.update(self._venue_features(home_stats, away_stats))
   316→
   317→        # 6. Market / odds features ────────────────────────────────────
   318→        feats.update(self._market_features(odds))
   319→
   320→        # 7. Temporal features ─────────────────────────────────────────
   321→        feats.update(self._temporal_features(match_data, home_stats, away_stats))
   322→
   323→        # 8. Advanced features ─────────────────────────────────────────
   324→        feats.update(self._advanced_features(home_stats, away_stats))
   325→
   326→        # Guarantee ordered output matching FEATURE_NAMES
   327→        return {name: feats.get(name, 0.0) for name in FEATURE_NAMES}

   329→    # ------------------------------------------------------------------
   330→    def build_training_dataset(
   331→        self,
   332→        matches: list[dict[str, Any]],
   333→        window: int = 10,
   334→    ) -> pd.DataFrame:
   335→        """Build a training dataset from historical matches.
   336→
   337→        Each match in *matches* must contain:
   338→
   339→        - ``home_team``, ``away_team``
   340→        - ``home_goals``, ``away_goals``  (for target + stats)
   341→        - ``date``
   342→        - ``home_odds``, ``draw_odds``, ``away_odds``  (optional)
   343→
   344→        The method computes a sliding-window of *window* previous matches
   345→        per team to derive form, streak, and trend features.  Missing
   346→        historical data is back-filled with league-average defaults.
   347→
   348→        Parameters
   349→        ----------
   350→        matches : list[dict]
   351→            Historical match records.  Must be **chronologically sorted**
   352→            (oldest first) for correct sliding-window behaviour.
   353→        window : int
   354→            Number of recent matches to use for rolling form stats.
   355→
   356→        Returns
   357→        -------
   358→        pd.DataFrame
   359→            One row per match.  Columns are the 50 features plus
   360→            ``target`` (0=away, 1=draw, 2=home).
   361→        """
   362→        if not matches:
   363→            return pd.DataFrame(columns=[*FEATURE_NAMES, "target"])
   364→
   365→        # Index: team_id → list of (date, goals_scored, goals_conceded, result)
   366→        team_history: dict[str, list[tuple]] = defaultdict(list)
   367→        # team_id → list of home-only / away-only records
   368→        team_home_history: dict[str, list[tuple]] = defaultdict(list)
   369→        team_away_history: dict[str, list[tuple]] = defaultdict(list)
   370→
   371→        rows: list[dict[str, float]] = []
   372→
   373→        for match in matches:
   374→            home_team = str(match.get("home_team", ""))
   375→            away_team = str(match.get("away_team", ""))
   376→            hg = int(match.get("home_goals", 0))
   377→            ag = int(match.get("away_goals", 0))
   378→
   379→            # Derive result codes (from home perspective)
   380→            if hg > ag:
   381→                home_result, away_result = "W", "L"
   382→            elif hg < ag:
   383→                home_result, away_result = "L", "W"
   384→            else:
   385→                home_result, away_result = "D", "D"
   386→
   387→            match_date = _parse_date(match.get("date")) or datetime.now()
   388→
   389→            # ── Build per-team stats dicts from rolling history ───────
   390→            home_stats = self._stats_from_history(
   391→                team_history[home_team],
   392→                team_home_history[home_team],
   393→                team_away_history[home_team],
   394→                window=window,
   395→            )
   396→            away_stats = self._stats_from_history(
   397→                team_history[away_team],
   398→                team_home_history[away_team],
   399→                team_away_history[away_team],
   400→                window=window,
   401→            )
   402→
   403→            # Set last-match date for rest-day calculation
   404→            if team_history[home_team]:
   405→                home_stats["last_match_date"] = team_history[home_team][-1][0].isoformat()
   406→            if team_history[away_team]:
   407→                away_stats["last_match_date"] = team_history[away_team][-1][0].isoformat()
   408→
   409→            # ── Head-to-head (use previous matches between these teams) ─
   410→            h2h: list[dict[str, Any]] = []
   411→            for prev in matches:
   412→                if prev is match:
   413→                    continue
   414→                if (
   415→                    str(prev.get("home_team")) == home_team
   416→                    and str(prev.get("away_team")) == away_team
   417→                ) or (
   418→                    str(prev.get("home_team")) == away_team
   419→                    and str(prev.get("away_team")) == home_team
   420→                ):
   421→                    phg = int(prev.get("home_goals", 0))
   422→                    pag = int(prev.get("away_goals", 0))
   423→                    # Normalise to current home perspective
   424→                    if str(prev.get("home_team")) == home_team:
   425→                        h2h.append({"home_goals": phg, "away_goals": pag})
   426→                    else:
   427→                        h2h.append({"home_goals": pag, "away_goals": phg})
   428→
   429→            # ── Odds ──────────────────────────────────────────────────
   430→            odds: dict[str, Any] | None = None
   431→            if match.get("home_odds") and match.get("draw_odds") and match.get("away_odds"):
   432→                odds = {
   433→                    "home_odds": float(match["home_odds"]),
   434→                    "draw_odds": float(match["draw_odds"]),
   435→                    "away_odds": float(match["away_odds"]),
   436→                }
   437→
   438→            # ── Build features ────────────────────────────────────────
   439→            feats = self.build_match_features(
   440→                match_data={"date": match_date},
   441→                home_stats=home_stats,
   442→                away_stats=away_stats,
   443→                h2h_data=h2h if h2h else None,
   444→                odds=odds,
   445→            )
   446→
   447→            # Target: 0=away, 1=draw, 2=home
   448→            if hg > ag:
   449→                target = 2
   450→            elif hg < ag:
   451→                target = 0
   452→            else:
   453→                target = 1
   454→
   455→            feats["target"] = float(target)
   456→            rows.append(feats)
   457→
   458→            # ── Update histories for future windows ───────────────────
   459→            team_history[home_team].append(
   460→                (match_date, hg, ag, home_result)
   461→            )
   462→            team_history[away_team].append(
   463→                (match_date, ag, hg, away_result)
   464→            )
   465→            team_home_history[home_team].append(
   466→                (match_date, hg, ag, home_result)
   467→            )
   468→            team_away_history[away_team].append(
   469→                (match_date, ag, hg, away_result)
   470→            )
   471→
   472→        df = pd.DataFrame(rows)
   473→        # Ensure column order
   474→        cols = [*FEATURE_NAMES, "target"]
   475→        for c in cols:
   476→            if c not in df.columns:
   477→                df[c] = 0.0
   478→        return df[cols]
   479→

   495→    def _elo_features(
   496→        self, home_stats: dict[str, Any], away_stats: dict[str, Any]
   497→    ) -> dict[str, float]:
   498→        elo_home = float(home_stats.get("elo", 1500.0))
   499→        elo_away = float(away_stats.get("elo", 1500.0))
   500→        elo_diff = elo_home - elo_away
   501→        elo_haa = elo_diff + self.home_advantage_elo
   502→
   503→        # Form trend: slope of recent ELO history
   504→        home_history = home_stats.get("elo_history", [])
   505→        elo_trend = _linear_trend(home_history, default=0.0)
   506→
   507→        return {
   508→            "elo_home": elo_home,
   509→            "elo_away": elo_away,
   510→            "elo_diff": elo_diff,
   511→            "elo_home_advantage_adjusted": elo_haa,
   512→            "elo_form_trend": elo_trend,
   513→        }

   516→
   517→    def _goal_features(
   518→        self, home_stats: dict[str, Any], away_stats: dict[str, Any]
   519→    ) -> dict[str, float]:
   520→        lag = self.league_avg_goals
   521→
   522→        h_gs_home = float(home_stats.get("goals_scored_home", lag))
   523→        h_gs_away = float(home_stats.get("goals_scored_away", lag))
   524→        h_gs = float(home_stats.get("goals_scored", lag))
   525→        h_gc_home = float(home_stats.get("goals_conceded_home", lag))
   526→        h_gc_away = float(home_stats.get("goals_conceded_away", lag))
   527→        h_gc = float(home_stats.get("goals_conceded", lag))
   528→
   529→        a_gs_home = float(away_stats.get("goals_scored_home", lag))
   530→        a_gs_away = float(away_stats.get("goals_scored_away", lag))
   531→        a_gs = float(away_stats.get("goals_scored", lag))
   532→        a_gc_home = float(away_stats.get("goals_conceded_home", lag))
   533→        a_gc_away = float(away_stats.get("goals_conceded_away", lag))
   534→        a_gc = float(away_stats.get("goals_conceded", lag))
   535→
   536→        gd = h_gs - h_gc
   537→
   538→        xg_h = float(home_stats.get("xg_for_home", lag))
   539→        xg_a = float(away_stats.get("xg_for_away", lag))
   540→        xg_diff = xg_h - xg_a
   541→
   542→        return {
   543→            "avg_goals_scored_home": h_gs_home,
   544→            "avg_goals_scored_away": a_gs_away,
   545→            "avg_goals_conceded_home": h_gc_home,
   546→            "avg_goals_conceded_away": a_gc_away,
   547→            "avg_goals_scored": h_gs,
   548→            "avg_goals_conceded": h_gc,
   549→            "goal_difference": gd,
   550→            "xg_for_home": xg_h,
   551→            "xg_for_away": xg_a,
   552→            "xg_diff": xg_diff,
   553→        }
   554→
   555→    # ---------- 3. Form features ---------------------------------------
   556→
   557→    def _form_features(
   558→        self, home_stats: dict[str, Any], away_stats: dict[str, Any]
   559→    ) -> dict[str, float]:
   560→        h_results = home_stats.get("recent_results", [])
   561→        a_results = away_stats.get("recent_results", [])
   562→
   563→        h_form = _form_points(h_results)
   564→        a_form = _form_points(a_results)
   565→
   566→        h_win, h_unbeaten, _ = _compute_streaks(h_results)
   567→        a_win, a_unbeaten, _ = _compute_streaks(a_results)
   568→
   569→        h_goals_trend_raw = home_stats.get("recent_goals_scored", [])
   570→        goals_trend = _linear_trend(h_goals_trend_raw, default=0.0)
   571→
   572→        return {
   573→            "last_5_form_points_home": float(h_form),
   574→            "last_5_form_points_away": float(a_form),
   575→            "form_points_diff": float(h_form - a_form),
   576→            "win_streak_home": float(h_win),
   577→            "win_streak_away": float(a_win),
   578→            "unbeaten_streak_home": float(h_unbeaten),
   579→            "unbeaten_streak_away": float(a_unbeaten),
   580→            "goals_per_game_trend": goals_trend,
   581→        }
   582→
   583→    # ---------- 4. Head-to-head ----------------------------------------
   584→
   585→    def _h2h_features(
   586→        self, h2h_data: list[dict[str, Any]] | None
   587→    ) -> dict[str, float]:
   588→        if not h2h_data:
   589→            return {
   590→                "h2h_home_win_pct": 0.0,
   591→                "h2h_away_win_pct": 0.0,
   592→                "h2h_draw_pct": 0.0,
   593→                "h2h_avg_goals": self.league_avg_goals * 2,
   594→            }
   595→
   596→        n = len(h2h_data)
   597→        home_wins = sum(1 for m in h2h_data if m.get("home_goals", 0) > m.get("away_goals", 0))
   598→        away_wins = sum(1 for m in h2h_data if m.get("home_goals", 0) < m.get("away_goals", 0))
   599→        draws = n - home_wins - away_wins
   600→
   601→        total_goals_list = [m.get("home_goals", 0) + m.get("away_goals", 0) for m in h2h_data]
   602→        avg_goals = _safe_avg(total_goals_list, default=self.league_avg_goals * 2)
   603→
   604→        return {
   605→            "h2h_home_win_pct": _safe_div(home_wins, n),
   606→            "h2h_away_win_pct": _safe_div(away_wins, n),
   607→            "h2h_draw_pct": _safe_div(draws, n),
   608→            "h2h_avg_goals": avg_goals,
   609→        }

   611→    # ---------- 5. Venue splits ----------------------------------------
   612→
   613→    def _venue_features(
   614→        self, home_stats: dict[str, Any], away_stats: dict[str, Any]
   615→    ) -> dict[str, float]:
   616→        lag = self.league_avg_goals
   617→
   618→        # Home team's home record
   619→        hw = int(home_stats.get("home_wins", 0))
   620→        hd = int(home_stats.get("home_draws", 0))
   621→        hl = int(home_stats.get("home_losses", 0))
   622→        h_home_total = hw + hd + hl
   623→        h_home_goals = float(home_stats.get("home_goals", lag))
   624→
   625→        # Away team's away record
   626→        aw = int(away_stats.get("away_wins", 0))
   627→        ad = int(away_stats.get("away_draws", 0))
   628→        al = int(away_stats.get("away_losses", 0))
   629→        a_away_total = aw + ad + al
   630→        a_away_goals = float(away_stats.get("away_goals", lag))
   631→
   632→        return {
   633→            "home_win_pct": _safe_div(hw, h_home_total),
   634→            "away_win_pct": _safe_div(aw, a_away_total),
   635→            "home_goals_avg": h_home_goals,
   636→            "away_goals_avg": a_away_goals,
   637→        }

   639→    # ---------- 6. Market / odds features ------------------------------
   640→
   641→    def _market_features(
   642→        self, odds: dict[str, Any] | None
   643→    ) -> dict[str, float]:
   644→        default_odds = 1.0
   645→        if odds is None:
   646→            return {
   647→                "implied_prob_home": 1.0 / 3.0,
   648→                "implied_prob_draw": 1.0 / 3.0,
   649→                "implied_prob_away": 1.0 / 3.0,
   650→                "overround": 1.0,
   651→            }
   652→
   653→        ho = float(odds.get("home_odds", default_odds))
   654→        do = float(odds.get("draw_odds", default_odds))
   655→        ao = float(odds.get("away_odds", default_odds))
   656→
   657→        # Guard against zero
   658→        ho = max(ho, 1.01)
   659→        do = max(do, 1.01)
   660→        ao = max(ao, 1.01)
   661→
   662→        ip_h = _safe_div(1.0, ho, 1.0 / 3.0)
   663→        ip_d = _safe_div(1.0, do, 1.0 / 3.0)
   664→        ip_a = _safe_div(1.0, ao, 1.0 / 3.0)
   665→        overround = ip_h + ip_d + ip_a
   666→
   667→        return {
   668→            "implied_prob_home": ip_h,
   669→            "implied_prob_draw": ip_d,
   670→            "implied_prob_away": ip_a,
   671→            "overround": overround,
   672→        }
   673→
   674→    # ---------- 7. Temporal features -----------------------------------
   675→
   676→    def _temporal_features(
   677→        self,
   678→        match_data: dict[str, Any],
   679→        home_stats: dict[str, Any],
   680→        away_stats: dict[str, Any],
   681→    ) -> dict[str, float]:
   682→        match_date = _parse_date(match_data.get("date")) or datetime.now()
   683→        dow = float(match_date.weekday())
   684→        month = float(match_date.month)
   685→        is_weekend = 1.0 if dow >= 5 else 0.0
   686→
   687→        # Rest days
   688→        h_last = _parse_date(home_stats.get("last_match_date"))
   689→        a_last = _parse_date(away_stats.get("last_match_date"))
   690→
   691→        default_rest = 7.0
   692→        h_rest = default_rest
   693→        a_rest = default_rest
   694→        if h_last:
   695→            delta = (match_date - h_last).days
   696→            h_rest = float(max(delta, 0)) if delta is not None else default_rest
   697→        if a_last:
   698→            delta = (match_date - a_last).days
   699→            a_rest = float(max(delta, 0)) if delta is not None else default_rest
   700→
   701→        return {
   702→            "day_of_week": dow,
   703→            "month": month,
   704→            "is_weekend": is_weekend,
   705→            "days_since_last_match_home": h_rest,
   706→            "days_since_last_match_away": a_rest,
   707→        }
   708→
   709→    # ---------- 8. Advanced features -----------------------------------
   710→
   711→    def _advanced_features(
   712→        self, home_stats: dict[str, Any], away_stats: dict[str, Any]
   713→    ) -> dict[str, float]:
   714→        lag = self.league_avg_goals
   715→
   716→        # Attacking / defensive strength (relative to league average)
   717→        h_gs = float(home_stats.get("goals_scored", lag))
   718→        h_gc = float(home_stats.get("goals_conceded", lag))
   719→        a_gs = float(away_stats.get("goals_scored", lag))
   720→        a_gc = float(away_stats.get("goals_conceded", lag))
   721→
   722→        atk_h = _safe_div(h_gs, lag, 1.0)
   723→        atk_a = _safe_div(a_gs, lag, 1.0)
   724→        def_h = _safe_div(h_gc, lag, 1.0)
   725→        def_a = _safe_div(a_gc, lag, 1.0)
   726→
   727→        # Poisson lambda: attack * opponent defence * league average
   728→        poisson_h = atk_h * def_a * lag
   729→        poisson_a = atk_a * def_h * lag
   730→
   731→        # Shot conversion rates
   732→        h_shots = float(home_stats.get("shots_total", 0.0))
   733→        h_sot = float(home_stats.get("shots_on_target", 0.0))
   734→        h_matches = max(float(home_stats.get("matches_played", 1)), 1.0)
   735→        a_shots = float(away_stats.get("shots_total", 0.0))
   736→        a_sot = float(away_stats.get("shots_on_target", 0.0))
   737→        a_matches = max(float(away_stats.get("matches_played", 1)), 1.0)
   738→
   739→        h_conv = _safe_div(h_sot, h_shots, 0.33)
   740→        a_conv = _safe_div(a_sot, a_shots, 0.33)
   741→
   742→        # Clean-sheet percentage
   743→        h_cs = float(home_stats.get("clean_sheets_home", 0.0))
   744→        h_home_m = max(float(home_stats.get("matches_played_home", 1)), 1.0)
   745→        a_cs = float(away_stats.get("clean_sheets_away", 0.0))
   746→        a_away_m = max(float(away_stats.get("matches_played_away", 1)), 1.0)
   747→
   748→        return {
   749→            "attacking_strength_home": atk_h,
   750→            "attacking_strength_away": atk_a,
   751→            "defensive_strength_home": def_h,
   752→            "defensive_strength_away": def_a,
   753→            "poisson_lambda_home": poisson_h,
   754→            "poisson_lambda_away": poisson_a,
   755→            "shot_conversion_rate_home": h_conv,
   756→            "shot_conversion_rate_away": a_conv,
   757→            "clean_sheet_pct_home": _safe_div(h_cs, h_home_m),
   758→            "clean_sheet_pct_away": _safe_div(a_cs, a_away_m),
   759→        }

   764→
   765→    @staticmethod
   766→    def _stats_from_history(
   767→        overall: list[tuple],
   768→        home_only: list[tuple],
   769→        away_only: list[tuple],
   770→        window: int = 10,
   771→    ) -> dict[str, Any]:
   772→        """Derive a stats dict from rolling match history.
   773→
   774→        Each tuple is ``(date, goals_scored, goals_conceded, result_char)``.
   775→        Uses the last *window* entries for form/streak, and full history
   776→        for season averages.
   777→        """
   778→        stats: dict[str, Any] = {}
   779→
   780→        if not overall:
   781→            return stats
   782→
   783→        # Full-season averages (all matches)
   784→        gs_all = [m[1] for m in overall]
   785→        gc_all = [m[2] for m in overall]
   786→        stats["goals_scored"] = _safe_avg(gs_all)
   787→        stats["goals_conceded"] = _safe_avg(gc_all)
   788→        stats["matches_played"] = float(len(overall))
   789→
   790→        # Venue-split averages
   791→        if home_only:
   792→            stats["goals_scored_home"] = _safe_avg([m[1] for m in home_only])
   793→            stats["goals_conceded_home"] = _safe_avg([m[2] for m in home_only])
   794→            stats["home_wins"] = sum(1 for m in home_only if m[3] == "W")
   795→            stats["home_draws"] = sum(1 for m in home_only if m[3] == "D")
   796→            stats["home_losses"] = sum(1 for m in home_only if m[3] == "L")
   797→            stats["home_goals"] = _safe_avg([m[1] for m in home_only])
   798→            stats["matches_played_home"] = float(len(home_only))
   799→            stats["clean_sheets_home"] = float(
   800→                sum(1 for m in home_only if m[2] == 0)
   801→            )
   802→
   803→        if away_only:
   804→            stats["goals_scored_away"] = _safe_avg([m[1] for m in away_only])
   805→            stats["goals_conceded_away"] = _safe_avg([m[2] for m in away_only])
   806→            stats["away_wins"] = sum(1 for m in away_only if m[3] == "W")
   807→            stats["away_draws"] = sum(1 for m in away_only if m[3] == "D")
   808→            stats["away_losses"] = sum(1 for m in away_only if m[3] == "L")
   809→            stats["away_goals"] = _safe_avg([m[1] for m in away_only])
   810→            stats["matches_played_away"] = float(len(away_only))
   811→            stats["clean_sheets_away"] = float(
   812→                sum(1 for m in away_only if m[2] == 0)
   813→            )
   814→
   815→        # Recent form (sliding window)
   816→        recent = overall[-window:] if len(overall) >= window else overall
   817→        stats["recent_results"] = [m[3] for m in recent]
   818→        stats["recent_goals_scored"] = [m[1] for m in recent]
   819→
   820→        return stats
