    58→function StatCard({ title, value, icon: Icon, change, color = 'emerald', suffix = '' }: {
    59→  title: string; value: string | number; icon: React.ElementType; change?: number; color?: string; suffix?: string
    60→}) {
    61→  const isPositive = change !== undefined && change >= 0
    62→  const colorMap: Record<string, string> = {
    63→    emerald: 'bg-emerald-500/10 text-emerald-400',
    64→    gold: 'bg-yellow-500/10 text-yellow-400',
    65→    red: 'bg-red-500/10 text-red-400',
    66→    blue: 'bg-cyan-500/10 text-cyan-400',
    67→    purple: 'bg-purple-500/10 text-purple-400',
    68→    orange: 'bg-orange-500/10 text-orange-400',
    69→  }
    70→  return (
    71→    <Card className="glass-card glass-card-hover animate-fade-in-up">
    72→      <CardContent className="p-4">
    73→        <div className="flex items-center justify-between">
    74→          <div className="space-y-1 min-w-0">
    75→            <p className="text-xs text-muted-foreground uppercase tracking-wider flex items-center gap-1.5">
    76→              <Icon className="h-3 w-3" />
    77→              {title}
    78→            </p>
    79→            <p className="text-2xl font-bold truncate">{value}{suffix}</p>
    80→            {change !== undefined && (
    81→              <div className={`flex items-center gap-1 text-xs ${isPositive ? 'text-emerald-400' : 'text-red-400'}`}>
    82→                {isPositive ? <ArrowUpRight className="h-3 w-3" /> : <ArrowDownRight className="h-3 w-3" />}
    83→                <span>{Math.abs(change)}%</span>
    84→                <span className="text-muted-foreground">vs last period</span>
    85→              </div>
    86→            )}
    87→          </div>
    88→          <div className={`p-3 rounded-xl shrink-0 ${colorMap[color] || colorMap.emerald}`}>
    89→            <Icon className="h-5 w-5" />
    90→          </div>
    91→        </div>
    92→      </CardContent>
    93→    </Card>
    94→  )
    95→}

   111→function ChartTooltip({ active, payload, label }: any) {
   112→  if (!active || !payload?.length) return null
   113→  return (
   114→    <div className="glass-card p-2 rounded-lg text-xs border border-border">
   115→      <p className="text-muted-foreground mb-1">{label}</p>
   116→      {payload.map((p: any, i: number) => (
   117→        <p key={i} className="flex items-center gap-1.5">
   118→          <span className="w-2 h-2 rounded-full" style={{ backgroundColor: p.color }} />
   119→          <span className="text-foreground">{p.name}:</span>
   120→          <span className="font-bold">{p.value}</span>
   121→        </p>
   122→      ))}
   123→    </div>
