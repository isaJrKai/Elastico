    62→// ── Types ──────────────────────────────────────────────────────────────────────
    63→
    64→interface EnhancedPlayer extends Player {
    65→  teamName?: string
    66→  teamCode?: string
    67→  teamColor?: string
    68→  nationality?: string
    69→  appearances?: number
    70→  minutesPlayed?: number
    71→}
    72→// ── Helpers ────────────────────────────────────────────────────────────────────
    73→
    74→function generateRadarStats(p: EnhancedPlayer) {
    75→  return {
    76→    Pace: p.position === 'FWD' || p.position === 'MID' ? 75 : 55,
    77→    Shooting: p.position === 'FWD' ? 75 : 50,
    78→    Passing: p.position === 'MID' ? 75 : 55,
    79→    Defending: p.position === 'DEF' ? 75 : 45,
    80→    Physical: 65,
    81→    Dribbling: p.position === 'FWD' || p.position === 'MID' ? 72 : 50,
    82→  }
    83→}

    89→  const token = useElasticoStore(s => s.token)
    90→  const [players, setPlayers] = useState<EnhancedPlayer[]>([])

    92→  const [positionFilter, setPositionFilter] = useState<string>('all')
    93→  const [teamFilter, setTeamFilter] = useState<string>('all')
    94→  const [sortBy, setSortBy] = useState<string>('rating')

    98→  const perPage = 12

   100→  // Fetch players — try DB first, fallback to ESPN
   101→  useEffect(() => {
   102→    async function fetchPlayers() {
   103→      try {
   104→        const headers: Record<string, string> = {}
   105→        if (token) headers['Authorization'] = `Bearer ${token}`
   106→        const params = new URLSearchParams()
   107→        if (sortBy) params.set('sortBy', sortBy)
   108→        if (positionFilter !== 'all') params.set('position', positionFilter)
   109→        if (search) params.set('search', search)
   110→        params.set('limit', '100')
   111→        const res = await fetch(`/api/players?${params}`, { headers })
   112→        if (res.ok) {
   113→          const data = await res.json()
   114→          if (data.players && data.players.length > 0) {
   115→            setPlayers(data.players.map((p: Record<string, unknown>) => ({
   116→              ...p,
   117→              teamName: (p.team as Record<string, unknown>)?.name as string || p.teamName,
   118→              teamCode: (p.team as Record<string, unknown>)?.code as string || p.teamCode,
   119→              teamColor: (p.team as Record<string, unknown>)?.primaryColor as string || '#00e676',
   120→            } as EnhancedPlayer)))
   121→            return
   122→          }
   123→        }
   124→      } catch {
   125→        // DB empty — fallback to ESPN
   126→      }
   127→
   128→      // ESPN fallback: fetch teams for selected league, then fetch rosters
   129→      try {
   130→        const leagueCode = teamFilter !== 'all' ? undefined : 'PL'
   131→        const teamsRes = await fetch(`/api/live?action=teams&league=${teamFilter !== 'all' ? teamFilter : 'PL'}`)
   132→        const teamsData = await teamsRes.json()
   133→        const espnTeams = teamsData.data || teamsData.teams || []
   134→
   135→        if (espnTeams.length === 0) return
   136→
   137→        // Fetch rosters for up to 6 teams (to avoid too many requests)
   138→        const allPlayers: EnhancedPlayer[] = []
   139→        const teamsToFetch = espnTeams.slice(0, 6)
   140→
   141→        await Promise.allSettled(teamsToFetch.map(async (t: any) => {
   142→          try {
   143→            const rosterRes = await fetch(`/api/live?action=roster&league=PL&team=${t.id}`)
   144→            if (!rosterRes.ok) return
   145→            const rosterData = await rosterRes.json()
   146→            const athletes = rosterData.data || rosterData.athletes || []
   147→            for (const a of athletes) {
   148→              allPlayers.push({
   149→                id: String(a.id || ''),
   150→                name: a.name || a.displayName || 'Unknown',
   151→                position: a.position || 'MID',
   152→                age: a.age || 25,
   153→                nationality: a.nationality || '',
   154→                rating: 0, // ESPN roster API does not provide match ratings
   155→                goals: a.goals || 0,
   156→                assists: a.assists || 0,
   157→                teamName: t.name,
   158→                teamCode: t.abbreviation || t.code,
   159→                teamColor: t.color || '#00e676',
   160→                teamId: String(t.id || ''),
   161→                appearances: a.appearances || 0,
   162→                minutesPlayed: 0,
   163→                shirtNumber: a.shirtNumber || a.jersey || 0,
   164→              } as unknown as EnhancedPlayer)
   165→            }
   166→          } catch { /* skip this team */ }
   167→        }))
   168→
   169→        if (allPlayers.length > 0) setPlayers(allPlayers)
   170→      } catch {
   171→        // ESPN fallback also failed
   172→      }
   173→    }
   174→    fetchPlayers()
   175→  }, [token, sortBy, positionFilter, search, teamFilter])
   176→

   182→    if (teamFilter !== 'all') result = result.filter(p => p.teamName === teamFilter)
   183→    result.sort((a, b) => {
   184→      if (sortBy === 'rating') return (b.rating || 0) - (a.rating || 0)
   185→      if (sortBy === 'goals') return b.goals - a.goals
   186→      if (sortBy === 'assists') return b.assists - a.assists
   187→      if (sortBy === 'name') return a.name.localeCompare(b.name)
   188→      if (sortBy === 'age') return (a.age || 0) - (b.age || 0)
   189→      if (sortBy === 'marketValue') return (b.marketValue || 0) - (a.marketValue || 0)
   190→      return 0
   191→    })
   192→    return result

   213→      const age = p.age || 25
   214→      if (age <= 21) buckets['18-21']++
   215→      else if (age <= 25) buckets['22-25']++
   216→      else if (age <= 29) buckets['26-29']++
   217→      else if (age <= 33) buckets['30-33']++
   218→      else buckets['34+']++
   219→    }

   244→          Math.abs(stats.Physical - ps.Physical) + Math.abs(stats.Dribbling - ps.Dribbling)
   245→        return { player: p, similarity: 100 - Math.round(diff / 6) }

   262→    a.download = 'elastico-players.csv'

   271→      case 'DEF': return 'text-blue-400 bg-blue-400/10 border-blue-400/30'

   274→      default: return ''
   275→    }
   276→  }

   281→    if (r >= 6.0) return 'text-yellow-400'
   282→    return 'text-red-400'
   283→  }
   284→

   286→    <motion.div
   287→      initial={{ opacity: 0, y: 20 }}
   288→      animate={{ opacity: 1, y: 0 }}
   289→      className="space-y-6"
   290→    >
   291→      {/* Header */}
   292→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   293→        <div>
   294→        <p className="text-muted-foreground text-sm">Search, analyze, and compare football players</p>
   295→        </div>
   296→        <Button variant="outline" size="sm" onClick={exportCSV} className="gap-2">
   297→          <Download className="size-4" /> Export CSV
   298→        </Button>
   299→      </div>
   300→
   301→      {/* Search & Filters */}
   302→      <div className="flex flex-col sm:flex-row gap-3">
   303→        <div className="relative flex-1">
   304→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />

   306→            placeholder="Search players..."

   309→            className="pl-9"
   310→          />
   311→        </div>
   312→        <Select value={positionFilter} onValueChange={(v) => { setPositionFilter(v); setPage(0) }}>

   314→            <SelectValue placeholder="Position" />
   315→          </SelectTrigger>
   316→          <SelectContent>
   317→            <SelectItem value="all">All Positions</SelectItem>
   318→            <SelectItem value="GK">GK</SelectItem>
   319→            <SelectItem value="DEF">DEF</SelectItem>
   320→            <SelectItem value="MID">MID</SelectItem>
   321→            <SelectItem value="FWD">FWD</SelectItem>
   322→          </SelectContent>
   323→        </Select>
   324→        <Select value={teamFilter} onValueChange={(v) => { setTeamFilter(v); setPage(0) }}>
   325→          <SelectTrigger className="w-full sm:w-40">
   326→            <SelectValue placeholder="Team" />
   327→          </SelectTrigger>
   328→          <SelectContent>
   329→            <SelectItem value="all">All Teams</SelectItem>

   331→              <SelectItem key={t} value={t!}>{t}</SelectItem>
   332→            ))}

   336→          <SelectTrigger className="w-full sm:w-36">
   337→            <SelectValue placeholder="Sort by" />
   338→          </SelectTrigger>

   340→            <SelectItem value="rating">Rating</SelectItem>

   346→          </SelectContent>
   347→        </Select>
   348→      </div>
   349→
   350→      <Tabs defaultValue="grid" className="space-y-4">
   351→        <TabsList className="flex flex-wrap gap-1 h-auto bg-card/50 p-1 rounded-lg">
   352→          <TabsTrigger value="grid" className="text-xs">Player Grid</TabsTrigger>
   353→          <TabsTrigger value="leaderboard" className="text-xs">Top Scorers</TabsTrigger>
   354→          <TabsTrigger value="radar" className="text-xs">Radar Charts</TabsTrigger>
   355→          <TabsTrigger value="compare" className="text-xs">Compare</TabsTrigger>
   356→          <TabsTrigger value="value" className="text-xs">Market Value</TabsTrigger>
   357→          <TabsTrigger value="positions" className="text-xs">By Position</TabsTrigger>
   358→          <TabsTrigger value="age" className="text-xs">Age Dist.</TabsTrigger>
   359→          <TabsTrigger value="nationality" className="text-xs">Nationality</TabsTrigger>

   437→        {/* Top Scorers Leaderboard */}

   439→          <Card className="glass-card">

   441→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   442→                <Star className="size-4 text-yellow-400" /> Top Scorers Leaderboard
   443→              </CardTitle>
   444→            </CardHeader>
   445→            <CardContent>
   446→              <div className="max-h-96 overflow-y-auto">
   447→                <Table>
   448→                  <TableHeader>
   449→                    <TableRow>
   450→                      <TableHead className="w-10">#</TableHead>
   451→                      <TableHead>Player</TableHead>

   458→                  </TableHeader>
   459→                  <TableBody>
   460→                    {topScorers.map((p, i) => (

   463→                        <TableCell>
   464→                          <div className="flex items-center gap-2">
   465→                            <FlagIcon nationality={p.nationality} size={16} />

   469→                          <span className="text-[10px] text-muted-foreground">{p.teamName}</span>
   470→                        </TableCell>
   471→                        <TableCell className="text-center font-bold text-primary">{p.goals}</TableCell>
   472→                        <TableCell className="text-center">{p.assists}</TableCell>
