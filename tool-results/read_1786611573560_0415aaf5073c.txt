    61→    61→interface EnhancedPlayer extends Player {
    62→    62→  teamName?: string
    63→    63→  teamCode?: string

    71→    71→function generateRadarStats(p: EnhancedPlayer) {
    72→    72→  return {
    73→    73→    Pace: p.position === 'FWD' || p.position === 'MID' ? 75 : 55,
    74→    74→    Shooting: p.position === 'FWD' ? 75 : 50,
    75→    75→    Passing: p.position === 'MID' ? 75 : 55,
    76→    76→    Defending: p.position === 'DEF' ? 75 : 45,
    77→    77→    Physical: 65,
    78→    78→    Dribbling: p.position === 'FWD' || p.position === 'MID' ? 72 : 50,
    79→    79→  }
    80→    80→}

   102→   102→        if (token) headers['Authorization'] = `Bearer ${token}`

   104→   104→        if (sortBy) params.set('sortBy', sortBy)
   105→   105→        if (positionFilter !== 'all') params.set('position', positionFilter)
   106→   106→        if (search) params.set('search', search)
   107→   107→        params.set('limit', '100')

   109→   109→        if (res.ok) {
   110→   110→          const data = await res.json()
   111→   111→          if (data.players && data.players.length > 0) {
   112→   112→            setPlayers(data.players.map((p: Record<string, unknown>) => ({
   113→   113→              ...p,
   114→   114→              teamName: (p.team as Record<string, unknown>)?.name as string || p.teamName,
   115→   115→              teamCode: (p.team as Record<string, unknown>)?.code as string || p.teamCode,
   116→   116→              teamColor: (p.team as Record<string, unknown>)?.primaryColor as string || '#00e676',
   117→   117→            } as EnhancedPlayer)))
   118→   118→            return
   119→   119→          }
   120→   120→        }
   121→   121→      } catch {
   122→   122→        // DB empty — fallback to ESPN
   123→   123→      }
   124→   124→
   125→   125→      // ESPN fallback: fetch teams for selected league, then fetch rosters
   126→   126→      try {
   127→   127→        const leagueCode = teamFilter !== 'all' ? undefined : 'PL'
   128→   128→        const teamsRes = await fetch(`/api/live?action=teams&league=${teamFilter !== 'all' ? teamFilter : 'PL'}`)
   129→   129→        const teamsData = await teamsRes.json()
   130→   130→        const espnTeams = teamsData.data || teamsData.teams || []
   131→   131→
   132→   132→        if (espnTeams.length === 0) return
   133→   133→
   134→   134→        // Fetch rosters for up to 6 teams (to avoid too many requests)
   135→   135→        const allPlayers: EnhancedPlayer[] = []
   136→   136→        const teamsToFetch = espnTeams.slice(0, 6)
   137→   137→
   138→   138→        await Promise.allSettled(teamsToFetch.map(async (t: any) => {
   139→   139→          try {
   140→   140→            const rosterRes = await fetch(`/api/live?action=roster&league=PL&team=${t.id}`)
   141→   141→            if (!rosterRes.ok) return
   142→   142→            const rosterData = await rosterRes.json()
   143→   143→            const athletes = rosterData.data || rosterData.athletes || []
   144→   144→            for (const a of athletes) {
   145→   145→              allPlayers.push({
   146→   146→                id: String(a.id || ''),
   147→   147→                name: a.name || a.displayName || 'Unknown',
   148→   148→                position: a.position || 'MID',
   149→   149→                age: a.age || 25,
   150→   150→                nationality: a.nationality || '',
   151→   151→                rating: 70 + Math.floor(Math.random() * 15),
   152→   152→                goals: a.goals || 0,
   153→   153→                assists: a.assists || 0,
   154→   154→                teamName: t.name,
   155→   155→                teamCode: t.abbreviation || t.code,
   156→   156→                teamColor: t.color || '#00e676',
   157→   157→                teamId: String(t.id || ''),
   158→   158→                appearances: a.appearances || 0,
   159→   159→                minutesPlayed: 0,
   160→   160→                shirtNumber: a.shirtNumber || a.jersey || 0,
   161→   161→              } as unknown as EnhancedPlayer)
   162→   162→            }
   163→   163→          } catch { /* skip this team */ }
   164→   164→        }))
   165→   165→
   166→   166→        if (allPlayers.length > 0) setPlayers(allPlayers)
   167→   167→      } catch {
   168→   168→        // ESPN fallback also failed
   169→   169→      }
   170→   170→    }
   171→   171→    fetchPlayers()

   213→   213→      else if (age <= 29) buckets['26-29']++
   214→   214→      else if (age <= 33) buckets['30-33']++
   215→   215→      else buckets['34+']++

   217→   217→    return Object.entries(buckets).map(([range, count]) => ({ range, count }))

   224→   224→      map[nat] = (map[nat] || 0) + 1

   230→   230→

   241→   241→          Math.abs(stats.Physical - ps.Physical) + Math.abs(stats.Dribbling - ps.Dribbling)

   243→   243→      })
   244→   244→      .sort((a, b) => b.similarity - a.similarity)

   280→   280→  }
