    59→// ── Types ──────────────────────────────────────────────────────────────────────
    60→
    61→interface EnhancedPlayer extends Player {
    62→  teamName?: string
    63→  teamCode?: string
    64→  teamColor?: string
    65→  nationality?: string
    66→  appearances?: number
    67→  minutesPlayed?: number
    68→}
    69→// ── Helpers ────────────────────────────────────────────────────────────────────
    70→
    71→function generateRadarStats(p: EnhancedPlayer) {
    72→  return {
    73→    Pace: p.position === 'FWD' || p.position === 'MID' ? 75 : 55,
    74→    Shooting: p.position === 'FWD' ? 75 : 50,
    75→    Passing: p.position === 'MID' ? 75 : 55,
    76→    Defending: p.position === 'DEF' ? 75 : 45,
    77→    Physical: 65,
    78→    Dribbling: p.position === 'FWD' || p.position === 'MID' ? 72 : 50,
    79→  }
    80→}

    84→export function PlayerView() {
    85→  const teams = useElasticoStore(s => s.teams)
    86→  const token = useElasticoStore(s => s.token)
    87→  const [players, setPlayers] = useState<EnhancedPlayer[]>([])
    88→  const [search, setSearch] = useState('')
    89→  const [positionFilter, setPositionFilter] = useState<string>('all')
    90→  const [teamFilter, setTeamFilter] = useState<string>('all')
    91→  const [sortBy, setSortBy] = useState<string>('rating')
    92→  const [selectedPlayer, setSelectedPlayer] = useState<EnhancedPlayer | null>(null)
    93→  const [comparePlayer, setComparePlayer] = useState<EnhancedPlayer | null>(null)
    94→  const [page, setPage] = useState(0)
    95→  const perPage = 12
    96→
    97→  // Fetch players — try DB first, fallback to ESPN
    98→  useEffect(() => {
    99→    async function fetchPlayers() {
   100→      try {
   101→        const headers: Record<string, string> = {}
   102→        if (token) headers['Authorization'] = `Bearer ${token}`
   103→        const params = new URLSearchParams()
   104→        if (sortBy) params.set('sortBy', sortBy)
   105→        if (positionFilter !== 'all') params.set('position', positionFilter)
   106→        if (search) params.set('search', search)
   107→        params.set('limit', '100')
   108→        const res = await fetch(`/api/players?${params}`, { headers })
   109→        if (res.ok) {
   110→          const data = await res.json()
   111→          if (data.players && data.players.length > 0) {
   112→            setPlayers(data.players.map((p: Record<string, unknown>) => ({
   113→              ...p,
   114→              teamName: (p.team as Record<string, unknown>)?.name as string || p.teamName,
   115→              teamCode: (p.team as Record<string, unknown>)?.code as string || p.teamCode,
   116→              teamColor: (p.team as Record<string, unknown>)?.primaryColor as string || '#00e676',
   117→            } as EnhancedPlayer)))
   118→            return
   119→          }
   120→        }
   121→      } catch {
   122→        // DB empty — fallback to ESPN
   123→      }
   124→
   125→      // ESPN fallback: fetch teams for selected league, then fetch rosters
   126→      try {
   127→        const leagueCode = teamFilter !== 'all' ? undefined : 'PL'
   128→        const teamsRes = await fetch(`/api/live?action=teams&league=${teamFilter !== 'all' ? teamFilter : 'PL'}`)
   129→        const teamsData = await teamsRes.json()
   130→        const espnTeams = teamsData.data || teamsData.teams || []
   131→
   132→        if (espnTeams.length === 0) return
   133→
   134→        // Fetch rosters for up to 6 teams (to avoid too many requests)
   135→        const allPlayers: EnhancedPlayer[] = []
   136→        const teamsToFetch = espnTeams.slice(0, 6)
   137→
   138→        await Promise.allSettled(teamsToFetch.map(async (t: any) => {
   139→          try {
   140→            const rosterRes = await fetch(`/api/live?action=roster&league=PL&team=${t.id}`)
   141→            if (!rosterRes.ok) return
   142→            const rosterData = await rosterRes.json()
   143→            const athletes = rosterData.data || rosterData.athletes || []
   144→            for (const a of athletes) {
   145→              allPlayers.push({
   146→                id: String(a.id || ''),
   147→                name: a.name || a.displayName || 'Unknown',
   148→                position: a.position || 'MID',
   149→                age: a.age || 25,
   150→                nationality: a.nationality || '',
   151→                rating: 0, // ESPN roster API does not provide match ratings
   152→                goals: a.goals || 0,
   153→                assists: a.assists || 0,
   154→                teamName: t.name,
   155→                teamCode: t.abbreviation || t.code,
   156→                teamColor: t.color || '#00e676',
   157→                teamId: String(t.id || ''),
   158→                appearances: a.appearances || 0,
   159→                minutesPlayed: 0,
   160→                shirtNumber: a.shirtNumber || a.jersey || 0,
   161→              } as unknown as EnhancedPlayer)
   162→            }
   163→          } catch { /* skip this team */ }
   164→        }))
   165→
   166→        if (allPlayers.length > 0) setPlayers(allPlayers)
   167→      } catch {
   168→        // ESPN fallback also failed
   169→      }
   170→    }
   171→    fetchPlayers()
   172→  }, [token, sortBy, positionFilter, search, teamFilter])
   173→
   174→  // Derived data
   175→  const filteredPlayers = useMemo(() => {
   176→    let result = [...players]
   177→    if (search) result = result.filter(p => p.name.toLowerCase().includes(search.toLowerCase()))
   178→    if (positionFilter !== 'all') result = result.filter(p => p.position === positionFilter)
   179→    if (teamFilter !== 'all') result = result.filter(p => p.teamName === teamFilter)
   180→    result.sort((a, b) => {
   181→      if (sortBy === 'rating') return (b.rating || 0) - (a.rating || 0)
   182→      if (sortBy === 'goals') return b.goals - a.goals
   183→      if (sortBy === 'assists') return b.assists - a.assists
   184→      if (sortBy === 'name') return a.name.localeCompare(b.name)
   185→      if (sortBy === 'age') return (a.age || 0) - (b.age || 0)
   186→      if (sortBy === 'marketValue') return (b.marketValue || 0) - (a.marketValue || 0)
   187→      return 0
   188→    })
   189→    return result
   190→  }, [players, search, positionFilter, teamFilter, sortBy])
   191→
   192→  const pagedPlayers = filteredPlayers.slice(page * perPage, (page + 1) * perPage)
   193→  const totalPages = Math.ceil(filteredPlayers.length / perPage)
   194→
   195→  const topScorers = useMemo(() => [...players].sort((a, b) => b.goals - a.goals).slice(0, 10), [players])
   196→  const topByValue = useMemo(() => [...players].filter(p => p.marketValue).sort((a, b) => (b.marketValue || 0) - (a.marketValue || 0)).slice(0, 10), [players])
   197→
   198→  const positionalGroups = useMemo(() => {
   199→    const groups: Record<string, EnhancedPlayer[]> = {}
   200→    for (const p of players) {
   201→      if (!groups[p.position]) groups[p.position] = []
   202→      groups[p.position].push(p)
   203→    }
   204→    return groups
   205→  }, [players])
   206→
   207→  const ageData = useMemo(() => {
   208→    const buckets: Record<string, number> = { '18-21': 0, '22-25': 0, '26-29': 0, '30-33': 0, '34+': 0 }
   209→    for (const p of players) {
   210→      const age = p.age || 25
   211→      if (age <= 21) buckets['18-21']++
   212→      else if (age <= 25) buckets['22-25']++
   213→      else if (age <= 29) buckets['26-29']++
   214→      else if (age <= 33) buckets['30-33']++
   215→      else buckets['34+']++
   216→    }
   217→    return Object.entries(buckets).map(([range, count]) => ({ range, count }))
   218→  }, [players])
   219→
   220→  const nationalities = useMemo(() => {
   221→    const map: Record<string, number> = {}
   222→    for (const p of players) {
   223→      const nat = p.nationality || 'Unknown'
   224→      map[nat] = (map[nat] || 0) + 1
   225→    }
   226→    return Object.entries(map).sort((a, b) => b[1] - a[1]).slice(0, 10)
   227→  }, [players])
   228→
   229→  const uniqueTeams = useMemo(() => [...new Set(players.map(p => p.teamName).filter(Boolean))], [players])
   230→
   231→  // Similar players
   232→  const similarPlayers = useMemo(() => {
   233→    if (!selectedPlayer) return []
   234→    const stats = generateRadarStats(selectedPlayer)
   235→    return players
   236→      .filter(p => p.id !== selectedPlayer.id && p.position === selectedPlayer.position)
   237→      .map(p => {
   238→        const ps = generateRadarStats(p)
   239→        const diff = Math.abs(stats.Pace - ps.Pace) + Math.abs(stats.Shooting - ps.Shooting) +
   240→          Math.abs(stats.Passing - ps.Passing) + Math.abs(stats.Defending - ps.Defending) +
   241→          Math.abs(stats.Physical - ps.Physical) + Math.abs(stats.Dribbling - ps.Dribbling)
   242→        return { player: p, similarity: 100 - Math.round(diff / 6) }
   243→      })
   244→      .sort((a, b) => b.similarity - a.similarity)
   245→      .slice(0, 5)
   246→  }, [selectedPlayer, players])
   247→
   248→  // CSV Export
   249→  const exportCSV = useCallback(() => {
   250→    const headers = ['Name', 'Position', 'Team', 'Age', 'Goals', 'Assists', 'Rating', 'Market Value (€M)', 'Yellow Cards', 'Red Cards', 'Nationality']
   251→    const rows = filteredPlayers.map(p => [
   252→      p.name, p.position, p.teamName || '', p.age || '', p.goals, p.assists, p.rating, p.marketValue || '', p.yellowCards, p.redCards, p.nationality || ''
   253→    ])
   254→    const csv = [headers.join(','), ...rows.map(r => r.map(v => `"${v}"`).join(','))].join('\n')
   255→    const blob = new Blob([csv], { type: 'text/csv' })
   256→    const url = URL.createObjectURL(blob)
   257→    const a = document.createElement('a')
   258→    a.href = url
   259→    a.download = 'elastico-players.csv'
   260→    a.click()
   261→    URL.revokeObjectURL(url)
   262→    toast.success('Players exported to CSV')
   263→  }, [filteredPlayers])
   264→
   265→  const getPositionColor = (pos: string) => {
   266→    switch (pos) {
   267→      case 'GK': return 'text-yellow-400 bg-yellow-400/10 border-yellow-400/30'
   268→      case 'DEF': return 'text-blue-400 bg-blue-400/10 border-blue-400/30'
   269→      case 'MID': return 'text-emerald-400 bg-emerald-400/10 border-emerald-400/30'
   270→      case 'FWD': return 'text-red-400 bg-red-400/10 border-red-400/30'
   271→      default: return ''
   272→    }
   273→  }
   274→
   275→  const getRatingColor = (r: number) => {
   276→    if (r >= 8.0) return 'text-emerald-400'
   277→    if (r >= 7.0) return 'text-emerald-300'
   278→    if (r >= 6.0) return 'text-yellow-400'
   279→    return 'text-red-400'
   280→  }
   281→
   282→  return (
   283→    <motion.div
   284→      initial={{ opacity: 0, y: 20 }}
   285→      animate={{ opacity: 1, y: 0 }}
   286→      className="space-y-6"
   287→    >
   288→      {/* Header */}
   289→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   290→        <div>
   291→          <h1 className="text-2xl font-bold flex items-center gap-2">
   292→            <Users className="text-primary" /> Player Analytics
   293→          </h1>
   294→          <p className="text-muted-foreground text-sm mt-1">Search, analyze, and compare football players</p>
   295→        </div>
   296→        <Button variant="outline" size="sm" onClick={exportCSV} className="gap-2">
   297→          <Download className="size-4" /> Export CSV
   298→        </Button>
   299→      </div>
   300→
   301→      {/* Search & Filters */}
   302→      <div className="flex flex-col sm:flex-row gap-3">
   303→        <div className="relative flex-1">
   304→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   305→          <Input
   306→            placeholder="Search players..."
   307→            value={search}
   308→            onChange={(e) => { setSearch(e.target.value); setPage(0) }}
   309→            className="pl-9"
   310→          />
   311→        </div>
   312→        <Select value={positionFilter} onValueChange={(v) => { setPositionFilter(v); setPage(0) }}>
   313→          <SelectTrigger className="w-full sm:w-36">
   314→            <SelectValue placeholder="Position" />
   315→          </SelectTrigger>
   316→          <SelectContent>
   317→            <SelectItem value="all">All Positions</SelectItem>
   318→            <SelectItem value="GK">GK</SelectItem>
   319→            <SelectItem value="DEF">DEF</SelectItem>
   320→            <SelectItem value="MID">MID</SelectItem>
   321→            <SelectItem value="FWD">FWD</SelectItem>
   322→          </SelectContent>
   323→        </Select>
   324→        <Select value={teamFilter} onValueChange={(v) => { setTeamFilter(v); setPage(0) }}>
   325→          <SelectTrigger className="w-full sm:w-40">
   326→            <SelectValue placeholder="Team" />
   327→          </SelectTrigger>
   328→          <SelectContent>
   329→            <SelectItem value="all">All Teams</SelectItem>
   330→            {uniqueTeams.map(t => (
   331→              <SelectItem key={t} value={t!}>{t}</SelectItem>
   332→            ))}
   333→          </SelectContent>
   334→        </Select>
   335→        <Select value={sortBy} onValueChange={setSortBy}>
   336→          <SelectTrigger className="w-full sm:w-36">
   337→            <SelectValue placeholder="Sort by" />
   338→          </SelectTrigger>
   339→          <SelectContent>
   340→            <SelectItem value="rating">Rating</SelectItem>
   341→            <SelectItem value="goals">Goals</SelectItem>
   342→            <SelectItem value="assists">Assists</SelectItem>
   343→            <SelectItem value="marketValue">Market Value</SelectItem>
   344→            <SelectItem value="age">Age</SelectItem>
   345→            <SelectItem value="name">Name</SelectItem>
   346→          </SelectContent>
   347→        </Select>
   348→      </div>
   349→
   350→      <Tabs defaultValue="grid" className="space-y-4">
   351→        <TabsList className="flex flex-wrap gap-1 h-auto bg-card/50 p-1 rounded-lg">
   352→          <TabsTrigger value="grid" className="text-xs">Player Grid</TabsTrigger>
   353→          <TabsTrigger value="leaderboard" className="text-xs">Top Scorers</TabsTrigger>
   354→          <TabsTrigger value="radar" className="text-xs">Radar Charts</TabsTrigger>
   355→          <TabsTrigger value="compare" className="text-xs">Compare</TabsTrigger>
   356→          <TabsTrigger value="value" className="text-xs">Market Value</TabsTrigger>
   357→          <TabsTrigger value="positions" className="text-xs">By Position</TabsTrigger>
   358→          <TabsTrigger value="age" className="text-xs">Age Dist.</TabsTrigger>
   359→          <TabsTrigger value="nationality" className="text-xs">Nationality</TabsTrigger>
   360→          <TabsTrigger value="cards" className="text-xs">Cards</TabsTrigger>
   361→          <TabsTrigger value="subs" className="text-xs">Substitutions</TabsTrigger>
   362→          <TabsTrigger value="similar" className="text-xs">Similar</TabsTrigger>
   363→          <TabsTrigger value="form" className="text-xs">Form</TabsTrigger>
   364→        </TabsList>
   365→
   366→        {/* Player Cards Grid */}
   367→        <TabsContent value="grid">
   368→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4">
   369→            {pagedPlayers.map((player) => (
   370→              <motion.div
   371→                key={player.id}
   372→                whileHover={{ scale: 1.02, y: -4 }}
   373→                whileTap={{ scale: 0.98 }}
   374→                className="cursor-pointer"
   375→                onClick={() => setSelectedPlayer(player)}
   376→              >
   377→                <Card className="glass-card h-full overflow-hidden">
   378→                  <div className="h-2" style={{ backgroundColor: player.teamColor || '#00e676' }} />
   379→                  <CardContent className="p-4">
   380→                    <div className="flex items-start gap-3">
   381→                      <div className="w-14 h-14 rounded-full bg-muted flex items-center justify-center text-lg font-bold shrink-0 border-2 border-border">
   382→                        {player.number}
   383→                      </div>
   384→                      <div className="min-w-0 flex-1">
   385→                        <h3 className="text-sm font-semibold truncate">{player.name}</h3>
   386→                        <div className="flex items-center gap-2 mt-1">
   387→                          <Badge variant="outline" className={cn('text-[10px]', getPositionColor(player.position))}>
   388→                            {player.position}

   390→                          <span className="text-[10px] text-muted-foreground truncate">{player.teamName}</span>

   396→                    </div>
   397→                    <div className="grid grid-cols-3 gap-2 mt-4">
   398→                      <div className="text-center">
   399→                        <div className="text-lg font-bold text-primary">{player.goals}</div>
   400→                        <div className="text-[10px] text-muted-foreground">Goals</div>
   401→                      </div>
   402→                      <div className="text-center">
   403→                        <div className="text-lg font-bold text-primary">{player.assists}</div>
   404→                        <div className="text-[10px] text-muted-foreground">Assists</div>
   405→                      </div>
   406→                      <div className="text-center">
   407→                        <div className="text-lg font-bold text-primary">{player.yellowCards + player.redCards}</div>
   408→                        <div className="text-[10px] text-muted-foreground">Cards</div>
   409→                      </div>
   410→                    </div>
   411→                    {player.marketValue && (
   412→                      <div className="mt-3 pt-3 border-t border-border/50 flex items-center gap-1">
   413→                        <TrendingUp className="size-3 text-muted-foreground" />
   414→                        <span className="text-xs text-muted-foreground">€{player.marketValue}M</span>
   415→                      </div>
   416→                    )}
   417→                  </CardContent>
   418→                </Card>
   419→              </motion.div>
   420→            ))}
   421→          </div>
   422→          {/* Pagination */}
   423→          {totalPages > 1 && (
   424→            <div className="flex items-center justify-center gap-2 mt-4">
   425→              <Button variant="outline" size="icon" className="h-8 w-8" onClick={() => setPage(Math.max(0, page - 1))} disabled={page === 0}>
   426→                <ChevronLeft className="size-4" />
   427→              </Button>
   428→              <span className="text-sm text-muted-foreground">{page + 1} / {totalPages}</span>
   429→              <Button variant="outline" size="icon" className="h-8 w-8" onClick={() => setPage(Math.min(totalPages - 1, page + 1))} disabled={page >= totalPages - 1}>
   430→                <ChevronRight className="size-4" />
   431→              </Button>
   432→            </div>
   433→          )}
   434→        </TabsContent>
   435→
   436→        {/* Top Scorers Leaderboard */}
   437→        <TabsContent value="leaderboard">
   438→          <Card className="glass-card">
   439→            <CardHeader className="pb-2">
   440→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   441→                <Star className="size-4 text-yellow-400" /> Top Scorers Leaderboard
   442→              </CardTitle>
   443→            </CardHeader>
   444→            <CardContent>
   445→              <div className="max-h-96 overflow-y-auto">
   446→                <Table>
   447→                  <TableHeader>
   448→                    <TableRow>
   449→                      <TableHead className="w-10">#</TableHead>
   450→                      <TableHead>Player</TableHead>
   451→                      <TableHead className="text-center">Goals</TableHead>
   452→                      <TableHead className="text-center">Assists</TableHead>
   453→                      <TableHead className="text-center">Mins</TableHead>
   454→                      <TableHead className="text-center">G/90</TableHead>
   455→                      <TableHead className="text-center">Rating</TableHead>
   456→                    </TableRow>
   457→                  </TableHeader>
   458→                  <TableBody>
   459→                    {topScorers.map((p, i) => (
   460→                      <TableRow key={p.id} className="cursor-pointer hover:bg-accent/50" onClick={() => setSelectedPlayer(p)}>
   461→                        <TableCell className="font-bold text-muted-foreground">{i + 1}</TableCell>
   462→                        <TableCell>
   463→                          <div className="flex items-center gap-2">
   464→                            <span className="text-sm font-medium">{p.name}</span>
   465→                            <Badge variant="outline" className={cn('text-[9px]', getPositionColor(p.position))}>{p.position}</Badge>
   466→                          </div>
   467→                          <span className="text-[10px] text-muted-foreground">{p.teamName}</span>
   468→                        </TableCell>
   469→                        <TableCell className="text-center font-bold text-primary">{p.goals}</TableCell>
   470→                        <TableCell className="text-center">{p.assists}</TableCell>
   471→                        <TableCell className="text-center text-muted-foreground">{p.minutesPlayed || (p.appearances || 0) * 90}</TableCell>
   472→                        <TableCell className="text-center font-medium">
   473→                          {((p.goals / ((p.minutesPlayed || (p.appearances || 0) * 90) / 90))).toFixed(2)}
   474→                        </TableCell>
