    57→function computeEloProb(homeElo: number, awayElo: number, outcome: string): number {
    58→  const ex = 1 / (1 + Math.pow(10, (awayElo - homeElo) / 400))
    59→  if (outcome === 'home') return Math.round(ex * 100)
    60→  if (outcome === 'away') return Math.round((1 - ex) * 100 * 0.75)
    61→  return Math.round(100 - ex * 100 - (1 - ex) * 100 * 0.75)
    62→}

    70→function StatBar({ label, homeValue, awayValue, suffix = '', decimals = 0 }: {
    71→  label: string; homeValue: number | null; awayValue: number | null; suffix?: string; decimals?: number
    72→}) {
    73→  if (homeValue === null || awayValue === null) {
    74→    return (
    75→      <div className="flex items-center gap-4 py-1">
    76→        <span className="flex-1 text-right el-body-sm text-muted-foreground tabular-nums">--</span>
    77→        <span className="w-20 text-center el-micro text-muted-foreground/60">{label}</span>
    78→        <span className="flex-1 el-body-sm text-muted-foreground tabular-nums">--</span>
    79→      </div>
    80→    )
    81→  }
    82→  const total = homeValue + awayValue || 1
    83→  const hp = (homeValue / total) * 100
    84→  const fmt = (v: number) => (decimals > 0 ? v.toFixed(decimals) : String(Math.round(v)))
    85→  return (
    86→    <div className="space-y-1 py-1">
    87→      <div className="flex items-center gap-4">
    88→        <span className={cn('flex-1 text-right el-body-sm font-semibold tabular-nums', hp >= 50 ? 'text-foreground' : 'text-muted-foreground')}>{fmt(homeValue)}{suffix}</span>
    89→        <span className="w-20 text-center el-micro text-muted-foreground/60">{label}</span>
    90→        <span className={cn('flex-1 el-body-sm font-semibold tabular-nums', (100 - hp) > hp ? 'text-foreground' : 'text-muted-foreground')}>{fmt(awayValue)}{suffix}</span>
    91→      </div>
    92→      <div className="flex h-1.5 gap-px rounded-full overflow-hidden bg-white/[0.04]">
    93→        <div className="h-full rounded-full transition-all duration-700 ease-out" style={{ width: `${hp}%`, backgroundColor: CHART_COLORS[0] }} />
    94→        <div className="h-full rounded-full transition-all duration-700 ease-out ml-auto" style={{ width: `${100 - hp}%`, backgroundColor: CHART_COLORS[1] }} />
    95→      </div>
    96→    </div>
    97→  )
    98→}

   100→function eventIcon(type: string) {
   101→  const c = 'size-3 shrink-0'
   102→  if (type === 'goal' || type === 'penalty' || type === 'own_goal') return <Target className={cn(c, 'text-emerald-400')} />
   103→  if (type === 'yellow_card') return <AlertCircle className={cn(c, 'text-amber-400')} />
   104→  if (type === 'red_card') return <AlertCircle className={cn(c, 'text-red-400')} />
   105→  if (type === 'substitution') return <Activity className={cn(c, 'text-muted-foreground')} />
   106→  return <Zap className={cn(c, 'text-muted-foreground')} />
   107→}
