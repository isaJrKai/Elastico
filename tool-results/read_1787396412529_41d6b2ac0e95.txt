    31→// ── Types ──────────────────────────────────────────────────────────────────
    32→
    33→/** Shape returned by GET /api/predictions */
    34→interface PredictionRecord {
    35→  id: string
    36→  matchId: string
    37→  homeTeam: string
    38→  awayTeam: string
    39→  competition: string
    40→  predictedHomeGoals: number
    41→  predictedAwayGoals: number
    42→  predictedOutcome: string
    43→  confidence: number
    44→  model: string
    45→  actualHomeGoals: number | null
    46→  actualAwayGoals: number | null
    47→  isCorrect: boolean | null
    48→  points: number
    49→  matchDate: string | null
    50→  createdAt: string
    51→}
    52→
    53→interface LeaderboardEntry {
    54→  rank: number
    55→  id: string
    56→  name: string | null
    57→  predictionAccuracy: number
    58→  predictionStreak: number
    59→  bestStreak: number
    60→  totalPredictions: number
    61→}
    62→
    63→type ViewState = 'loading' | 'error' | 'empty' | 'success'
    64→
    65→/** Shape returned by GET /api/predictions/compute */
    66→interface ComputePrediction {
    67→  matchId: string
    68→  competition: string
    69→  homeTeam: { name: string; abbreviation?: string; logo?: string }
    70→  awayTeam: { name: string; abbreviation?: string; logo?: string }
    71→  homeScore: number | null
    72→  awayScore: number | null
    73→  status: string
    74→  date: string
    75→  ensemble: { homeWin: number; draw: number; awayWin: number }
    76→  elo: { home: number; draw: number; away: number; expHome: string; expAway: string }
    77→  poisson: { home: number; draw: number; away: number; over25: number; btts: number }
    78→  dixonColes: { home: number; draw: number; away: number }
    79→  overUnder25: number
    80→  btts: number
    81→  mostLikelyScore: string
    82→  expectedGoals: { home: string; away: string; total: string }
    83→  confidence?: number
    84→  volatility?: number
    85→}
    86→
    87→type ComputeState = 'idle' | 'loading' | 'error' | 'success'

    90→
    91→export default function PredictionsView() {
    92→  const user = useElasticoStore(s => s.user)
    93→  const token = useElasticoStore(s => s.token)
    94→  const matches = useElasticoStore(s => s.matches)
    95→
    96→  // ── State ────────────────────────────────────────────────────────────────
    97→  const [predictions, setPredictions] = useState<PredictionRecord[]>([])
    98→  const [leaderboardPos, setLeaderboardPos] = useState<number>(0)
    99→  const [leaderboardCount, setLeaderboardCount] = useState<number>(0)
   100→  const [viewState, setViewState] = useState<ViewState>('loading')
   101→  const [errorMessage, setErrorMessage] = useState('')
   102→  const [sortBy, setSortBy] = useState<'date' | 'confidence' | 'model'>('date')
   103→  const [modelFilter, setModelFilter] = useState('all')
   104→  const [resultFilter, setResultFilter] = useState<'all' | 'correct' | 'incorrect' | 'pending'>('all')
   105→
   106→  // Quick Predict state
   107→  const [quickConfidence, setQuickConfidence] = useState(50)
   108→  const [submittingMatchId, setSubmittingMatchId] = useState<string | null>(null)
   109→  // Per-match score inputs: { [matchId]: { home: number, away: number } }
   110→  const [quickScores, setQuickScores] = useState<Record<string, { home: string; away: string }>>({})
   111→
   112→  // Model Predictions (from /api/predictions/compute)
   113→  const [computePredictions, setComputePredictions] = useState<ComputePrediction[]>([])
   114→  const [computeState, setComputeState] = useState<ComputeState>('idle')
   115→  const [computeError, setComputeError] = useState('')
   116→
   117→  const fetchComputePredictions = useCallback(async () => {
   118→    setComputeState('loading')
   119→    setComputeError('')
   120→    try {
   121→      const headers: Record<string, string> = {}
   122→      if (token) headers['Authorization'] = `Bearer ${token}`
   123→      const res = await fetch('/api/predictions/compute', { headers })
   124→      if (!res.ok) {
   125→        if (res.status === 429) throw new Error('Rate limited — max 5 requests per minute. Wait a moment and try again.')
   126→        if (res.status === 401) throw new Error('Sign in to load model predictions.')
   127→        const body = await res.json().catch(() => ({}))
   128→        throw new Error(body.error || `Prediction engine returned ${res.status}`)
   129→      }
   130→      const data = await res.json()
   131→      setComputePredictions(data.predictions || [])
   132→      setComputeState('success')
   133→    } catch (err) {
   134→      setComputeError(err instanceof Error ? err.message : 'Failed to compute predictions.')
   135→      setComputeState('error')
   136→    }
   137→  }, [token])
   138→
   139→  // ── Data Fetching ────────────────────────────────────────────────────────
   140→  const fetchData = useCallback(async () => {
   141→    setViewState('loading')
   142→    setErrorMessage('')
   143→
   144→    try {
   145→      const headers: Record<string, string> = {}
   146→      if (token) headers['Authorization'] = `Bearer ${token}`
   147→
   148→      // Fetch predictions
   149→      const res = await fetch('/api/predictions', { headers })
   150→      if (!res.ok) {
   151→        const body = await res.json().catch(() => ({}))
   152→        throw new Error(body.error || `Request failed (${res.status})`)
   153→      }
   154→      const data = await res.json()
   155→      const fetched: PredictionRecord[] = Array.isArray(data) ? data : data.predictions || []
   156→      setPredictions(fetched)
   157→
   158→      // Fetch leaderboard
   159→      const lbRes = await fetch('/api/leaderboard')
   160→      if (lbRes.ok) {
   161→        const lbData = await lbRes.json()
   162→        const entries: LeaderboardEntry[] = Array.isArray(lbData) ? lbData : lbData.leaderboard || []
   163→        const pos = entries.findIndex((e: LeaderboardEntry) => e.id === user?.id)
   164→        setLeaderboardPos(pos >= 0 ? pos + 1 : 0)
   165→        setLeaderboardCount(entries.length)
   166→      }
   167→
   168→      // Determine view state
   169→      if (fetched.length === 0) {
   170→        setViewState('empty')
   171→      } else {
   172→        setViewState('success')
   173→      }
   174→    } catch (err) {
   175→      const msg = err instanceof Error ? err.message : 'Failed to load predictions. Check your connection and try again.'
   176→      setErrorMessage(msg)
   177→      setViewState('error')
   178→    }
   179→  }, [token, user?.id])
   180→
   181→  useEffect(() => { fetchData() }, [fetchData])
   182→  // Auto-fetch compute predictions when user has predictions loaded
   183→  useEffect(() => { if (viewState === 'success' || viewState === 'empty') fetchComputePredictions() }, [viewState, fetchComputePredictions])
   184→
   185→  // ── Derived data ─────────────────────────────────────────────────────────
   186→
   187→  // Match map from store for cross-referencing (status, team colors)
   188→  const matchMap = useMemo(() => {
   189→    const map: Record<string, (typeof matches)[number]> = {}
   190→    for (const m of matches) map[m.id] = m
   191→    return map
   192→  }, [matches])
   193→
   194→  const upcomingMatches = useMemo(() => matches.filter(m => m.status === 'upcoming'), [matches])
   195→
   196→  // Separate predictions: resolved (match played) vs pending
   197→  const resolvedPredictions = useMemo(
   198→    () => predictions.filter(p => p.actualHomeGoals != null && p.actualAwayGoals != null),
   199→    [predictions],
   200→  )
   201→  const pendingPredictions = useMemo(
   202→    () => predictions.filter(p => p.actualHomeGoals == null || p.actualAwayGoals == null),
   203→    [predictions],
   204→  )
   205→
   206→  // User's model values (for filter dropdown)
   207→  const usedModels = useMemo(() => {
   208→    const set = new Set<string>()
   209→    for (const p of predictions) set.add(p.model)
   210→    return Array.from(set).sort()
   211→  }, [predictions])
   212→
   213→  const filteredResolved = useMemo(() => {
   214→    let arr = [...resolvedPredictions]
   215→    if (modelFilter !== 'all') arr = arr.filter(p => p.model === modelFilter)
   216→    if (resultFilter === 'correct') arr = arr.filter(p => p.isCorrect === true)
   217→    if (resultFilter === 'incorrect') arr = arr.filter(p => p.isCorrect === false)
   218→    if (sortBy === 'date') arr.sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   219→    if (sortBy === 'confidence') arr.sort((a, b) => b.confidence - a.confidence)
   220→    if (sortBy === 'model') arr.sort((a, b) => a.model.localeCompare(b.model))
   221→    return arr
   222→  }, [resolvedPredictions, modelFilter, resultFilter, sortBy])
   223→
   224→  // Stats from DB user record
   225→  const accuracy = user?.predictionAccuracy ?? 0
   226→  const streak = user?.predictionStreak ?? 0
   227→  const bestStreak = user?.bestStreak ?? 0
   228→  const totalPredictions = user?.totalPredictions ?? 0
   229→
   230→  // Accuracy by Model — computed from real resolved predictions
   231→  const accuracyByModel = useMemo(() => {
   232→    const map: Record<string, { total: number; correct: number }> = {}
   233→    for (const p of resolvedPredictions) {
   234→      if (!map[p.model]) map[p.model] = { total: 0, correct: 0 }
   235→      map[p.model].total++
   236→      if (p.isCorrect) map[p.model].correct++
   237→    }
   238→    return Object.entries(map)
   239→      .map(([model, { total, correct }]) => ({
   240→        model,
   241→        accuracy: total > 0 ? Math.round((correct / total) * 100) : 0,
   242→        total,
   243→      }))
   244→      .sort((a, b) => b.accuracy - a.accuracy)
   245→  }, [resolvedPredictions])
   246→
   247→  // ── Actions ──────────────────────────────────────────────────────────────
   248→
   249→  const handleExport = useCallback(() => {
   250→    if (filteredResolved.length === 0) {
   251→      toast.error('Nothing to export', { description: 'No resolved predictions to export.' })
   252→      return
   253→    }
   254→    const data = filteredResolved.map(p => ({
   255→      Date: p.matchDate ? new Date(p.matchDate).toLocaleDateString() : new Date(p.createdAt).toLocaleDateString(),
   256→      Match: `${p.homeTeam} vs ${p.awayTeam}`,
   257→      Competition: p.competition,
   258→      Prediction: p.predictedOutcome,
   259→      'Predicted Score': `${p.predictedHomeGoals}-${p.predictedAwayGoals}`,
   260→      'Actual Score': `${p.actualHomeGoals ?? '?'}-${p.actualAwayGoals ?? '?'}`,
   261→      Confidence: `${p.confidence}%`,
   262→      Model: p.model,
   263→      Result: p.isCorrect === true ? 'Correct' : p.isCorrect === false ? 'Incorrect' : 'Pending',
   264→      Points: p.points,
   265→    }))
   266→    generateCSV(data, 'elastico-predictions')
   267→    toast.success('Exported!', { description: `${data.length} predictions exported as CSV` })
   268→  }, [filteredResolved])
   269→
   270→  const handleQuickPredict = useCallback(async (matchId: string, choice: string) => {
   271→    if (!token) {
   272→      toast.error('Not authenticated')
   273→      return
   274→    }
   275→
   276→    const scores = quickScores[matchId]
   277→    if (!scores || scores.home === '' || scores.away === '') {
   278→      toast.error('Score required', { description: 'Enter your predicted score for both teams.' })
   279→      return
   280→    }
   281→
   282→    const homeGoals = parseInt(scores.home, 10)
   283→    const awayGoals = parseInt(scores.away, 10)
   284→    if (isNaN(homeGoals) || isNaN(awayGoals) || homeGoals < 0 || awayGoals < 0) {
   285→      toast.error('Invalid score', { description: 'Scores must be non-negative integers.' })
   286→      return
   287→    }
   288→
   289→    const storeMatch = matchMap[matchId]
   290→    setSubmittingMatchId(matchId)
   291→    try {
   292→      const res = await fetch('/api/predictions', {
   293→        method: 'POST',
   294→        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
   295→        body: JSON.stringify({
   296→          matchId,
   297→          predictedHomeGoals: homeGoals,
   298→          predictedAwayGoals: awayGoals,
   299→          predictedOutcome: choice,
   300→          confidence: quickConfidence,
   301→          model: 'user',
   302→          homeTeam: storeMatch?.homeTeam?.name ?? null,
   303→          awayTeam: storeMatch?.awayTeam?.name ?? null,
   304→          competition: storeMatch?.competition ?? null,
   305→          matchDate: storeMatch?.date ?? null,
   306→        }),
   307→      })
   308→      if (!res.ok) {
   309→        const body = await res.json().catch(() => ({}))
   310→        throw new Error(body.error || `Submission failed (${res.status})`)
   311→      }
   312→      toast.success('Prediction submitted!')
   313→      fetchData() // refresh
   314→    } catch (err) {
   315→      const msg = err instanceof Error ? err.message : 'Failed to submit prediction'
   316→      toast.error('Prediction failed', { description: msg })
   317→    } finally {
   318→      setSubmittingMatchId(null)
   319→    }
   320→  }, [token, quickScores, quickConfidence, matchMap, fetchData])
   321→
   322→  const updateScore = useCallback((matchId: string, side: 'home' | 'away', value: string) => {
   323→    // Allow only digits
   324→    const cleaned = value.replace(/[^0-9]/g, '')
   325→    setQuickScores(prev => ({
   326→      ...prev,
   327→      [matchId]: {
   328→        home: side === 'home' ? cleaned : (prev[matchId]?.home ?? ''),
   329→        away: side === 'away' ? cleaned : (prev[matchId]?.away ?? ''),
   330→      },
   331→    }))
   332→  }, [])
   333→
   334→  // Check which upcoming matches already have predictions
   335→  const predictedMatchIds = useMemo(() => new Set(predictions.map(p => p.matchId)), [predictions])
   336→
   337→  // ── Render: LOADING ─────────────────────────────────────────────────────
   338→
   339→  if (viewState === 'loading') {
   340→    return (
   341→      <div className="space-y-5 animate-fade-in-up">
   342→        <div className="flex items-center gap-3">
   343→          <Skeleton className="size-10 rounded-xl" />
   344→          <div className="flex-1 space-y-2">
   345→            <Skeleton className="h-4 w-48" />
   346→            <Skeleton className="h-3 w-72" />
   347→          </div>
   348→        </div>
   349→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   350→          {Array.from({ length: 4 }).map((_, i) => (
   351→            <Skeleton key={i} className="h-20 rounded-xl" />
   352→          ))}
   353→        </div>
   354→        <Skeleton className="h-72 rounded-xl" />
   355→        <Skeleton className="h-64 rounded-xl" />
   356→      </div>
   357→    )
   358→  }
   359→
   360→  // ── Render: ERROR ───────────────────────────────────────────────────────
   361→
   362→  if (viewState === 'error') {
   363→    return (
   364→      <div className="space-y-5 animate-fade-in-up">
   365→        <div className="flex items-center gap-3">
   366→          <div className="flex size-10 items-center justify-center rounded-xl bg-primary/15">
   367→            <Target className="size-5 text-primary" />
   368→          </div>
   369→          <div>
   370→            <h2 className="text-base font-bold text-primary">Predictions</h2>
   371→            <p className="text-sm text-muted-foreground">Track, analyze, and improve your prediction performance</p>
   372→          </div>
   373→        </div>
   374→        <DataState
   375→          type="error"
   376→          message={errorMessage || 'Failed to load predictions'}
   377→          onRetry={fetchData}
   378→        />
   379→      </div>
   380→    )
   381→  }
   382→
   383→  // ── Render: EMPTY ───────────────────────────────────────────────────────
   384→
   385→  if (viewState === 'empty') {
   386→    return (
   387→      <div className="space-y-5 animate-fade-in-up">
   388→        {/* Header */}
   389→        <div className="flex items-center gap-3">
   390→          <div className="flex size-10 items-center justify-center rounded-xl bg-primary/15">
   391→            <Target className="size-5 text-primary" />
   392→          </div>
   393→          <div className="flex-1">
   394→            <h2 className="text-base font-bold text-primary">Predictions</h2>
   395→            <p className="text-sm text-muted-foreground">Track, analyze, and improve your prediction performance</p>
   396→          </div>
   397→        </div>
   398→
   399→        {/* Stats (all zeros — honest) */}
   400→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   401→          {[
   402→            { icon: Target, label: 'Accuracy', value: `${accuracy}%`, color: 'text-primary' },
   403→            { icon: Zap, label: 'Total Predictions', value: totalPredictions, color: 'text-cyan-400' },
   404→            { icon: Flame, label: 'Current Streak', value: streak, color: 'text-orange-400' },
   405→            { icon: Crown, label: 'Best Streak', value: bestStreak, color: 'text-amber-400' },
   406→          ].map(s => (
   407→            <Card key={s.label} className="glass-card-premium card-hover-lift rounded-xl">
   408→              <CardContent className="flex items-center gap-3 p-4">
   409→                <div className="flex size-9 shrink-0 items-center justify-center rounded-lg bg-muted/50">
   410→                  <s.icon className={cn('size-4', s.color)} />
   411→                </div>
   412→                <div>
   413→                  <p className="text-lg font-bold leading-tight">{s.value}</p>
   414→                  <p className="text-[10px] text-muted-foreground">{s.label}</p>
   415→                </div>
   416→              </CardContent>
   417→            </Card>
   418→          ))}
   419→        </div>
   420→
   421→        {/* Quick Predict — still visible when empty so user can start */}
   422→        {upcomingMatches.length > 0 && (
   423→          <Card className="glass-card-premium rounded-xl">
   424→            <CardHeader className="pb-3">
   425→              <CardTitle className="text-sm font-bold flex items-center gap-2">
   426→                <Send className="size-4 text-primary" />
   427→                Quick Predict — Upcoming Matches
   428→              </CardTitle>
   429→            </CardHeader>
   430→            <CardContent className="space-y-3">
   431→              <div className="flex items-center gap-3 px-1 pb-2">
   432→                <Label className="text-xs text-muted-foreground whitespace-nowrap">Your confidence:</Label>
   433→                <Slider
   434→                  value={[quickConfidence]}
   435→                  onValueChange={([v]) => setQuickConfidence(v)}
   436→                  min={10}
   437→                  max={100}
   438→                  step={5}
   439→                  className="flex-1"
   440→                />
   441→                <span className="text-xs font-mono text-primary tabular-nums w-8 text-right">{quickConfidence}%</span>
   442→              </div>
   443→              {upcomingMatches.slice(0, 4).map((m) => (
   444→                <QuickPredictRow
   445→                  key={m.id}
   446→                  match={m}
   447→                  scores={quickScores[m.id] ?? { home: '', away: '' }}
   448→                  onScoreChange={(side, val) => updateScore(m.id, side, val)}
   449→                  onPredict={(choice) => handleQuickPredict(m.id, choice)}
   450→                  isSubmitting={submittingMatchId === m.id}
   451→                  alreadyPredicted={predictedMatchIds.has(m.id)}
   452→                />
   453→              ))}
   454→            </CardContent>
   455→          </Card>
   456→        )}
   457→
   458→        <DataState
   459→          type="empty"
   460→          message="No predictions yet. Use Quick Predict above or the Prediction Engine to get started."
   461→          actionLabel="Refresh"
   462→          actionOnClick={fetchData}
   463→        />
   464→      </div>
   465→    )
   466→  }
   467→
   468→  // ── Render: SUCCESS ─────────────────────────────────────────────────────
   469→
   470→  return (
   471→    <div className="space-y-5 animate-fade-in-up">
   472→      {/* Header */}
   473→      <div className="flex items-center gap-3">
   474→        <div className="flex size-10 items-center justify-center rounded-xl bg-primary/15">
   475→          <Target className="size-5 text-primary" />
   476→        </div>
   477→        <div className="flex-1">
   478→          <h2 className="text-base font-bold text-primary">Predictions</h2>
   479→          <p className="text-sm text-muted-foreground">Track, analyze, and improve your prediction performance</p>
   480→        </div>
   481→        <Button
   482→          variant="outline"
   483→          size="sm"
   484→          className="gap-1.5 border-border text-xs"
   485→          onClick={handleExport}
   486→          disabled={filteredResolved.length === 0}
   487→        >
   488→          <Download className="size-3.5" />Export CSV
   489→        </Button>
   490→      </div>
   491→
   492→      {/* ── ACCURACY STATS CARDS ── */}
   493→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   494→        {[
   495→          { icon: Target, label: 'Accuracy', value: `${accuracy}%`, color: 'text-primary' },
   496→          { icon: Zap, label: 'Total Predictions', value: totalPredictions, color: 'text-cyan-400' },
   497→          { icon: Flame, label: 'Current Streak', value: streak, color: 'text-orange-400' },
   498→          { icon: Crown, label: 'Best Streak', value: bestStreak, color: 'text-amber-400' },
   499→        ].map(s => (
   500→          <Card key={s.label} className="glass-card-premium card-hover-lift rounded-xl">
   501→            <CardContent className="flex items-center gap-3 p-4">
   502→              <div className="flex size-9 shrink-0 items-center justify-center rounded-lg bg-muted/50">
   503→                <s.icon className={cn('size-4', s.color)} />
   504→              </div>
   505→              <div>
   506→                <p className="text-lg font-bold leading-tight">{s.value}</p>
   507→                <p className="text-[10px] text-muted-foreground">{s.label}</p>
