   133→   133→   228→class FeatureEngineer:
   134→   134→   229→    """Transform raw match/team/odds data into ~50 ML features.
   135→   135→   230→
   136→   136→   231→    Parameters
   137→   137→   232→    ----------
   138→   138→   233→    league_avg_goals : float
   139→   139→   234→        Average goals per team per match in the league.
   140→   140→   235→        Used as a baseline for strength calculations and as a
   141→   141→   236→        default fallback when team data is missing.
   142→   142→   237→    home_advantage_elo : float
   143→   143→   238→        ELO points added to home team (standard ≈ 65).
   144→   144→   239→    """
   145→   145→   240→
   146→   146→   241→    def __init__(
   147→   147→   242→        self,
   148→   148→   243→        league_avg_goals: float = 1.35,
   149→   149→   244→        home_advantage_elo: float = 65.0,
   150→   150→   245→    ) -> None:
   151→   151→   246→        self.league_avg_goals = league_avg_goals
   152→   152→   247→        self.home_advantage_elo = home_advantage_elo
   153→   153→   248→
   154→   154→   249→    # ------------------------------------------------------------------
   155→   155→   250→    # Public API
   156→   156→   251→    # ------------------------------------------------------------------
   157→   157→   252→
   158→   158→   253→    def build_match_features(
   159→   159→   254→        self,
   160→   160→   255→        match_data: dict[str, Any],
   161→   161→   256→        home_stats: dict[str, Any],
   162→   162→   257→        away_stats: dict[str, Any],
   163→   163→   258→        h2h_data: list[dict[str, Any]] | None = None,
   164→   164→   259→        odds: dict[str, Any] | None = None,
   165→   165→   260→    ) -> dict[str, float]:
   166→   166→   261→        """Build a feature vector (50 values) for a **single** match prediction.
   167→   167→   262→
   168→   168→   263→        Parameters
   169→   169→   264→        ----------
   170→   170→   265→        match_data : dict
   171→   171→   266→            Must contain ``date`` (str / datetime / timestamp).
   172→   172→   267→        home_stats, away_stats : dict
   173→   173→   268→            Per-team season statistics.  Expected keys (all optional):
   174→   174→   269→
   175→   175→   270→            ELO: ``elo``, ``elo_history`` (list[float])
   176→   176→   271→            Goals: ``goals_scored``, ``goals_conceded``,
   177→   177→   272→                   ``goals_scored_home``, ``goals_conceded_home`` (or ``_away``)
   178→   178→   273→            xG: ``xg_for``, ``xg_for_home`` (or ``_away``)
   179→   179→   274→            Form: ``recent_results`` (list[str] 'W'/'D'/'L', most-recent first),
   180→   180→   275→                  ``recent_goals_scored`` (list[float]),
   181→   181→   276→                  ``last_match_date`` (str/datetime)
   182→   182→   277→            Shots: ``shots_total``, ``shots_on_target``
   183→   183→   278→            Venue: ``home_wins``, ``home_draws``, ``home_losses``,
   184→   184→   279→                   ``away_wins``, ``away_draws``, ``away_losses``,
   185→   185→   280→                   ``home_goals``, ``away_goals``,
   186→   186→   281→                   ``clean_sheets_home``, ``clean_sheets_away``,
   187→   187→   282→                   ``matches_played_home``, ``matches_played_away``
   188→   188→   283→        h2h_data : list[dict], optional
   189→   189→   284→            List of previous head-to-head match dicts.  Each needs
   190→   190→   285→            ``home_goals`` and ``away_goals`` (or ``score`` like '2-1').
   191→   191→   286→        odds : dict, optional
   192→   192→   287→            ``home_odds``, ``draw_odds``, ``away_odds`` (decimal).
   193→   193→   288→
   194→   194→   289→        Returns
   195→   195→   290→        -------
   196→   196→   291→        dict[str, float]
   197→   197→   292→            Mapping of all 50 feature names to float values.
   198→   198→   293→        """
   199→   199→   294→        feats: dict[str, float] = {}
   200→   200→   295→
   201→   201→   296→        # 1. ELO features ─────────────────────────────────────────────
   202→   202→   297→        feats.update(self._elo_features(home_stats, away_stats))
   203→   203→   298→
   204→   204→   299→        # 2. Goal features ────────────────────────────────────────────
   205→   205→   300→        feats.update(self._goal_features(home_stats, away_stats))
   206→   206→   301→
   207→   207→   302→        # 3. Form features ────────────────────────────────────────────
   208→   208→   303→        feats.update(self._form_features(home_stats, away_stats))
   209→   209→   304→
   210→   210→   305→        # 4. Head-to-head features ─────────────────────────────────────
   211→   211→   306→        feats.update(self._h2h_features(h2h_data))
   212→   212→   307→
   213→   213→   308→        # 5. Home/away split features ──────────────────────────────────
   214→   214→   309→        feats.update(self._venue_features(home_stats, away_stats))
   215→   215→   310→
   216→   216→   311→        # 6. Market / odds features ────────────────────────────────────
   217→   217→   312→        feats.update(self._market_features(odds))
   218→   218→   313→
   219→   219→   314→        # 7. Temporal features ─────────────────────────────────────────
   220→   220→   315→        feats.update(self._temporal_features(match_data, home_stats, away_stats))
   221→   221→   316→
   222→   222→   317→        # 8. Advanced features ─────────────────────────────────────────
   223→   223→   318→        feats.update(self._advanced_features(home_stats, away_stats))
   224→   224→   319→
   225→   225→   320→        # Guarantee ordered output matching FEATURE_NAMES
   226→   226→   321→        return {name: feats.get(name, 0.0) for name in FEATURE_NAMES}
   227→   227→   322→
   228→   228→   323→    # ------------------------------------------------------------------
   229→   229→   324→    def build_training_dataset(
   230→   230→   325→        self,
   231→   231→   326→        matches: list[dict[str, Any]],
   232→   232→   327→        window: int = 10,
   233→   233→   328→    ) -> pd.DataFrame:
   234→   234→   329→        """Build a training dataset from historical matches.
   235→   235→   330→
   236→   236→   331→        Each match in *matches* must contain:
   237→   237→   332→
   238→   238→   333→        - ``home_team``, ``away_team``
   239→   239→   334→        - ``home_goals``, ``away_goals``  (for target + stats)
   240→   240→   335→        - ``date``
   241→   241→   336→        - ``home_odds``, ``draw_odds``, ``away_odds``  (optional)
   242→   242→   337→
   243→   243→   338→        The method computes a sliding-window of *window* previous matches
   244→   244→   339→        per team to derive form, streak, and trend features.  Missing
   245→   245→   340→        historical data is back-filled with league-average defaults.
   246→   246→   341→
   247→   247→   342→        Parameters
   248→   248→   343→        ----------
   249→   249→   344→        matches : list[dict]
   250→   250→   345→            Historical match records.  Must be **chronologically sorted**
   251→   251→   346→            (oldest first) for correct sliding-window behaviour.
   252→   252→   347→        window : int
   253→   253→   348→            Number of recent matches to use for rolling form stats.
   254→   254→   349→
   255→   255→   350→        Returns
   256→   256→   351→        -------
   257→   257→   352→        pd.DataFrame
   258→   258→   353→            One row per match.  Columns are the 50 features plus
   259→   259→   354→            ``target`` (0=away, 1=draw, 2=home).
   260→   260→   355→        """
   261→   261→   356→        if not matches:
   262→   262→   357→            return pd.DataFrame(columns=[*FEATURE_NAMES, "target"])
   263→   263→   358→
   264→   264→   359→        # Index: team_id → list of (date, goals_scored, goals_conceded, result)
   265→   265→   360→        team_history: dict[str, list[tuple]] = defaultdict(list)
   266→   266→   361→        # team_id → list of home-only / away-only records
   267→   267→   362→        team_home_history: dict[str, list[tuple]] = defaultdict(list)
   268→   268→   363→        team_away_history: dict[str, list[tuple]] = defaultdict(list)
   269→   269→   364→
   270→   270→   365→        rows: list[dict[str, float]] = []
   271→   271→   366→
   272→   272→   367→        for match in matches:
   273→   273→   368→            home_team = str(match.get("home_team", ""))
   274→   274→   369→            away_team = str(match.get("away_team", ""))
   275→   275→   370→            hg = int(match.get("home_goals", 0))
   276→   276→   371→            ag = int(match.get("away_goals", 0))
   277→   277→   372→
   278→   278→   373→            # Derive result codes (from home perspective)
   279→   279→   374→            if hg > ag:
   280→   280→   375→                home_result, away_result = "W", "L"
   281→   281→   376→            elif hg < ag:
   282→   282→   377→                home_result, away_result = "L", "W"
   283→   283→   378→            else:
   284→   284→   379→                home_result, away_result = "D", "D"
   285→   285→   380→
   286→   286→   381→            match_date = _parse_date(match.get("date")) or datetime.now()
   287→   287→   382→
   288→   288→   383→            # ── Build per-team stats dicts from rolling history ───────
   289→   289→   384→            home_stats = self._stats_from_history(
   290→   290→   385→                team_history[home_team],
   291→   291→   386→                team_home_history[home_team],
   292→   292→   387→                team_away_history[home_team],
   293→   293→   388→                window=window,
   294→   294→   389→            )
   295→   295→   390→            away_stats = self._stats_from_history(
   296→   296→   391→                team_history[away_team],
   297→   297→   392→                team_home_history[away_team],
   298→   298→   393→                team_away_history[away_team],
   299→   299→   394→                window=window,
   300→   300→   395→            )
   301→   301→   396→
   302→   302→   397→            # Set last-match date for rest-day calculation
   303→   303→   398→            if team_history[home_team]:
   304→   304→   399→                home_stats["last_match_date"] = team_history[home_team][-1][0].isoformat()
   305→   305→   400→            if team_history[away_team]:
   306→   306→   401→                away_stats["last_match_date"] = team_history[away_team][-1][0].isoformat()
   307→   307→   402→
   308→   308→   403→            # ── Head-to-head (use previous matches between these teams) ─
   309→   309→   404→            h2h: list[dict[str, Any]] = []
   310→   310→   405→            for prev in matches:
   311→   311→   406→                if prev is match:
   312→   312→   407→                    continue
   313→   313→   408→                if (
   314→   314→   409→                    str(prev.get("home_team")) == home_team
   315→   315→   410→                    and str(prev.get("away_team")) == away_team
   316→   316→   411→                ) or (
   317→   317→   412→                    str(prev.get("home_team")) == away_team
   318→   318→   413→                    and str(prev.get("away_team")) == home_team
   319→   319→   414→                ):
   320→   320→   415→                    phg = int(prev.get("home_goals", 0))
   321→   321→   416→                    pag = int(prev.get("away_goals", 0))
   322→   322→   417→                    # Normalise to current home perspective
   323→   323→   418→                    if str(prev.get("home_team")) == home_team:
   324→   324→   419→                        h2h.append({"home_goals": phg, "away_goals": pag})
   325→   325→   420→                    else:
   326→   326→   421→                        h2h.append({"home_goals": pag, "away_goals": phg})
   327→   327→   422→
   328→   328→   423→            # ── Odds ──────────────────────────────────────────────────
   329→   329→   424→            odds: dict[str, Any] | None = None
   330→   330→   425→            if match.get("home_odds") and match.get("draw_odds") and match.get("away_odds"):
   331→   331→   426→                odds = {
   332→   332→   427→                    "home_odds": float(match["home_odds"]),
   333→   333→   428→                    "draw_odds": float(match["draw_odds"]),
   334→   334→   429→                    "away_odds": float(match["away_odds"]),
   335→   335→   430→                }
   336→   336→   431→
   337→   337→   432→            # ── Build features ────────────────────────────────────────
   338→   338→   433→            feats = self.build_match_features(
   339→   339→   434→                match_data={"date": match_date},
   340→   340→   435→                home_stats=home_stats,
   341→   341→   436→                away_stats=away_stats,
   342→   342→   437→                h2h_data=h2h if h2h else None,
   343→   343→   438→                odds=odds,
   344→   344→   439→            )
   345→   345→   440→
   346→   346→   441→            # Target: 0=away, 1=draw, 2=home
   347→   347→   442→            if hg > ag:
   348→   348→   443→                target = 2
   349→   349→   444→            elif hg < ag:
   350→   350→   445→                target = 0
   351→   351→   446→            else:
   352→   352→   447→                target = 1
   353→   353→   448→
   354→   354→   449→            feats["target"] = float(target)
   355→   355→   450→            rows.append(feats)
   356→   356→   451→
   357→   357→   452→            # ── Update histories for future windows ───────────────────
   358→   358→   453→            team_history[home_team].append(
   359→   359→   454→                (match_date, hg, ag, home_result)
   360→   360→   455→            )
   361→   361→   456→            team_history[away_team].append(
   362→   362→   457→                (match_date, ag, hg, away_result)
   363→   363→   458→            )
   364→   364→   459→            team_home_history[home_team].append(
   365→   365→   460→                (match_date, hg, ag, home_result)
   366→   366→   461→            )
   367→   367→   462→            team_away_history[away_team].append(
   368→   368→   463→                (match_date, ag, hg, away_result)
   369→   369→   464→            )
   370→   370→   465→
   371→   371→   466→        df = pd.DataFrame(rows)
   372→   372→   467→        # Ensure column order
   373→   373→   468→        cols = [*FEATURE_NAMES, "target"]
   374→   374→   469→        for c in cols:
   375→   375→   470→            if c not in df.columns:
   376→   376→   471→                df[c] = 0.0
   377→   377→   472→        return df[cols]
   378→   378→   473→
   379→   379→   474→    # ------------------------------------------------------------------
   380→   380→   475→    def get_feature_names(self) -> list[str]:
   381→   381→   476→        """Return ordered list of 50 feature names."""
   382→   382→   477→        return list(FEATURE_NAMES)

   397→   397→   492→        elo_home = float(home_stats.get("elo", 1500.0))

   414→   414→   509→    # ---------- 2. Goal features ---------------------------------------
   415→   415→   510→
   416→   416→   511→    def _goal_features(
   417→   417→   512→        self, home_stats: dict[str, Any], away_stats: dict[str, Any]
   418→   418→   513→    ) -> dict[str, float]:
   419→   419→   514→        lag = self.league_avg_goals
   420→   420→   515→
   421→   421→   516→        h_gs_home = float(home_stats.get("goals_scored_home", lag))
   422→   422→   517→        h_gs_away = float(home_stats.get("goals_scored_away", lag))
   423→   423→   518→        h_gs = float(home_stats.get("goals_scored", lag))
   424→   424→   519→        h_gc_home = float(home_stats.get("goals_conceded_home", lag))
   425→   425→   520→        h_gc_away = float(home_stats.get("goals_conceded_away", lag))
   426→   426→   521→        h_gc = float(home_stats.get("goals_conceded", lag))
   427→   427→   522→
   428→   428→   523→        a_gs_home = float(away_stats.get("goals_scored_home", lag))
   429→   429→   524→        a_gs_away = float(away_stats.get("goals_scored_away", lag))
   430→   430→   525→        a_gs = float(away_stats.get("goals_scored", lag))
   431→   431→   526→        a_gc_home = float(away_stats.get("goals_conceded_home", lag))
   432→   432→   527→        a_gc_away = float(away_stats.get("goals_conceded_away", lag))
   433→   433→   528→        a_gc = float(away_stats.get("goals_conceded", lag))
   434→   434→   529→
   435→   435→   530→        gd = h_gs - h_gc
   436→   436→   531→
   437→   437→   532→        xg_h = float(home_stats.get("xg_for_home", lag))
   438→   438→   533→        xg_a = float(away_stats.get("xg_for_away", lag))
   439→   439→   534→        xg_diff = xg_h - xg_a
   440→   440→   535→
   441→   441→   536→        return {
   442→   442→   537→            "avg_goals_scored_home": h_gs_home,
   443→   443→   538→            "avg_goals_scored_away": a_gs_away,
   444→   444→   539→            "avg_goals_conceded_home": h_gc_home,
   445→   445→   540→            "avg_goals_conceded_away": a_gc_away,
   446→   446→   541→            "avg_goals_scored": h_gs,
   447→   447→   542→            "avg_goals_conceded": h_gc,
   448→   448→   543→            "goal_difference": gd,
   449→   449→   544→            "xg_for_home": xg_h,
   450→   450→   545→            "xg_for_away": xg_a,
   451→   451→   546→            "xg_diff": xg_diff,
   452→   452→   547→        }

   459→   459→   554→        h_results = home_stats.get("recent_results", [])

   482→   482→   577→    # ---------- 4. Head-to-head ----------------------------------------
   483→   483→   578→
   484→   484→   579→    def _h2h_features(
   485→   485→   580→        self, h2h_data: list[dict[str, Any]] | None
   486→   486→   581→    ) -> dict[str, float]:
   487→   487→   582→        if not h2h_data:
   488→   488→   583→            return {
   489→   489→   584→                "h2h_home_win_pct": 0.0,
   490→   490→   585→                "h2h_away_win_pct": 0.0,
   491→   491→   586→                "h2h_draw_pct": 0.0,
   492→   492→   587→                "h2h_avg_goals": self.league_avg_goals * 2,
   493→   493→   588→            }
   494→   494→   589→
   495→   495→   590→        n = len(h2h_data)
   496→   496→   591→        home_wins = sum(1 for m in h2h_data if m.get("home_goals", 0) > m.get("away_goals", 0))
   497→   497→   592→        away_wins = sum(1 for m in h2h_data if m.get("home_goals", 0) < m.get("away_goals", 0))
   498→   498→   593→        draws = n - home_wins - away_wins
   499→   499→   594→
   500→   500→   595→        total_goals_list = [m.get("home_goals", 0) + m.get("away_goals", 0) for m in h2h_data]
   501→   501→   596→        avg_goals = _safe_avg(total_goals_list, default=self.league_avg_goals * 2)
   502→   502→   597→
   503→   503→   598→        return {
   504→   504→   599→            "h2h_home_win_pct": _safe_div(home_wins, n),
   505→   505→   600→            "h2h_away_win_pct": _safe_div(away_wins, n),
   506→   506→   601→            "h2h_draw_pct": _safe_div(draws, n),
   507→   507→   602→            "h2h_avg_goals": avg_goals,
   508→   508→   603→        }
   509→   509→   604→
   510→   510→   605→    # ---------- 5. Venue splits ----------------------------------------
   511→   511→   606→
   512→   512→   607→    def _venue_features(
   513→   513→   608→        self, home_stats: dict[str, Any], away_stats: dict[str, Any]
   514→   514→   609→    ) -> dict[str, float]:
   515→   515→   610→        lag = self.league_avg_goals
   516→   516→   611→
   517→   517→   612→        # Home team's home record
   518→   518→   613→        hw = int(home_stats.get("home_wins", 0))
   519→   519→   614→        hd = int(home_stats.get("home_draws", 0))
   520→   520→   615→        hl = int(home_stats.get("home_losses", 0))
   521→   521→   616→        h_home_total = hw + hd + hl
   522→   522→   617→        h_home_goals = float(home_stats.get("home_goals", lag))
   523→   523→   618→
   524→   524→   619→        # Away team's away record
   525→   525→   620→        aw = int(away_stats.get("away_wins", 0))
   526→   526→   621→        ad = int(away_stats.get("away_draws", 0))
   527→   527→   622→        al = int(away_stats.get("away_losses", 0))
   528→   528→   623→        a_away_total = aw + ad + al
   529→   529→   624→        a_away_goals = float(away_stats.get("away_goals", lag))
   530→   530→   625→
   531→   531→   626→        return {
   532→   532→   627→            "home_win_pct": _safe_div(hw, h_home_total),
   533→   533→   628→            "away_win_pct": _safe_div(aw, a_away_total),
   534→   534→   629→            "home_goals_avg": h_home_goals,
   535→   535→   630→            "away_goals_avg": a_away_goals,
   536→   536→   631→        }
   537→   537→   632→
   538→   538→   633→    # ---------- 6. Market / odds features ------------------------------
   539→   539→   634→
   540→   540→   635→    def _market_features(
   541→   541→   636→        self, odds: dict[str, Any] | None
   542→   542→   637→    ) -> dict[str, float]:
   543→   543→   638→        default_odds = 1.0
   544→   544→   639→        if odds is None:
   545→   545→   640→            return {
   546→   546→   641→                "implied_prob_home": 1.0 / 3.0,
   547→   547→   642→                "implied_prob_draw": 1.0 / 3.0,
   548→   548→   643→                "implied_prob_away": 1.0 / 3.0,
   549→   549→   644→                "overround": 1.0,
   550→   550→   645→            }
   551→   551→   646→
   552→   552→   647→        ho = float(odds.get("home_odds", default_odds))
   553→   553→   648→        do = float(odds.get("draw_odds", default_odds))
   554→   554→   649→        ao = float(odds.get("away_odds", default_odds))
   555→   555→   650→
   556→   556→   651→        # Guard against zero
   557→   557→   652→        ho = max(ho, 1.01)
   558→   558→   653→        do = max(do, 1.01)
   559→   559→   654→        ao = max(ao, 1.01)
   560→   560→   655→
   561→   561→   656→        ip_h = _safe_div(1.0, ho, 1.0 / 3.0)
   562→   562→   657→        ip_d = _safe_div(1.0, do, 1.0 / 3.0)
   563→   563→   658→        ip_a = _safe_div(1.0, ao, 1.0 / 3.0)
   564→   564→   659→        overround = ip_h + ip_d + ip_a
   565→   565→   660→
   566→   566→   661→        return {
   567→   567→   662→            "implied_prob_home": ip_h,
   568→   568→   663→            "implied_prob_draw": ip_d,
   569→   569→   664→            "implied_prob_away": ip_a,
   570→   570→   665→            "overround": overround,
   571→   571→   666→        }
   572→   572→   667→
   573→   573→   668→    # ---------- 7. Temporal features -----------------------------------
   574→   574→   669→
   575→   575→   670→    def _temporal_features(
   576→   576→   671→        self,
   577→   577→   672→        match_data: dict[str, Any],
   578→   578→   673→        home_stats: dict[str, Any],
   579→   579→   674→        away_stats: dict[str, Any],
   580→   580→   675→    ) -> dict[str, float]:
   581→   581→   676→        match_date = _parse_date(match_data.get("date")) or datetime.now()
   582→   582→   677→        dow = float(match_date.weekday())
   583→   583→   678→        month = float(match_date.month)
   584→   584→   679→        is_weekend = 1.0 if dow >= 5 else 0.0
   585→   585→   680→
   586→   586→   681→        # Rest days
   587→   587→   682→        h_last = _parse_date(home_stats.get("last_match_date"))
   588→   588→   683→        a_last = _parse_date(away_stats.get("last_match_date"))
   589→   589→   684→
   590→   590→   685→        default_rest = 7.0
   591→   591→   686→        h_rest = default_rest
   592→   592→   687→        a_rest = default_rest
   593→   593→   688→        if h_last:
   594→   594→   689→            delta = (match_date - h_last).days
   595→   595→   690→            h_rest = float(max(delta, 0)) if delta is not None else default_rest
   596→   596→   691→        if a_last:
   597→   597→   692→            delta = (match_date - a_last).days
   598→   598→   693→            a_rest = float(max(delta, 0)) if delta is not None else default_rest
   599→   599→   694→
   600→   600→   695→        return {
   601→   601→   696→            "day_of_week": dow,
   602→   602→   697→            "month": month,
   603→   603→   698→            "is_weekend": is_weekend,
   604→   604→   699→            "days_since_last_match_home": h_rest,
   605→   605→   700→            "days_since_last_match_away": a_rest,
   606→   606→   701→        }

   608→   608→   703→    # ---------- 8. Advanced features -----------------------------------
   609→   609→   704→
   610→   610→   705→    def _advanced_features(
   611→   611→   706→        self, home_stats: dict[str, Any], away_stats: dict[str, Any]
   612→   612→   707→    ) -> dict[str, float]:
   613→   613→   708→        lag = self.league_avg_goals
   614→   614→   709→
   615→   615→   710→        # Attacking / defensive strength (relative to league average)
   616→   616→   711→        h_gs = float(home_stats.get("goals_scored", lag))
   617→   617→   712→        h_gc = float(home_stats.get("goals_conceded", lag))
   618→   618→   713→        a_gs = float(away_stats.get("goals_scored", lag))
   619→   619→   714→        a_gc = float(away_stats.get("goals_conceded", lag))
   620→   620→   715→
   621→   621→   716→        atk_h = _safe_div(h_gs, lag, 1.0)
   622→   622→   717→        atk_a = _safe_div(a_gs, lag, 1.0)
   623→   623→   718→        def_h = _safe_div(h_gc, lag, 1.0)
   624→   624→   719→        def_a = _safe_div(a_gc, lag, 1.0)
   625→   625→   720→
   626→   626→   721→        # Poisson lambda: attack * opponent defence * league average
   627→   627→   722→        poisson_h = atk_h * def_a * lag
   628→   628→   723→        poisson_a = atk_a * def_h * lag
   629→   629→   724→
   630→   630→   725→        # Shot conversion rates
   631→   631→   726→        h_shots = float(home_stats.get("shots_total", 0.0))
   632→   632→   727→        h_sot = float(home_stats.get("shots_on_target", 0.0))
   633→   633→   728→        h_matches = max(float(home_stats.get("matches_played", 1)), 1.0)
   634→   634→   729→        a_shots = float(away_stats.get("shots_total", 0.0))
   635→   635→   730→        a_sot = float(away_stats.get("shots_on_target", 0.0))
   636→   636→   731→        a_matches = max(float(away_stats.get("matches_played", 1)), 1.0)
   637→   637→   732→
   638→   638→   733→        h_conv = _safe_div(h_sot, h_shots, 0.33)
   639→   639→   734→        a_conv = _safe_div(a_sot, a_shots, 0.33)
   640→   640→   735→
   641→   641→   736→        # Clean-sheet percentage
   642→   642→   737→        h_cs = float(home_stats.get("clean_sheets_home", 0.0))
   643→   643→   738→        h_home_m = max(float(home_stats.get("matches_played_home", 1)), 1.0)
   644→   644→   739→        a_cs = float(away_stats.get("clean_sheets_away", 0.0))
   645→   645→   740→        a_away_m = max(float(away_stats.get("matches_played_away", 1)), 1.0)
   646→   646→   741→
   647→   647→   742→        return {
   648→   648→   743→            "attacking_strength_home": atk_h,
   649→   649→   744→            "attacking_strength_away": atk_a,
   650→   650→   745→            "defensive_strength_home": def_h,
   651→   651→   746→            "defensive_strength_away": def_a,
   652→   652→   747→            "poisson_lambda_home": poisson_h,
   653→   653→   748→            "poisson_lambda_away": poisson_a,
   654→   654→   749→            "shot_conversion_rate_home": h_conv,
   655→   655→   750→            "shot_conversion_rate_away": a_conv,
   656→   656→   751→            "clean_sheet_pct_home": _safe_div(h_cs, h_home_m),
   657→   657→   752→            "clean_sheet_pct_away": _safe_div(a_cs, a_away_m),
   658→   658→   753→        }
   659→   659→   754→

   663→   663→   758→
   664→   664→   759→    @staticmethod
   665→   665→   760→    def _stats_from_history(
   666→   666→   761→        overall: list[tuple],
   667→   667→   762→        home_only: list[tuple],
   668→   668→   763→        away_only: list[tuple],
   669→   669→   764→        window: int = 10,
   670→   670→   765→    ) -> dict[str, Any]:
   671→   671→   766→        """Derive a stats dict from rolling match history.
   672→   672→   767→
   673→   673→   768→        Each tuple is ``(date, goals_scored, goals_conceded, result_char)``.
   674→   674→   769→        Uses the last *window* entries for form/streak, and full history
   675→   675→   770→        for season averages.
   676→   676→   771→        """
   677→   677→   772→        stats: dict[str, Any] = {}
   678→   678→   773→
   679→   679→   774→        if not overall:
   680→   680→   775→            return stats
   681→   681→   776→
   682→   682→   777→        # Full-season averages (all matches)
   683→   683→   778→        gs_all = [m[1] for m in overall]
   684→   684→   779→        gc_all = [m[2] for m in overall]
   685→   685→   780→        stats["goals_scored"] = _safe_avg(gs_all)
   686→   686→   781→        stats["goals_conceded"] = _safe_avg(gc_all)
   687→   687→   782→        stats["matches_played"] = float(len(overall))
   688→   688→   783→
   689→   689→   784→        # Venue-split averages
   690→   690→   785→        if home_only:
   691→   691→   786→            stats["goals_scored_home"] = _safe_avg([m[1] for m in home_only])
   692→   692→   787→            stats["goals_conceded_home"] = _safe_avg([m[2] for m in home_only])
   693→   693→   788→            stats["home_wins"] = sum(1 for m in home_only if m[3] == "W")
   694→   694→   789→            stats["home_draws"] = sum(1 for m in home_only if m[3] == "D")
   695→   695→   790→            stats["home_losses"] = sum(1 for m in home_only if m[3] == "L")
   696→   696→   791→            stats["home_goals"] = _safe_avg([m[1] for m in home_only])
   697→   697→   792→            stats["matches_played_home"] = float(len(home_only))
   698→   698→   793→            stats["clean_sheets_home"] = float(
   699→   699→   794→                sum(1 for m in home_only if m[2] == 0)
   700→   700→   795→            )
   701→   701→   796→
   702→   702→   797→        if away_only:
   703→   703→   798→            stats["goals_scored_away"] = _safe_avg([m[1] for m in away_only])
   704→   704→   799→            stats["goals_conceded_away"] = _safe_avg([m[2] for m in away_only])
   705→   705→   800→            stats["away_wins"] = sum(1 for m in away_only if m[3] == "W")
   706→   706→   801→            stats["away_draws"] = sum(1 for m in away_only if m[3] == "D")
   707→   707→   802→            stats["away_losses"] = sum(1 for m in away_only if m[3] == "L")
   708→   708→   803→            stats["away_goals"] = _safe_avg([m[1] for m in away_only])
   709→   709→   804→            stats["matches_played_away"] = float(len(away_only))
   710→   710→   805→            stats["clean_sheets_away"] = float(
   711→   711→   806→                sum(1 for m in away_only if m[2] == 0)
   712→   712→   807→            )
   713→   713→   808→
   714→   714→   809→        # Recent form (sliding window)
   715→   715→   810→        recent = overall[-window:] if len(overall) >= window else overall
   716→   716→   811→        stats["recent_results"] = [m[3] for m in recent]
   717→   717→   812→        stats["recent_goals_scored"] = [m[1] for m in recent]
   718→   718→   813→
   719→   719→   814→        return stats
