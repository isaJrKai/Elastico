   157→// ── NVIDIA API Status Badge ────────────────────────────────────────────────────

   159→function NvidiaApiStatusBadge({ token }: { token?: string }) {
   160→  const [status, setStatus] = React.useState<'loading' | 'connected' | 'disconnected'>('loading')
   161→
   162→  React.useEffect(() => {
   163→    fetch('/api/chat', {
   164→      method: 'POST',
   165→      headers: {
   166→        'Content-Type': 'application/json',
   167→        ...(token ? { Authorization: `Bearer ${token}` } : {}),
   168→      },
   169→      body: JSON.stringify({ message: '__status_check__', model: 'pro', stream: false }),
   170→    })
   171→      .then((res) => res.json())
   172→      .then((data) => {
   173→        if (data.model === 'pro') {
   174→          setStatus('connected')
   175→        } else if (data.model === 'mock-fallback') {
   176→          setStatus('disconnected')
   177→        } else {
   178→          setStatus('disconnected')
   179→        }
   180→      })
   181→      .catch(() => setStatus('disconnected'))
   182→  }, [])
   183→
   184→  if (status === 'loading') {
   185→    return (
   186→      <Badge variant="outline" className="gap-1.5 text-xs border-border">
   187→        <Loader2 className="size-3 animate-spin text-muted-foreground" />
   188→        Checking...
   189→      </Badge>
   190→    )
   191→  }
   192→
   193→  if (status === 'connected') {
   194→    return (
   195→      <Badge variant="outline" className="gap-1.5 text-xs border-emerald-500/30 text-emerald-400 bg-emerald-500/5">
   196→        <CheckCircle2 className="size-3" />
   197→        Connected
   198→      </Badge>
   199→    )
   200→  }
   201→
   202→  return (
   203→    <Badge variant="outline" className="gap-1.5 text-xs border-amber-500/30 text-amber-400 bg-amber-500/5">
   204→      <XCircle className="size-3" />
   205→      Not Configured
   206→    </Badge>
   207→  )
   208→}

   258→function BandwidthSection() {
   259→  const [bandwidth, setBandwidth] = useState<{ totalKB: number; requests: number; avgBytesPerRequest: number } | null>(null)
   260→  const { isOffline, isRegistered, isStandalone, clearCache } = usePWA()
   261→
   262→  useEffect(() => {
   263→    // Poll bandwidth tracker every 2 seconds
   264→    const load = async () => {
   265→      try {
   266→        const mod = await import('@/lib/compressed-data-stream')
   267→        setBandwidth(mod.BandwidthTracker.summary())
   268→      } catch {}
   269→    }
   270→    load()
   271→    const interval = setInterval(load, 2000)
   272→    return () => clearInterval(interval)
   273→  }, [])
   274→
   275→  return (
   276→    <Card className="glass-card border-border">
   277→      <CardHeader className="pb-4">
   278→        <SectionHeader
   279→          icon={Radio}
   280→          title="Bandwidth & Offline"
   281→          description="PWA caching & data usage monitoring"
   282→        />
   283→      </CardHeader>
   284→      <CardContent className="space-y-4">
   285→        {/* PWA Status Grid */}
   286→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   287→          <StatusPill
   288→            label="Network"
   289→            value={isOffline ? 'Offline' : 'Online'}
   290→            active={!isOffline}
   291→            icon={isOffline ? WifiOff : Wifi}
   292→          />
   293→          <StatusPill
   294→            label="App Cached"
   295→            value={isRegistered ? 'Yes' : 'No'}
   296→            active={isRegistered}
   297→            icon={Smartphone}
   298→          />
   299→          <StatusPill
   300→            label="Standalone"
   301→            value={isStandalone ? 'Yes' : 'No'}
   302→            active={isStandalone}
   303→            icon={Monitor}
   304→          />
   305→          <StatusPill
   306→            label="Data Used"
   307→            value={bandwidth ? `${bandwidth.totalKB.toFixed(1)} KB` : '—'}
   308→            active={bandwidth ? bandwidth.totalKB < 100 : true}
   309→            icon={Zap}
   310→          />
   311→        </div>
   312→
   313→        {/* Bandwidth Details */}
   314→        {bandwidth && (
   315→          <div className="rounded-lg border border-border bg-secondary/30 p-4 space-y-3">
   316→            <div className="flex items-center justify-between text-sm">
   317→              <span className="text-muted-foreground">API Requests</span>
   318→              <span className="font-mono text-foreground">{bandwidth.requests}</span>
   319→            </div>
   320→            <div className="flex items-center justify-between text-sm">
   321→              <span className="text-muted-foreground">Avg per Request</span>
   322→              <span className="font-mono text-foreground">{bandwidth.avgBytesPerRequest} B</span>
   323→            </div>
   324→            <div className="flex items-center justify-between text-sm">
   325→              <span className="text-muted-foreground">Total Today</span>
   326→              <span className={cn(
   327→                'font-mono font-medium',
   328→                bandwidth.totalKB < 50 ? 'text-emerald-400' : (bandwidth.totalKB ?? 0) < 200 ? 'text-amber-400' : 'text-red-400'
   329→              )}>
   330→                {(bandwidth.totalKB ?? 0).toFixed(1)} KB
   331→              </span>
   332→            </div>
   333→          </div>
   334→        )}
   335→
   336→        {/* Compression Features */}
   337→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   338→          {[
   339→            { label: 'Null Stripping', desc: 'Removes 30-60% empty fields' },
   340→            { label: 'Key Compaction', desc: '2-char keys in compact mode' },
   341→            { label: 'Service Worker', desc: '0 MB data on app reopen' },
   342→            { label: 'Diff Updates', desc: 'Only changed data sent' },
   343→          ].map((f) => (
   344→            <div key={f.label} className="flex items-center gap-2.5 rounded-md border border-border bg-secondary/20 px-3 py-2">
   345→              <CheckCircle2 className="size-3.5 text-primary shrink-0" />
   346→              <div>
   347→                <p className="text-xs font-medium text-foreground">{f.label}</p>
   348→                <p className="text-[10px] text-muted-foreground">{f.desc}</p>
   349→              </div>
   350→            </div>
   351→          ))}
   352→        </div>
   353→
   354→        {/* Clear Cache Button */}
   355→        {isRegistered && (
   356→          <Button
   357→            variant="outline"
   358→            size="sm"
   359→            className="w-full border-border text-xs"
   360→            onClick={async () => {
   361→              await clearCache()

   363→            }}
   364→          >
   365→            <Trash2 className="size-3.5 mr-2" />

   372→          <p className="text-[11px] leading-relaxed text-muted-foreground">
   373→            After the first visit, ELASTICO caches itself (~5 MB) to your device storage.
   374→            Subsequent opens load instantly from your phone — using <strong className="text-foreground">0 MB mobile data</strong>.
   375→            Live match updates use compressed diff payloads under 5 KB per cycle.
