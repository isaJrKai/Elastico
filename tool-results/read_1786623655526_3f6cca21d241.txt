    88→export function ExportView() {
    89→  const token = useElasticoStore(s => s.token)
    90→  const user = useElasticoStore(s => s.user)
    91→  const [exportType, setExportType] = useState('matches')
    92→  const [exportFormat, setExportFormat] = useState('csv')
    93→  const [exporting, setExporting] = useState(false)
    94→  const [exportHistory, setExportHistory] = useState<ExportRecord[]>([])
    95→  const [scheduledReports] = useState<{ id: string; name: string; frequency: string; type: string; format: string; active: boolean; nextRun: string }[]>([])
    96→  const [chartRef, setChartRef] = useState<HTMLElement | null>(null)
    97→
    98→  const handleExport = useCallback(async () => {
    99→    setExporting(true)
   100→    try {
   101→      const headers: Record<string, string> = { 'Content-Type': 'application/json' }
   102→      if (token) headers['Authorization'] = `Bearer ${token}`
   103→
   104→      const res = await fetch('/api/export', {
   105→        method: 'POST',
   106→        headers,
   107→        body: JSON.stringify({ type: exportType, format: exportFormat, filters: {} }),
   108→      })
   109→
   110→      if (!res.ok) throw new Error('Export failed')
   111→
   112→      if (exportFormat === 'csv' && res.headers.get('content-type')?.includes('text/csv')) {
   113→        const blob = await res.blob()
   114→        const url = URL.createObjectURL(blob)
   115→        const a = document.createElement('a')
   116→        a.href = url
   117→        a.download = `elastico-${exportType}-${Date.now()}.csv`
   118→        a.click()
   119→        URL.revokeObjectURL(url)
   120→      } else {
   121→        const data = await res.json()
   122→        const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' })
   123→        const url = URL.createObjectURL(blob)
   124→        const a = document.createElement('a')
   125→        a.href = url
   126→        a.download = `elastico-${exportType}-${Date.now()}.json`
   127→        a.click()
   128→        URL.revokeObjectURL(url)
   129→      }
   130→
   131→      const newRecord: ExportRecord = {
   132→        id: `e${Date.now()}`,
   133→        type: exportType.charAt(0).toUpperCase() + exportType.slice(1),
   134→        format: exportFormat.toUpperCase(),
   135→        rows: 0,
   136→        createdAt: new Date().toLocaleString(),
   137→        status: 'completed',
   138→      }
   139→      setExportHistory(prev => [newRecord, ...prev])
   140→      toast.success(`Exported ${exportType} as ${exportFormat.toUpperCase()}`)
   141→    } catch {
   142→      toast.error('Export failed. Please try again.')
   143→      setExportHistory(prev => [{
   144→        id: `e${Date.now()}`,
   145→        type: exportType,
   146→        format: exportFormat.toUpperCase(),
   147→        rows: 0,
   148→        createdAt: new Date().toLocaleString(),
   149→        status: 'failed',
   150→      }, ...prev])
   151→    } finally {
   152→      setExporting(false)
   153→    }
   154→  }, [exportType, exportFormat, token])
   155→
   156→  const handleReportTemplate = useCallback((template: typeof REPORT_TEMPLATES[0]) => {
   157→    toast.info(`Generating ${template.name}...`)
   158→    // Use browser print as PDF generation
   159→    setTimeout(() => {
   160→      window.print()
   161→      toast.success(`${template.name} generated!`)
   162→    }, 500)
   163→  }, [])
   164→
   165→  const handleExportChartAsPNG = useCallback(() => {
   166→    if (!chartRef) {
   167→      toast.error('No chart to export. Visit a chart view first.')
   168→      return
   169→    }
   170→    toast.info('Chart export uses the browser Print dialog. Use "Save as PDF" to capture charts.')
   171→    window.print()
   172→  }, [chartRef])
   173→
   174→  const handleShareReport = useCallback((record: ExportRecord) => {
   175→    const link = `${window.location.origin}/shared/report/${record.id}`
   176→    if (navigator.clipboard?.writeText) {
   177→      navigator.clipboard.writeText(link)
   178→    } else {
   179→      const ta = document.createElement('textarea'); ta.value = link; document.body.appendChild(ta); ta.select(); document.execCommand('copy'); document.body.removeChild(ta)
   180→    }
   181→    toast.success('Shareable link copied to clipboard!')
   182→  }, [])
   183→
   184→  const handleCopyEndpoint = useCallback((path: string) => {
   185→    const fullUrl = `${window.location.origin}${path}`
   186→    if (navigator.clipboard?.writeText) {
   187→      navigator.clipboard.writeText(fullUrl)
   188→    } else {
   189→      const ta = document.createElement('textarea'); ta.value = fullUrl; document.body.appendChild(ta); ta.select(); document.execCommand('copy'); document.body.removeChild(ta)
   190→    }
   191→    toast.success('API endpoint URL copied!')
   192→  }, [])
   193→
   194→  return (
   195→    <div
   196→      className="space-y-6"
   197→    >
   198→      {/* Header */}
   199→      <div>
   200→        <h1 className="text-2xl font-bold flex items-center gap-2">
   201→          <Download className="text-primary" /> Data Export & Reports
   202→        </h1>
   203→        <p className="text-muted-foreground text-sm mt-1">Export data, generate reports, and access the API</p>
   204→      </div>
   205→
   206→      <Tabs defaultValue="custom" className="space-y-4">
   207→        <TabsList className="flex flex-wrap gap-1 h-auto bg-card/50 p-1 rounded-lg">
   208→          <TabsTrigger value="templates" className="text-xs">Templates</TabsTrigger>
   209→          <TabsTrigger value="custom" className="text-xs">Custom Export</TabsTrigger>
   210→          <TabsTrigger value="scheduled" className="text-xs">Scheduled</TabsTrigger>
   211→          <TabsTrigger value="history" className="text-xs">History</TabsTrigger>
   212→          <TabsTrigger value="charts" className="text-xs">Chart Export</TabsTrigger>
   213→          <TabsTrigger value="api" className="text-xs">API Access</TabsTrigger>
   214→          <TabsTrigger value="bulk" className="text-xs">Bulk Export</TabsTrigger>
   215→          <TabsTrigger value="share" className="text-xs">Share</TabsTrigger>
   216→        </TabsList>
   217→
   218→        {/* Report Templates */}
   219→        <TabsContent value="templates">
   220→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   221→            {REPORT_TEMPLATES.map((template) => (
   222→              <div
   223→                key={template.id}
   224→                className="glass-card p-6 rounded-xl border border-border/30 cursor-pointer group transition-all duration-200 hover:-translate-y-1"
   225→                onClick={() => handleReportTemplate(template)}
   226→              >
   227→                {(() => { const Icon = ICON_MAP[template.icon] || FileText; return <Icon className="size-8 text-primary/60 mb-3" /> })()}
   228→                <h3 className="text-sm font-semibold group-hover:text-primary transition-colors">{template.name}</h3>
   229→                <p className="text-[11px] text-muted-foreground mt-2 leading-relaxed">{template.description}</p>
   230→                <div className="flex items-center gap-2 mt-4">
   231→                  <Badge variant="outline" className="text-[9px]">{template.category}</Badge>
   232→                  <span className="text-[10px] text-muted-foreground ml-auto flex items-center gap-1">
   233→                    <FileDown className="size-3" /> PDF
   234→                  </span>
   235→                </div>
   236→              </div>
   237→            ))}
   238→          </div>
   239→        </TabsContent>
   240→
   241→        {/* Custom Report Builder */}
   242→        <TabsContent value="custom">
   243→          <Card className="glass-card">
   244→            <CardHeader className="pb-2">
   245→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   246→                <FileSpreadsheet className="size-4 text-primary" /> Custom Export Builder
   247→              </CardTitle>
   248→            </CardHeader>
   249→            <CardContent className="space-y-6">
   250→              <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
   251→                <div className="space-y-2">
   252→                  <Label className="text-xs">Data Type</Label>
   253→                  <Select value={exportType} onValueChange={setExportType}>
   254→                    <SelectTrigger>
   255→                      <SelectValue />
   256→                    </SelectTrigger>
   257→                    <SelectContent>
   258→                      <SelectItem value="matches">Matches</SelectItem>
   259→                      <SelectItem value="players">Players</SelectItem>
   260→                      <SelectItem value="predictions">My Predictions</SelectItem>
   261→                      <SelectItem value="teams">Teams</SelectItem>
   262→                    </SelectContent>
   263→                  </Select>
   264→                </div>
   265→                <div className="space-y-2">
   266→                  <Label className="text-xs">Export Format</Label>
   267→                  <Select value={exportFormat} onValueChange={setExportFormat}>
   268→                    <SelectTrigger>
   269→                      <SelectValue />
   270→                    </SelectTrigger>
   271→                    <SelectContent>
   272→                      <SelectItem value="csv">CSV</SelectItem>
   273→                      <SelectItem value="json">JSON</SelectItem>
   274→                    </SelectContent>
   275→                  </Select>
   276→                </div>
   277→                <div className="space-y-2">
   278→                  <Label className="text-xs">Date Range</Label>
   279→                  <Input type="text" placeholder="All dates" disabled />
   280→                </div>
   281→              </div>
   282→
   283→              <div className="flex flex-col sm:flex-row items-start sm:items-center gap-3">
   284→                <Button onClick={handleExport} disabled={exporting} className="gap-2">
   285→                  {exporting ? <Loader2 className="size-4 animate-spin" /> : <Download className="size-4" />}
   286→                  {exporting ? 'Exporting...' : `Export ${exportType} as ${exportFormat.toUpperCase()}`}
   287→                </Button>
   288→                <span className="text-xs text-muted-foreground">
   289→                  {exportType === 'matches' && 'Includes scores, xG, possession, shots, and team data'}
   290→                  {exportType === 'players' && 'Includes stats, ratings, market values, and team info'}
   291→                  {exportType === 'predictions' && 'Includes your predictions with outcomes and accuracy'}
   292→                  {exportType === 'teams' && 'Includes ELO, xG, style, and squad data'}
   293→                </span>
   294→              </div>
   295→            </CardContent>
   296→          </Card>
   297→        </TabsContent>
   298→
   299→        {/* Scheduled Reports */}
   300→        <TabsContent value="scheduled">
   301→          <Card className="glass-card">
   302→            <CardHeader className="pb-2">
   303→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   304→                <Clock className="size-4 text-primary" /> Scheduled Reports
   305→              </CardTitle>
   306→            </CardHeader>
   307→            <CardContent className="space-y-3">
   308→              {scheduledReports.length === 0 ? (
   309→                <div className="text-center py-12 text-muted-foreground">
   310→                  <Clock className="size-12 mx-auto mb-3 opacity-30" />
   311→                  <p className="text-sm">No scheduled reports yet.</p>
   312→                  <p className="text-xs text-muted-foreground/70 mt-1">Scheduled report automation is coming soon.</p>
   313→                </div>
   314→              ) : (
   315→                scheduledReports.map(report => (
   316→                  <div key={report.id} className="flex items-center justify-between p-4 rounded-lg bg-card/50 border border-border/30">
   317→                    <div className="flex items-center gap-3">
   318→                      <div className={cn('w-3 h-3 rounded-full', report.active ? 'bg-primary' : 'bg-muted-foreground/30')} />
   319→                      <div>
   320→                        <div className="text-sm font-medium">{report.name}</div>
   321→                        <div className="text-[10px] text-muted-foreground">
   322→                          {report.frequency} · {report.type} · {report.format} · Next: {report.nextRun}
   323→                        </div>
   324→                      </div>
   325→                    </div>
   326→                    <div className="flex items-center gap-2">
   327→                      <Badge variant="outline" className={cn('text-[10px]', report.active ? 'text-primary border-primary/30' : 'text-muted-foreground')}>
   328→                        {report.active ? 'Active' : 'Paused'}
   329→                      </Badge>
   330→                    </div>
   331→                  </div>
   332→                ))
   333→              )}
   334→              <Button variant="outline" className="w-full gap-2 text-xs" disabled>
   335→                <Calendar className="size-4" /> Schedule New Report
   336→              </Button>
   337→            </CardContent>
   338→          </Card>
   339→        </TabsContent>
   340→
   341→        {/* Export History */}
   342→        <TabsContent value="history">
   343→          <Card className="glass-card">
   344→            <CardHeader className="pb-2">
   345→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   346→                <History className="size-4 text-primary" /> Export History
   347→              </CardTitle>
   348→            </CardHeader>
   349→            <CardContent>
   350→              <div className="space-y-2 max-h-96 overflow-y-auto">
   351→                {exportHistory.length === 0 ? (
   352→                  <div className="text-center py-12 text-muted-foreground">
   353→                    <History className="size-12 mx-auto mb-3 opacity-30" />
   354→                    <p className="text-sm">No exports yet.</p>
   355→                    <p className="text-xs text-muted-foreground/70 mt-1">Use the Custom Export tab to generate your first export.</p>
   356→                  </div>
   357→                ) : exportHistory.map(record => (
   358→                  <div key={record.id} className="flex items-center gap-3 p-3 rounded-lg bg-card/50 border border-border/30">
   359→                    {record.status === 'completed' ? (
   360→                      <CheckCircle2 className="size-4 text-primary shrink-0" />
   361→                    ) : (
   362→                      <AlertCircle className="size-4 text-red-400 shrink-0" />
   363→                    )}
   364→                    <div className="flex-1 min-w-0">
   365→                      <div className="text-sm font-medium">{record.type} Export</div>
   366→                      <div className="text-[10px] text-muted-foreground">
   367→                        {record.format} · {record.rows} rows · {record.createdAt}
   368→                        {record.size && ` · ${record.size}`}
   369→                      </div>
   370→                    </div>
   371→                    <Badge variant="outline" className={cn(
   372→                      'text-[9px] shrink-0',
   373→                      record.status === 'completed' ? 'text-primary border-primary/30' : 'text-red-400 border-red-400/30'
   374→                    )}>
   375→                      {record.status}
   376→                    </Badge>
   377→                    {record.status === 'completed' && (
   378→                      <Button variant="ghost" size="icon" className="h-7 w-7 shrink-0" onClick={() => handleShareReport(record)}>
   379→                        <Share2 className="size-3" />
   380→                      </Button>
   381→                    )}
   382→                  </div>
   383→                ))}
   384→              </div>
   385→            </CardContent>
   386→          </Card>
   387→        </TabsContent>
   388→
   389→        {/* Chart Export */}
   390→        <TabsContent value="charts">
   391→          <Card className="glass-card">
   392→            <CardHeader className="pb-2">
   393→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   394→                <FileDown className="size-4 text-primary" /> Data Visualization Export
   395→              </CardTitle>
   396→            </CardHeader>
   397→            <CardContent className="space-y-4">
   398→              <p className="text-sm text-muted-foreground">
   399→                Export any chart or visualization as a high-quality image using your browser&apos;s built-in capabilities.
   400→              </p>
   401→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   402→                {[
   403→                  { name: 'Match xG Chart', view: 'tactical' },
   404→                  { name: 'Player Radar Chart', view: 'players' },
   405→                  { name: 'Team Comparison', view: 'compare' },
   406→                  { name: 'Leaderboard Chart', view: 'leaderboard' },
   407→                  { name: 'Tournament Standings', view: 'tournament' },
   408→                  { name: 'Prediction Accuracy', view: 'predictions' },
   409→                ].map(chart => (
   410→                  <div
   411→                    key={chart.name}
   412→                    className="glass-card p-4 rounded-lg border border-border/30 cursor-pointer transition-all duration-200 hover:-translate-y-1"
   413→                    onClick={() => toast.info(`Navigate to ${chart.name} view, then use Ctrl+P to save as PDF`)}
   414→                  >
   415→                    <BarChartIcon className="size-5 text-primary mb-2" />
   416→                    <h4 className="text-sm font-medium">{chart.name}</h4>
   417→                    <p className="text-[10px] text-muted-foreground mt-1">Click to navigate, then print to save</p>
   418→                  </div>
   419→                ))}
   420→              </div>
   421→              <Button variant="outline" onClick={handleExportChartAsPNG} className="gap-2">
   422→                <FileDown className="size-4" /> Export Current View as PDF
   423→              </Button>
   424→            </CardContent>
   425→          </Card>
   426→        </TabsContent>
   427→
   428→        {/* API Access */}
   429→        <TabsContent value="api">
   430→          <Card className="glass-card">
   431→            <CardHeader className="pb-2">
   432→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   433→                <Code2 className="size-4 text-primary" /> API Documentation
   434→              </CardTitle>
   435→              <p className="text-xs text-muted-foreground">
   436→                Programmatic access to ELASTICO data. Use Bearer token authentication for protected endpoints.
   437→              </p>
   438→            </CardHeader>
   439→            <CardContent>
   440→              {/* Auth info */}
   441→              <div className="glass-card p-3 rounded-lg border border-border/30 mb-4">
   442→                <div className="text-xs font-medium mb-1">Authentication</div>
   443→                <code className="text-[11px] text-primary bg-primary/5 px-2 py-1 rounded block overflow-x-auto">
   444→                  Authorization: Bearer {'{'}token{'}'}
   445→                </code>
   446→                <div className="text-[10px] text-muted-foreground mt-1">Your API token is the same as your session token.</div>
   447→              </div>
   448→
   449→              <div className="space-y-2 max-h-96 overflow-y-auto">
   450→                {API_ENDPOINTS.map((endpoint) => (
   451→                  <div
   452→                    key={endpoint.path + endpoint.method}
   453→                    className="flex items-center gap-3 p-3 rounded-lg bg-card/50 border border-border/30 group cursor-pointer"
   454→                    onClick={() => handleCopyEndpoint(endpoint.path)}
   455→                  >
   456→                    <Badge variant="outline" className={cn(
   457→                      'text-[10px] font-mono shrink-0',
   458→                      endpoint.method === 'GET' ? 'text-emerald-400 border-emerald-400/30' : 'text-orange-400 border-orange-400/30'
   459→                    )}>
   460→                      {endpoint.method}
   461→                    </Badge>
   462→                    <code className="text-xs font-mono text-foreground/80 flex-1 truncate">{endpoint.path}</code>
   463→                    <span className="text-[10px] text-muted-foreground hidden md:block max-w-48 truncate">{endpoint.description}</span>
   464→                    <Copy className="size-3 text-muted-foreground opacity-0 group-hover:opacity-100 transition-opacity shrink-0" />
   465→                  </div>
   466→                ))}
   467→              </div>
   468→            </CardContent>
   469→          </Card>
   470→        </TabsContent>
   471→
   472→        {/* Bulk Export */}
   473→        <TabsContent value="bulk">
   474→          <Card className="glass-card">
   475→            <CardHeader className="pb-2">
   476→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   477→                <Database className="size-4 text-primary" /> Bulk Data Export
   478→              </CardTitle>
   479→            </CardHeader>
   480→            <CardContent className="space-y-4">
   481→              {user?.role === 'admin' ? (
