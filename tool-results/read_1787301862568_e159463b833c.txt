     3→   142→class SuperEnsemble:
     4→   143→    """Combines all prediction models into a calibrated super-ensemble.
     5→   144→
     6→   145→    Weighted average with:
     7→   146→    1. Per-model calibration factors (based on recent accuracy)
     8→   147→    2. Market signal integration (odds movement detection)
     9→   148→    3. Kelly Criterion for bankroll management
    10→   149→    4. Auto-weight optimisation when enough historical data
    11→   150→
    12→   151→    Falls back gracefully when ML models aren't trained.
    13→   152→    """
    14→   153→
    15→   154→    def __init__(self, config: EnsembleConfig | None = None) -> None:
    16→   155→        self.config = config or EnsembleConfig()
    17→   156→        self.model_dir = Path(self.config.model_dir)
    18→   157→
    19→   158→        # Lazy-loaded ML models
    20→   159→        self._xgboost_engine = None
    21→   160→        self._lstm_engine = None
    22→   161→        self._xgboost_loaded = False
    23→   162→        self._lstm_loaded = False
    24→   163→
    25→   164→    # ==================================================================
    26→   165→    # Public API
    27→   166→    # ==================================================================
    28→   167→
    29→   168→    def predict(self, match_data: dict) -> SuperEnsembleResult:
    30→   169→        """Run full super-ensemble prediction.
    31→   170→
    32→   171→        match_data should contain:
    33→   172→        - home_team, away_team (names)
    34→   173→        - home_elo, away_elo
    35→   174→        - home_avg_goals, away_avg_goals
    36→   175→        - home_avg_conceded, away_avg_conceded
    37→   176→        - home_xg, away_xg (optional, defaults to avg_goals)
    38→   177→        - odds_home, odds_draw, odds_away (optional, for market analysis)
    39→   178→        - home_form, away_form (list of recent results, optional)
    40→   179→        - league (optional)
    41→   180→        - match_id (optional)
    42→   181→
    43→   182→        Returns
    44→   183→        -------
    45→   184→        SuperEnsembleResult
    46→   185→            Comprehensive prediction result with probabilities,
    47→   186→            markets, Kelly analysis, and signals.
    48→   187→        """
    49→   188→        match_id = match_data.get("match_id")
    50→   189→        home_team = match_data.get("home_team", "Home")
    51→   190→        away_team = match_data.get("away_team", "Away")
    52→   191→
    53→   192→        home_elo = float(match_data.get("home_elo", 1500.0))
    54→   193→        away_elo = float(match_data.get("away_elo", 1500.0))
    55→   194→        home_avg_goals = float(match_data.get("home_avg_goals", 1.3))
    56→   195→        away_avg_goals = float(match_data.get("away_avg_goals", 1.1))
    57→   196→        home_avg_conceded = float(match_data.get("home_avg_conceded", 1.0))
    58→   197→        away_avg_conceded = float(match_data.get("away_avg_conceded", 1.2))
    59→   198→        home_xg = float(match_data.get("home_xg", home_avg_goals))
    60→   199→        away_xg = float(match_data.get("away_xg", away_avg_goals))
    61→   200→        odds_home = match_data.get("odds_home")
    62→   201→        odds_draw = match_data.get("odds_draw")
    63→   202→        odds_away = match_data.get("odds_away")
    64→   203→
    65→   204→        # Compute expected goals for Poisson-based models
    66→   205→        lambda_h = (home_avg_goals + away_avg_conceded) / 2.0
    67→   206→        lambda_a = (away_avg_goals + home_avg_conceded) / 2.0
    68→   207→
    69→   208→        per_model: dict[str, dict] = {}
    70→   209→        available_models: list[str] = []
    71→   210→
    72→   211→        # ── 1. ELO Model ────────────────────────────────────────────────
    73→   212→        try:
    74→   213→            elo_probs = compute_elo_prob(home_elo, away_elo)
    75→   214→            per_model["elo"] = {
    76→   215→                "home_win": elo_probs["home_win"],
    77→   216→                "draw": elo_probs["draw"],
    78→   217→                "away_win": elo_probs["away_win"],
    79→   218→            }
    80→   219→            available_models.append("elo")
    81→   220→        except Exception as exc:
    82→   221→            logger.warning("ELO model failed: %s", exc)
    83→   222→
    84→   223→        # ── 2. Poisson Model ────────────────────────────────────────────
    85→   224→        try:
    86→   225→            poisson_probs = compute_poisson_prob(lambda_h, lambda_a)
    87→   226→            per_model["poisson"] = {
    88→   227→                "home_win": poisson_probs["home_win"],
    89→   228→                "draw": poisson_probs["draw"],
    90→   229→                "away_win": poisson_probs["away_win"],
    91→   230→            }
    92→   231→            available_models.append("poisson")
    93→   232→        except Exception as exc:
    94→   233→            logger.warning("Poisson model failed: %s", exc)
    95→   234→
    96→   235→        # ── 3. Dixon-Coles Model ────────────────────────────────────────
    97→   236→        try:
    98→   237→            dc_probs = compute_dixon_coles_prob(lambda_h, lambda_a)
    99→   238→            per_model["dixon_coles"] = {
   100→   239→                "home_win": dc_probs["home_win"],
   101→   240→                "draw": dc_probs["draw"],
   102→   241→                "away_win": dc_probs["away_win"],
   103→   242→            }
   104→   243→            available_models.append("dixon_coles")
   105→   244→        except Exception as exc:
   106→   245→            logger.warning("Dixon-Coles model failed: %s", exc)
   107→   246→
   108→   247→        # ── 4. Stochastic (Merton + GARCH Monte Carlo) ──────────────────
   109→   248→        stochastic_result: StochasticResult | None = None
   110→   249→        try:
   111→   250→            stochastic_result = run_stochastic_simulation(
   112→   251→                home_xg=home_xg,
   113→   252→                away_xg=away_xg,
   114→   253→                home_conceded=home_avg_conceded,
   115→   254→                away_conceded=away_avg_conceded,
   116→   255→                home_elo=home_elo,
   117→   256→                away_elo=away_elo,
   118→   257→                config=EngineConfig(num_simulations=50_000),
   119→   258→            )
   120→   259→            per_model["stochastic"] = {
   121→   260→                "home_win": stochastic_result.match_probabilities["home_victory"],
   122→   261→                "draw": stochastic_result.match_probabilities["draw"],
   123→   262→                "away_win": stochastic_result.match_probabilities["away_victory"],
   124→   263→            }
   125→   264→            available_models.append("stochastic")
   126→   265→        except Exception as exc:
   127→   266→            logger.warning("Stochastic model failed: %s", exc)
   128→   267→
   129→   268→        # ── 5. XGBoost ───────────────────────────────────────────────────
   130→   269→        try:
   131→   270→            xgb_probs = self._run_xgboost(match_data)
   132→   271→            if xgb_probs:
   133→   272→                per_model["xgboost"] = xgb_probs
   134→   273→                available_models.append("xgboost")
   135→   274→        except Exception as exc:
   136→   275→            logger.debug("XGBoost not available: %s", exc)
   137→   276→
   138→   277→        # ── 6. LSTM ─────────────────────────────────────────────────────
   139→   278→        try:
   140→   279→            lstm_probs = self._run_lstm(match_data)
   141→   280→            if lstm_probs:
   142→   281→                per_model["lstm"] = lstm_probs
   143→   282→                available_models.append("lstm")
   144→   283→        except Exception as exc:
   145→   284→            logger.debug("LSTM not available: %s", exc)
   146→   285→
   147→   286→        # ── 7. Weighted Ensemble ────────────────────────────────────────
   148→   287→        weights = self._get_model_weights(available_models)
   149→   288→        final_probs = self._compute_weighted_average(per_model, weights)
   150→   289→
   151→   290→        # ── 8. Market Signals ───────────────────────────────────────────
   152→   291→        market_signals: list[dict] = []
   153→   292→        if odds_home and odds_draw and odds_away:
   154→   293→            try:
   155→   294→                # Use current odds as both opening and current when we
   156→   295→                # don't have opening odds — signals will be empty, but
   157→   296→                # we can still compute Kelly
   158→   297→                opening_odds = match_data.get("opening_odds_home")
   159→   298→                if opening_odds and match_data.get("opening_odds_draw"):
   160→   299→                    opening = {
   161→   300→                        "home": float(match_data["opening_odds_home"]),
   162→   301→                        "draw": float(match_data["opening_odds_draw"]),
   163→   302→                        "away": float(match_data["opening_odds_away"]),
   164→   303→                    }
   165→   304→                    current = {
   166→   305→                        "home": float(odds_home),
   167→   306→                        "draw": float(odds_draw),
   168→   307→                        "away": float(odds_away),
   169→   308→                    }
   170→   309→                    detected = analyze_market_signals(opening, current)
   171→   310→                    for sig in detected:
   172→   311→                        market_signals.append(sig.model_dump())
   173→   312→            except Exception as exc:
   174→   313→                logger.debug("Market signal analysis failed: %s", exc)
   175→   314→
   176→   315→        # ── 9. Kelly Criterion ──────────────────────────────────────────
   177→   316→        kelly_results: dict[str, dict] = {}
   178→   317→        if odds_home and odds_draw and odds_away:
   179→   318→            kelly_results["home_win"] = calculate_kelly(
   180→   319→                final_probs["home_win"],
   181→   320→                float(odds_home),
   182→   321→                bankroll=self.config.bankroll,
   183→   322→                fraction=self.config.kelly_fraction,
   184→   323→            ).model_dump()
   185→   324→            kelly_results["draw"] = calculate_kelly(
   186→   325→                final_probs["draw"],
   187→   326→                float(odds_draw),
   188→   327→                bankroll=self.config.bankroll,
   189→   328→                fraction=self.config.kelly_fraction,
   190→   329→            ).model_dump()
   191→   330→            kelly_results["away_win"] = calculate_kelly(
   192→   331→                final_probs["away_win"],
   193→   332→                float(odds_away),
   194→   333→                bankroll=self.config.bankroll,
   195→   334→                fraction=self.config.kelly_fraction,
   196→   335→            ).model_dump()
   197→   336→
   198→   337→        # ── 10. Signals & Risks ────────────────────────────────────────
   199→   338→        signals, risks = self._generate_signals_and_risks(
   200→   339→            match_data, final_probs
   201→   340→        )
   202→   341→
   203→   342→        # Add market signal descriptions to signals list
   204→   343→        for sig in market_signals:
   205→   344→            signals.append(f"[{sig.get('signal_type', '').upper()}] {sig.get('description', '')}")
   206→   345→
   207→   346→        # ── 11. Markets (derived) ──────────────────────────────────────
   208→   347→        markets: dict = {}
   209→   348→        if stochastic_result:
   210→   349→            markets["over_25"] = stochastic_result.totals_market.get("over25", 0.0)
   211→   350→            markets["over_15"] = stochastic_result.totals_market.get("over15", 0.0)
   212→   351→            markets["over_35"] = stochastic_result.totals_market.get("over35", 0.0)
   213→   352→            markets["btts"] = stochastic_result.both_teams_to_score
   214→   353→            markets["asian_handicap"] = stochastic_result.asian_handicap
   215→   354→        else:
   216→   355→            # Fallback: estimate from Poisson
   217→   356→            from scipy.stats import poisson as poisson_dist
   218→   357→            over_25 = 0.0
   219→   358→            btts = 0.0
   220→   359→            for i in range(11):
   221→   360→                for j in range(11):
   222→   361→                    p = poisson_dist.pmf(i, lambda_h) * poisson_dist.pmf(j, lambda_a)
   223→   362→                    if i + j > 2:
   224→   363→                        over_25 += p
   225→   364→                    if i > 0 and j > 0:
   226→   365→                        btts += p
   227→   366→            markets["over_25"] = round(over_25, 4)
   228→   367→            markets["over_15"] = 0.0
   229→   368→            markets["over_35"] = 0.0
   230→   369→            markets["btts"] = round(btts, 4)
   231→   370→            markets["asian_handicap"] = {
   232→   371→                "0": round(final_probs["home_win"], 4),
   233→   372→                "-0.5": round(final_probs["home_win"], 4),
   234→   373→                "-1.0": 0.0,
   235→   374→                "-1.5": 0.0,
   236→   375→            }
   237→   376→
   238→   377→        # ── 12. Recommended Bet ────────────────────────────────────────
   239→   378→        recommended_bet = None
   240→   379→        if odds_home and odds_draw and odds_away:
   241→   380→            best_edge = -1.0
   242→   381→            for outcome_key, odds_val in [
   243→   382→                ("home_win", odds_home),
   244→   383→                ("draw", odds_draw),
   245→   384→                ("away_win", odds_away),
   246→   385→            ]:
   247→   386→                kr = kelly_results.get(outcome_key, {})
   248→   387→                edge = kr.get("edge", 0.0)
   249→   388→                kelly_frac = kr.get("optimal_fraction", 0.0)
   250→   389→                if edge > best_edge and kelly_frac > 0.001:
   251→   390→                    best_edge = edge
   252→   391→                    recommended_bet = {
   253→   392→                        "outcome": outcome_key,
   254→   393→                        "edge": round(edge, 4),
   255→   394→                        "kelly_fraction": round(kelly_frac, 6),
   256→   395→                        "odds": float(odds_val),
   257→   396→                        "recommended_stake": kr.get("recommended_stake", 0.0),
   258→   397→                    }
   259→   398→
   260→   399→        # ── 13. Expected Goals ──────────────────────────────────────────
   261→   400→        if stochastic_result:
   262→   401→            expected_goals = {
   263→   402→                "home": stochastic_result.expected_means["home"],
   264→   403→                "away": stochastic_result.expected_means["away"],
   265→   404→                "total": stochastic_result.expected_means["total"],
   266→   405→            }
   267→   406→            volatility_index = stochastic_result.volatility_index
   268→   407→        else:
   269→   408→            expected_goals = {
   270→   409→                "home": round(lambda_h, 3),
   271→   410→                "away": round(lambda_a, 3),
   272→   411→                "total": round(lambda_h + lambda_a, 3),
   273→   412→            }
   274→   413→            volatility_index = 0.0
   275→   414→
   276→   415→        # ── 14. Confidence ──────────────────────────────────────────────
   277→   416→        confidence = self._compute_confidence(final_probs)
   278→   417→
   279→   418→        return SuperEnsembleResult(
   280→   419→            match_id=match_id,
   281→   420→            home_team=home_team,
   282→   421→            away_team=away_team,
   283→   422→            probabilities=final_probs,
   284→   423→            per_model=per_model,
   285→   424→            expected_goals=expected_goals,
   286→   425→            markets=markets,
   287→   426→            kelly=kelly_results,
   288→   427→            market_signals=market_signals,
   289→   428→            confidence=confidence,
   290→   429→            signals=signals,
   291→   430→            risks=risks,
   292→   431→            recommended_bet=recommended_bet,
   293→   432→            volatility_index=volatility_index,
   294→   433→        )

   296→   435→    def predict_batch(self, matches: list[dict]) -> list[SuperEnsembleResult]:
   297→   436→        """Predict for multiple matches.
   298→   437→
   299→   438→        Parameters
   300→   439→        ----------

   303→   442→
   304→   443→        Returns
   305→   444→        -------

   319→   458→                    exc,
   320→   459→                )
   321→   460→                # Insert a placeholder with error info
   322→   461→                results.append(
   323→   462→                    SuperEnsembleResult(
   324→   463→                        match_id=match.get("match_id"),
   325→   464→                        home_team=match.get("home_team"),

   343→   482→    def _get_model_weights(self, available_models: list[str]) -> dict[str, float]:
   344→   483→        """Get normalized weights for available models only.
   345→   484→
   346→   485→        If some models are unavailable, their weight is redistributed
   347→   486→        proportionally among the remaining models.
   348→   487→
   349→   488→        Parameters
   350→   489→        ----------
   351→   490→        available_models : list[str]
   352→   491→            List of model names that produced valid results.
   353→   492→
   354→   493→        Returns
   355→   494→        -------
   356→   495→        dict[str, float]
   357→   496→            Normalized weights summing to 1.0, keyed by model name.
   358→   497→        """
   359→   498→        base_weights = self.config.weights
   360→   499→
   361→   500→        # Filter to available models
   362→   501→        filtered = {
   363→   502→            name: base_weights[name]
   364→   503→            for name in available_models
   365→   504→            if name in base_weights
   366→   505→        }
   367→   506→
   368→   507→        if not filtered:
   369→   508→            # Fallback: equal weight for all available
   370→   509→            n = len(available_models)
   371→   510→            return {name: 1.0 / n for name in available_models} if n > 0 else {}
   372→   511→
   373→   512→        # Normalize to sum to 1.0
   374→   513→        total = sum(filtered.values())
   375→   514→        return {name: w / total for name, w in filtered.items()}

   377→   516→    def _compute_weighted_average(
   378→   517→        self,
   379→   518→        probs: dict[str, dict],
   380→   519→        weights: dict[str, float],
   381→   520→    ) -> dict[str, float]:
   382→   521→        """Weighted average of model probabilities.
   383→   522→
   384→   523→        Parameters
   385→   524→        ----------
   386→   525→        probs : dict[str, dict]
   387→   526→            Per-model probability dicts: {"elo": {"home_win": 0.5, ...}, ...}
   388→   527→        weights : dict[str, float]
   389→   528→            Normalized weights per model.
   390→   529→
   391→   530→        Returns
   392→   531→        -------
   393→   532→        dict[str, float]
   394→   533→            Combined {"home_win": ..., "draw": ..., "away_win": ...}.
   395→   534→        """
   396→   535→        if not probs or not weights:
   397→   536→            return {"home_win": 1/3, "draw": 1/3, "away_win": 1/3}
   398→   537→
   399→   538→        combined = {"home_win": 0.0, "draw": 0.0, "away_win": 0.0}
   400→   539→
   401→   540→        for model_name, model_probs in probs.items():
   402→   541→            w = weights.get(model_name, 0.0)
   403→   542→            if w <= 0:
   404→   543→                continue
   405→   544→            for key in ("home_win", "draw", "away_win"):
   406→   545→                combined[key] += w * model_probs.get(key, 0.0)
   407→   546→
   408→   547→        # Normalize to exactly 1.0
   409→   548→        total = sum(combined.values())
   410→   549→        if total > 0:
   411→   550→            combined = {k: round(v / total, 6) for k, v in combined.items()}
   412→   551→
   413→   552→        return combined

   419→   558→    def _compute_confidence(self, probs: dict) -> str:
   420→   559→        """Classify prediction confidence based on max probability.
   421→   560→
   422→   561→        - high: max probability >= 0.55
   423→   562→        - medium: max probability >= 0.42
   424→   563→        - low: otherwise
   425→   564→        """
   426→   565→        max_prob = max(probs.get("home_win", 0), probs.get("draw", 0), probs.get("away_win", 0))
   427→   566→        if max_prob >= 0.55:
   428→   567→            return "high"
   429→   568→        elif max_prob >= 0.42:
   430→   569→            return "medium"
   431→   570→        return "low"

   433→   572→    def _generate_signals_and_risks(
   434→   573→        self,
   435→   574→        match_data: dict,
   436→   575→        probs: dict,
   437→   576→    ) -> tuple[list[str], list[str]]:
   438→   577→        """Generate human-readable signals and risk factors.
   439→   578→
   440→   579→        Parameters
   441→   580→        ----------
   442→   581→        match_data : dict
   443→   582→            Original match data dict.
   444→   583→        probs : dict
   445→   584→            Final ensemble probabilities.
   446→   585→
   447→   586→        Returns
   448→   587→        -------
   449→   588→        tuple[list[str], list[str]]
   450→   589→            (signals, risks) lists.
   451→   590→        """
   452→   591→        signals: list[str] = []
   453→   592→        risks: list[str] = []
   454→   593→
   455→   594→        home_elo = float(match_data.get("home_elo", 1500))
   456→   595→        away_elo = float(match_data.get("away_elo", 1500))
   457→   596→        elo_diff = home_elo - away_elo
   458→   597→        home_xg = float(match_data.get("home_xg", match_data.get("home_avg_goals", 1.3)))
   459→   598→        away_xg = float(match_data.get("away_xg", match_data.get("away_avg_goals", 1.1)))
   460→   599→        odds_home = match_data.get("odds_home")
   461→   600→
   462→   601→        # ELO signals
   463→   602→        if elo_diff > 150:
   464→   603→            signals.append(f"Strong ELO advantage for {match_data.get('home_team', 'home')} (+{elo_diff:.0f})")
   465→   604→        elif elo_diff < -150:
   466→   605→            signals.append(f"Strong ELO advantage for {match_data.get('away_team', 'away')} (+{abs(elo_diff):.0f})")
   467→   606→        elif abs(elo_diff) > 80:
   468→   607→            team = "home" if elo_diff > 0 else "away"
   469→   608→            signals.append(f"Moderate ELO edge for {team} team")
   470→   609→
   471→   610→        # xG signals
   472→   611→        if home_xg > away_xg * 1.3:
   473→   612→            signals.append(f"Home xG ({home_xg:.2f}) significantly higher than away ({away_xg:.2f})")
   474→   613→        elif away_xg > home_xg * 1.3:
   475→   614→            signals.append(f"Away xG ({away_xg:.2f}) significantly higher than home ({home_xg:.2f})")
   476→   615→
   477→   616→        # Form signals
   478→   617→        home_form = match_data.get("home_form", [])
   479→   618→        away_form = match_data.get("away_form", [])
   480→   619→        if len(home_form) >= 3:
   481→   620→            recent_wins = sum(1 for r in home_form[-3:] if r == "W")
   482→   621→            if recent_wins == 3:
   483→   622→                signals.append(f"{match_data.get('home_team', 'Home')} on a 3-match winning streak")
   484→   623→        if len(away_form) >= 3:
   485→   624→            recent_wins = sum(1 for r in away_form[-3:] if r == "W")
   486→   625→            if recent_wins == 3:
   487→   626→                signals.append(f"{match_data.get('away_team', 'Away')} on a 3-match winning streak")
   488→   627→
   489→   628→        # Model agreement signal
   490→   629→        max_prob = max(probs["home_win"], probs["draw"], probs["away_win"])
   491→   630→        if max_prob >= 0.55:
   492→   631→            signals.append(f"High model agreement ({max_prob*100:.1f}% on most likely outcome)")
   493→   632→
   494→   633→        # Risks
   495→   634→        if odds_home is None:
   496→   635→            risks.append("No market odds available — Kelly criterion and market signals excluded")
   497→   636→
   498→   637→        if odds_home and float(odds_home) < 1.3:
   499→   638→            risks.append("Heavy favorite — low value in outcome market")
   500→   639→
   501→   640→        if abs(elo_diff) < 30:
   502→   641→            risks.append("Teams are very evenly matched on ELO — high draw probability")
   503→   642→
   504→   643→        # Check if any model is missing
   505→   644→        expected_models = {"elo", "poisson", "dixon_coles", "stochastic"}
   506→   645→        # We don't risk-flag missing xgboost/lstm since they're optional
   507→   646→
   508→   647→        # Derive risk from model disagreement
   509→   648→        if probs["draw"] > 0.32:
   510→   649→            risks.append("High draw probability — consider Double Chance market")
   511→   650→
   512→   651→        return signals, risks

   518→   657→    def _run_xgboost(self, match_data: dict) -> dict[str, float] | None:
   519→   658→        """Try to load and run XGBoost model.
   520→   659→
   521→   660→        Returns home_win/draw/away_win probabilities or None if
   522→   661→        the model is not available.
   523→   662→        """
   524→   663→        if not self._xgboost_loaded:
   525→   664→            try:
   526→   665→                from src.ml.xgboost_engine import XGBoostEngine
   527→   666→
   528→   667→                self._xgboost_engine = XGBoostEngine(
   529→   668→                    model_dir=str(self.model_dir),
   530→   669→                )
   531→   670→                loaded = self._xgboost_engine.load("xgboost_v1")
   532→   671→                self._xgboost_loaded = True  # mark as attempted regardless
   533→   672→                if not loaded:
   534→   673→                    logger.info("XGBoost model not found — skipping")
   535→   674→                    return None
   536→   675→            except Exception as exc:
   537→   676→                self._xgboost_loaded = True
   538→   677→                logger.debug("XGBoost load failed: %s", exc)
   539→   678→                return None
   540→   679→
   541→   680→        if self._xgboost_engine is None or not self._xgboost_engine.is_trained:
   542→   681→            return None
   543→   682→
   544→   683→        # Build a feature vector from match_data
   545→   684→        # The XGBoost model expects the same feature order as training
   546→   685→        # We construct a basic feature vector; if the model was trained with
   547→   686→        # the full feature engineer, it will need the full 50-feature vector.
   548→   687→        # For the ensemble, we use a simplified 6-feature approximation.
   549→   688→        from src.ml.feature_schema import FEATURE_NAMES, FEATURE_DEFAULTS
   550→   689→
   551→   690→        # Build feature vector using canonical feature schema
   552→   691→        home_avg_goals = float(match_data.get("home_avg_goals", 1.3))
   553→   692→        away_avg_goals = float(match_data.get("away_avg_goals", 1.1))
   554→   693→        home_avg_conceded = float(match_data.get("home_avg_conceded", 1.0))
   555→   694→        away_avg_conceded = float(match_data.get("away_avg_conceded", 1.2))
   556→   695→        home_elo = float(match_data.get("home_elo", 1500))
   557→   696→        away_elo = float(match_data.get("away_elo", 1500))
   558→   697→
   559→   698→        # Build feature vector using canonical feature schema
   560→   699→        from src.ml.feature_schema import FEATURE_NAMES, FEATURE_DEFAULTS
   561→   700→
   562→   701→        feature_dict = dict(FEATURE_DEFAULTS)  # start with all defaults
   563→   702→
   564→   703→        # Override with available match data (using CANONICAL names)
   565→   704→        feature_dict["elo_home"] = home_elo
   566→   705→        feature_dict["elo_away"] = away_elo
   567→   706→        feature_dict["elo_diff"] = home_elo - away_elo
   568→   707→        feature_dict["elo_home_advantage_adjusted"] = home_elo + 65.0  # standard home advantage
   569→   708→        feature_dict["avg_goals_scored_home"] = home_avg_goals
   570→   709→        feature_dict["avg_goals_scored_away"] = away_avg_goals
   571→   710→        feature_dict["avg_goals_conceded_home"] = home_avg_conceded
   572→   711→        feature_dict["avg_goals_conceded_away"] = away_avg_conceded
   573→   712→        feature_dict["avg_goals_scored"] = home_avg_goals
   574→   713→        feature_dict["avg_goals_conceded"] = home_avg_conceded
   575→   714→        feature_dict["poisson_lambda_home"] = (home_avg_goals + away_avg_conceded) / 2.0
   576→   715→        feature_dict["poisson_lambda_away"] = (away_avg_goals + home_avg_conceded) / 2.0
   577→   716→
   578→   717→        # Handle optional xG data
   579→   718→        home_xg = match_data.get("home_xg")
   580→   719→        away_xg = match_data.get("away_xg")
   581→   720→        if home_xg is not None:
   582→   721→            feature_dict["xg_for_home"] = float(home_xg)
   583→   722→        if away_xg is not None:
   584→   723→            feature_dict["xg_for_away"] = float(away_xg)
   585→   724→
   586→   725→        # Handle optional odds data for market features
   587→   726→        odds_home = match_data.get("odds_home")
   588→   727→        odds_draw = match_data.get("odds_draw")
   589→   728→        odds_away = match_data.get("odds_away")
   590→   729→        if odds_home and odds_draw and odds_away:
   591→   730→            total_implied = 1.0/float(odds_home) + 1.0/float(odds_draw) + 1.0/float(odds_away)
   592→   731→            feature_dict["implied_prob_home"] = (1.0/float(odds_home)) / total_implied
   593→   732→            feature_dict["implied_prob_draw"] = (1.0/float(odds_draw)) / total_implied
   594→   733→            feature_dict["implied_prob_away"] = (1.0/float(odds_away)) / total_implied
   595→   734→            feature_dict["overround"] = total_implied - 1.0
   596→   735→
   597→   736→        # Order features correctly
   598→   737→        feature_vector = np.array(
   599→   738→            [feature_dict[fname] for fname in FEATURE_NAMES],
   600→   739→            dtype=np.float32,
   601→   740→        )
   602→   741→
   603→   742→        try:
   604→   743→            pred = self._xgboost_engine.predict(
   605→   744→                feature_vector, feature_names=FEATURE_NAMES
   606→   745→            )
   607→   746→            return {
   608→   747→                "home_win": pred.home_win_prob,
   609→   748→                "draw": pred.draw_prob,
   610→   749→                "away_win": pred.away_win_prob,
   611→   750→            }
   612→   751→        except Exception as exc:
   613→   752→            logger.warning("XGBoost prediction failed: %s", exc)
   614→   753→            return None

   616→   755→    def _run_lstm(self, match_data: dict) -> dict[str, float] | None:
   617→   756→        """Try to load and run LSTM model.
   618→   757→
   619→   758→        Returns home_win/draw/away_win probabilities or None.
   620→   759→
   621→   760→        CRITICAL: The LSTM is a SEQUENCE model.  It requires genuine temporal
   622→   761→        match histories — not a single feature vector repeated N times.
   623→   762→        If real sequences are not available, this method returns None
   624→   763→        rather than producing meaningless output from padded data.
   625→   764→        """
   626→   765→        if not self._lstm_loaded:
   627→   766→            try:
   628→   767→                from src.ml.lstm_engine import LSTMEngine
   629→   768→
   630→   769→                self._lstm_engine = LSTMEngine(
   631→   770→                    model_dir=str(self.model_dir),
   632→   771→                )
   633→   772→                loaded = self._lstm_engine.load("lstm_v1")
   634→   773→                self._lstm_loaded = True  # mark as attempted regardless
   635→   774→                if not loaded:
   636→   775→                    logger.info("LSTM model not found — skipping")
   637→   776→                    return None
   638→   777→            except Exception as exc:
   639→   778→                self._lstm_loaded = True
   640→   779→                logger.debug("LSTM load failed: %s", exc)
   641→   780→                return None
   642→   781→
   643→   782→        if self._lstm_engine is None or not self._lstm_engine.is_trained:
   644→   783→            return None
   645→   784→
   646→   785→        # ── HONEST BEHAVIOR: require real sequences ─────────────────────
   647→   786→        # The LSTM model requires (seq_len, input_size) tensors built from
   648→   787→        # actual past match histories.  Repeating a single feature vector
   649→   788→        # across timesteps is NOT a valid sequence — it destroys the
   650→   789→        # temporal structure the model was trained on.
   651→   790→        #
   652→   791→        # Real sequences must be provided via match_data["home_sequence"]
   653→   792→        # and match_data["away_sequence"] (each a list of dicts or a
   654→   793→        # 2-D numpy array of shape (seq_len, n_features)).
   655→   794→        #
   656→   795→        # If these are not present, we honestly return None.
   657→   796→        home_seq = match_data.get("home_sequence")
   658→   797→        away_seq = match_data.get("away_sequence")
   659→   798→
   660→   799→        if home_seq is None or away_seq is None:
   661→   800→            logger.debug(
   662→   801→                "LSTM: no real sequences in match_data — honestly skipping. "
   663→   802→                "Pass home_sequence and away_sequence (each a list of dicts "
   664→   803→                "or np.ndarray of shape (seq_len, n_features))."
   665→   804→            )
   666→   805→            return None
   667→   806→
   668→   807→        try:
   669→   808→            import numpy as np
   670→   809→
   671→   810→            X_home = np.asarray(home_seq, dtype=np.float32)
   672→   811→            X_away = np.asarray(away_seq, dtype=np.float32)
   673→   812→
   674→   813→            # Ensure 3-D: (1, seq_len, n_features)
   675→   814→            if X_home.ndim == 2:
   676→   815→                X_home = X_home[np.newaxis, ...]
   677→   816→            if X_away.ndim == 2:
   678→   817→                X_away = X_away[np.newaxis, ...]
   679→   818→
   680→   819→            # Validate shape matches what model expects
   681→   820→            expected_features = len(self._lstm_engine.feature_cols)
   682→   821→            if X_home.shape[2] != expected_features:
   683→   822→                logger.warning(
   684→   823→                    "LSTM: feature mismatch — model expects %d features, "
   685→   824→                    "got %d. Cannot produce valid prediction.",
   686→   825→                    expected_features, X_home.shape[2],
   687→   826→                )
   688→   827→                return None
   689→   828→
   690→   829→            result = self._lstm_engine.predict(X_home[0], X_away[0])
   691→   830→
   692→   831→            if isinstance(result, dict) and "batch" not in result:
   693→   832→                return {
   694→   833→                    "home_win": result["home_win"],
   695→   834→                    "draw": result["draw"],
   696→   835→                    "away_win": result["away_win"],
   697→   836→                }
   698→   837→
   699→   838→            return None
   700→   839→        except Exception as exc:
   701→   840→            logger.warning("LSTM prediction failed: %s", exc)
   702→   841→            return None
