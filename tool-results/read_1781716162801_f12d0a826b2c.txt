   112→// ── Stat Bar Row ─────────────────────────────────────────────────────────────
   113→
   114→function StatBarRow({
   115→  label,
   116→  homeValue,
   117→  awayValue,
   118→  homeColor,
   119→  awayColor,
   120→  suffix = '',
   121→  decimals = 0,
   122→}: {
   123→  label: string
   124→  homeValue: number
   125→  awayValue: number
   126→  homeColor?: string
   127→  awayColor?: string
   128→  suffix?: string
   129→  decimals?: number
   130→}) {
   131→  const total = homeValue + awayValue || 1
   132→  const homePercent = (homeValue / total) * 100
   133→  const awayPercent = (awayValue / total) * 100
   134→
   135→  const formatVal = (v: number) =>
   136→    decimals > 0 ? v.toFixed(decimals) : String(Math.round(v))
   137→
   138→  return (
   139→    <div className="space-y-1.5">
   140→      <div className="flex items-center justify-between text-xs">
   141→        <span className={cn('font-semibold tabular-nums', homePercent >= 50 ? 'text-foreground' : 'text-muted-foreground')}>
   142→          {formatVal(homeValue)}{suffix}
   143→        </span>
   144→        <span className="text-[10px] text-muted-foreground font-medium uppercase tracking-wider">{label}</span>
   145→        <span className={cn('font-semibold tabular-nums', awayPercent > homePercent ? 'text-foreground' : 'text-muted-foreground')}>
   146→          {formatVal(awayValue)}{suffix}
   147→        </span>
   148→      </div>
   149→      <div className="flex h-2 gap-0.5 rounded-full overflow-hidden bg-muted">
   150→        <div
   151→          className="h-full rounded-full transition-all duration-700 ease-out"
   152→          style={{
   153→            width: `${homePercent}%`,
   154→            backgroundColor: homeColor || 'oklch(0.72 0.19 163)',
   155→          }}
   156→        />
   157→        <div
   158→          className="h-full rounded-full transition-all duration-700 ease-out ml-auto"
   159→          style={{
   160→            width: `${awayPercent}%`,
   161→            backgroundColor: awayColor || 'oklch(0.68 0.15 230)',
   162→          }}
   163→        />
   164→      </div>
   165→    </div>
   166→  )
   167→}

   248→function getWeatherIcon(weather: string | null) {
   249→  if (!weather) return <Cloud className="size-3.5 text-muted-foreground" />
   250→  const w = weather.toLowerCase()
   251→  if (w.includes('sun') || w.includes('clear')) return <span className="text-sm">☀️</span>
   252→  if (w.includes('rain')) return <span className="text-sm">🌧️</span>
   253→  if (w.includes('cloud')) return <span className="text-sm">⛅</span>
   254→  if (w.includes('hot') || w.includes('warm')) return <span className="text-sm">🔥</span>
   255→  if (w.includes('cold') || w.includes('snow')) return <span className="text-sm">❄️</span>
   256→  return <Cloud className="size-3.5 text-muted-foreground" />
   257→}

   261→export function MatchDetailView() {
   262→  const { selectedMatchId, setView, user, token } = useElasticoStore()
   263→
   264→  const [match, setMatch] = useState<MatchDetail | null>(null)
   265→  const [loading, setLoading] = useState(true)
   266→  const [error, setError] = useState<string | null>(null)
   267→  const [activeTab, setActiveTab] = useState('statistics')
   268→
   269→  // AI Analysis
   270→  const [aiAnalysis, setAiAnalysis] = useState<string | null>(null)
   271→  const [aiLoading, setAiLoading] = useState(false)
   272→  const [aiError, setAiError] = useState<string | null>(null)
   273→
   274→  // Vote state
   275→  const [voting, setVoting] = useState(false)
   276→
   277→  // Simulate state
   278→  const [simulating, setSimulating] = useState(false)
   279→  const [simulateError, setSimulateError] = useState<string | null>(null)
   280→
   281→  // Bookmark state
   282→  const [bookmarked, setBookmarked] = useState(false)
   283→
   284→  // ── Fetch match detail ─────────────────────────────────────────────────
   285→  const fetchMatch = useCallback(async () => {
   286→    if (!selectedMatchId) return
   287→    setLoading(true)
   288→    setError(null)
   289→
   290→    try {
   291→      const headers: Record<string, string> = {}
   292→      if (token) headers['Authorization'] = `Bearer ${token}`
   293→
   294→      const res = await fetch(`/api/matches/${selectedMatchId}`, { headers })
   295→      if (!res.ok) {
   296→        throw new Error(`Failed to fetch match (status: ${res.status})`)
   297→      }
   298→      const data = await res.json()
   299→      setMatch(data.match)
   300→
   301→      // Check if user has bookmarked
   302→      if (user && data.match._count) {
   303→        setBookmarked(data.match._count.bookmarks > 0)
   304→      }
   305→    } catch (err) {
   306→      console.error('Match detail fetch error:', err)
   307→      setError(err instanceof Error ? err.message : 'Failed to load match')
   308→    } finally {
   309→      setLoading(false)
   310→    }
   311→  }, [selectedMatchId, token, user])
   312→
   313→  useEffect(() => {
   314→    fetchMatch()
   315→  }, [fetchMatch])
   316→
   317→  // ── Fetch AI Analysis ─────────────────────────────────────────────────
   318→  const fetchAiAnalysis = useCallback(async () => {
   319→    if (!selectedMatchId || !token) return
   320→    setAiLoading(true)
   321→    setAiError(null)
   322→
   323→    try {
   324→      const res = await fetch('/api/chat', {
   325→        method: 'POST',
   326→        headers: {
   327→          'Content-Type': 'application/json',
   328→          Authorization: `Bearer ${token}`,
   329→        },
   330→        body: JSON.stringify({
   331→          message: 'Analyze this match tactically',
   332→          matchId: selectedMatchId,
   333→        }),
   334→      })
   335→
   336→      if (!res.ok) throw new Error('Failed to get AI analysis')
   337→      const data = await res.json()
   338→      setAiAnalysis(data.response)
   339→    } catch (err) {
   340→      console.error('AI analysis error:', err)
   341→      setAiError('Could not load AI analysis. Try again later.')
   342→    } finally {
   343→      setAiLoading(false)
   344→    }
   345→  }, [selectedMatchId, token])
   346→
   347→  // ── Handle Vote ─────────────────────────────────────────────────────────
   348→  const handleVote = useCallback(
   349→    async (choice: 'home' | 'draw' | 'away') => {
   350→      if (!selectedMatchId || !token) return
   351→      setVoting(true)
   352→
   353→      try {
   354→        const res = await fetch(`/api/matches/${selectedMatchId}`, {
   355→          method: 'POST',
   356→          headers: {
   357→            'Content-Type': 'application/json',
   358→            Authorization: `Bearer ${token}`,
   359→          },
   360→          body: JSON.stringify({ choice }),
   361→        })
   362→
   363→        if (!res.ok) throw new Error('Failed to vote')
   364→        // Re-fetch match to update vote distribution
   365→        await fetchMatch()
   366→      } catch (err) {
   367→        console.error('Vote error:', err)
   368→      } finally {
   369→        setVoting(false)
   370→      }
   371→    },
   372→    [selectedMatchId, token, fetchMatch],
   373→  )
   374→
   375→  // ── Handle Simulate ────────────────────────────────────────────────────
   376→  const handleSimulate = useCallback(async () => {
   377→    if (!selectedMatchId || !token) return
   378→    setSimulating(true)
   379→    setSimulateError(null)
   380→
   381→    try {
   382→      const res = await fetch(`/api/matches/${selectedMatchId}/simulate`, {
   383→        method: 'POST',
   384→        headers: {
   385→          Authorization: `Bearer ${token}`,
   386→        },
   387→      })
   388→
   389→      if (!res.ok) {
   390→        const data = await res.json().catch(() => ({}))
   391→        throw new Error(data.error || 'Simulation failed')
   392→      }
   393→
   394→      // Re-fetch match with simulated data
   395→      await fetchMatch()
   396→    } catch (err) {
   397→      console.error('Simulate error:', err)
   398→      setSimulateError(err instanceof Error ? err.message : 'Simulation failed')
   399→    } finally {
   400→      setSimulating(false)
   401→    }
   402→  }, [selectedMatchId, token, fetchMatch])
   403→
   404→  // ── Handle Bookmark ─────────────────────────────────────────────────────
   405→  const handleBookmark = useCallback(async () => {
   406→    if (!selectedMatchId || !token) return
   407→    // Toggle bookmark
   408→    setBookmarked((prev) => !prev)
   409→    // In a real app, this would POST/DELETE to /api/bookmarks
   410→  }, [selectedMatchId, token])
   411→
   412→  // ── Handle Share ────────────────────────────────────────────────────────
   413→  const handleShare = useCallback(() => {
   414→    if (navigator.share && match) {
   415→      navigator.share({
   416→        title: `${match.homeTeam?.name} vs ${match.awayTeam?.name} - ELASTICO`,
   417→        text: `Check out this match on ELASTICO!`,
   418→        url: window.location.href,
   419→      }).catch(() => {})
   420→    }
   421→  }, [match])
   422→
   423→  // ── xG Chart Data ─────────────────────────────────────────────────────
   424→  const xgChartData = useMemo(() => {
   425→    if (!match || !match.events || match.events.length === 0) return []
   426→
   427→    // Build cumulative xG over time based on goal events
   428→    const goalEvents = match.events
   429→      .filter((e) => e.type === 'goal')
   430→      .sort((a, b) => a.minute - b.minute)
   431→
   432→    let homeCum = 0
   433→    let awayCum = 0
   434→    const dataPoints: { minute: string; 'Home xG': number; 'Away xG': number }[] = [
   435→      { minute: "0'", 'Home xG': 0, 'Away xG': 0 },
   436→    ]
   437→
   438→    // Create data points at each goal event and intermediate points
   439→    const goalXgPerGoal = match.homeXg > 0 || match.awayXg > 0
   440→      ? (match.homeXg / Math.max((match.events || []).filter(e => e.team === 'home' && e.type === 'goal').length, 1))
   441→      : 0
   442→    const awayGoalXgPerGoal = match.awayXg > 0 || match.homeXg > 0
   443→      ? (match.awayXg / Math.max((match.events || []).filter(e => e.team === 'away' && e.type === 'goal').length, 1))
   444→      : 0
   445→
   446→    for (const event of goalEvents) {
   447→      if (event.team === 'home') {
   448→        homeCum += goalXgPerGoal
   449→      } else {
   450→        awayCum += awayGoalXgPerGoal
   451→      }
   452→      dataPoints.push({
   453→        minute: `${event.minute}'`,
   454→        'Home xG': Math.round(homeCum * 100) / 100,
   455→        'Away xG': Math.round(awayCum * 100) / 100,
   456→      })
   457→    }
   458→
   459→    // Final totals
   460→    dataPoints.push({
   461→      minute: "90'",
   462→      'Home xG': Math.round(match.homeXg * 100) / 100,
   463→      'Away xG': Math.round(match.awayXg * 100) / 100,
   464→    })
   465→
   466→    return dataPoints
   467→  }, [match])
   468→
   469→  // ── User's existing vote ─────────────────────────────────────────────────
   470→  const userVote = useMemo(() => {
   471→    if (!match || !user) return null
   472→    return match.votes?.find((v) => v.userId === user.id) || null
   473→  }, [match, user])
   474→
   475→  // ── User's existing prediction ─────────────────────────────────────────
   476→  const userPrediction = useMemo(() => {
   477→    if (!match || !user) return null
   478→    return match.predictions?.find((p) => p.userId === user.id) || null
   479→  }, [match, user])
   480→
   481→  // ── Get prediction result for finished matches ──────────────────────────
   482→  const predictionResult = useMemo(() => {
   483→    if (!userPrediction || match?.status !== 'finished') return null
   484→    const actual =
   485→      (match?.homeScore ?? 0) > (match?.awayScore ?? 0)
   486→        ? 'home'
   487→        : (match?.homeScore ?? 0) === (match?.awayScore ?? 0)
   488→          ? 'draw'
   489→          : 'away'
   490→    return userPrediction.predictedOutcome === actual ? 'correct' : 'incorrect'
   491→  }, [userPrediction, match])
   492→
   493→  // ── Loading State ──────────────────────────────────────────────────────
   494→  if (loading) {
   495→    return (
   496→      <div className="flex flex-col gap-4 animate-fade-in-up">
   497→        <Skeleton className="h-44 w-full rounded-xl" />
   498→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   499→          <Skeleton className="h-64 w-full rounded-xl" />
   500→          <Skeleton className="h-64 w-full rounded-xl" />
   501→        </div>
   502→        <Skeleton className="h-48 w-full rounded-xl" />
   503→      </div>
   504→    )
   505→  }
   506→
   507→  // ── Error State ────────────────────────────────────────────────────────
   508→  if (error || !match) {
   509→    return (
   510→      <div className="flex flex-col items-center justify-center py-20 animate-fade-in-up">
   511→        <AlertCircle className="size-12 text-destructive mb-4" />
   512→        <h3 className="text-lg font-semibold mb-2">Failed to Load Match</h3>
   513→        <p className="text-sm text-muted-foreground mb-6 max-w-md text-center">{error || 'Match not found'}</p>
   514→        <Button
   515→          variant="outline"
   516→          onClick={() => setView('matches')}
   517→          className="gap-2 border-border"
   518→        >
   519→          <ArrowLeft className="size-4" />
   520→          Back to Matches
   521→        </Button>
   522→      </div>
   523→    )
   524→  }
   525→
   526→  const statusConfig = getStatusConfig(match.status)
   527→  const isUpcoming = match.status === 'upcoming'
   528→  const isLive = match.status === 'live' || match.status === 'halftime'
   529→  const isFinished = match.status === 'finished'
   530→  const isProOrElite = user && (user.plan === 'pro' || user.plan === 'elite')
   531→  const homeTeam = match.homeTeam
   532→  const awayTeam = match.awayTeam
   533→
   534→  // ── Compute ELO probabilities if not stored ────────────────────────────
   535→  const homeWinProb = match.homeWinProb ?? computeEloProb(
   536→    match.homeEloBefore ?? homeTeam?.eloRating ?? 1500,
   537→    match.awayEloBefore ?? awayTeam?.eloRating ?? 1500,
   538→    'home'
   539→  )
   540→  const drawProb = match.drawProb ?? computeEloProb(
   541→    match.homeEloBefore ?? homeTeam?.eloRating ?? 1500,
   542→    match.awayEloBefore ?? awayTeam?.eloRating ?? 1500,
   543→    'draw'
   544→  )
   545→  const awayWinProb = match.awayWinProb ?? computeEloProb(
   546→    match.homeEloBefore ?? homeTeam?.eloRating ?? 1500,
   547→    match.awayEloBefore ?? awayTeam?.eloRating ?? 1500,
   548→    'away'
   549→  )
   550→
   551→  const totalVotes = (match.voteDistribution?.home ?? 0) + (match.voteDistribution?.draw ?? 0) + (match.voteDistribution?.away ?? 0)
   552→
   553→  return (
   554→    <div className="flex flex-col gap-4 animate-fade-in-up">
   555→      {/* ══════════════════════════════════════════════════════════════════════
   556→          ACTION BUTTONS BAR
   557→      ══════════════════════════════════════════════════════════════════════ */}
   558→      <div className="flex items-center gap-2 flex-wrap">
   559→        <Button
   560→          variant="outline"
   561→          size="sm"
   562→          className="h-8 gap-1.5 border-border text-xs"
   563→          onClick={() => setView('matches')}
   564→        >
   565→          <ArrowLeft className="size-3.5" />
   566→          Back
   567→        </Button>
   568→
   569→        <div className="flex-1" />
   570→
   571→        <Button
   572→          variant="ghost"
   573→          size="sm"
   574→          className={cn('h-8 gap-1.5 text-xs', bookmarked && 'text-primary')}
   575→          onClick={handleBookmark}
   576→          aria-label={bookmarked ? 'Remove bookmark' : 'Bookmark match'}
   577→        >
   578→          {bookmarked ? <BookmarkCheck className="size-3.5" /> : <Bookmark className="size-3.5" />}
   579→          {bookmarked ? 'Saved' : 'Bookmark'}
   580→        </Button>
   581→
   582→        <Button
   583→          variant="ghost"
   584→          size="sm"
   585→          className="h-8 gap-1.5 text-xs border-border"
   586→          onClick={handleShare}
   587→        >
   588→          <Share2 className="size-3.5" />
   589→          Share
   590→        </Button>
   591→
   592→        {isProOrElite && isUpcoming && (
   593→          <Button
   594→            variant="outline"
   595→            size="sm"
   596→            className="h-8 gap-1.5 text-xs bg-primary/10 text-primary border-primary/30 hover:bg-primary/20 hover:text-primary"
   597→            onClick={handleSimulate}
   598→            disabled={simulating}

  1546→// ── Helper: Compute ELO-based probability ────────────────────────────────────
  1547→
  1548→function computeEloProb(homeElo: number, awayElo: number, outcome: 'home' | 'draw' | 'away'): number {
  1549→  const expectedHome = 1 / (1 + Math.pow(10, (awayElo - homeElo) / 400))
  1550→  const homeProb = expectedHome
  1551→  const awayProb = 1 - expectedHome
  1552→  const drawProb = (1 - Math.abs(homeProb - awayProb)) * 0.3
  1553→
  1554→  const total = homeProb + drawProb + awayProb
  1555→  if (outcome === 'home') return Math.round((homeProb / total) * 1000) / 10
  1556→  if (outcome === 'away') return Math.round((awayProb / total) * 1000) / 10
  1557→  return Math.round((drawProb / total) * 1000) / 10
  1558→}

  1569→      </span>
  1570→    ) : (
  1571→      <span key={idx}>{part}</span>
  1572→    ),
  1573→  )
  1574→}
