   100→   100→function CircGauge({ value, max = 100, size = 120, label, strokeColor }: {
   101→   101→  value: number; max?: number; size?: number; label: string; strokeColor?: string
   102→   102→}) {
   103→   103→  const pct = Math.min(value / max, 1)
   104→   104→  const r = (size - 16) / 2
   105→   105→  const circ = 2 * Math.PI * r
   106→   106→  const offset = circ * (1 - pct)
   107→   107→  const color = strokeColor || (pct > 0.7 ? '#00e676' : pct > 0.4 ? '#eab308' : '#ef4444')
   108→   108→
   109→   109→  return (
   110→   110→    <div className="flex flex-col items-center gap-2">
   111→   111→      <svg width={size} height={size} className="drop-shadow-lg">
   112→   112→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="oklch(0.15 0.02 260)" strokeWidth={8} />
   113→   113→        <circle
   114→   114→          cx={size / 2} cy={size / 2} r={r} fill="none"
   115→   115→          stroke={color} strokeWidth={8} strokeLinecap="round"
   116→   116→          strokeDasharray={circ} strokeDashoffset={offset}
   117→   117→          transform={`rotate(-90 ${size / 2} ${size / 2})`}
   118→   118→          className="transition-all duration-1000 ease-out"
   119→   119→        />
   120→   120→        <text x={size / 2} y={size / 2 - 4} textAnchor="middle" className="fill-foreground text-2xl font-bold" dominantBaseline="middle">
   121→   121→          {Math.round(value)}
   122→   122→        </text>
   123→   123→        <text x={size / 2} y={size / 2 + 18} textAnchor="middle" className="fill-muted-foreground text-xs" dominantBaseline="middle">
   124→   124→          / {max}
   125→   125→        </text>
   126→   126→      </svg>
   127→   127→      <span className="text-xs text-muted-foreground font-medium">{label}</span>
   128→   128→    </div>
