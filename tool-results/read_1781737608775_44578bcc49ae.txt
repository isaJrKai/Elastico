   141→        const headers: Record<string, string> = {}
   142→        if (token) headers['Authorization'] = `Bearer ${token}`
   143→        const params = new URLSearchParams()

   145→        if (positionFilter !== 'all') params.set('position', positionFilter)
   146→        if (search) params.set('search', search)
   147→        params.set('limit', '100')

   149→        if (res.ok) {
   150→          const data = await res.json()
   151→          if (data.players && data.players.length > 0) {
   152→            setPlayers(data.players.map((p: Record<string, unknown>) => ({
   153→              ...p,
   154→              teamName: (p.team as Record<string, unknown>)?.name as string || p.teamName,
   155→              teamCode: (p.team as Record<string, unknown>)?.code as string || p.teamCode,
   156→              teamColor: (p.team as Record<string, unknown>)?.primaryColor as string || '#00e676',
   157→            } as EnhancedPlayer)))
   158→            return
   159→          }
   160→        }
   161→      } catch {
   162→        // fallback to mock

   173→    if (teamFilter !== 'all') result = result.filter(p => p.teamName === teamFilter)
   174→    result.sort((a, b) => {
   175→      if (sortBy === 'rating') return b.rating - a.rating
   176→      if (sortBy === 'goals') return b.goals - a.goals
   177→      if (sortBy === 'assists') return b.assists - a.assists
   178→      if (sortBy === 'name') return a.name.localeCompare(b.name)
   179→      if (sortBy === 'age') return (a.age || 0) - (b.age || 0)
   180→      if (sortBy === 'marketValue') return (b.marketValue || 0) - (a.marketValue || 0)
   181→      return 0
   182→    })
   183→    return result
   184→  }, [players, search, positionFilter, teamFilter, sortBy])

   197→    }
   198→    return groups

   201→  const ageData = useMemo(() => {
   202→    const buckets: Record<string, number> = { '18-21': 0, '22-25': 0, '26-29': 0, '30-33': 0, '34+': 0 }
   203→    for (const p of players) {
   204→      const age = p.age || 25
   205→      if (age <= 21) buckets['18-21']++
   206→      else if (age <= 25) buckets['22-25']++
   207→      else if (age <= 29) buckets['26-29']++
   208→      else if (age <= 33) buckets['30-33']++
   209→      else buckets['34+']++
   210→    }
   211→    return Object.entries(buckets).map(([range, count]) => ({ range, count }))
   212→  }, [players])
