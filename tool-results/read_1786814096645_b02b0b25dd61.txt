   138→   138→function normalRandom(): number {
   139→   139→  // Box-Muller transform for standard normal distribution
   140→   140→  let u = 0, v = 0
   141→   141→  while (u === 0) u = Math.random()
   142→   142→  while (v === 0) v = Math.random()
   143→   143→  return Math.sqrt(-2.0 * Math.log(u)) * Math.cos(2.0 * Math.PI * v)
   144→   144→}
   145→   145→
   146→   146→function poissonRandom(lambda: number): number {
   147→   147→  // Knuth algorithm for Poisson distribution
   148→   148→  if (lambda <= 0) return 0
   149→   149→  const L = Math.exp(-lambda)
   150→   150→  let k = 0
   151→   151→  let p = 1.0
   152→   152→  do {
   153→   153→    k++
   154→   154→    p *= Math.random()
   155→   155→  } while (p > L)
   156→   156→  return k - 1
   157→   157→}

   163→   163→// ── GARCH(1,1) Volatility Calibration ───────────────────────────────────────────
   164→   164→
   165→   165→export function calibrateGARCH(residuals: number[]): number {
   166→   166→  /**
   167→   167→   * Estimates conditional variance using a simplified GARCH(1,1) model.
   168→   168→   * Returns a volatility scaling factor.
   169→   169→   * 
   170→   170→   * GARCH(1,1): σ²_t = ω + α * ε²_{t-1} + β * σ²_{t-1}
   171→   171→   * We use maximum likelihood estimation with simplified parameters.
   172→   172→   */
   173→   173→  if (residuals.length < 15) return 1.0
   174→   174→
   175→   175→  const n = residuals.length
   176→   176→  const mean = residuals.reduce((s, v) => s + v, 0) / n
   177→   177→
   178→   178→  // Estimate GARCH parameters using method of moments
   179→   179→  const variance = residuals.reduce((s, v) => s + (v - mean) ** 2, 0) / (n - 1)
   180→   180→  const squaredResiduals = residuals.map(r => r * r)
   181→   181→  const autoCorrLag1 = squaredResiduals.slice(0, -1).reduce((s, r, i) =>
   182→   182→    s + (r - variance) * (squaredResiduals[i + 1] - variance), 0) /
   183→   183→    (n - 1) / (variance * variance || 1)
   184→   184→
   185→   185→  // Simplified GARCH parameters
   186→   186→  const alpha = clamp(autoCorrLag1 * 0.4, 0.01, 0.3)  // ARCH parameter
   187→   187→  const beta = clamp(1 - alpha - 0.05, 0.6, 0.95)     // GARCH parameter
   188→   188→  const omega = 0.05 * variance                          // Constant term
   189→   189→
   190→   190→  // Forecast next-step conditional variance
   191→   191→  const lastSquaredResidual = squaredResiduals[n - 1]
   192→   192→  const prevVariance = variance
   193→   193→  const forecastVariance = omega + alpha * lastSquaredResidual + beta * prevVariance
   194→   194→
   195→   195→  // Return volatility scaling factor (forecast / historical baseline)
   196→   196→  return Math.sqrt(forecastVariance / (variance || 1))
   197→   197→}

   199→   199→// ── Merton Jump-Diffusion Parameter Extraction ──────────────────────────────────
   200→   200→
   201→   201→export function extractJumpDiffusionParams(residuals: number[]): JumpDiffusionParams {
   202→   202→  /**
   203→   203→   * Separates continuous diffusion from structural jump shocks.
   204→   204→   * Uses a 2-standard-deviation threshold to identify outlier events
   205→   205→   * (red cards, early injuries, unexpected tactical shifts).
   206→   206→   */
   207→   207→  if (residuals.length < 20) {
   208→   208→    return { sigma: 0.25, jumpLambda: 0.10, jumpMu: -0.05, jumpSigma: 0.15 }
   209→   209→  }
   210→   210→
   211→   211→  const threshold = 2.0 * (residuals.reduce((s, r) => s + r * r, 0) / residuals.length) ** 0.5
   212→   212→  const normalShocks = residuals.filter(r => Math.abs(r) <= threshold)
   213→   213→  const jumpShocks = residuals.filter(r => Math.abs(r) > threshold)
   214→   214→
   215→   215→  const sigma = normalShocks.length > 0
   216→   216→    ? normalShocks.reduce((s, r) => s + r * r, 0) / normalShocks.length
   217→   217→    : 0.25
   218→   218→
   219→   219→  const jumpLambda = jumpShocks.length / residuals.length
   220→   220→  const jumpMu = jumpShocks.length > 0
   221→   221→    ? jumpShocks.reduce((s, r) => s + r, 0) / jumpShocks.length
   222→   222→    : 0.0
   223→   223→  const jumpSigma = jumpShocks.length > 1
   224→   224→    ? Math.sqrt(jumpShocks.reduce((s, r) => s + (r - jumpMu) ** 2, 0) / (jumpShocks.length - 1))
   225→   225→    : 0.20
   226→   226→
   227→   227→  return {
   228→   228→    sigma: Math.sqrt(sigma),
   229→   229→    jumpLambda: clamp(jumpLambda, 0, 1),
   230→   230→    jumpMu,
   231→   231→    jumpSigma: clamp(jumpSigma, 0.01, 2),
   232→   232→  }
   233→   233→}

   235→   235→// ── Injury Adjustment Layer ────────────────────────────────────────────────────
   236→   236→
   237→   237→export function applyInjuryAdjustments(
   238→   238→  baseXg: number,
   239→   239→  injuries: InjuryAdjustment[]
   240→   240→): number {
   241→   241→  /**
   242→   242→   * Modifies projected expected goals based on missing players.
   243→   243→   * Star player out: -15% xG
   244→   244→   * Key player out: -8% xG
   245→   245→   * Rotation player out: -3% xG
   246→   246→   */
   247→   247→  let adjustedXg = baseXg
   248→   248→  for (const injury of injuries) {
   249→   249→    if (injury.status === 'out') {
   250→   250→      adjustedXg *= (1 - injury.xgImpact)
   251→   251→    } else if (injury.status === 'doubtful') {
   252→   252→      adjustedXg *= (1 - injury.xgImpact * 0.5)
   253→   253→    } else if (injury.status === 'questionable') {
   254→   254→      adjustedXg *= (1 - injury.xgImpact * 0.25)
   255→   255→    }
   256→   256→  }
   257→   257→  return Math.max(0.1, adjustedXg)
   258→   258→}

   260→   260→// ── Core Stochastic Simulation Engine ──────────────────────────────────────────
   261→   261→
   262→   262→export function runStochasticSimulation(
   263→   263→  input: MatchInput,
   264→   264→  config: EngineConfig = DEFAULT_CONFIG
   265→   265→): StochasticMatchResult {
   266→   266→  /**
   267→   267→   * Executes a high-iteration Monte Carlo simulation using a
   268→   268→   * Bivariate Merton Jump-Diffusion Process with GARCH volatility scaling.
   269→   269→   * 
   270→   270→   * This models match outcomes as continuous stochastic differential
   271→   271→   * processes interrupted by Poisson-distributed structural jumps,
   272→   272→   * matching the methodology used by elite quantitative sports funds.
   273→   273→   */
   274→   274→  const { simulationRuns, garchEnabled, jumpDiffusionEnabled } = config
   275→   275→
   276→   276→  // Step 1: Calibrate base intensity parameters from xG + ELO
   277→   277→  const eloDiff = input.homeElo - input.awayElo
   278→   278→  const eloFactor = 1 / (1 + Math.pow(10, -eloDiff / 400))
   279→   279→  const avgXg = (input.homeXg + input.awayXg) / 2
   280→   280→
   281→   281→  // Home advantage adjustment (+0.25 goals typical for home field)
   282→   282→  let muHome = input.homeXg * (0.7 + 0.3 * eloFactor) + 0.25
   283→   283→  let muAway = input.awayXg * (0.7 + 0.3 * (1 - eloFactor))
   284→   284→
   285→   285→  // Step 2: Apply injury adjustments
   286→   286→  const homeInjuries = input.injuries?.filter(i => i.teamId === input.homeTeamId) || []
   287→   287→  const awayInjuries = input.injuries?.filter(i => i.teamId === input.awayTeamId) || []
   288→   288→  muHome = applyInjuryAdjustments(muHome, homeInjuries)
   289→   289→  muAway = applyInjuryAdjustments(muAway, awayInjuries)
   290→   290→
   291→   291→  // Step 3: GARCH volatility calibration
   292→   292→  const homeResiduals = input.historicalResiduals?.home || []
   293→   293→  const awayResiduals = input.historicalResiduals?.away || []
   294→   294→  const volScalarHome = garchEnabled ? calibrateGARCH(homeResiduals) : 1.0
   295→   295→  const volScalarAway = garchEnabled ? calibrateGARCH(awayResiduals) : 1.0
   296→   296→
   297→   297→  // Step 4: Extract jump-diffusion parameters
   298→   298→  const homeJd = jumpDiffusionEnabled
   299→   299→    ? extractJumpDiffusionParams(homeResiduals)
   300→   300→    : { sigma: 0.20, jumpLambda: 0.05, jumpMu: 0, jumpSigma: 0.15 }
   301→   301→  const awayJd = jumpDiffusionEnabled
   302→   302→    ? extractJumpDiffusionParams(awayResiduals)
   303→   303→    : { sigma: 0.20, jumpLambda: 0.05, jumpMu: 0, jumpSigma: 0.15 }
   304→   304→
   305→   305→  // Step 5: Compute inter-team correlation
   306→   306→  const rho = input.correlation ?? 0.12
   307→   307→
   308→   308→  // Step 6: Execute Monte Carlo simulation
   309→   309→  let homeWins = 0
   310→   310→  let awayWins = 0
   311→   311→  let draws = 0
   312→   312→  let over25 = 0
   313→   313→  let over15 = 0
   314→   314→  let over35 = 0
   315→   315→  let btts = 0
   316→   316→  const scorelineCounts: Record<string, number> = {}
   317→   317→  let totalHomeGoals = 0
   318→   318→  let totalAwayGoals = 0
   319→   319→
   320→   320→  // Pre-compute correlation transformation constants
   321→   321→  const sqrtOneMinusRho2 = Math.sqrt(Math.max(0, 1 - rho * rho))
   322→   322→
   323→   323→  for (let i = 0; i < simulationRuns; i++) {
   324→   324→    // Generate correlated Brownian motion paths via Cholesky decomposition
   325→   325→    const zHomeRaw = normalRandom()
   326→   326→    const zAwayRaw = normalRandom()
   327→   327→    const zHome = zHomeRaw
   328→   328→    const zAway = rho * zHomeRaw + sqrtOneMinusRho2 * zAwayRaw
   329→   329→
   330→   330→    // Apply GARCH volatility scaling to base intensities
   331→   331→    const scaledMuHome = muHome * volScalarHome
   332→   332→    const scaledMuAway = muAway * volScalarAway
   333→   333→
   334→   334→    // Merton Jump-Diffusion: continuous diffusion + Poisson jumps
   335→   335→    const hSigma = homeJd.sigma
   336→   336→    const aSigma = awayJd.sigma
   337→   337→    const dt = 1.0 // One match unit
   338→   338→
   339→   339→    // Poisson jump processes for structural shocks
   340→   340→    const jumpCountHome = poissonRandom(homeJd.jumpLambda * dt)
   341→   341→    const jumpCountAway = poissonRandom(awayJd.jumpLambda * dt)
   342→   342→
   343→   343→    // Calculate jump magnitudes with drift + volatility
   344→   344→    const jumpMagHome = jumpCountHome > 0
   345→   345→      ? jumpCountHome * homeJd.jumpMu + Math.sqrt(jumpCountHome) * homeJd.jumpSigma * normalRandom()
   346→   346→      : 0
   347→   347→    const jumpMagAway = jumpCountAway > 0
   348→   348→      ? jumpCountAway * awayJd.jumpMu + Math.sqrt(jumpCountAway) * awayJd.jumpSigma * normalRandom()
   349→   349→      : 0
   350→   350→
   351→   351→    // Synthesize continuous diffusion with structural shocks
   352→   352→    const stochasticHome = scaledMuHome *
   353→   353→      Math.exp((hSigma * hSigma / -2.0) * dt + hSigma * Math.sqrt(dt) * zHome + jumpMagHome)
   354→   354→    const stochasticAway = scaledMuAway *
   355→   355→      Math.exp((aSigma * aSigma / -2.0) * dt + aSigma * Math.sqrt(dt) * zAway + jumpMagAway)
   356→   356→
   357→   357→    // Convert intensity paths to observable goal outputs via Poisson distribution
   358→   358→    const goalsHome = poissonRandom(Math.max(0.01, stochasticHome))
   359→   359→    const goalsAway = poissonRandom(Math.max(0.01, stochasticAway))
   360→   360→
   361→   361→    // Accumulate statistics
   362→   362→    totalHomeGoals += goalsHome
   363→   363→    totalAwayGoals += goalsAway
   364→   364→
   365→   365→    if (goalsHome > goalsAway) homeWins++
   366→   366→    else if (goalsAway > goalsHome) awayWins++
   367→   367→    else draws++
   368→   368→
   369→   369→    const total = goalsHome + goalsAway
   370→   370→    if (total > 2.5) over25++
   371→   371→    if (total > 1.5) over15++
   372→   372→    if (total > 3.5) over35++
   373→   373→    if (goalsHome > 0 && goalsAway > 0) btts++
   374→   374→
   375→   375→    const key = `${goalsHome}-${goalsAway}`
   376→   376→    scorelineCounts[key] = (scorelineCounts[key] || 0) + 1
   377→   377→  }
   378→   378→
   379→   379→  // Step 7: Compute Asian Handicap probabilities
   380→   380→  const computeAH = (line: number) => {
   381→   381→    let homeCovers = 0
   382→   382→    let awayCovers = 0
   383→   383→    // Re-simulate for AH (using same parameters, smaller run)
   384→   384→    const ahRuns = Math.min(50000, simulationRuns)
   385→   385→    for (let i = 0; i < ahRuns; i++) {
   386→   386→      const zH = normalRandom()
   387→   387→      const zA = rho * zH + sqrtOneMinusRho2 * normalRandom()
   388→   388→      const sH = poissonRandom(Math.max(0.01, muHome * volScalarHome *
   389→   389→        Math.exp((homeJd.sigma ** 2 / -2.0) + homeJd.sigma * zH)))
   390→   390→      const sA = poissonRandom(Math.max(0.01, muAway * volScalarAway *
   391→   391→        Math.exp((awayJd.sigma ** 2 / -2.0) + awayJd.sigma * zA)))
   392→   392→      const diff = sH - sA - line
   393→   393→      if (diff > 0) homeCovers++
   394→   394→      else if (diff < 0) awayCovers++
   395→   395→    }
   396→   396→    return { home: homeCovers / ahRuns, away: awayCovers / ahRuns }
   397→   397→  }
   398→   398→
   399→   399→  // Step 8: Determine confidence level
   400→   400→  const maxProb = Math.max(homeWins, draws, awayWins) / simulationRuns
   401→   401→  const confidence: StochasticMatchResult['confidence'] =
   402→   402→    maxProb > 0.55 ? 'very-high' :
   403→   403→    maxProb > 0.48 ? 'high' :
   404→   404→    maxProb > 0.42 ? 'medium' : 'low'
   405→   405→
   406→   406→  // Step 9: Calculate volatility index (0-100)
   407→   407→  const goalVariance = Array.from({ length: 100 }, () => {
   408→   408→    const gH = poissonRandom(muHome)
   409→   409→    const gA = poissonRandom(muAway)
   410→   410→    return Math.abs(gH + gA - (muHome + muAway))
   411→   411→  })
   412→   412→  const avgDeviation = goalVariance.reduce((s, v) => s + v, 0) / 100
   413→   413→  const volatilityIndex = clamp((avgDeviation / (muHome + muAway || 1)) * 100, 0, 100)
   414→   414→
   415→   415→  return {
   416→   416→    matchProbabilities: {
   417→   417→      homeVictory: homeWins / simulationRuns,
   418→   418→      draw: draws / simulationRuns,
   419→   419→      awayVictory: awayWins / simulationRuns,
   420→   420→    },
   421→   421→    totalsMarket: {
   422→   422→      over25: over25 / simulationRuns,
   423→   423→      under25: 1 - over25 / simulationRuns,
   424→   424→      over15: over15 / simulationRuns,
   425→   425→      over35: over35 / simulationRuns,
   426→   426→    },
   427→   427→    exoticScorelines: Object.fromEntries(
   428→   428→      Object.entries(scorelineCounts)
   429→   429→        .sort(([, a], [, b]) => b - a)
   430→   430→        .slice(0, 10)
   431→   431→        .map(([key, count]) => [key, count / simulationRuns])
   432→   432→    ),
   433→   433→    expectedMeans: {
   434→   434→      home: totalHomeGoals / simulationRuns,
   435→   435→      away: totalAwayGoals / simulationRuns,
   436→   436→      total: (totalHomeGoals + totalAwayGoals) / simulationRuns,
   437→   437→    },
   438→   438→    asianHandicap: {
   439→   439→      line0: computeAH(0),
   440→   440→      lineHalf: computeAH(-0.5),
   441→   441→      line1: computeAH(-1),
   442→   442→      line15: computeAH(-1.5),
   443→   443→    },
   444→   444→    bothTeamsToScore: btts / simulationRuns,
   445→   445→    confidence,
   446→   446→    volatilityIndex,
   447→   447→  }
   448→   448→}

   450→   450→// ── Kelly Criterion Engine ──────────────────────────────────────────────────────
   451→   451→
   452→   452→export function calculateKelly(
   453→   453→  modelProbability: number,
   454→   454→  bookmakerOdds: number,
   455→   455→  bankroll: number,
   456→   456→  fractionalKelly: number = 0.25
   457→   457→): KellyResult {
   458→   458→  /**
   459→   459→   * Implements the Kelly Criterion for optimal bankroll allocation.
   460→   460→   * Uses fractional Kelly (default 25%) for safer variance management.
   461→   461→   * 
   462→   462→   * Kelly Formula: f* = (p*b - q) / b
   463→   463→   * Where: p = model probability, b = decimal odds - 1, q = 1 - p
   464→   464→   */
   465→   465→  const impliedProb = 1 / bookmakerOdds
   466→   466→  const edge = modelProbability - impliedProb
   467→   467→
   468→   468→  if (edge <= 0) {
   469→   469→    return {
   470→   470→      action: 'NO_EDGE',
   471→   471→      edgePercentage: -Math.abs(edge) * 100,
   472→   472→      impliedProbability: impliedProb,
   473→   473→      modelProbability,
   474→   474→      suggestedFraction: 0,
   475→   475→      suggestedWager: 0,
   476→   476→      kellyFraction: 0,
   477→   477→      odds: bookmakerOdds,
   478→   478→    }
   479→   479→  }
   480→   480→
   481→   481→  const b = bookmakerOdds - 1  // Net odds
   482→   482→  const q = 1 - modelProbability
   483→   483→  const rawKelly = (modelProbability * b - q) / b
   484→   484→  const safeKelly = rawKelly * fractionalKelly
   485→   485→  const wager = bankroll * safeKelly
   486→   486→
   487→   487→  // Risk classification
   488→   488→  const action: KellyResult['action'] =
   489→   489→    edge > 0.10 ? 'BET' :
   490→   490→    edge > 0.03 ? 'BET' : 'RISKY'
   491→   491→
   492→   492→  return {
   493→   493→    action,
   494→   494→    edgePercentage: edge * 100,
   495→   495→    impliedProbability: impliedProb,
   496→   496→    modelProbability,
   497→   497→    suggestedFraction: safeKelly,
   498→   498→    suggestedWager: Math.round(wager * 100) / 100,
   499→   499→    kellyFraction: rawKelly,
   500→   500→    odds: bookmakerOdds,
   501→   501→  }
   502→   502→}

   504→   504→// ── Multi-Asset Portfolio Allocation ────────────────────────────────────────────
   505→   505→
   506→   506→export function calculatePortfolioAllocation(
   507→   507→  outcomes: { label: string; modelProb: number; odds: number }[],
   508→   508→  bankroll: number,
   509→   509→  correlationMatrix: number[][] = [
   510→   510→    [1.00, 0.55, 0.15],
   511→   511→    [0.55, 1.00, 0.20],
   512→   512→    [0.15, 0.20, 1.00],
   513→   513→  ]
   514→   514→): PortfolioAllocation {
   515→   515→  /**
   516→   516→   * Implements a multi-asset Kelly Criterion with covariance adjustment.
   517→   517→   * Accounts for correlation between bet outcomes to prevent overexposure
   518→   518→   * on correlated positions (e.g., home win + over 2.5 goals).
   519→   519→   */
   520→   520→
   521→   521→  // Invert correlation matrix for covariance-weighted allocation
   522→   522→  const n = correlationMatrix.length
   523→   523→  const covInv = invertMatrix(correlationMatrix)
   524→   524→
   525→   525→  const results = outcomes.map((out, i) => {
   526→   526→    const impliedProb = 1 / out.odds
   527→   527→    const edge = out.modelProb - impliedProb
   528→   528→
   529→   529→    // Covariance-adjusted edge
   530→   530→    const adjustedEdge = covInv
   531→   531→      ? covInv[i]?.reduce((sum, c, j) => sum + c * (outcomes[j].modelProb - 1 / outcomes[j].odds), 0) || edge
   532→   532→      : edge
   533→   533→
   534→   534→    const b = out.odds - 1
   535→   535→    const q = 1 - out.modelProb
   536→   536→    const rawKelly = (out.modelProb * b - q) / b
   537→   537→    const safeKelly = Math.max(0, rawKelly * 0.10) // 10% fractional Kelly
   538→   538→
   539→   539→    return {
   540→   540→      label: out.label,
   541→   541→      modelProb: out.modelProb,
   542→   542→      marketOdds: out.odds,
   543→   543→      edge: edge * 100,
   544→   544→      kellyFraction: safeKelly,
   545→   545→      wagerAmount: Math.round(bankroll * safeKelly * 100) / 100,
   546→   546→      action: edge > 0.02 ? 'BET' as const : edge > 0 ? 'RISKY' as const : 'NO_EDGE' as const,
   547→   547→    }
   548→   548→  })
   549→   549→
   550→   550→  const totalExposure = results.reduce((s, r) => s + r.wagerAmount, 0)
   551→   551→  const totalEV = results.reduce((s, r) => s + r.wagerAmount * r.edge / 100, 0)
   552→   552→  const avgReturn = totalExposure > 0 ? totalEV / totalExposure : 0
   553→   553→  const sharpeRatio = results.length > 1 ? avgReturn / (Math.sqrt(results.reduce((s, r) => s + (r.edge / 100 - avgReturn) ** 2, 0) / results.length) || 0.01) : 0
   554→   554→
   555→   555→  return {
   556→   556→    outcomes: results,
   557→   557→    totalBankroll: bankroll,
   558→   558→    totalExposure: Math.round(totalExposure * 100) / 100,
   559→   559→    totalExpectedValue: Math.round(totalEV * 100) / 100,
   560→   560→    sharpeRatio: Math.round(sharpeRatio * 100) / 100,
   561→   561→  }
   562→   562→}

   564→   564→// ── Matrix Inversion (3x3) ─────────────────────────────────────────────────────
   565→   565→
   566→   566→function invertMatrix(m: number[][]): number[][] | null {
   567→   567→  if (m.length !== 3 || m[0].length !== 3) return null
   568→   568→
   569→   569→  const [a, b, c] = m[0]
   570→   570→  const [d, e, f] = m[1]
   571→   571→  const [g, h, i] = m[2]
   572→   572→
   573→   573→  const det = a * (e * i - f * h) - b * (d * i - f * g) + c * (d * h - e * g)
   574→   574→  if (Math.abs(det) < 1e-10) return null
   575→   575→
   576→   576→  const invDet = 1 / det
   577→   577→  return [
   578→   578→    [(e * i - f * h) * invDet, (c * h - b * i) * invDet, (b * f - c * e) * invDet],
   579→   579→    [(f * g - d * i) * invDet, (a * i - c * g) * invDet, (c * d - a * f) * invDet],
   580→   580→    [(d * h - e * g) * invDet, (b * g - a * h) * invDet, (a * e - b * d) * invDet],
   581→   581→  ]
   582→   582→}

   584→   584→// ── Market Signal Tracker ───────────────────────────────────────────────────────
   585→   585→
   586→   586→export function analyzeMarketSignals(
   587→   587→  openingOdds: { home: number; draw: number; away: number },
   588→   588→  currentOdds: { home: number; draw: number; away: number },
   589→   589→  matchId: string,
   590→   590→  homeTeam: string,
   591→   591→  awayTeam: string
   592→   592→): MarketSignal {
   593→   593→  /**
   594→   594→   * Tracks line movement velocity to detect sharp money action.
   595→   595→   * 
   596→   596→   * Key signals:
   597→   597→   * - Steam Move: Rapid odds movement in one direction
   598→   598→   * - Reverse Line Movement (RLM): Line moves AGAINST public betting
   599→   599→   * - Sharp Action: Professional syndicate money detected
   600→   600→   */
   601→   601→
   602→   602→  // Calculate line velocity (negative = odds dropping = money coming in)
   603→   603→  const homeVelocity = (currentOdds.home - openingOdds.home) / openingOdds.home
   604→   604→  const drawVelocity = (currentOdds.draw - openingOdds.draw) / openingOdds.draw
   605→   605→  const awayVelocity = (currentOdds.away - openingOdds.away) / openingOdds.away
   606→   606→
   607→   607→  // Detect significant moves (>3% change is notable)
   608→   608→  const threshold = 0.03
   609→   609→  const homeSteam = homeVelocity < -threshold
   610→   610→  const awaySteam = awayVelocity < -threshold
   611→   611→  const drawSteam = drawVelocity < -threshold
   612→   612→
   613→   613→  // Detect RLM: odds drop on one side despite public likely betting the other
   614→   614→  const rlmDetected = (homeVelocity < -threshold && awayVelocity > threshold) ||
   615→   615→    (awayVelocity < -threshold && homeVelocity > threshold)
   616→   616→
   617→   617→  // Determine sharp action direction
   618→   618→  const velocities = [
   619→   619→    { dir: 'home' as const, vel: Math.abs(homeVelocity) },
   620→   620→    { dir: 'draw' as const, vel: Math.abs(drawVelocity) },
   621→   621→    { dir: 'away' as const, vel: Math.abs(awayVelocity) },
   622→   622→  ]
   623→   623→  velocities.sort((a, b) => b.vel - a.vel)
   624→   624→
   625→   625→  const sharpAction: MarketSignal['sharpAction'] =
   626→   626→    velocities[0].vel > threshold ? velocities[0].dir : 'neutral'
   627→   627→
   628→   628→  // Confidence score based on magnitude of movement
   629→   629→  const maxMove = Math.max(Math.abs(homeVelocity), Math.abs(drawVelocity), Math.abs(awayVelocity))
   630→   630→  const confidence = clamp(maxMove * 500, 0, 100)
   631→   631→
   632→   632→  return {
   633→   633→    matchId,
   634→   634→    homeTeam,
   635→   635→    awayTeam,
   636→   636→    openingOdds,
   637→   637→    currentOdds,
   638→   638→    lineVelocity: {
   639→   639→      home: Math.round(homeVelocity * 10000) / 100,
   640→   640→      draw: Math.round(drawVelocity * 10000) / 100,
   641→   641→      away: Math.round(awayVelocity * 10000) / 100,
   642→   642→    },
   643→   643→    sharpAction,
   644→   644→    steamMove: homeSteam || awaySteam || drawSteam,
   645→   645→    rlmDetected,
   646→   646→    confidence: Math.round(confidence),
   647→   647→    timestamp: new Date().toISOString(),
   648→   648→  }
   649→   649→}

   651→   651→// ── xG Luck Adjustment ─────────────────────────────────────────────────────────
   652→   652→
   653→   653→export function adjustForLuck(
   654→   654→  rawGoals: number,
   655→   655→  xg: number,
   656→   656→  sampleSize: number = 10
   657→   657→): { luckAdjustedGoals: number; luckFactor: number } {
   658→   658→  /**
   659→   659→   * Removes "luck" from raw scores by regressing toward xG.
   660→   660→   * Uses a Bayesian shrinkage estimator weighted by sample size.
   661→   661→   * 
   662→   662→   * Small sample → more regression toward xG (more luck adjustment)
   663→   663→   * Large sample → trust raw goals more
   664→   664→   */
   665→   665→  const weight = Math.min(sampleSize / 30, 1) // Full weight at 30+ games
   666→   666→  const luckAdjustedGoals = rawGoals * weight + xg * (1 - weight)
   667→   667→  const luckFactor = rawGoals > 0 ? xg / rawGoals : 1
   668→   668→
   669→   669→  return {
   670→   670→    luckAdjustedGoals: Math.round(luckAdjustedGoals * 100) / 100,
   671→   671→    luckFactor: Math.round(luckFactor * 100) / 100,
   672→   672→  }
   673→   673→}

   675→   675→// ── Complete Match Analysis Pipeline ────────────────────────────────────────────
   676→   676→
   677→   677→export interface FullMatchAnalysis {
   678→   678→  matchInput: MatchInput
   679→   679→  simulation: StochasticMatchResult
   680→   680→  kellyResults: {
   681→   681→    home: KellyResult
   682→   682→    draw: KellyResult
   683→   683→    away: KellyResult
   684→   684→    over25: KellyResult
   685→   685→    under25: KellyResult
   686→   686→  }
   687→   687→  portfolioAllocation: PortfolioAllocation
   688→   688→  marketSignal: MarketSignal | null
   689→   689→  luckAdjustment: {
   690→   690→    home: ReturnType<typeof adjustForLuck>
   691→   691→    away: ReturnType<typeof adjustForLuck>
   692→   692→  }
   693→   693→  recommendation: string
   694→   694→  riskRating: 'low' | 'medium' | 'high' | 'very-high'
   695→   695→}
   696→   696→
   697→   697→export function runFullMatchAnalysis(
   698→   698→  input: MatchInput,
   699→   699→  bankroll: number = 1000,
   700→   700→  openingOdds?: { home: number; draw: number; away: number },
   701→   701→  config: EngineConfig = DEFAULT_CONFIG
   702→   702→): FullMatchAnalysis {
   703→   703→  /**
   704→   704→   * The complete ELASTICO analysis pipeline:
   705→   705→   * 1. Luck-adjust historical data
   706→   706→   * 2. Run stochastic simulation
   707→   707→   * 3. Calculate Kelly Criterion for all markets
   708→   708→   * 4. Generate portfolio allocation
   709→   709→   * 5. Analyze market signals
   710→   710→   * 6. Produce final recommendation
   711→   711→   */
   712→   712→
   713→   713→  // Step 1: Luck adjustment
   714→   714→  const homeLuck = adjustForLuck(
   715→   715→    input.homeXg * 1.1, // Approximate actual goals from xG
   716→   716→    input.homeXg
   717→   717→  )
   718→   718→  const awayLuck = adjustForLuck(
   719→   719→    input.awayXg * 0.9,
   720→   720→    input.awayXg
   721→   721→  )
   722→   722→
   723→   723→  // Step 2: Run simulation with luck-adjusted inputs
   724→   724→  const adjustedInput: MatchInput = {
   725→   725→    ...input,
   726→   726→    homeXg: homeLuck.luckAdjustedGoals,
   727→   727→    awayXg: awayLuck.luckAdjustedGoals,
   728→   728→  }
   729→   729→  const simulation = runStochasticSimulation(adjustedInput, config)
   730→   730→
   731→   731→  // Step 3: Kelly Criterion for each market
   732→   732→  const kellyResults = {
   733→   733→    home: calculateKelly(
   734→   734→      simulation.matchProbabilities.homeVictory,
   735→   735→      input.bookmakerOdds.home,
   736→   736→      bankroll,
   737→   737→      config.kellyFraction
   738→   738→    ),
   739→   739→    draw: calculateKelly(
   740→   740→      simulation.matchProbabilities.draw,
   741→   741→      input.bookmakerOdds.draw,
   742→   742→      bankroll,
   743→   743→      config.kellyFraction
   744→   744→    ),
   745→   745→    away: calculateKelly(
   746→   746→      simulation.matchProbabilities.awayVictory,
   747→   747→      input.bookmakerOdds.away,
   748→   748→      bankroll,
   749→   749→      config.kellyFraction
   750→   750→    ),
   751→   751→    over25: calculateKelly(
   752→   752→      simulation.totalsMarket.over25,
   753→   753→      input.bookmakerOdds.home * 0.9, // Approximate over/under odds
   754→   754→      bankroll,
   755→   755→      config.kellyFraction
   756→   756→    ),
   757→   757→    under25: calculateKelly(
   758→   758→      simulation.totalsMarket.under25,
   759→   759→      input.bookmakerOdds.home * 1.1,
   760→   760→      bankroll,
   761→   761→      config.kellyFraction
   762→   762→    ),
   763→   763→  }
   764→   764→
   765→   765→  // Step 4: Portfolio allocation
   766→   766→  const portfolioAllocation = calculatePortfolioAllocation([
   767→   767→    { label: `Home Win (${input.homeTeam})`, modelProb: simulation.matchProbabilities.homeVictory, odds: input.bookmakerOdds.home },
   768→   768→    { label: `Draw`, modelProb: simulation.matchProbabilities.draw, odds: input.bookmakerOdds.draw },
   769→   769→    { label: `Away Win (${input.awayTeam})`, modelProb: simulation.matchProbabilities.awayVictory, odds: input.bookmakerOdds.away },
   770→   770→  ], bankroll)
   771→   771→
   772→   772→  // Step 5: Market signals
   773→   773→  const marketSignal = openingOdds
   774→   774→    ? analyzeMarketSignals(openingOdds, input.bookmakerOdds, input.homeTeamId || '', input.homeTeam, input.awayTeam)
   775→   775→    : null
   776→   776→
   777→   777→  // Step 6: Generate recommendation
   778→   778→  const bestBet = [kellyResults.home, kellyResults.draw, kellyResults.away]
   779→   779→    .filter(k => k.action === 'BET')
   780→   780→    .sort((a, b) => b.edgePercentage - a.edgePercentage)[0]
   781→   781→
   782→   782→  const recommendation = bestBet
   783→   783→    ? `RECOMMENDED: ${bestBet.edgePercentage.toFixed(1)}% edge detected. Suggested wager: $${bestBet.suggestedWager.toFixed(2)} (${(bestBet.suggestedFraction * 100).toFixed(2)}% of bankroll). Model probability: ${(bestBet.modelProbability * 100).toFixed(1)}% vs market implied: ${(bestBet.impliedProbability * 100).toFixed(1)}%.`
   784→   784→    : 'No positive expected value bets found for this match. The market appears efficiently priced.'
   785→   785→
   786→   786→  // Risk rating
   787→   787→  const maxEdge = Math.max(
   788→   788→    kellyResults.home.edgePercentage,
   789→   789→    kellyResults.draw.edgePercentage,
   790→   790→    kellyResults.away.edgePercentage
   791→   791→  )
   792→   792→  const riskRating: FullMatchAnalysis['riskRating'] =
   793→   793→    simulation.volatilityIndex > 70 ? 'very-high' :
   794→   794→    simulation.volatilityIndex > 50 ? 'high' :
   795→   795→    simulation.volatilityIndex > 30 ? 'medium' : 'low'
   796→   796→
   797→   797→  return {
   798→   798→    matchInput: adjustedInput,
   799→   799→    simulation,
   800→   800→    kellyResults,
   801→   801→    portfolioAllocation,
   802→   802→    marketSignal,
   803→   803→    luckAdjustment: { home: homeLuck, away: awayLuck },
   804→   804→    recommendation,
   805→   805→    riskRating,
   806→   806→  }
   807→   807→}
