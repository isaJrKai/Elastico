   134→   134→def _safe_div(numerator: float, denominator: float, default: float = 0.0) -> float:
   135→   135→    """Divide with safe fallback for zero denominators."""
   136→   136→    if denominator == 0 or not math.isfinite(denominator):
   137→   137→        return default
   138→   138→    result = numerator / denominator
   139→   139→    return result if math.isfinite(result) else default

   142→   142→def _safe_avg(values: list[float] | np.ndarray, default: float = 0.0) -> float:
   143→   143→    """Mean of a numeric sequence, returning *default* for empty input."""
   144→   144→    arr = np.asarray(values, dtype=float)
   145→   145→    if arr.size == 0:
   146→   146→        return default
   147→   147→    return float(np.nanmean(arr))

   150→   150→def _linear_trend(values: list[float] | np.ndarray, default: float = 0.0) -> float:
   151→   151→    """Slope of a simple linear fit (most recent → oldest = positive trend)."""
   152→   152→    arr = np.asarray(values, dtype=float)
   153→   153→    if arr.size < 2:
   154→   154→        return default
   155→   155→    x = np.arange(len(arr), dtype=float)
   156→   156→    valid = np.isfinite(arr)
   157→   157→    if valid.sum() < 2:
   158→   158→        return default
   159→   159→    x = x[valid]
   160→   160→    y = arr[valid]
   161→   161→    slope = np.polyfit(x, y, 1)[0]
   162→   162→    return float(slope) if math.isfinite(slope) else default

   165→   165→def _compute_streaks(
   166→   166→    results: list[str] | list[int] | None,
   167→   167→) -> tuple[int, int, int]:
   168→   168→    """Return (win_streak, unbeaten_streak, loss_streak) from recent results.
   169→   169→
   170→   170→    *results* is ordered **most recent first**.  Each element is one of:
   171→   171→    - 'W' / 1  → win
   172→   172→    - 'D' / 0  → draw
   173→   173→    - 'L' / -1 → loss
   174→   174→    """
   175→   175→    if not results:
   176→   176→        return 0, 0, 0
   177→   177→    win_s = unbeaten_s = loss_s = 0
   178→   178→    for r in results:
   179→   179→        if r in ('W', 1):
   180→   180→            win_s += 1
   181→   181→            unbeaten_s += 1
   182→   182→        elif r in ('D', 0):
   183→   183→            unbeaten_s += 1
   184→   184→            break
   185→   185→        elif r in ('L', -1):
   186→   186→            break
   187→   187→        else:
   188→   188→            break
   189→   189→    for r in results:
   190→   190→        if r in ('L', -1):
   191→   191→            loss_s += 1
   192→   192→        else:
   193→   193→            break
   194→   194→    return win_s, unbeaten_s, loss_s

   197→   197→def _form_points(results: list[str] | list[int] | None) -> float:
   198→   198→    """Sum of 3/1/0 for W/D/L over recent results (max 15 for 5 matches)."""
   199→   199→    if not results:
   200→   200→        return 0.0
   201→   201→    mapping = {'W': 3, 'D': 1, 'L': 0, 1: 3, 0: 1, -1: 0}
   202→   202→    return sum(mapping.get(r, 0) for r in results)

   205→   205→def _parse_date(value: Any) -> Optional[datetime]:
   206→   206→    """Coerce *value* to a ``datetime``.  Returns ``None`` on failure."""
   207→   207→    if isinstance(value, datetime):
   208→   208→        return value
   209→   209→    if isinstance(value, str):
   210→   210→        for fmt in ("%Y-%m-%d", "%Y-%m-%dT%H:%M:%S", "%Y-%m-%d %H:%M:%S"):
   211→   211→            try:
   212→   212→                return datetime.strptime(value, fmt)
   213→   213→            except ValueError:
   214→   214→                continue
   215→   215→    if isinstance(value, (int, float)):
   216→   216→        try:
   217→   217→            return datetime.fromtimestamp(value)
   218→   218→        except (OSError, ValueError):
   219→   219→            pass
   220→   220→    return None

   248→   248→
   249→   249→    # ------------------------------------------------------------------
   250→   250→    # Public API

   252→   252→
   253→   253→    def build_match_features(
   254→   254→        self,
   255→   255→        match_data: dict[str, Any],
   256→   256→        home_stats: dict[str, Any],
   257→   257→        away_stats: dict[str, Any],
   258→   258→        h2h_data: list[dict[str, Any]] | None = None,
   259→   259→        odds: dict[str, Any] | None = None,
   260→   260→    ) -> dict[str, float]:
   261→   261→        """Build a feature vector (50 values) for a **single** match prediction.
   262→   262→
   263→   263→        Parameters
   264→   264→        ----------
   265→   265→        match_data : dict
   266→   266→            Must contain ``date`` (str / datetime / timestamp).
   267→   267→        home_stats, away_stats : dict
   268→   268→            Per-team season statistics.  Expected keys (all optional):
   269→   269→
   270→   270→            ELO: ``elo``, ``elo_history`` (list[float])
   271→   271→            Goals: ``goals_scored``, ``goals_conceded``,
   272→   272→                   ``goals_scored_home``, ``goals_conceded_home`` (or ``_away``)
   273→   273→            xG: ``xg_for``, ``xg_for_home`` (or ``_away``)
   274→   274→            Form: ``recent_results`` (list[str] 'W'/'D'/'L', most-recent first),
   275→   275→                  ``recent_goals_scored`` (list[float]),
   276→   276→                  ``last_match_date`` (str/datetime)
   277→   277→            Shots: ``shots_total``, ``shots_on_target``
   278→   278→            Venue: ``home_wins``, ``home_draws``, ``home_losses``,
   279→   279→                   ``away_wins``, ``away_draws``, ``away_losses``,
   280→   280→                   ``home_goals``, ``away_goals``,
   281→   281→                   ``clean_sheets_home``, ``clean_sheets_away``,
   282→   282→                   ``matches_played_home``, ``matches_played_away``
   283→   283→        h2h_data : list[dict], optional
   284→   284→            List of previous head-to-head match dicts.  Each needs
   285→   285→            ``home_goals`` and ``away_goals`` (or ``score`` like '2-1').
   286→   286→        odds : dict, optional
   287→   287→            ``home_odds``, ``draw_odds``, ``away_odds`` (decimal).
   288→   288→
   289→   289→        Returns

   292→   292→            Mapping of all 50 feature names to float values.

   295→   295→
   296→   296→        # 1. ELO features ─────────────────────────────────────────────
   297→   297→        feats.update(self._elo_features(home_stats, away_stats))
   298→   298→
   299→   299→        # 2. Goal features ────────────────────────────────────────────
   300→   300→        feats.update(self._goal_features(home_stats, away_stats))
   301→   301→
   302→   302→        # 3. Form features ────────────────────────────────────────────
   303→   303→        feats.update(self._form_features(home_stats, away_stats))
   304→   304→
   305→   305→        # 4. Head-to-head features ─────────────────────────────────────

   307→   307→
   308→   308→        # 5. Home/away split features ──────────────────────────────────
   309→   309→        feats.update(self._venue_features(home_stats, away_stats))
   310→   310→
   311→   311→        # 6. Market / odds features ────────────────────────────────────
   312→   312→        feats.update(self._market_features(odds))
   313→   313→
   314→   314→        # 7. Temporal features ─────────────────────────────────────────
   315→   315→        feats.update(self._temporal_features(match_data, home_stats, away_stats))
   316→   316→
   317→   317→        # 8. Advanced features ─────────────────────────────────────────

   323→   323→    # ------------------------------------------------------------------
   324→   324→    def build_training_dataset(
   325→   325→        self,
   326→   326→        matches: list[dict[str, Any]],
   327→   327→        window: int = 10,
   328→   328→    ) -> pd.DataFrame:
   329→   329→        """Build a training dataset from historical matches.
   330→   330→
   331→   331→        Each match in *matches* must contain:
   332→   332→
   333→   333→        - ``home_team``, ``away_team``
   334→   334→        - ``home_goals``, ``away_goals``  (for target + stats)
   335→   335→        - ``date``
   336→   336→        - ``home_odds``, ``draw_odds``, ``away_odds``  (optional)
   337→   337→
   338→   338→        The method computes a sliding-window of *window* previous matches
   339→   339→        per team to derive form, streak, and trend features.  Missing
   340→   340→        historical data is back-filled with league-average defaults.
   341→   341→
   342→   342→        Parameters
   343→   343→        ----------
   344→   344→        matches : list[dict]
   345→   345→            Historical match records.  Must be **chronologically sorted**
   346→   346→            (oldest first) for correct sliding-window behaviour.
   347→   347→        window : int
   348→   348→            Number of recent matches to use for rolling form stats.
   349→   349→
   350→   350→        Returns
   351→   351→        -------
   352→   352→        pd.DataFrame
   353→   353→            One row per match.  Columns are the 50 features plus
   354→   354→            ``target`` (0=away, 1=draw, 2=home).
   355→   355→        """
   356→   356→        if not matches:
   357→   357→            return pd.DataFrame(columns=[*FEATURE_NAMES, "target"])
   358→   358→
   359→   359→        # Index: team_id → list of (date, goals_scored, goals_conceded, result)
   360→   360→        team_history: dict[str, list[tuple]] = defaultdict(list)
   361→   361→        # team_id → list of home-only / away-only records
   362→   362→        team_home_history: dict[str, list[tuple]] = defaultdict(list)
   363→   363→        team_away_history: dict[str, list[tuple]] = defaultdict(list)
   364→   364→
   365→   365→        rows: list[dict[str, float]] = []
   366→   366→
   367→   367→        for match in matches:
   368→   368→            home_team = str(match.get("home_team", ""))
   369→   369→            away_team = str(match.get("away_team", ""))
   370→   370→            hg = int(match.get("home_goals", 0))
   371→   371→            ag = int(match.get("away_goals", 0))
   372→   372→
   373→   373→            # Derive result codes (from home perspective)
   374→   374→            if hg > ag:
   375→   375→                home_result, away_result = "W", "L"
   376→   376→            elif hg < ag:
   377→   377→                home_result, away_result = "L", "W"
   378→   378→            else:
   379→   379→                home_result, away_result = "D", "D"
   380→   380→
   381→   381→            match_date = _parse_date(match.get("date")) or datetime.now()
   382→   382→
   383→   383→            # ── Build per-team stats dicts from rolling history ───────
   384→   384→            home_stats = self._stats_from_history(
   385→   385→                team_history[home_team],
   386→   386→                team_home_history[home_team],
   387→   387→                team_away_history[home_team],
   388→   388→                window=window,
   389→   389→            )
   390→   390→            away_stats = self._stats_from_history(
   391→   391→                team_history[away_team],
   392→   392→                team_home_history[away_team],
   393→   393→                team_away_history[away_team],
   394→   394→                window=window,
   395→   395→            )
   396→   396→
   397→   397→            # Set last-match date for rest-day calculation
   398→   398→            if team_history[home_team]:
   399→   399→                home_stats["last_match_date"] = team_history[home_team][-1][0].isoformat()
   400→   400→            if team_history[away_team]:
   401→   401→                away_stats["last_match_date"] = team_history[away_team][-1][0].isoformat()
   402→   402→
   403→   403→            # ── Head-to-head (use previous matches between these teams) ─
   404→   404→            h2h: list[dict[str, Any]] = []
   405→   405→            for prev in matches:
   406→   406→                if prev is match:
   407→   407→                    continue
   408→   408→                if (
   409→   409→                    str(prev.get("home_team")) == home_team
   410→   410→                    and str(prev.get("away_team")) == away_team
   411→   411→                ) or (
   412→   412→                    str(prev.get("home_team")) == away_team
   413→   413→                    and str(prev.get("away_team")) == home_team
   414→   414→                ):
   415→   415→                    phg = int(prev.get("home_goals", 0))
   416→   416→                    pag = int(prev.get("away_goals", 0))
   417→   417→                    # Normalise to current home perspective
   418→   418→                    if str(prev.get("home_team")) == home_team:
   419→   419→                        h2h.append({"home_goals": phg, "away_goals": pag})
   420→   420→                    else:
   421→   421→                        h2h.append({"home_goals": pag, "away_goals": phg})
   422→   422→
   423→   423→            # ── Odds ──────────────────────────────────────────────────
   424→   424→            odds: dict[str, Any] | None = None
   425→   425→            if match.get("home_odds") and match.get("draw_odds") and match.get("away_odds"):
   426→   426→                odds = {
   427→   427→                    "home_odds": float(match["home_odds"]),
   428→   428→                    "draw_odds": float(match["draw_odds"]),
   429→   429→                    "away_odds": float(match["away_odds"]),
   430→   430→                }
   431→   431→
   432→   432→            # ── Build features ────────────────────────────────────────
   433→   433→            feats = self.build_match_features(
   434→   434→                match_data={"date": match_date},
   435→   435→                home_stats=home_stats,
   436→   436→                away_stats=away_stats,
   437→   437→                h2h_data=h2h if h2h else None,
   438→   438→                odds=odds,
   439→   439→            )
   440→   440→
   441→   441→            # Target: 0=away, 1=draw, 2=home
   442→   442→            if hg > ag:
   443→   443→                target = 2
   444→   444→            elif hg < ag:
   445→   445→                target = 0
   446→   446→            else:
   447→   447→                target = 1
   448→   448→
   449→   449→            feats["target"] = float(target)
   450→   450→            rows.append(feats)
   451→   451→
   452→   452→            # ── Update histories for future windows ───────────────────
   453→   453→            team_history[home_team].append(
   454→   454→                (match_date, hg, ag, home_result)
   455→   455→            )
   456→   456→            team_history[away_team].append(
   457→   457→                (match_date, ag, hg, away_result)
   458→   458→            )
   459→   459→            team_home_history[home_team].append(
   460→   460→                (match_date, hg, ag, home_result)
   461→   461→            )
   462→   462→            team_away_history[away_team].append(
   463→   463→                (match_date, ag, hg, away_result)
   464→   464→            )
   465→   465→
   466→   466→        df = pd.DataFrame(rows)
   467→   467→        # Ensure column order
   468→   468→        cols = [*FEATURE_NAMES, "target"]
   469→   469→        for c in cols:
   470→   470→            if c not in df.columns:
   471→   471→                df[c] = 0.0
   472→   472→        return df[cols]

   634→   634→
   635→   635→    def _market_features(
   636→   636→        self, odds: dict[str, Any] | None
   637→   637→    ) -> dict[str, float]:
   638→   638→        default_odds = 1.0
   639→   639→        if odds is None:
   640→   640→            return {
   641→   641→                "implied_prob_home": 1.0 / 3.0,
   642→   642→                "implied_prob_draw": 1.0 / 3.0,
   643→   643→                "implied_prob_away": 1.0 / 3.0,
   644→   644→                "overround": 1.0,
   645→   645→            }
   646→   646→
   647→   647→        ho = float(odds.get("home_odds", default_odds))
   648→   648→        do = float(odds.get("draw_odds", default_odds))
   649→   649→        ao = float(odds.get("away_odds", default_odds))
   650→   650→
   651→   651→        # Guard against zero
   652→   652→        ho = max(ho, 1.01)
   653→   653→        do = max(do, 1.01)
   654→   654→        ao = max(ao, 1.01)
   655→   655→
   656→   656→        ip_h = _safe_div(1.0, ho, 1.0 / 3.0)
   657→   657→        ip_d = _safe_div(1.0, do, 1.0 / 3.0)
   658→   658→        ip_a = _safe_div(1.0, ao, 1.0 / 3.0)
   659→   659→        overround = ip_h + ip_d + ip_a
   660→   660→
   661→   661→        return {
   662→   662→            "implied_prob_home": ip_h,
   663→   663→            "implied_prob_draw": ip_d,
   664→   664→            "implied_prob_away": ip_a,
   665→   665→            "overround": overround,
   666→   666→        }

   670→   670→    def _temporal_features(
   671→   671→        self,
   672→   672→        match_data: dict[str, Any],
   673→   673→        home_stats: dict[str, Any],
   674→   674→        away_stats: dict[str, Any],
   675→   675→    ) -> dict[str, float]:
   676→   676→        match_date = _parse_date(match_data.get("date")) or datetime.now()
   677→   677→        dow = float(match_date.weekday())

   680→   680→

   682→   682→        h_last = _parse_date(home_stats.get("last_match_date"))
   683→   683→        a_last = _parse_date(away_stats.get("last_match_date"))

   688→   688→        if h_last:
   689→   689→            delta = (match_date - h_last).days
   690→   690→            h_rest = float(max(delta, 0)) if delta is not None else default_rest
   691→   691→        if a_last:
   692→   692→            delta = (match_date - a_last).days
   693→   693→            a_rest = float(max(delta, 0)) if delta is not None else default_rest
   694→   694→
   695→   695→        return {
   696→   696→            "day_of_week": dow,
   697→   697→            "month": month,
   698→   698→            "is_weekend": is_weekend,
   699→   699→            "days_since_last_match_home": h_rest,
   700→   700→            "days_since_last_match_away": a_rest,
   701→   701→        }

   703→   703→    # ---------- 8. Advanced features -----------------------------------
   704→   704→
   705→   705→    def _advanced_features(
   706→   706→        self, home_stats: dict[str, Any], away_stats: dict[str, Any]
   707→   707→    ) -> dict[str, float]:
   708→   708→        lag = self.league_avg_goals
   709→   709→
   710→   710→        # Attacking / defensive strength (relative to league average)
   711→   711→        h_gs = float(home_stats.get("goals_scored", lag))
   712→   712→        h_gc = float(home_stats.get("goals_conceded", lag))
   713→   713→        a_gs = float(away_stats.get("goals_scored", lag))
   714→   714→        a_gc = float(away_stats.get("goals_conceded", lag))
   715→   715→
   716→   716→        atk_h = _safe_div(h_gs, lag, 1.0)
   717→   717→        atk_a = _safe_div(a_gs, lag, 1.0)
   718→   718→        def_h = _safe_div(h_gc, lag, 1.0)
   719→   719→        def_a = _safe_div(a_gc, lag, 1.0)
   720→   720→
   721→   721→        # Poisson lambda: attack * opponent defence * league average
   722→   722→        poisson_h = atk_h * def_a * lag
   723→   723→        poisson_a = atk_a * def_h * lag
   724→   724→
   725→   725→        # Shot conversion rates
   726→   726→        h_shots = float(home_stats.get("shots_total", 0.0))
   727→   727→        h_sot = float(home_stats.get("shots_on_target", 0.0))
   728→   728→        h_matches = max(float(home_stats.get("matches_played", 1)), 1.0)
   729→   729→        a_shots = float(away_stats.get("shots_total", 0.0))
   730→   730→        a_sot = float(away_stats.get("shots_on_target", 0.0))
   731→   731→        a_matches = max(float(away_stats.get("matches_played", 1)), 1.0)
   732→   732→
   733→   733→        h_conv = _safe_div(h_sot, h_shots, 0.33)
   734→   734→        a_conv = _safe_div(a_sot, a_shots, 0.33)
   735→   735→
   736→   736→        # Clean-sheet percentage
   737→   737→        h_cs = float(home_stats.get("clean_sheets_home", 0.0))
   738→   738→        h_home_m = max(float(home_stats.get("matches_played_home", 1)), 1.0)
   739→   739→        a_cs = float(away_stats.get("clean_sheets_away", 0.0))
   740→   740→        a_away_m = max(float(away_stats.get("matches_played_away", 1)), 1.0)
   741→   741→
   742→   742→        return {
   743→   743→            "attacking_strength_home": atk_h,
   744→   744→            "attacking_strength_away": atk_a,
   745→   745→            "defensive_strength_home": def_h,
   746→   746→            "defensive_strength_away": def_a,
   747→   747→            "poisson_lambda_home": poisson_h,
   748→   748→            "poisson_lambda_away": poisson_a,
   749→   749→            "shot_conversion_rate_home": h_conv,
   750→   750→            "shot_conversion_rate_away": a_conv,
   751→   751→            "clean_sheet_pct_home": _safe_div(h_cs, h_home_m),
   752→   752→            "clean_sheet_pct_away": _safe_div(a_cs, a_away_m),
   753→   753→        }

   759→   759→    @staticmethod
   760→   760→    def _stats_from_history(
   761→   761→        overall: list[tuple],
   762→   762→        home_only: list[tuple],
   763→   763→        away_only: list[tuple],
   764→   764→        window: int = 10,
   765→   765→    ) -> dict[str, Any]:
   766→   766→        """Derive a stats dict from rolling match history.
   767→   767→
   768→   768→        Each tuple is ``(date, goals_scored, goals_conceded, result_char)``.
   769→   769→        Uses the last *window* entries for form/streak, and full history
   770→   770→        for season averages.
   771→   771→        """
   772→   772→        stats: dict[str, Any] = {}
   773→   773→
   774→   774→        if not overall:
   775→   775→            return stats
   776→   776→
   777→   777→        # Full-season averages (all matches)
   778→   778→        gs_all = [m[1] for m in overall]
   779→   779→        gc_all = [m[2] for m in overall]
   780→   780→        stats["goals_scored"] = _safe_avg(gs_all)
   781→   781→        stats["goals_conceded"] = _safe_avg(gc_all)
   782→   782→        stats["matches_played"] = float(len(overall))
   783→   783→
   784→   784→        # Venue-split averages
   785→   785→        if home_only:
   786→   786→            stats["goals_scored_home"] = _safe_avg([m[1] for m in home_only])
   787→   787→            stats["goals_conceded_home"] = _safe_avg([m[2] for m in home_only])
   788→   788→            stats["home_wins"] = sum(1 for m in home_only if m[3] == "W")
   789→   789→            stats["home_draws"] = sum(1 for m in home_only if m[3] == "D")
   790→   790→            stats["home_losses"] = sum(1 for m in home_only if m[3] == "L")
   791→   791→            stats["home_goals"] = _safe_avg([m[1] for m in home_only])
   792→   792→            stats["matches_played_home"] = float(len(home_only))
   793→   793→            stats["clean_sheets_home"] = float(
   794→   794→                sum(1 for m in home_only if m[2] == 0)
   795→   795→            )
   796→   796→
   797→   797→        if away_only:
   798→   798→            stats["goals_scored_away"] = _safe_avg([m[1] for m in away_only])
   799→   799→            stats["goals_conceded_away"] = _safe_avg([m[2] for m in away_only])
   800→   800→            stats["away_wins"] = sum(1 for m in away_only if m[3] == "W")
   801→   801→            stats["away_draws"] = sum(1 for m in away_only if m[3] == "D")
   802→   802→            stats["away_losses"] = sum(1 for m in away_only if m[3] == "L")
   803→   803→            stats["away_goals"] = _safe_avg([m[1] for m in away_only])
   804→   804→            stats["matches_played_away"] = float(len(away_only))
   805→   805→            stats["clean_sheets_away"] = float(
   806→   806→                sum(1 for m in away_only if m[2] == 0)
   807→   807→            )
   808→   808→
   809→   809→        # Recent form (sliding window)
   810→   810→        recent = overall[-window:] if len(overall) >= window else overall
   811→   811→        stats["recent_results"] = [m[3] for m in recent]
   812→   812→        stats["recent_goals_scored"] = [m[1] for m in recent]
   813→   813→
   814→   814→        return stats
   815→   815→
