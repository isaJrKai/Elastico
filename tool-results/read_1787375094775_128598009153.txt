   421→   421→async function fetchScoreboardForLeague(league: { espnId: string; name: string; code: string }) {
   422→   422→  const SITE_V2 = 'https://site.api.espn.com/apis/v2/sports/soccer'
   423→   423→  const url = `${SITE_V2}/${league.espnId}/scoreboard`
   424→   424→  try {
   425→   425→    const res = await fetch(url, { cache: 'no-store' })
   426→   426→    if (!res.ok) return []
   427→   427→    const data = await res.json()
   428→   428→    const events: any[] = []
   429→   429→    for (const event of data.events || []) {
   430→   430→      const comp = event.competitions?.[0]
   431→   431→      if (!comp) continue
   432→   432→      const competitors = comp.competitors || []
   433→   433→      if (competitors.length < 2) continue
   434→   434→      const home = competitors.find((c: any) => c.homeAway === 'home') || competitors[0]
   435→   435→      const away = competitors.find((c: any) => c.homeAway === 'away') || competitors[1]
   436→   436→      const statusDetail = comp.status?.type?.detail || ''
   437→   437→      const minuteMatch = statusDetail.match(/(\d+)\s*'/)
   438→   438→      events.push({
   439→   439→        id: event.id?.toString() || '',
   440→   440→        homeTeamId: home.team?.id?.toString() || '',
   441→   441→        homeTeamName: home.team?.displayName || 'Unknown',
   442→   442→        homeTeamAbbr: home.team?.abbreviation || '???',
   443→   443→        homeTeamLogo: home.team?.logo || '',
   444→   444→        homeTeamColor: home.team?.color || '#00e676',
   445→   445→        awayTeamId: away.team?.id?.toString() || '',
   446→   446→        awayTeamName: away.team?.displayName || 'Unknown',
   447→   447→        awayTeamAbbr: away.team?.abbreviation || '???',
   448→   448→        awayTeamLogo: away.team?.logo || '',
   449→   449→        awayTeamColor: away.team?.color || '#ffffff',
   450→   450→        homeScore: parseInt(home.score) || 0,
   451→   451→        awayScore: parseInt(away.score) || 0,
   452→   452→        status: comp.status?.type?.name || 'STATUS_SCHEDULED',
   453→   453→        date: event.date || '',
   454→   454→        venue: comp.venue?.fullName || '',
   455→   455→        competition: event.name || league.name,
   456→   456→        competitionCode: league.code,
   457→   457→        minute: minuteMatch ? parseInt(minuteMatch[1]) : undefined,
   458→   458→      })
   459→   459→    }
   460→   460→    return events
   461→   461→  } catch (err) {
   462→   462→    console.error(`[SYNC/ESPN] Failed to fetch ${league.espnId}:`, err)
   463→   463→    return []
   464→   464→  }
   465→   465→}

   531→   531→      const matchData = {

   548→   548→      try {

   558→   558→    }
   559→   559→  }
   560→   560→
   561→   561→  return result
   562→   562→}

   566→   566→async function syncStandingsFromESPN(): Promise<{
   567→   567→  created: number
   568→   568→}> {
   569→   569→  const topLeagues = ['PL', 'LIGA', 'SA', 'BL', 'L1']
   570→   570→  let totalCreated = 0
   571→   571→  const season = String(new Date().getFullYear())
   572→   572→
   573→   573→  const results = await Promise.allSettled(
   574→   574→    topLeagues.map(async (code) => {
   575→   575→      const league = ESPN_LEAGUES.find(l => l.code === code)
   576→   576→      if (!league) return []
   577→   577→      const standings = await fetchESPNStandings(code)
   578→   578→      return standings.map(s => ({
   579→   579→        leagueCode: code,
   580→   580→        leagueName: league.name,
   581→   581→        ...s,
   582→   582→      }))
   583→   583→    })
   584→   584→  )
   585→   585→
   586→   586→  for (const r of results) {
   587→   587→    if (r.status !== 'fulfilled') continue
   588→   588→    for (const entry of r.value) {
   589→   589→      checkTimeout()
   590→   590→      const code = entry.leagueCode

   603→   603→            teamLogo: entry.logo,
   604→   604→            competition: leagueName,
   605→   605→            rank: entry.rank,
   606→   606→            played: entry.played,
   607→   607→            wins: entry.wins,
   608→   608→            draws: entry.draws,
   609→   609→            losses: entry.losses,
   610→   610→            goalsFor: entry.goalsFor,
   611→   611→            goalsAgainst: entry.goalsAgainst,
   612→   612→            goalDiff: entry.goalDiff,
   613→   613→            points: entry.points,
   614→   614→            form: entry.form || null,
   615→   615→            homeRecord: entry.homeRecord || null,
   616→   616→            source: 'espn',
   617→   617→            lastSyncedAt: new Date(),
   618→   618→          },
   619→   619→          create: {
   620→   620→            teamName: entry.team,
   621→   621→            teamCode: entry.code,
   622→   622→            teamLogo: entry.logo,
   623→   623→            competition: leagueName,
   624→   624→            competitionCode: code,
   625→   625→            season,
   626→   626→            rank: entry.rank,

   629→   629→            draws: entry.draws,

   631→   631→            goalsFor: entry.goalsFor,
   632→   632→            goalsAgainst: entry.goalsAgainst,
   633→   633→            goalDiff: entry.goalDiff,
   634→   634→            points: entry.points,
   635→   635→            form: entry.form || null,
   636→   636→            homeRecord: entry.homeRecord || null,
   637→   637→            source: 'espn',

   683→   683→export async function GET(req: NextRequest) {
   684→   684→  // 1. Validate cron secret
   685→   685→  if (!validateCron(req)) {
   686→   686→    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
   687→   687→  }
   688→   688→
   689→   689→  const summary: any = {
   690→   690→    success: true,
   691→   691→    syncs: {
   692→   692→      fixtures: { source: 'none', created: 0, updated: 0 },
   693→   693→      standings: { source: 'none', created: 0, updated: 0 },
   694→   694→      teams: { source: 'none', created: 0, updated: 0 },
   695→   695→      players: { source: 'none', created: 0, updated: 0 },
   696→   696→    },
   697→   697→    durationMs: 0,
   698→   698→  }
   699→   699→
   700→   700→  try {
   701→   701→    // ═══ 2. API-Sports: Fixtures + Standings + Teams + Players ═══
   702→   702→    let apiSportsAvailable = false
   703→   703→
   704→   704→    // 2a. Fixtures (live + today's top 5 leagues)
   705→   705→    try {
   706→   706→      const fixResult = await syncFixturesFromApiSports()
   707→   707→      if (fixResult.processed > 0) {
   708→   708→        apiSportsAvailable = true
   709→   709→        summary.syncs.fixtures = {
   710→   710→          source: 'api-sports',
   711→   711→          created: fixResult.created,
   712→   712→          updated: fixResult.updated,
   713→   713→        }
   714→   714→        await logSync('api-sports', 'fixtures', 'success', fixResult.processed, fixResult.created, fixResult.updated)
   715→   715→      }
   716→   716→    } catch (err: any) {
   717→   717→      if (err.message?.includes('timeout')) {
   718→   718→        console.warn('[SYNC] API-Sports fixtures timeout')
   719→   719→        await logSync('api-sports', 'fixtures', 'partial', 0, 0, 0, err.message)
   720→   720→      } else {
   721→   721→        console.warn('[SYNC] API-Sports fixtures failed:', err.message)
   722→   722→      }
   723→   723→    }
   724→   724→
   725→   725→    // 2b. Standings (top 10 leagues — only if API-Sports is working)
   726→   726→    if (apiSportsAvailable) {
   727→   727→      try {
   728→   728→        const stResult = await syncStandingsFromApiSports()
   729→   729→        summary.syncs.standings = {
   730→   730→          source: 'api-sports',
   731→   731→          created: stResult.created,
   732→   732→          updated: stResult.updated,
   733→   733→        }
   734→   734→        await logSync('api-sports', 'standings', 'success', stResult.created, stResult.created, stResult.updated)
   735→   735→      } catch (err: any) {
   736→   736→        if (err.message?.includes('timeout')) {
   737→   737→          console.warn('[SYNC] API-Sports standings timeout')
   738→   738→          await logSync('api-sports', 'standings', 'partial', 0, 0, 0, err.message)
   739→   739→        } else {
   740→   740→          console.warn('[SYNC] API-Sports standings failed:', err.message)
   741→   741→        }
   742→   742→      }
   743→   743→    }
   744→   744→
   745→   745→    // 2c. Teams (top 5 leagues — venue info, country, etc.)
   746→   746→    if (apiSportsAvailable) {
   747→   747→      try {
   748→   748→        const tmResult = await syncTeamsFromApiSports()
   749→   749→        summary.syncs.teams = {
   750→   750→          source: 'api-sports',
   751→   751→          created: tmResult.created,
   752→   752→          updated: tmResult.updated,
   753→   753→        }
   754→   754→        await logSync('api-sports', 'teams', 'success', tmResult.created + tmResult.updated, tmResult.created, tmResult.updated)
   755→   755→      } catch (err: any) {
   756→   756→        if (err.message?.includes('timeout')) {
   757→   757→          console.warn('[SYNC] API-Sports teams timeout')
   758→   758→        } else {
   759→   759→          console.warn('[SYNC] API-Sports teams failed:', err.message)
   760→   760→        }
   761→   761→      }
   762→   762→    }
   763→   763→
   764→   764→    // 2d. Player stats (top scorers, top 5 leagues)
   765→   765→    if (apiSportsAvailable) {
   766→   766→      try {
   767→   767→        const plResult = await syncPlayersFromApiSports()
   768→   768→        summary.syncs.players = {
   769→   769→          source: 'api-sports',
   770→   770→          created: plResult.created,
   771→   771→          updated: plResult.updated,
   772→   772→        }
   773→   773→        await logSync('api-sports', 'players', 'success', plResult.created + plResult.updated, plResult.created, plResult.updated)
   774→   774→      } catch (err: any) {
   775→   775→        if (err.message?.includes('timeout')) {
   776→   776→          console.warn('[SYNC] API-Sports players timeout')
   777→   777→        } else {
   778→   778→          console.warn('[SYNC] API-Sports players failed:', err.message)
   779→   779→        }
   780→   780→      }
   781→   781→    }
   782→   782→
   783→   783→    // ═══ 3. ESPN Fallback — only if API-Sports didn't return fixtures ═══
   784→   784→    if (!apiSportsAvailable) {
   785→   785→      try {
   786→   786→        const espnResult = await syncFixturesFromESPN()
   787→   787→        summary.syncs.fixtures = {
   788→   788→          source: 'espn',
   789→   789→          created: espnResult.created,
   790→   790→          updated: espnResult.updated,
   791→   791→        }
   792→   792→        await logSync('espn', 'fixtures', 'success', espnResult.processed, espnResult.created, espnResult.updated)
   793→   793→      } catch (err: any) {
   794→   794→        console.error('[SYNC] ESPN fixtures failed:', err)
   795→   795→        await logSync('espn', 'fixtures', 'error', 0, 0, 0, String(err))
   796→   796→        summary.syncs.fixtures.error = String(err)
   797→   797→      }
   798→   798→
   799→   799→      // ESPN Standings fallback
   800→   800→      try {
   801→   801→        const standingsResult = await syncStandingsFromESPN()
   802→   802→        summary.syncs.standings = {
   803→   803→          source: 'espn',
   804→   804→          created: standingsResult.created,
   805→   805→          updated: 0,
   806→   806→        }
   807→   807→        await logSync('espn', 'standings', 'success', standingsResult.created, standingsResult.created, 0)
   808→   808→      } catch (err: any) {
   809→   809→        if (err.message?.includes('timeout')) {
   810→   810→          console.warn('[SYNC] ESPN standings timeout')
   811→   811→          await logSync('espn', 'standings', 'partial', 0, 0, 0, err.message)
   812→   812→        } else {
   813→   813→          console.error('[SYNC] ESPN standings failed:', err)
   814→   814→          await logSync('espn', 'standings', 'error', 0, 0, 0, String(err))
   815→   815→        }
   816→   816→      }
   817→   817→    }
   818→   818→
   819→   819→    summary.durationMs = elapsed()
   820→   820→    console.log(`[SYNC] Complete in ${summary.durationMs}ms`, JSON.stringify(summary.syncs))
   821→   821→
   822→   822→    return NextResponse.json(summary)
   823→   823→  } catch (error) {
   824→   824→    const duration = elapsed()
   825→   825→    console.error(`[SYNC] Fatal error after ${duration}ms:`, error)
   826→   826→    summary.success = false
   827→   827→    summary.durationMs = duration
   828→   828→    summary.error = error instanceof Error ? error.message : String(error)
   829→   829→    return NextResponse.json(summary, { status: 500 })
   830→   830→  }
   831→   831→}
