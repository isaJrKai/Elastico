   308→def fit_poisson_model(train_df):
   309→    """
   310→    Fit independent Poisson model per league using simple averages.
   311→    lambda_home = (home_team_avg_goals_scored + away_team_avg_goals_conceded) / 2
   312→    lambda_away = (away_team_avg_goals_scored + home_team_avg_goals_conceded) / 2
   313→    Per-team stats computed from training data.
   314→    """
   315→    params = {}
   316→    for league in train_df['league_code'].unique():
   317→        ld = train_df[train_df['league_code'] == league]
   318→        if len(ld) < 50:
   319→            continue
   320→
   321→        teams = sorted(set(ld['HomeTeam'].unique()) | set(ld['AwayTeam'].unique()))
   322→
   323→        home_stats = {}
   324→        away_stats = {}
   325→        for t in teams:
   326→            home_matches = ld[ld['HomeTeam'] == t]
   327→            away_matches = ld[ld['AwayTeam'] == t]
   328→            home_stats[t] = {
   329→                'scored': home_matches['FTHG'].mean() if len(home_matches) > 0 else ld['FTHG'].mean(),
   330→                'conceded': home_matches['FTAG'].mean() if len(home_matches) > 0 else ld['FTAG'].mean(),
   331→                'n': len(home_matches),
   332→            }
   333→            away_stats[t] = {
   334→                'scored': away_matches['FTAG'].mean() if len(away_matches) > 0 else ld['FTAG'].mean(),
   335→                'conceded': away_matches['FTHG'].mean() if len(away_matches) > 0 else ld['FTHG'].mean(),
   336→                'n': len(away_matches),
   337→            }
   338→
   339→        params[league] = {
   340→            'home_stats': home_stats,
   341→            'away_stats': away_stats,
   342→            'avg_home_goals': float(ld['FTHG'].mean()),
   343→            'avg_away_goals': float(ld['FTAG'].mean()),
   344→        }
   345→    return params
   346→

   348→def poisson_predict(test_df, poisson_params, max_goals=8):
   349→    """
   350→    Predict match probabilities using fitted Poisson model.
   351→    Vectorized per-league implementation.
   352→    """
   353→    test_reset = test_df.reset_index(drop=True)
   354→    n = len(test_reset)
   355→    proba = np.full((n, 3), 1/3)
   356→
   357→    for league, p in poisson_params.items():
   358→        hs = p['home_stats']
   359→        as_ = p['away_stats']
   360→
   361→        mask = test_reset['league_code'] == league
   362→        if mask.sum() == 0:
   363→            continue
   364→
   365→        homes = test_reset.loc[mask, 'HomeTeam'].values
   366→        aways = test_reset.loc[mask, 'AwayTeam'].values
   367→
   368→        for i in range(mask.sum()):
   369→            h, a = homes[i], aways[i]
   370→            lambda_h = (hs.get(h, {}).get('scored', 1.4) + as_.get(a, {}).get('conceded', 1.1)) / 2
   371→            lambda_a = (as_.get(a, {}).get('scored', 1.1) + hs.get(h, {}).get('conceded', 1.1)) / 2
   372→            lambda_h = max(lambda_h, 0.3)
   373→            lambda_a = max(lambda_a, 0.2)
   374→
   375→            goal_range = np.arange(max_goals + 1)
   376→            pmf_h = poisson_dist.pmf(goal_range, lambda_h)
   377→            pmf_a = poisson_dist.pmf(goal_range, lambda_a)
   378→            joint = np.outer(pmf_h, pmf_a)
   379→            p_home_win = joint[np.triu_indices(max_goals + 1, k=1)].sum()
   380→            p_draw = np.diag(joint).sum()
   381→            p_away_win = joint[np.tril_indices(max_goals + 1, k=-1)].sum()
   382→
   383→            proba[mask.values, 0] = p_home_win
   384→            proba[mask.values, 1] = p_draw
   385→            proba[mask.values, 2] = p_away_win
   386→
   387→    proba = np.clip(proba, 1e-6, 1 - 1e-6)
   388→    proba = proba / proba.sum(axis=1, keepdims=True)
   389→    pred = np.argmax(proba, axis=1)
   390→    return proba, pred

   393→def fit_elo_model(train_df, k_factor=20.0, home_advantage=65.0):
   394→    """
   395→    Compute ELO ratings from scratch on training data.
   396→    Returns final ELO per (team, league) for use in prediction.
   397→    """
   398→    elo_ratings = {}
   399→
   400→    for league in train_df['league_code'].unique():
   401→        league_data = train_df[train_df['league_code'] == league].sort_values('Date').reset_index(drop=True)
   402→        ratings = {}
   403→
   404→        home_teams = league_data['HomeTeam'].values
   405→        away_teams = league_data['AwayTeam'].values
   406→        ftrs = league_data['FTR'].values
   407→
   408→        for i in range(len(league_data)):
   409→            home = home_teams[i]
   410→            away = away_teams[i]
   411→            ftr = ftrs[i]
   412→
   413→            if home not in ratings:
   414→                ratings[home] = 1500.0
   415→            if away not in ratings:
   416→                ratings[away] = 1500.0
   417→
   418→            r_home = ratings[home] + home_advantage
   419→            r_away = ratings[away]
   420→
   421→            exp_home = 1.0 / (1.0 + 10 ** ((r_away - r_home) / 400.0))
   422→            exp_away = 1.0 - exp_home
   423→
   424→            if ftr == 'H':
   425→                actual_home, actual_away = 1.0, 0.0
   426→            elif ftr == 'A':
   427→                actual_home, actual_away = 0.0, 1.0
   428→            else:
   429→                actual_home, actual_away = 0.5, 0.5
   430→
   431→            ratings[home] += k_factor * (actual_home - exp_home)
   432→            ratings[away] += k_factor * (actual_away - exp_away)
   433→
   434→        elo_ratings[league] = dict(ratings)
   435→
   436→    return elo_ratings
   437→

   439→def elo_predict(test_df, elo_ratings, home_advantage=65.0, draw_rate=0.25):
   440→    """
   441→    Predict using ELO ratings. Convert ELO difference to probabilities.
   442→    Vectorized implementation.
   443→    """
   444→    test_reset = test_df.reset_index(drop=True)
   445→    n = len(test_reset)
   446→    proba = np.full((n, 3), 1/3)
   447→
   448→    for league, r in elo_ratings.items():
   449→        mask = test_reset['league_code'] == league
   450→        if mask.sum() == 0:
   451→            continue
   452→
   453→        home_teams = test_reset.loc[mask, 'HomeTeam'].values
   454→        away_teams = test_reset.loc[mask, 'AwayTeam'].values
   455→
   456→        r_home = np.array([r.get(h, 1500.0) for h in home_teams])
   457→        r_away = np.array([r.get(a, 1500.0) for a in away_teams])
   458→
   459→        p_home_win = 1.0 / (1.0 + np.power(10, (r_away - r_home - home_advantage) / 400.0))
   460→        p_away_win = 1.0 - p_home_win
   461→
   462→        proba[mask.values, 0] = p_home_win * (1 - draw_rate)
   463→        proba[mask.values, 1] = draw_rate
   464→        proba[mask.values, 2] = p_away_win * (1 - draw_rate)
   465→
   466→    proba = np.clip(proba, 1e-6, 1 - 1e-6)
   467→    proba = proba / proba.sum(axis=1, keepdims=True)
   468→    pred = np.argmax(proba, axis=1)
   469→    return proba, pred

   476→def calibrate_platt(y_true, y_proba, cal_proba, n_classes=3):
   477→    """
   478→    Platt/logistic calibration: fit a logistic regression per class
   479→    on the calibration set probabilities, then apply to target probabilities.
   480→    """
   481→    calibrated = np.zeros_like(cal_proba)
   482→    for c in range(n_classes):
   483→        lr = LogisticRegression(C=1.0, max_iter=1000)
   484→        lr.fit(y_proba[:, c:c+1], (y_true == c).astype(int))
   485→        calibrated[:, c] = lr.predict_proba(cal_proba[:, c:c+1])[:, 1]
   486→    # Normalize
   487→    calibrated = np.clip(calibrated, 1e-6, 1 - 1e-6)
   488→    calibrated = calibrated / calibrated.sum(axis=1, keepdims=True)
   489→    return calibrated
   490→
   491→
   492→def calibrate_isotonic(y_true, y_proba, cal_proba, n_classes=3):
   493→    """
   494→    Isotonic calibration: fit isotonic regression per class
   495→    on the calibration set probabilities, then apply to target probabilities.
   496→    """
   497→    calibrated = np.zeros_like(cal_proba)
   498→    for c in range(n_classes):
   499→        ir = IsotonicRegression(y_min=1e-6, y_max=1 - 1e-6, out_of_bounds='clip')
   500→        ir.fit(y_proba[:, c], (y_true == c).astype(float))
   501→        calibrated[:, c] = ir.predict(cal_proba[:, c])
   502→    # Normalize
   503→    calibrated = np.clip(calibrated, 1e-6, 1 - 1e-6)
   504→    calibrated = calibrated / calibrated.sum(axis=1, keepdims=True)
   505→    return calibrated
   506→
   507→
   508→def calibrate_temperature(y_true, y_proba, cal_proba, n_classes=3):
   509→    """
   510→    Temperature scaling: learn a single temperature parameter T.
   511→    p_calibrated = softmax(log(p) / T)
   512→    Fit T on calibration set (y_true, y_proba) via NLL minimization.
   513→    Apply T to cal_proba.
   514→    """
   515→    def nll_temperature(log_T):
   516→        # T stored as log for positivity constraint
   517→        T = np.exp(log_T[0])
   518→        logits = np.log(np.clip(y_proba, 1e-10, 1.0)) / T
   519→        max_logits = logits.max(axis=1, keepdims=True)
   520→        exp_logits = np.exp(logits - max_logits)
   521→        softmax = exp_logits / exp_logits.sum(axis=1, keepdims=True)
   522→        correct = softmax[np.arange(len(y_true)), y_true]
   523→        correct = np.clip(correct, 1e-10, 1.0)
   524→        return -np.log(correct).mean()
   525→
   526→    result = minimize(nll_temperature, x0=[0.0], method='Nelder-Mead',
   527→                      options={'maxiter': 1000, 'xatol': 0.001})
   528→    T = float(np.exp(result.x[0]))
   529→    T = max(T, 0.01)
   530→
   531→    # Apply to target
   532→    logits = np.log(np.clip(cal_proba, 1e-10, 1.0)) / T
   533→    max_logits = logits.max(axis=1, keepdims=True)
   534→    exp_logits = np.exp(logits - max_logits)
   535→    calibrated = exp_logits / exp_logits.sum(axis=1, keepdims=True)
   536→    calibrated = np.clip(calibrated, 1e-6, 1 - 1e-6)
   537→    return calibrated, T

   544→def analyze_model_disagreement(market_proba, football_proba, y_true, class_idx=0):
   545→    """
   546→    Analyze disagreement between market and football model predictions.
   547→    Bucket by |market_prob - football_prob| for a given class.
   548→    Measure actual outcome frequency in each bucket.
   549→    """
   550→    prob_diff = np.abs(market_proba[:, class_idx] - football_proba[:, class_idx])
   551→    buckets = [
   552→        ('<5pp', 0, 5),
   553→        ('5-10pp', 5, 10),
   554→        ('10-15pp', 10, 15),
   555→        ('15-20pp', 15, 20),
   556→        ('20+pp', 20, 100),
   557→    ]
   558→
   559→    results = []
   560→    for label, lo, hi in buckets:
   561→        mask = (prob_diff >= lo / 100) & (prob_diff < hi / 100)
   562→        n = mask.sum()
   563→        if n < 10:
   564→            results.append({
   565→                'bucket': label, 'n': n,
   566→                'outcome_freq': None, 'avg_market_prob': None,
   567→                'avg_football_prob': None, 'avg_diff': None,
   568→            })
   569→            continue
   570→
   571→        outcome_freq = (y_true[mask] == class_idx).mean()  # frequency of this class occurring
   572→        avg_market = market_proba[mask, class_idx].mean()
   573→        avg_football = football_proba[mask, class_idx].mean()
   574→        avg_diff = prob_diff[mask].mean() * 100  # in percentage points
   575→
   576→        results.append({
   577→            'bucket': label, 'n': int(n),
   578→            'outcome_freq': float(outcome_freq),
   579→            'avg_market_prob': float(avg_market),
   580→            'avg_football_prob': float(avg_football),
   581→            'avg_diff_pp': float(avg_diff),
   582→        })
   583→    return results

   587→# 5E — Market Edge
   588→# ═══════════════════════════════════════════════════════════════════════
   589→
   590→def compute_market_edge(model_proba, market_proba, y_true):
   591→    """
   592→    Compute model edge = model_prob - market_prob for the predicted class.
   593→    Analyze whether edges correspond to actual value.
   594→    """
   595→    pred_class = np.argmax(model_proba, axis=1)
   596→    model_confidence = model_proba[np.arange(len(y_true)), pred_class]
   597→    market_confidence = market_proba[np.arange(len(y_true)), pred_class]
   598→    edge = model_confidence - market_confidence
   599→    correct = (pred_class == y_true).astype(float)
   600→
   601→    # Bucket by edge size
   602→    edge_buckets = [
   603→        ('<-10pp', -100, -10),
   604→        ('-10 to -5pp', -10, -5),
   605→        ('-5 to 0pp', -5, 0),
   606→        ('0 to 5pp', 0, 5),
   607→        ('5 to 10pp', 5, 10),
   608→        ('>10pp', 10, 100),
   609→    ]
   610→
   611→    results = []
   612→    for label, lo, hi in edge_buckets:
   613→        mask = (edge >= lo / 100) & (edge < hi / 100)
   614→        n = mask.sum()
   615→        if n < 10:
   616→            results.append({
   617→                'bucket': label, 'n': n,
   618→                'accuracy': None, 'avg_edge_pp': None,
   619→                'avg_model_conf': None, 'avg_market_conf': None,
   620→            })
   621→            continue
   622→        results.append({
   623→            'bucket': label, 'n': int(n),
   624→            'accuracy': float(correct[mask].mean()),
   625→            'avg_edge_pp': float(edge[mask].mean() * 100),
   626→            'avg_model_conf': float(model_confidence[mask].mean()),
   627→            'avg_market_conf': float(market_confidence[mask].mean()),
   628→        })
   629→    return results

   633→# 5F — Uncertainty
   634→# ═══════════════════════════════════════════════════════════════════════
   635→
   636→def analyze_uncertainty(model_proba, market_proba, y_true, test_df):
   637→    """
   638→    Investigate whether prediction confidence relates to accuracy.
   639→    Bucket by max probability (confidence) and measure accuracy.
   640→    """
   641→    confidence = model_proba.max(axis=1)
   642→    pred = np.argmax(model_proba, axis=1)
   643→    correct = (pred == y_true).astype(float)
   644→
   645→    # Also compute market-model agreement
   646→    market_pred = np.argmax(market_proba, axis=1)
   647→    agree = (pred == market_pred).astype(float)
   648→
   649→    conf_buckets = [
   650→        ('30-40%', 0.30, 0.40),
   651→        ('40-45%', 0.40, 0.45),
   652→        ('45-50%', 0.45, 0.50),
   653→        ('50-55%', 0.50, 0.55),
   654→        ('55-60%', 0.55, 0.60),
   655→        ('60-70%', 0.60, 0.70),
   656→        ('>70%', 0.70, 1.01),
   657→    ]
   658→
   659→    results = []
   660→    for label, lo, hi in conf_buckets:
   661→        mask = (confidence >= lo) & (confidence < hi)
   662→        n = mask.sum()
   663→        if n < 10:
   664→            results.append({
   665→                'bucket': label, 'n': n, 'accuracy': None,
   666→                'market_agreement': None, 'avg_confidence': None,
   667→            })
   668→            continue
   669→        results.append({
   670→            'bucket': label, 'n': int(n),
   671→            'accuracy': float(correct[mask].mean()),
   672→            'market_agreement': float(agree[mask].mean()),
   673→            'avg_confidence': float(confidence[mask].mean()),
   674→        })
   675→    return results

  1013→# 5B — Market Probability Audit (computed from data inspection)

  1016→def audit_market_probabilities(df):
  1017→    """Audit how market probabilities are constructed in the dataset."""
  1018→    audit = {}
  1019→
  1020→    # Check overround distribution
  1021→    overround = df['overround'].dropna()
  1022→    audit['overround_stats'] = {
  1023→        'mean': float(overround.mean()),
  1024→        'std': float(overround.std()),
  1025→        'min': float(overround.min()),
  1026→        'max': float(overround.max()),
  1027→        'median': float(overround.median()),
  1028→        'n_available': int(overround.count()),
  1029→        'n_missing': int(df['overround'].isna().sum()),
  1030→    }
  1031→
  1032→    # Check probability sum (should be ~100 if normalized)
  1033→    prob_sum = df[['implied_prob_home', 'implied_prob_draw', 'implied_prob_away']].dropna().sum(axis=1)
  1034→    audit['probability_sum_stats'] = {
  1035→        'mean': float(prob_sum.mean()),
  1036→        'std': float(prob_sum.std()),
  1037→        'min': float(prob_sum.min()),
  1038→        'max': float(prob_sum.max()),
  1039→    }
  1040→
  1041→    # Check per-class distribution
  1042→    for feat in ['implied_prob_home', 'implied_prob_draw', 'implied_prob_away']:
  1043→        vals = df[feat].dropna() / 100.0  # Convert to [0,1]
  1044→        audit[f'{feat}_stats'] = {
  1045→            'mean': float(vals.mean()),
  1046→            'std': float(vals.std()),
  1047→            'min': float(vals.min()),
  1048→            'max': float(vals.max()),
  1049→        }
  1050→
  1051→    return audit
