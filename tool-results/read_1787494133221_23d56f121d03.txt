   267→   267→// ═════════════════════════════════════════════════════════════════════════
   268→   268→// UNDERSTAT ANALYTICAL ENRICHMENT (Phase 1: team xG/xGA only)
   269→   269→// Runs AFTER API-Sports sync so that canonical teams exist in DB.
   270→   270→// ═════════════════════════════════════════════════════════════════════════
   271→   271→
   272→   272→const UNDERSTAT_LEAGUES = ['PL', 'LIGA', 'SA', 'BL', 'L1'] as const
   273→   273→const UNDERSTAT_SEASON = '2024' // Understat uses calendar year for 24/25 season
   274→   274→
   275→   275→async function syncUnderstatAnalytics(): Promise<{
   276→   276→  resolved: number
   277→   277→  unresolved: number
   278→   278→  updated: number
   279→   279→}> {
   280→   280→  const result = { resolved: 0, unresolved: 0, updated: 0 }
   281→   281→
   282→   282→  for (const leagueCode of UNDERSTAT_LEAGUES) {
   283→   283→    checkTimeout()
   284→   284→    try {
   285→   285→      const understatTeams = await fetchUnderstatLeagueTeams(leagueCode, parseInt(UNDERSTAT_SEASON))
   286→   286→      if (understatTeams.length === 0) continue
   287→   287→
   288→   288→      const { resolved, unresolved } = await resolveUnderstatTeams(
   289→   289→        db, understatTeams.map(t => ({ id: t.id, title: t.title, team_name: t.team_name || t.title })), leagueCode, UNDERSTAT_SEASON,
   290→   290→      )
   291→   291→
   292→   292→      result.resolved += resolved.length
   293→   293→      result.unresolved += unresolved.length
   294→   294→
   295→   295→      // Upsert resolved teams' analytics
   296→   296→      // NOTE: The new Understat API does NOT return xG/xGA at team level.
   297→   297→      // We compute per-game xG from match-level data via computeTeamXgFromMatches.
   298→   298→      // This makes one additional fetch per resolved team (rate-limited, so we
   299→   299→      // process only the first 3 resolved teams per league per sync cycle).
   300→   300→      const teamsToProcess = resolved.slice(0, 3)
   301→   301→
   302→   302→      for (const r of teamsToProcess) {
   303→   303→        checkTimeout()
   304→   304→        try {
   305→   305→          // Compute real xG from match history
   306→   306→          const xgData = await computeTeamXgFromMatches(
   307→   307→            r.understatTeamId, parseInt(UNDERSTAT_SEASON),
   308→   308→          )
   309→   309→
   310→   310→          if (!xgData || xgData.matchesPlayed === 0) {
   311→   311→            console.warn(`[SYNC/Understat] No xG data for ${r.matchedDbTeamName} (${r.understatTeamId})`)
   312→   312→            continue
   313→   313→          }
   314→   314→
   315→   315→          const xgPerGame = Math.round((xgData.totalXg / xgData.matchesPlayed) * 100) / 100
   316→   316→          const xgaPerGame = Math.round((xgData.totalXga / xgData.matchesPlayed) * 100) / 100
   317→   317→
   318→   318→          if (xgPerGame < 0 || xgaPerGame < 0) {
   319→   319→            console.warn(`[SYNC/Understat] Invalid xG for ${r.matchedDbTeamName}: xG=${xgPerGame}, xGA=${xgaPerGame}`)
   320→   320→            continue
   321→   321→          }
   322→   322→
   323→   323→          // Link understat identity to canonical team
   324→   324→          const canonicalId = await linkUnderstatToCanonical(
   325→   325→            db, r.dbTeamId, r.understatTeamId,
   326→   326→            r.understatTeamName, r.confidence, r.method,
   327→   327→          )
   328→   328→
   329→   329→          const now = new Date()
   330→   330→          const freshness = classifyFreshness(now)
   331→   331→
   332→   332→          await db.teamAnalytic.upsert({
   333→   333→            where: {
   334→   334→              teamId_source_season_leagueCode: {
   335→   335→                teamId: r.dbTeamId,
   336→   336→                source: 'understat',
   337→   337→                season: UNDERSTAT_SEASON,
   338→   338→                leagueCode,
   339→   339→              },
   340→   340→            },
   341→   341→            update: {
   342→   342→              xgPerGame,
   343→   343→              xgaPerGame,
   344→   344→              npxGPerGame: xgData.matchesPlayed > 0
   345→   345→                ? Math.round((xgData.npxG / xgData.matchesPlayed) * 100) / 100
   346→   346→                : null,
   347→   347→              truthClass: 'REAL',
   348→   348→              dataFreshness: freshness,
   349→   349→              canonicalTeamId: canonicalId,
   350→   350→              syncedAt: now,
   351→   351→            },
   352→   352→            create: {
   353→   353→              teamId: r.dbTeamId,
   354→   354→              source: 'understat',
   355→   355→              season: UNDERSTAT_SEASON,
   356→   356→              leagueCode,
   357→   357→              xgPerGame,
   358→   358→              xgaPerGame,
   359→   359→              npxGPerGame: xgData.matchesPlayed > 0
   360→   360→                ? Math.round((xgData.npxG / xgData.matchesPlayed) * 100) / 100
   361→   361→                : null,
   362→   362→              truthClass: 'REAL',
   363→   363→              dataFreshness: freshness,
   364→   364→              sourceTeamId: String(r.understatTeamId),
   365→   365→              sourceTeamName: r.understatTeamName,
   366→   366→              canonicalTeamId: canonicalId,
   367→   367→              syncedAt: now,
   368→   368→            },
   369→   369→          })
   370→   370→          result.updated++
   371→   371→          console.log(`[SYNC/Understat] ${r.matchedDbTeamName}: xG/g=${xgPerGame}, xGA/g=${xgaPerGame} (${xgData.matchesPlayed} matches, ${r.confidence})`)
   372→   372→        } catch (err) {
   373→   373→          console.warn(`[SYNC/Understat] Failed to upsert for ${r.matchedDbTeamName}:`, err)
   374→   374→        }
   375→   375→      }
   376→   376→
   377→   377→      if (resolved.length > 3) {
   378→   378→        console.log(`[SYNC/Understat] ${leagueCode}: ${resolved.length - 3} teams deferred to next sync cycle (rate limit)`)
   379→   379→      }
   380→   380→
   381→   381→      // Log unresolved teams
   382→   382→      for (const u of unresolved) {
   383→   383→        console.warn(
   384→   384→          `[SYNC/Understat] UNRESOLVED: ${u.understatTeamName} ` +
   385→   385→          `(id=${u.understatTeamId}, league=${leagueCode})`,
   386→   386→        )
   387→   387→      }
   388→   388→    } catch (err: any) {
   389→   389→      if (err.message?.includes('timeout')) throw err
   390→   390→      console.warn(`[SYNC/Understat] Failed for ${leagueCode}:`, err.message)
   391→   391→    }
   392→   392→  }
   393→   393→
   394→   394→  await logSync('understat', 'team_analytics', 'success',
   395→   395→    result.resolved + result.unresolved, result.resolved, result.updated)
   396→   396→
   397→   397→  return result
   398→   398→}

   561→   561→async function fetchScoreboardForLeague(league: { espnId: string; name: string; code: string }) {
   562→   562→  const SITE_V2 = 'https://site.api.espn.com/apis/v2/sports/soccer'
   563→   563→  const url = `${SITE_V2}/${league.espnId}/scoreboard`
   564→   564→  try {
   565→   565→    const res = await fetch(url, { cache: 'no-store' })
   566→   566→    if (!res.ok) return []
   567→   567→    const data = await res.json()
   568→   568→    const events: any[] = []
   569→   569→    for (const event of data.events || []) {
   570→   570→      const comp = event.competitions?.[0]
   571→   571→      if (!comp) continue
   572→   572→      const competitors = comp.competitors || []
   573→   573→      if (competitors.length < 2) continue
   574→   574→      const home = competitors.find((c: any) => c.homeAway === 'home') || competitors[0]
   575→   575→      const away = competitors.find((c: any) => c.homeAway === 'away') || competitors[1]
   576→   576→      const statusDetail = comp.status?.type?.detail || ''
   577→   577→      const minuteMatch = statusDetail.match(/(\d+)\s*'/)
   578→   578→      events.push({
   579→   579→        id: event.id?.toString() || '',
   580→   580→        homeTeamId: home.team?.id?.toString() || '',
   581→   581→        homeTeamName: home.team?.displayName || 'Unknown',
   582→   582→        homeTeamAbbr: home.team?.abbreviation || '???',
   583→   583→        homeTeamLogo: home.team?.logo || '',
   584→   584→        homeTeamColor: home.team?.color || '#00e676',
   585→   585→        awayTeamId: away.team?.id?.toString() || '',
   586→   586→        awayTeamName: away.team?.displayName || 'Unknown',
   587→   587→        awayTeamAbbr: away.team?.abbreviation || '???',
   588→   588→        awayTeamLogo: away.team?.logo || '',
   589→   589→        awayTeamColor: away.team?.color || '#ffffff',
   590→   590→        homeScore: parseInt(home.score) || 0,
   591→   591→        awayScore: parseInt(away.score) || 0,
   592→   592→        status: comp.status?.type?.name || 'STATUS_SCHEDULED',
   593→   593→        date: event.date || '',
   594→   594→        venue: comp.venue?.fullName || '',
   595→   595→        competition: event.name || league.name,
   596→   596→        competitionCode: league.code,
   597→   597→        minute: minuteMatch ? parseInt(minuteMatch[1]) : undefined,
   598→   598→      })
   599→   599→    }
   600→   600→    return events
   601→   601→  } catch (err) {
   602→   602→    console.error(`[SYNC/ESPN] Failed to fetch ${league.espnId}:`, err)
   603→   603→    return []
   604→   604→  }

   952→   952→        await logSync('understat', 'team_analytics', 'partial', 0, 0, 0, err.message)
   953→   953→      } else {
   954→   954→        console.warn('[SYNC] Understat failed:', err.message)
   955→   955→      }
   956→   956→    }
   957→   957→
   958→   958→    // ═══ 4. ESPN Fallback — only if API-Sports didn't return fixtures ═══
   959→   959→    if (!apiSportsAvailable) {
   960→   960→      try {
   961→   961→        const espnResult = await syncFixturesFromESPN()
   962→   962→        summary.syncs.fixtures = {
   963→   963→          source: 'espn',
   964→   964→          created: espnResult.created,
   965→   965→          updated: espnResult.updated,
   966→   966→        }
   967→   967→        await logSync('espn', 'fixtures', 'success', espnResult.processed, espnResult.created, espnResult.updated)
   968→   968→      } catch (err: any) {
   969→   969→        console.error('[SYNC] ESPN fixtures failed:', err)
   970→   970→        await logSync('espn', 'fixtures', 'error', 0, 0, 0, String(err))
   971→   971→        summary.syncs.fixtures.error = String(err)
   972→   972→      }
   973→   973→
   974→   974→      // ESPN Standings fallback
   975→   975→      try {
   976→   976→        const standingsResult = await syncStandingsFromESPN()
   977→   977→        summary.syncs.standings = {
   978→   978→          source: 'espn',
   979→   979→          created: standingsResult.created,
   980→   980→          updated: 0,
   981→   981→        }
   982→   982→        await logSync('espn', 'standings', 'success', standingsResult.created, standingsResult.created, 0)
   983→   983→      } catch (err: any) {
   984→   984→        if (err.message?.includes('timeout')) {
   985→   985→          console.warn('[SYNC] ESPN standings timeout')
   986→   986→          await logSync('espn', 'standings', 'partial', 0, 0, 0, err.message)
   987→   987→        } else {
   988→   988→          console.error('[SYNC] ESPN standings failed:', err)
   989→   989→          await logSync('espn', 'standings', 'error', 0, 0, 0, String(err))
   990→   990→        }
   991→   991→      }
   992→   992→    }
   993→   993→
   994→   994→    summary.durationMs = elapsed()
   995→   995→    console.log(`[SYNC] Complete in ${summary.durationMs}ms`, JSON.stringify(summary.syncs))
   996→   996→
   997→   997→    return NextResponse.json(summary)
   998→   998→  } catch (error) {
   999→   999→    const duration = elapsed()
  1000→  1000→    console.error(`[SYNC] Fatal error after ${duration}ms:`, error)
  1001→  1001→    summary.success = false
  1002→  1002→    summary.durationMs = duration
  1003→  1003→    summary.error = error instanceof Error ? error.message : String(error)
  1004→  1004→    return NextResponse.json(summary, { status: 500 })
  1005→  1005→  }
  1006→  1006→}
