   106→export default function TacticalInsight({ match }: TacticalInsightProps) {
   107→  const [activeTab, setActiveTab] = useState<"network" | "heatmap">("network");

   110→  // Controls / Modifiers
   111→  const [passingStyle, setPassingStyle] = useState<"possession" | "direct" | "counter">("possession");

   113→  const [heatmapType, setHeatmapType] = useState<"touches" | "shots" | "defends">("touches");
   114→  
   115→  // Focus state
   116→  const [selectedPlayer, setSelectedPlayer] = useState<PlayerNode | null>(null);
   117→  const [isD3Simulating, setIsD3Simulating] = useState(true);

   119→  // References
   120→  const d3ContainerRef = useRef<SVGSVGElement | null>(null);

   122→  const getTeamRoster = (teamName: string, isHome: boolean, color: string): PlayerNode[] => {
   123→    const list = defaultTeamRosters[teamName]?.starters || defaultList;
   124→    return list.map((p, idx) => ({
   125→      name: p.name,
   126→      role: p.role,
   127→      posName: p.posName,
   128→      rating: p.rating,
   129→      initialX: p.x,
   130→      initialY: p.y,
   131→      color,
   132→      isHome,
   133→    }));
   134→  };
   135→
   136→  const getRosters = () => {
   137→    const hColor = match.home_jersey_color || "#10b981";
   138→    const aColor = match.away_jersey_color || "#14b8a6";
   139→    const homePlayers = getTeamRoster(match.home_team, true, hColor);
   140→    const awayPlayers = getTeamRoster(match.away_team, false, aColor);
   141→    return { homePlayers, awayPlayers };
   142→  };
   143→
   144→  // Generate mock passing lines based on positions and rating synergy
   145→  const generatePassLinks = (nodes: PlayerNode[]): PassLink[] => {
   146→    const links: PassLink[] = [];
   147→    // Only connect logical neighbors or key patterns to avoid excessive clutter
   148→    for (let i = 0; i < nodes.length; i++) {
   149→      for (let j = i + 1; j < nodes.length; j++) {
   150→        const p1 = nodes[i];
   151→        const p2 = nodes[j];
   152→
   153→        // GK mostly passes only to DEF or DMs
   154→        if (p1.role === "GK" && p2.role === "FWD") continue;
   155→        if (p2.role === "GK" && p1.role === "FWD") continue;
   156→
   157→        const distance = Math.sqrt(
   158→          Math.pow(p1.initialX - p2.initialX, 2) + Math.pow(p1.initialY - p2.initialY, 2)
   159→        );
   160→
   161→        // Standard proximity link or playmaking connection
   162→        if (distance < 45) {
   163→          // Calculate weight
   164→          let baseWeight = (p1.rating + p2.rating) / 40;
   165→          if (passingStyle === "possession") {
   166→            baseWeight *= 1.4;
   167→          } else if (passingStyle === "direct" && (p1.role === "MID" || p1.role === "DEF") && p2.role === "FWD") {
   168→            baseWeight *= 1.8; // Boost direct passes
   169→          } else if (passingStyle === "counter" && distance > 30) {
   170→            baseWeight *= 1.3;
   171→          }
   172→
   173→          // Add a simple deterministic variation
   174→          const hashId = (p1.name.length * p2.name.length) % 5;
   175→          const weight = Math.max(1.5, baseWeight + hashId - (distance / 15));
   176→
   177→          links.push({
   178→            source: p1.name,
   179→            target: p2.name,
   180→            weight: parseFloat(weight.toFixed(1)),
   181→          });
   182→        }
   183→      }
   184→    }
   185→    return links;
   186→  };
   187→
   188→  // Complement Poisson parameters
   189→  // Returns tactical multipliers that visualizes alignment to expected goals
   190→  const getPoissonInfluence = () => {
   191→    const isTargetHome = selectedTeam === "home";
   192→    const baseMu = isTargetHome ? (match.mu_home || 1.35) : (match.mu_away || 1.15);
   193→    
   194→    // Calculate passing index score
   195→    let passVolumeMultiplier = 1.0;
   196→    let attackRatio = 0.5;
   197→
   198→    if (passingStyle === "possession") {
   199→      passVolumeMultiplier = 1.25;
   200→      attackRatio = 0.45;
   201→    } else if (passingStyle === "direct") {
   202→      passVolumeMultiplier = 0.9;
   203→      attackRatio = 0.70;
   204→    } else if (passingStyle === "counter") {
   205→      passVolumeMultiplier = 1.1;
   206→      attackRatio = 0.60;
   207→    }
   208→
   209→    const tempoFactor = 0.8 + (tempo / 150);
   210→    const resultPoissonGoals = baseMu * passVolumeMultiplier * tempoFactor;
   211→    const completenessRatio = Math.min(99, Math.round(70 + (tempo / 5) * passVolumeMultiplier));
   212→
   213→    return {
   214→      compliedGoals: resultPoissonGoals.toFixed(2),
   215→      completenessRatio,
   216→      possessionPct: passingStyle === "possession" ? 58 : passingStyle === "counter" ? 44 : 49,
   217→      directnessIndex: passingStyle === "direct" ? "HI-THRUST" : "ORGANIZED",
   218→    };
   219→  };
   220→
   221→  // Main D3 lifecycle renderer
   222→  useEffect(() => {
   223→    if (!d3ContainerRef.current) return;
   224→
   225→    // Clear previous elements
   226→    d3.select(d3ContainerRef.current).selectAll("*").remove();
   227→
   228→    const svg = d3.select(d3ContainerRef.current);
   229→    const width = 400;
   230→    const height = 500;
   231→
   232→    // Responsive attributes
   233→    svg
   234→      .attr("viewBox", `0 0 ${width} ${height}`)
   235→      .attr("width", "100%")
   236→      .attr("height", "100%");
   237→
   238→    // Fetch players
   239→    const { homePlayers, awayPlayers } = getRosters();
   240→    const currTeamPlayers = selectedTeam === "home" ? homePlayers : awayPlayers;
   241→    
   242→    // Map initial positions onto actual coordinates
   243→    const nodes: PlayerNode[] = currTeamPlayers.map(p => ({
   244→      ...p,
   245→      x: (p.initialX / 100) * width,
   246→      y: (p.initialY / 100) * height,
   247→    }));
   248→
   249→    if (activeTab === "network") {
   250→      // 1. RENDER PASSING NETWORK
   251→      const links = generatePassLinks(nodes);
   252→
   253→      // Link definitions
   254→      const linkGroup = svg.append("g").attr("class", "links-group");
   255→      const nodeGroup = svg.append("g").attr("class", "nodes-group");
   256→
   257→      // Draw links
   258→      const linkElements = linkGroup
   259→        .selectAll("line")
   260→        .data(links)
   261→        .enter()
   262→        .append("line")
   263→        .attr("stroke", selectedTeam === "home" ? "#10b981" : "#14b8a6")
   264→        .attr("stroke-opacity", (d) => {
   265→          if (selectedPlayer) {
   266→            const isRelated = d.source === selectedPlayer.name || d.target === selectedPlayer.name;
   267→            return isRelated ? 0.85 : 0.08;
   268→          }
   269→          return 0.35;
   270→        })
   271→        .attr("stroke-width", (d) => Math.max(1, d.weight * 0.95))
   272→        .attr("class", "transition-all duration-300");
   273→
   274→      // Draw node g wrappers
   275→      const nodeElements = nodeGroup
   276→        .selectAll("g")
   277→        .data(nodes)
   278→        .enter()
   279→        .append("g")
   280→        .attr("class", "cursor-pointer group")
   281→        .on("click", (event, d) => {
   282→          triggerHaptic("light");
   283→          setSelectedPlayer(d === selectedPlayer ? null : d);
   284→        });
   285→
   286→      // Nodes glows
   287→      nodeElements
   288→        .append("circle")
   289→        .attr("r", (d) => (selectedPlayer?.name === d.name ? 16 : 11))
   290→        .attr("fill", "#090d16")
   291→        .attr("stroke", (d) => d.color)
   292→        .attr("stroke-width", (d) => (selectedPlayer?.name === d.name ? 3 : 1.5))
   293→        .attr("class", "transition-all duration-300 shadow-xl");
   294→
   295→      // Inner glowing core
   296→      nodeElements
   297→        .append("circle")
   298→        .attr("r", 3)
   299→        .attr("fill", (d) => d.color)
   300→        .attr("opacity", 0.7);
   301→
   302→      // Player names inside network
   303→      nodeElements
   304→        .append("text")
   305→        .text((d) => `${d.posName} ${d.name.split(" ").pop()}`)
   306→        .attr("dx", 0)
   307→        .attr("dy", (d) => (selectedPlayer?.name === d.name ? -22 : -15))
   308→        .attr("text-anchor", "middle")
   309→        .attr("font-size", "7.5px")
   310→        .attr("font-family", "monospace")
   311→        .attr("fill", (d) => (selectedPlayer?.name === d.name ? "#ffffff" : "#64748b"))
   312→        .attr("font-weight", (d) => (selectedPlayer?.name === d.name ? "bold" : "normal"));
   313→
   314→      // Set force-directed simulation
   315→      if (isD3Simulating) {
   316→        const simulation = d3
   317→          .forceSimulation<PlayerNode>(nodes)
   318→          .force("link", d3.forceLink<PlayerNode, PassLink>(links).id((d) => d.name).distance(75))

   320→          .force("center", d3.forceCenter(width / 2, height / 2))
   321→          .force("collide", d3.forceCollide(25));

   324→        simulation.on("tick", () => {
   325→          linkElements
   326→            .attr("x1", (d) => (d.source as PlayerNode).x ?? 0)
   327→            .attr("y1", (d) => (d.source as PlayerNode).y ?? 0)

   329→            .attr("y2", (d) => (d.target as PlayerNode).y ?? 0);
   330→
   331→          nodeElements.attr("transform", (d) => `translate(${d.x ?? 0}, ${d.y ?? 0})`);
   332→        });

   335→        nodeElements.call(
   336→          d3.drag<SVGGElement, PlayerNode>()
   337→            .on("start", (event, d) => {
   338→              if (!event.active) simulation.alphaTarget(0.2).restart();
   339→              d.fx = d.x;
   340→              d.fy = d.y;
   341→            })
   342→            .on("drag", (event, d) => {
   343→              d.fx = event.x;
   344→              d.fy = event.y;
   345→            })
   346→            .on("end", (event, d) => {
   347→              if (!event.active) simulation.alphaTarget(0);
   348→              d.fx = null;
   349→              d.fy = null;
   350→            })
   351→        );
   352→      } else {
   353→        // Direct tactical projection view without simulation

   355→        linkElements
   356→          .attr("x1", (d) => {
   357→            const srcNode = nodes.find(n => n.name === d.source);
   358→            return srcNode ? srcNode.x ?? 0 : 0;
   359→          })
   360→          .attr("y1", (d) => {
   361→            const srcNode = nodes.find(n => n.name === d.source);
   362→            return srcNode ? srcNode.y ?? 0 : 0;
   363→          })
   364→          .attr("x2", (d) => {
   365→            const targetNode = nodes.find(n => n.name === (d.target as any).name || n.name === d.target);
   366→            return targetNode ? targetNode.x ?? 0 : 0;
   367→          })
   368→          .attr("y2", (d) => {
   369→            const targetNode = nodes.find(n => n.name === (d.target as any).name || n.name === d.target);
   370→            return targetNode ? targetNode.y ?? 0 : 0;
   371→          });
   372→      }
   373→
   374→    } else {
   375→      // 2. RENDER SPATIAL HEATMAP USING D3 GRID INJECTIONS
   376→      const heatmapGroup = svg.append("g").attr("class", "heatmap-group");
   377→      
   378→      // Determine hot spots coordinates depending on the selected heatmapType
   379→      const totalPoints = 90;

   392→          const cx = pl.x + Math.cos(angle) * r;
   393→          const cy = pl.y + Math.sin(angle) * r + offsetForward;
   394→
   395→          // Constraints to stay inside field boundaries
   396→          const finalX = Math.max(15, Math.min(width - 15, cx));
   397→          const finalY = Math.max(15, Math.min(height - 15, cy));
   398→
   399→          // Base factor based on stats
   400→          let w = pl.rating / 100;
   401→          if (heatmapType === "shots" && pl.role !== "FWD") w *= 0.15;
   402→          if (heatmapType === "defends" && pl.role !== "DEF") w *= 0.35;
   403→
   404→          points.push({ cx: finalX, cy: finalY, weight: w });
   405→        }
   406→      });
   407→
   408→      // Define radial color gradients
   409→      const defs = svg.append("defs");
   410→      const grad = defs
   411→        .append("radialGradient")
   412→        .attr("id", "heat-radial")
   413→        .attr("cx", "50%")
   414→        .attr("cy", "50%")
   415→        .attr("r", "50%");
   416→
   417→      const teamColor = selectedTeam === "home" ? "#10b981" : "#14b8a6";
   418→
   419→      grad.append("stop").attr("offset", "0%").attr("stop-color", teamColor).attr("stop-opacity", 0.45);
   420→      grad.append("stop").attr("offset", "100%").attr("stop-color", teamColor).attr("stop-opacity", 0.0);
   421→
   422→      // Render glowing soft heat points
   423→      heatmapGroup
   424→        .selectAll("circle")
   425→        .data(points)
   426→        .enter()
   427→        .append("circle")
   428→        .attr("cx", (d) => d.cx)
   429→        .attr("cy", (d) => d.cy)
   430→        .attr("r", (d) => Math.max(20, d.weight * 52))
   431→        .attr("fill", "url(#heat-radial)")
   432→        .attr("class", "animate-pulse transition-all duration-[3000ms]")
   433→        .attr("opacity", 0.8);
   434→
   435→      // Render smaller dense hot-point anchors
   436→      heatmapGroup
   437→        .selectAll(".core-point")
   438→        .data(points.filter((_, i) => i % 3 === 0))
   439→        .enter()
   440→        .append("circle")
   441→        .attr("class", "core-point")
   442→        .attr("cx", (d) => d.cx)
   443→        .attr("cy", (d) => d.cy)
   444→        .attr("r", 1.5)
   445→        .attr("fill", teamColor)
   446→        .attr("opacity", 0.25);
   447→        
   448→      // Overlay player tags lightly in heatmap
   449→      const pinsGroup = svg.append("g").attr("class", "pins-group");
   450→      pinsGroup
   451→        .selectAll("g")
   452→        .data(nodes)
   453→        .enter()
   454→        .append("g")
   455→        .attr("transform", (d) => `translate(${d.x}, ${d.y})`)
   456→        .append("text")
   457→        .text((d) => d.posName)
