    44→function computeEloProb(homeElo: number, awayElo: number, outcome: string): number {
    45→  const expectedHome = 1 / (1 + Math.pow(10, (awayElo - homeElo) / 400))
    46→  if (outcome === 'home') return Math.round(expectedHome * 100)
    47→  if (outcome === 'away') return Math.round((1 - expectedHome) * 100 * 0.75)
    48→  return Math.round(100 - expectedHome * 100 - (1 - expectedHome) * 100 * 0.75)
    49→}

    51→function StatBarRow({ label, homeValue, awayValue, homeColor, awayColor, suffix = '', decimals = 0 }: { label: string; homeValue: number; awayValue: number; homeColor?: string; awayColor?: string; suffix?: string; decimals?: number }) {
    52→  const total = homeValue + awayValue || 1
    53→  const hp = (homeValue / total) * 100
    54→  const fmt = (v: number) => decimals > 0 ? v.toFixed(decimals) : String(Math.round(v))
    55→  return (
    56→    <div className="space-y-1.5">
    57→      <div className="flex items-center justify-between text-xs">
    58→        <span className={cn('font-semibold tabular-nums', hp >= 50 ? 'text-foreground' : 'text-muted-foreground')}>{fmt(homeValue)}{suffix}</span>
    59→        <span className="text-[10px] text-muted-foreground font-medium uppercase tracking-wider">{label}</span>
    60→        <span className={cn('font-semibold tabular-nums', (100 - hp) > hp ? 'text-foreground' : 'text-muted-foreground')}>{fmt(awayValue)}{suffix}</span>
    61→      </div>
    62→      <div className="flex h-2 gap-0.5 rounded-full overflow-hidden bg-muted">
    63→        <div className="h-full rounded-full transition-all duration-700 ease-out" style={{ width: `${hp}%`, backgroundColor: homeColor || 'oklch(0.72 0.19 163)' }} />
    64→        <div className="h-full rounded-full transition-all duration-700 ease-out ml-auto" style={{ width: `${100 - hp}%`, backgroundColor: awayColor || 'oklch(0.68 0.15 230)' }} />
    65→      </div>
    66→    </div>
    67→  )
    68→}

    96→// Fallback: generate shot positions from match events when no deep data source
    97→function generateShotsFromEvents(events: MatchEvent[], homeXg: number, awayXg: number): ShotMapPoint[] {
    98→  const shots: ShotMapPoint[] = []
    99→  const homeGoals = events.filter(e => e.type === 'goal' && e.team === 'home')
   100→  const awayGoals = events.filter(e => e.type === 'goal' && e.team === 'away')
   101→  for (let i = 0; i < Math.ceil(homeXg * 3); i++) {
   102→    shots.push({ x: 15 + (i * 17 % 30), y: 10 + (i * 23 % 80), team: 'home', goal: i < homeGoals.length })
   103→  }
   104→  for (let i = 0; i < Math.ceil(awayXg * 3); i++) {
   105→    shots.push({ x: 55 + (i * 17 % 30), y: 10 + (i * 23 % 80), team: 'away', goal: i < awayGoals.length })
   106→  }
   107→  return shots
   108→}
