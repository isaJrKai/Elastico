   115→// ── Helper: format date ───────────────────────────────────────────────────────
   116→
   117→function formatDate(dateStr: string | null): string {
   118→  if (!dateStr) return 'TBD'
   119→  const d = new Date(dateStr)
   120→  const now = new Date()
   121→  const diffMs = d.getTime() - now.getTime()
   122→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24))
   123→
   124→  if (diffDays === 0) return 'Today'
   125→  if (diffDays === 1) return 'Tomorrow'
   126→  if (diffDays === -1) return 'Yesterday'
   127→  if (diffDays > 0 && diffDays <= 7) return `In ${diffDays} days`
   128→
   129→  return d.toLocaleDateString('en-US', {
   130→    month: 'short',
   131→    day: 'numeric',
   132→    hour: '2-digit',
   133→    minute: '2-digit',
   134→  })
   135→}
   136→
   137→function formatTimeAgo(dateStr: string | null): string {
   138→  if (!dateStr) return ''
   139→  const d = new Date(dateStr)
   140→  const now = new Date()
   141→  const diffMs = now.getTime() - d.getTime()
   142→  const diffMins = Math.floor(diffMs / 60000)
   143→  const diffHours = Math.floor(diffMs / 3600000)
   144→  const diffDays = Math.floor(diffMs / 86400000)
   145→
   146→  if (diffMins < 1) return 'Just now'
   147→  if (diffMins < 60) return `${diffMins}m ago`
   148→  if (diffHours < 24) return `${diffHours}h ago`
   149→  if (diffDays < 7) return `${diffDays}d ago`
   150→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   151→}

   186→function CustomTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ value: number; name: string; color: string }>; label?: string }) {
   187→  if (!active || !payload?.length) return null
   188→  return (
   189→    <div className="glass-card rounded-lg p-3 text-sm shadow-lg border border-border/50">
   190→      <p className="text-muted-foreground text-xs mb-1">{label}</p>
   191→      {payload.map((entry, i) => (
   192→        <p key={i} className="font-semibold" style={{ color: entry.color }}>
   193→          {entry.name}: {typeof entry.value === 'number' ? entry.value.toFixed(1) : entry.value}
   194→        </p>
   195→      ))}
   196→    </div>
   197→  )
   198→}
