   164→function NvidiaApiStatusBadge({ token }: { token?: string }) {
   165→  const [status, setStatus] = React.useState<'loading' | 'connected' | 'disconnected'>('loading')
   166→
   167→  React.useEffect(() => {
   168→    fetch('/api/chat', {
   169→      method: 'POST',
   170→      headers: {
   171→        'Content-Type': 'application/json',
   172→        ...(token ? { Authorization: `Bearer ${token}` } : {}),
   173→      },
   174→      body: JSON.stringify({ message: '__status_check__', model: 'pro', stream: false }),
   175→    })
   176→      .then((res) => res.json())
   177→      .then((data) => {
   178→        if (data.model === 'pro') {
   179→          setStatus('connected')
   180→        } else if (data.model === 'mock-fallback') {
   181→          setStatus('disconnected')
   182→        } else {
   183→          setStatus('disconnected')
   184→        }
   185→      })
   186→      .catch(() => setStatus('disconnected'))
   187→  }, [])
   188→
   189→  if (status === 'loading') {
   190→    return (
   191→      <Badge variant="outline" className="gap-1.5 text-xs border-border">
   192→        <Loader2 className="size-3 animate-spin text-muted-foreground" />
   193→        Checking...
   194→      </Badge>
   195→    )
   196→  }
   197→
   198→  if (status === 'connected') {
   199→    return (
   200→      <Badge variant="outline" className="gap-1.5 text-xs border-emerald-500/30 text-emerald-400 bg-emerald-500/5">
   201→        <CheckCircle2 className="size-3" />
   202→        Connected
   203→      </Badge>
   204→    )
   205→  }
   206→
   207→  return (
   208→    <Badge variant="outline" className="gap-1.5 text-xs border-amber-500/30 text-amber-400 bg-amber-500/5">
   209→      <XCircle className="size-3" />
   210→      Not Configured
   211→    </Badge>
   212→  )
   213→}

   263→function BandwidthSection() {
   264→  const [bandwidth, setBandwidth] = useState<{ totalKB: number; requests: number; avgBytesPerRequest: number } | null>(null)
   265→  const { isOffline, isRegistered, isStandalone, clearCache } = usePWA()
   266→
   267→  useEffect(() => {
   268→    // Poll bandwidth tracker every 2 seconds
   269→    const load = async () => {
   270→      try {
   271→        const mod = await import('@/lib/compressed-data-stream')
   272→        setBandwidth(mod.BandwidthTracker.summary())
   273→      } catch {}
   274→    }
   275→    load()
   276→    const interval = setInterval(load, 2000)
   277→    return () => clearInterval(interval)
   278→  }, [])
   279→
   280→  return (
   281→    <Card className="glass-card border-border">
   282→      <CardHeader className="pb-4">
   283→        <SectionHeader
   284→          icon={Radio}
   285→          title="Bandwidth & Offline"
   286→          description="PWA caching & data usage monitoring"
   287→        />
   288→      </CardHeader>
   289→      <CardContent className="space-y-4">
   290→        {/* PWA Status Grid */}
   291→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   292→          <StatusPill
   293→            label="Network"
   294→            value={isOffline ? 'Offline' : 'Online'}
   295→            active={!isOffline}
   296→            icon={isOffline ? WifiOff : Wifi}
   297→          />
   298→          <StatusPill
   299→            label="App Cached"
   300→            value={isRegistered ? 'Yes' : 'No'}
   301→            active={isRegistered}
   302→            icon={Smartphone}
   303→          />
   304→          <StatusPill
   305→            label="Standalone"
   306→            value={isStandalone ? 'Yes' : 'No'}
   307→            active={isStandalone}
   308→            icon={Monitor}
   309→          />
   310→          <StatusPill
   311→            label="Data Used"
   312→            value={bandwidth ? `${bandwidth.totalKB.toFixed(1)} KB` : '—'}
   313→            active={bandwidth ? bandwidth.totalKB < 100 : true}
   314→            icon={Zap}
   315→          />
   316→        </div>
   317→
   318→        {/* Bandwidth Details */}
   319→        {bandwidth && (
   320→          <div className="rounded-lg border border-border bg-secondary/30 p-4 space-y-3">
   321→            <div className="flex items-center justify-between text-sm">
   322→              <span className="text-muted-foreground">API Requests</span>
   323→              <span className="font-mono text-foreground">{bandwidth.requests}</span>
   324→            </div>
   325→            <div className="flex items-center justify-between text-sm">
   326→              <span className="text-muted-foreground">Avg per Request</span>
   327→              <span className="font-mono text-foreground">{bandwidth.avgBytesPerRequest} B</span>
   328→            </div>
   329→            <div className="flex items-center justify-between text-sm">
   330→              <span className="text-muted-foreground">Total Today</span>
   331→              <span className={cn(
   332→                'font-mono font-medium',
   333→                bandwidth.totalKB < 50 ? 'text-emerald-400' : bandwidth.totalKB < 200 ? 'text-amber-400' : 'text-red-400'
   334→              )}>
   335→                {bandwidth.totalKB.toFixed(1)} KB
   336→              </span>
   337→            </div>
   338→          </div>
   339→        )}
   340→
   341→        {/* Compression Features */}
   342→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   343→          {[
   344→            { label: 'Null Stripping', desc: 'Removes 30-60% empty fields' },
   345→            { label: 'Key Compaction', desc: '2-char keys in compact mode' },
   346→            { label: 'Service Worker', desc: '0 MB data on app reopen' },
   347→            { label: 'Diff Updates', desc: 'Only changed data sent' },
   348→          ].map((f) => (
   349→            <div key={f.label} className="flex items-center gap-2.5 rounded-md border border-border bg-secondary/20 px-3 py-2">
   350→              <CheckCircle2 className="size-3.5 text-primary shrink-0" />

   352→                <p className="text-xs font-medium text-foreground">{f.label}</p>
   353→                <p className="text-[10px] text-muted-foreground">{f.desc}</p>
   354→              </div>
   355→            </div>
   356→          ))}
   357→        </div>
   358→
   359→        {/* Clear Cache Button */}
   360→        {isRegistered && (
   361→          <Button
   362→            variant="outline"
   363→            size="sm"
   364→            className="w-full border-border text-xs"
   365→            onClick={async () => {
   366→              await clearCache()

   369→          >
   370→            <Trash2 className="size-3.5 mr-2" />

   378→            After the first visit, ELASTICO caches itself (~5 MB) to your device storage.
   379→            Subsequent opens load instantly from your phone — using <strong className="text-foreground">0 MB mobile data</strong>.
   380→            Live match updates use compressed diff payloads under 5 KB per cycle.
