    87→
    88→export function ExportView() {
    89→  const token = useElasticoStore(s => s.token)
    90→  const user = useElasticoStore(s => s.user)
    91→  const [exportType, setExportType] = useState('matches')
    92→  const [exportFormat, setExportFormat] = useState('csv')
    93→  const [exporting, setExporting] = useState(false)
    94→  const [exportHistory, setExportHistory] = useState<ExportRecord[]>([])
    95→  const [scheduledReports] = useState<{ id: string; name: string; frequency: string; type: string; format: string; active: boolean; nextRun: string }[]>([])
    96→  const [chartRef, setChartRef] = useState<HTMLElement | null>(null)
    97→
    98→  const handleExport = useCallback(async () => {
    99→    setExporting(true)
   100→    try {
   101→      const headers: Record<string, string> = { 'Content-Type': 'application/json' }
   102→      if (token) headers['Authorization'] = `Bearer ${token}`
   103→
   104→      const res = await fetch('/api/export', {
   105→        method: 'POST',
   106→        headers,
   107→        body: JSON.stringify({ type: exportType, format: exportFormat, filters: {} }),
   108→      })
   109→
   110→      if (!res.ok) throw new Error('Export failed')
   111→
   112→      if (exportFormat === 'csv' && res.headers.get('content-type')?.includes('text/csv')) {
   113→        const blob = await res.blob()
   114→        const url = URL.createObjectURL(blob)
   115→        const a = document.createElement('a')
   116→        a.href = url
   117→        a.download = `elastico-${exportType}-${Date.now()}.csv`
   118→        a.click()
   119→        URL.revokeObjectURL(url)
   120→      } else {
   121→        const data = await res.json()
   122→        const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' })
   123→        const url = URL.createObjectURL(blob)
   124→        const a = document.createElement('a')
   125→        a.href = url
   126→        a.download = `elastico-${exportType}-${Date.now()}.json`
   127→        a.click()
   128→        URL.revokeObjectURL(url)
   129→      }
   130→
   131→      const newRecord: ExportRecord = {
   132→        id: `e${Date.now()}`,
   133→        type: exportType.charAt(0).toUpperCase() + exportType.slice(1),
   134→        format: exportFormat.toUpperCase(),
   135→        rows: 0,
   136→        createdAt: new Date().toLocaleString(),
   137→        status: 'completed',
   138→      }
   139→      setExportHistory(prev => [newRecord, ...prev])
   140→      toast.success(`Exported ${exportType} as ${exportFormat.toUpperCase()}`)
   141→    } catch {
   142→      toast.error('Export failed. Please try again.')
   143→      setExportHistory(prev => [{
   144→        id: `e${Date.now()}`,
   145→        type: exportType,
   146→        format: exportFormat.toUpperCase(),
   147→        rows: 0,
   148→        createdAt: new Date().toLocaleString(),
   149→        status: 'failed',
   150→      }, ...prev])
   151→    } finally {
   152→      setExporting(false)
   153→    }
   154→  }, [exportType, exportFormat, token])
   155→
   156→  const handleReportTemplate = useCallback((template: typeof REPORT_TEMPLATES[0]) => {
   157→    toast.info(`Generating ${template.name}...`)
   158→    // Use browser print as PDF generation
   159→    setTimeout(() => {
   160→      window.print()
   161→      toast.success(`${template.name} generated!`)
   162→    }, 500)
   163→  }, [])
   164→
   165→  const handleExportChartAsPNG = useCallback(() => {
   166→    if (!chartRef) {
   167→      toast.error('No chart to export. Visit a chart view first.')
   168→      return
   169→    }
   170→    toast.info('Chart export uses the browser Print dialog. Use "Save as PDF" to capture charts.')
   171→    window.print()
   172→  }, [chartRef])
   173→
   174→  const handleShareReport = useCallback((record: ExportRecord) => {
   175→    const link = `${window.location.origin}/shared/report/${record.id}`
   176→    if (navigator.clipboard?.writeText) {
   177→      navigator.clipboard.writeText(link)
   178→    } else {
   179→      const ta = document.createElement('textarea'); ta.value = link; document.body.appendChild(ta); ta.select(); document.execCommand('copy'); document.body.removeChild(ta)
   180→    }
   181→    toast.success('Shareable link copied to clipboard!')
   182→  }, [])
   183→
   184→  const handleCopyEndpoint = useCallback((path: string) => {
   185→    const fullUrl = `${window.location.origin}${path}`
   186→    if (navigator.clipboard?.writeText) {
   187→      navigator.clipboard.writeText(fullUrl)
   188→    } else {
   189→      const ta = document.createElement('textarea'); ta.value = fullUrl; document.body.appendChild(ta); ta.select(); document.execCommand('copy'); document.body.removeChild(ta)
   190→    }
   191→    toast.success('API endpoint URL copied!')
   192→  }, [])
   193→
   194→  return (
   195→    <div
   196→      className="space-y-6"
   197→    >
   198→      {/* Header */}
   199→      <div>
   200→        <p className="text-muted-foreground text-sm">Export data, generate reports, and access the API</p>
   201→      </div>
   202→
   203→      <Tabs defaultValue="custom" className="space-y-4">
   204→        <TabsList className="flex flex-wrap gap-1 h-auto bg-card/50 p-1 rounded-lg">
   205→          <TabsTrigger value="templates" className="text-xs">Templates</TabsTrigger>
   206→          <TabsTrigger value="custom" className="text-xs">Custom Export</TabsTrigger>
   207→          <TabsTrigger value="scheduled" className="text-xs">Scheduled</TabsTrigger>
   208→          <TabsTrigger value="history" className="text-xs">History</TabsTrigger>
   209→          <TabsTrigger value="charts" className="text-xs">Chart Export</TabsTrigger>
   210→          <TabsTrigger value="api" className="text-xs">API Access</TabsTrigger>
   211→          <TabsTrigger value="bulk" className="text-xs">Bulk Export</TabsTrigger>
   212→          <TabsTrigger value="share" className="text-xs">Share</TabsTrigger>
   213→        </TabsList>
   214→
   215→        {/* Report Templates */}
   216→        <TabsContent value="templates">
   217→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   218→            {REPORT_TEMPLATES.map((template) => (
   219→              <div
   220→                key={template.id}
   221→                className="glass-card p-6 rounded-xl border border-border/30 cursor-pointer group transition-all duration-200 hover:-translate-y-1"
   222→                onClick={() => handleReportTemplate(template)}
   223→              >
   224→                {(() => { const Icon = ICON_MAP[template.icon] || FileText; return <Icon className="size-8 text-primary/60 mb-3" /> })()}
   225→                <h3 className="text-sm font-semibold group-hover:text-primary transition-colors">{template.name}</h3>
   226→                <p className="text-[11px] text-muted-foreground mt-2 leading-relaxed">{template.description}</p>
   227→                <div className="flex items-center gap-2 mt-4">
   228→                  <Badge variant="outline" className="text-[9px]">{template.category}</Badge>
   229→                  <span className="text-[10px] text-muted-foreground ml-auto flex items-center gap-1">
   230→                    <FileDown className="size-3" /> PDF
   231→                  </span>
   232→                </div>
   233→              </div>
   234→            ))}
   235→          </div>
   236→        </TabsContent>
   237→
   238→        {/* Custom Report Builder */}
   239→        <TabsContent value="custom">
   240→          <Card className="glass-card">
   241→            <CardHeader className="pb-2">
   242→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   243→                <FileSpreadsheet className="size-4 text-primary" /> Custom Export Builder
   244→              </CardTitle>
   245→            </CardHeader>
   246→            <CardContent className="space-y-6">
   247→              <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
   248→                <div className="space-y-2">
   249→                  <Label className="text-xs">Data Type</Label>
   250→                  <Select value={exportType} onValueChange={setExportType}>
   251→                    <SelectTrigger>
   252→                      <SelectValue />
   253→                    </SelectTrigger>
   254→                    <SelectContent>
   255→                      <SelectItem value="matches">Matches</SelectItem>
   256→                      <SelectItem value="players">Players</SelectItem>
   257→                      <SelectItem value="predictions">My Predictions</SelectItem>
   258→                      <SelectItem value="teams">Teams</SelectItem>
   259→                    </SelectContent>
   260→                  </Select>
   261→                </div>
   262→                <div className="space-y-2">
   263→                  <Label className="text-xs">Export Format</Label>
   264→                  <Select value={exportFormat} onValueChange={setExportFormat}>
   265→                    <SelectTrigger>
   266→                      <SelectValue />
   267→                    </SelectTrigger>
   268→                    <SelectContent>
   269→                      <SelectItem value="csv">CSV</SelectItem>
   270→                      <SelectItem value="json">JSON</SelectItem>
   271→                    </SelectContent>
   272→                  </Select>
   273→                </div>
   274→                <div className="space-y-2">
   275→                  <Label className="text-xs">Date Range</Label>
   276→                  <Input type="text" placeholder="All dates" disabled />
   277→                </div>
   278→              </div>
   279→
   280→              <div className="flex flex-col sm:flex-row items-start sm:items-center gap-3">
   281→                <Button onClick={handleExport} disabled={exporting} className="gap-2">
   282→                  {exporting ? <Loader2 className="size-4 animate-spin" /> : <Download className="size-4" />}
   283→                  {exporting ? 'Exporting...' : `Export ${exportType} as ${exportFormat.toUpperCase()}`}
   284→                </Button>
   285→                <span className="text-xs text-muted-foreground">
   286→                  {exportType === 'matches' && 'Includes scores, xG, possession, shots, and team data'}
   287→                  {exportType === 'players' && 'Includes stats, ratings, market values, and team info'}
   288→                  {exportType === 'predictions' && 'Includes your predictions with outcomes and accuracy'}
   289→                  {exportType === 'teams' && 'Includes ELO, xG, style, and squad data'}
   290→                </span>
   291→              </div>
   292→            </CardContent>
   293→          </Card>
   294→        </TabsContent>
   295→
   296→        {/* Scheduled Reports */}
   297→        <TabsContent value="scheduled">
   298→          <Card className="glass-card">
   299→            <CardHeader className="pb-2">
   300→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   301→                <Clock className="size-4 text-primary" /> Scheduled Reports
   302→              </CardTitle>
   303→            </CardHeader>
   304→            <CardContent className="space-y-3">
   305→              {scheduledReports.length === 0 ? (
   306→                <div className="text-center py-12 text-muted-foreground">
   307→                  <Clock className="size-12 mx-auto mb-3 opacity-30" />
   308→                  <p className="text-sm">No scheduled reports yet.</p>
   309→                  <p className="text-xs text-muted-foreground/70 mt-1">Scheduled report automation is coming soon.</p>
   310→                </div>
   311→              ) : (
   312→                scheduledReports.map(report => (
   313→                  <div key={report.id} className="flex items-center justify-between p-4 rounded-lg bg-card/50 border border-border/30">
   314→                    <div className="flex items-center gap-3">
   315→                      <div className={cn('w-3 h-3 rounded-full', report.active ? 'bg-primary' : 'bg-muted-foreground/30')} />
   316→                      <div>
   317→                        <div className="text-sm font-medium">{report.name}</div>
   318→                        <div className="text-[10px] text-muted-foreground">
   319→                          {report.frequency} · {report.type} · {report.format} · Next: {report.nextRun}
   320→                        </div>
   321→                      </div>
   322→                    </div>
   323→                    <div className="flex items-center gap-2">
   324→                      <Badge variant="outline" className={cn('text-[10px]', report.active ? 'text-primary border-primary/30' : 'text-muted-foreground')}>
   325→                        {report.active ? 'Active' : 'Paused'}
   326→                      </Badge>
   327→                    </div>
   328→                  </div>
   329→                ))
   330→              )}
   331→              <Button variant="outline" className="w-full gap-2 text-xs" disabled>
   332→                <Calendar className="size-4" /> Schedule New Report
   333→              </Button>
   334→            </CardContent>
   335→          </Card>
   336→        </TabsContent>
   337→
   338→        {/* Export History */}
   339→        <TabsContent value="history">
   340→          <Card className="glass-card">
   341→            <CardHeader className="pb-2">
   342→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   343→                <History className="size-4 text-primary" /> Export History
   344→              </CardTitle>
   345→            </CardHeader>
   346→            <CardContent>
   347→              <div className="space-y-2 max-h-96 overflow-y-auto">
   348→                {exportHistory.length === 0 ? (
   349→                  <div className="text-center py-12 text-muted-foreground">
   350→                    <History className="size-12 mx-auto mb-3 opacity-30" />
   351→                    <p className="text-sm">No exports yet.</p>
   352→                    <p className="text-xs text-muted-foreground/70 mt-1">Use the Custom Export tab to generate your first export.</p>
   353→                  </div>
   354→                ) : exportHistory.map(record => (
   355→                  <div key={record.id} className="flex items-center gap-3 p-3 rounded-lg bg-card/50 border border-border/30">
   356→                    {record.status === 'completed' ? (
   357→                      <CheckCircle2 className="size-4 text-primary shrink-0" />
   358→                    ) : (
   359→                      <AlertCircle className="size-4 text-red-400 shrink-0" />
   360→                    )}
   361→                    <div className="flex-1 min-w-0">
   362→                      <div className="text-sm font-medium">{record.type} Export</div>
   363→                      <div className="text-[10px] text-muted-foreground">
   364→                        {record.format} · {record.rows} rows · {record.createdAt}
   365→                        {record.size && ` · ${record.size}`}
   366→                      </div>
   367→                    </div>
   368→                    <Badge variant="outline" className={cn(
   369→                      'text-[9px] shrink-0',
   370→                      record.status === 'completed' ? 'text-primary border-primary/30' : 'text-red-400 border-red-400/30'
   371→                    )}>
   372→                      {record.status}
   373→                    </Badge>
   374→                    {record.status === 'completed' && (
   375→                      <Button variant="ghost" size="icon" className="h-7 w-7 shrink-0" onClick={() => handleShareReport(record)} aria-label="Share report">
   376→                        <Share2 className="size-3" />
   377→                      </Button>
   378→                    )}
   379→                  </div>
   380→                ))}
   381→              </div>
   382→            </CardContent>
   383→          </Card>
   384→        </TabsContent>
   385→
   386→        {/* Chart Export */}
   387→        <TabsContent value="charts">
   388→          <Card className="glass-card">
   389→            <CardHeader className="pb-2">
   390→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   391→                <FileDown className="size-4 text-primary" /> Data Visualization Export
   392→              </CardTitle>
   393→            </CardHeader>
   394→            <CardContent className="space-y-4">
   395→              <p className="text-sm text-muted-foreground">
   396→                Export any chart or visualization as a high-quality image using your browser&apos;s built-in capabilities.
   397→              </p>
   398→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   399→                {[
   400→                  { name: 'Match xG Chart', view: 'tactical' },
   401→                  { name: 'Player Radar Chart', view: 'players' },
   402→                  { name: 'Team Comparison', view: 'compare' },
   403→                  { name: 'Leaderboard Chart', view: 'leaderboard' },
   404→                  { name: 'Tournament Standings', view: 'tournament' },
   405→                  { name: 'Prediction Accuracy', view: 'predictions' },
   406→                ].map(chart => (
   407→                  <div
   408→                    key={chart.name}
   409→                    className="glass-card p-4 rounded-lg border border-border/30 cursor-pointer transition-all duration-200 hover:-translate-y-1"
   410→                    onClick={() => toast.info(`Navigate to ${chart.name} view, then use Ctrl+P to save as PDF`)}
   411→                  >
   412→                    <BarChartIcon className="size-5 text-primary mb-2" />
   413→                    <h4 className="text-sm font-medium">{chart.name}</h4>
   414→                    <p className="text-[10px] text-muted-foreground mt-1">Click to navigate, then print to save</p>
   415→                  </div>
   416→                ))}
   417→              </div>
   418→              <Button variant="outline" onClick={handleExportChartAsPNG} className="gap-2">
   419→                <FileDown className="size-4" /> Export Current View as PDF
   420→              </Button>
   421→            </CardContent>
   422→          </Card>
   423→        </TabsContent>
   424→
   425→        {/* API Access */}
   426→        <TabsContent value="api">
   427→          <Card className="glass-card">
   428→            <CardHeader className="pb-2">
   429→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   430→                <Code2 className="size-4 text-primary" /> API Documentation
   431→              </CardTitle>
   432→              <p className="text-xs text-muted-foreground">
   433→                Programmatic access to ELASTICO data. Use Bearer token authentication for protected endpoints.
   434→              </p>
   435→            </CardHeader>
   436→            <CardContent>
   437→              {/* Auth info */}
   438→              <div className="glass-card p-3 rounded-lg border border-border/30 mb-4">
   439→                <div className="text-xs font-medium mb-1">Authentication</div>
   440→                <code className="text-[11px] text-primary bg-primary/5 px-2 py-1 rounded block overflow-x-auto">
   441→                  Authorization: Bearer {'{'}token{'}'}
   442→                </code>
   443→                <div className="text-[10px] text-muted-foreground mt-1">Your API token is the same as your session token.</div>
   444→              </div>
   445→
   446→              <div className="space-y-2 max-h-96 overflow-y-auto">
   447→                {API_ENDPOINTS.map((endpoint) => (
   448→                  <div
   449→                    key={endpoint.path + endpoint.method}
   450→                    className="flex items-center gap-3 p-3 rounded-lg bg-card/50 border border-border/30 group cursor-pointer"
   451→                    onClick={() => handleCopyEndpoint(endpoint.path)}
   452→                  >
   453→                    <Badge variant="outline" className={cn(
   454→                      'text-[10px] font-mono shrink-0',
   455→                      endpoint.method === 'GET' ? 'text-emerald-400 border-emerald-400/30' : 'text-orange-400 border-orange-400/30'
   456→                    )}>
   457→                      {endpoint.method}
   458→                    </Badge>
   459→                    <code className="text-xs font-mono text-foreground/80 flex-1 truncate">{endpoint.path}</code>
   460→                    <span className="text-[10px] text-muted-foreground hidden md:block max-w-48 truncate">{endpoint.description}</span>
   461→                    <Copy className="size-3 text-muted-foreground opacity-0 group-hover:opacity-100 transition-opacity shrink-0" />
   462→                  </div>
   463→                ))}
   464→              </div>
   465→            </CardContent>
   466→          </Card>
   467→        </TabsContent>
   468→
   469→        {/* Bulk Export */}
   470→        <TabsContent value="bulk">
   471→          <Card className="glass-card">
   472→            <CardHeader className="pb-2">
   473→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   474→                <Database className="size-4 text-primary" /> Bulk Data Export
   475→              </CardTitle>
   476→            </CardHeader>
   477→            <CardContent className="space-y-4">
   478→              {user?.role === 'admin' ? (
   479→                <>
   480→                  <p className="text-sm text-muted-foreground">
