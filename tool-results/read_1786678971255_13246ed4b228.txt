    59→// ── Types ──────────────────────────────────────────────────────────────────────
    60→
    61→interface EnhancedPlayer extends Player {
    62→  teamName?: string
    63→  teamCode?: string
    64→  teamColor?: string
    65→  nationality?: string
    66→  appearances?: number
    67→  minutesPlayed?: number
    68→}
    69→// ── Helpers ────────────────────────────────────────────────────────────────────
    70→
    71→function generateRadarStats(p: EnhancedPlayer) {
    72→  return {
    73→    Pace: p.position === 'FWD' || p.position === 'MID' ? 75 : 55,
    74→    Shooting: p.position === 'FWD' ? 75 : 50,
    75→    Passing: p.position === 'MID' ? 75 : 55,
    76→    Defending: p.position === 'DEF' ? 75 : 45,
    77→    Physical: 65,
    78→    Dribbling: p.position === 'FWD' || p.position === 'MID' ? 72 : 50,
    79→  }
    80→}

    84→export function PlayerView() {
    85→  const teams = useElasticoStore(s => s.teams)
    86→  const token = useElasticoStore(s => s.token)
    87→  const [players, setPlayers] = useState<EnhancedPlayer[]>([])
    88→  const [search, setSearch] = useState('')
    89→  const [positionFilter, setPositionFilter] = useState<string>('all')
    90→  const [teamFilter, setTeamFilter] = useState<string>('all')
    91→  const [sortBy, setSortBy] = useState<string>('rating')
    92→  const [selectedPlayer, setSelectedPlayer] = useState<EnhancedPlayer | null>(null)
    93→  const [comparePlayer, setComparePlayer] = useState<EnhancedPlayer | null>(null)
    94→  const [page, setPage] = useState(0)
    95→  const perPage = 12
    96→
    97→  // Fetch players — try DB first, fallback to ESPN
    98→  useEffect(() => {
    99→    async function fetchPlayers() {
   100→      try {
   101→        const headers: Record<string, string> = {}
   102→        if (token) headers['Authorization'] = `Bearer ${token}`
   103→        const params = new URLSearchParams()
   104→        if (sortBy) params.set('sortBy', sortBy)
   105→        if (positionFilter !== 'all') params.set('position', positionFilter)
   106→        if (search) params.set('search', search)
   107→        params.set('limit', '100')
   108→        const res = await fetch(`/api/players?${params}`, { headers })
   109→        if (res.ok) {
   110→          const data = await res.json()
   111→          if (data.players && data.players.length > 0) {
   112→            setPlayers(data.players.map((p: Record<string, unknown>) => ({
   113→              ...p,
   114→              teamName: (p.team as Record<string, unknown>)?.name as string || p.teamName,
   115→              teamCode: (p.team as Record<string, unknown>)?.code as string || p.teamCode,
   116→              teamColor: (p.team as Record<string, unknown>)?.primaryColor as string || '#00e676',
   117→            } as EnhancedPlayer)))
   118→            return
   119→          }
   120→        }
   121→      } catch {
   122→        // DB empty — fallback to ESPN
   123→      }
   124→
   125→      // ESPN fallback: fetch teams for selected league, then fetch rosters
   126→      try {
   127→        const leagueCode = teamFilter !== 'all' ? undefined : 'PL'
   128→        const teamsRes = await fetch(`/api/live?action=teams&league=${teamFilter !== 'all' ? teamFilter : 'PL'}`)
   129→        const teamsData = await teamsRes.json()
   130→        const espnTeams = teamsData.data || teamsData.teams || []
   131→
   132→        if (espnTeams.length === 0) return
   133→
   134→        // Fetch rosters for up to 6 teams (to avoid too many requests)
   135→        const allPlayers: EnhancedPlayer[] = []
   136→        const teamsToFetch = espnTeams.slice(0, 6)
   137→
   138→        await Promise.allSettled(teamsToFetch.map(async (t: any) => {
   139→          try {
   140→            const rosterRes = await fetch(`/api/live?action=roster&league=PL&team=${t.id}`)
   141→            if (!rosterRes.ok) return
   142→            const rosterData = await rosterRes.json()
   143→            const athletes = rosterData.data || rosterData.athletes || []
   144→            for (const a of athletes) {
   145→              allPlayers.push({
   146→                id: String(a.id || ''),
   147→                name: a.name || a.displayName || 'Unknown',
   148→                position: a.position || 'MID',
   149→                age: a.age || 25,
   150→                nationality: a.nationality || '',
   151→                rating: 0, // ESPN roster API does not provide match ratings
   152→                goals: a.goals || 0,
   153→                assists: a.assists || 0,
   154→                teamName: t.name,
   155→                teamCode: t.abbreviation || t.code,
   156→                teamColor: t.color || '#00e676',
   157→                teamId: String(t.id || ''),
   158→                appearances: a.appearances || 0,
   159→                minutesPlayed: 0,
   160→                shirtNumber: a.shirtNumber || a.jersey || 0,
   161→              } as unknown as EnhancedPlayer)
   162→            }
   163→          } catch { /* skip this team */ }
   164→        }))
   165→
   166→        if (allPlayers.length > 0) setPlayers(allPlayers)
   167→      } catch {
   168→        // ESPN fallback also failed
   169→      }
   170→    }
   171→    fetchPlayers()
   172→  }, [token, sortBy, positionFilter, search, teamFilter])
   173→
   174→  // Derived data
   175→  const filteredPlayers = useMemo(() => {
   176→    let result = [...players]
   177→    if (search) result = result.filter(p => p.name.toLowerCase().includes(search.toLowerCase()))
   178→    if (positionFilter !== 'all') result = result.filter(p => p.position === positionFilter)
   179→    if (teamFilter !== 'all') result = result.filter(p => p.teamName === teamFilter)
   180→    result.sort((a, b) => {
   181→      if (sortBy === 'rating') return (b.rating || 0) - (a.rating || 0)
   182→      if (sortBy === 'goals') return b.goals - a.goals
   183→      if (sortBy === 'assists') return b.assists - a.assists
   184→      if (sortBy === 'name') return a.name.localeCompare(b.name)
   185→      if (sortBy === 'age') return (a.age || 0) - (b.age || 0)
   186→      if (sortBy === 'marketValue') return (b.marketValue || 0) - (a.marketValue || 0)
   187→      return 0
   188→    })
   189→    return result
   190→  }, [players, search, positionFilter, teamFilter, sortBy])
   191→
   192→  const pagedPlayers = filteredPlayers.slice(page * perPage, (page + 1) * perPage)
   193→  const totalPages = Math.ceil(filteredPlayers.length / perPage)
   194→
   195→  const topScorers = useMemo(() => [...players].sort((a, b) => b.goals - a.goals).slice(0, 10), [players])
   196→  const topByValue = useMemo(() => [...players].filter(p => p.marketValue).sort((a, b) => (b.marketValue || 0) - (a.marketValue || 0)).slice(0, 10), [players])
   197→
   198→  const positionalGroups = useMemo(() => {
   199→    const groups: Record<string, EnhancedPlayer[]> = {}
   200→    for (const p of players) {
   201→      if (!groups[p.position]) groups[p.position] = []
   202→      groups[p.position].push(p)
   203→    }
   204→    return groups
   205→  }, [players])
   206→
   207→  const ageData = useMemo(() => {
   208→    const buckets: Record<string, number> = { '18-21': 0, '22-25': 0, '26-29': 0, '30-33': 0, '34+': 0 }
   209→    for (const p of players) {
   210→      const age = p.age || 25
   211→      if (age <= 21) buckets['18-21']++
   212→      else if (age <= 25) buckets['22-25']++
   213→      else if (age <= 29) buckets['26-29']++
   214→      else if (age <= 33) buckets['30-33']++
   215→      else buckets['34+']++
   216→    }
   217→    return Object.entries(buckets).map(([range, count]) => ({ range, count }))
   218→  }, [players])
   219→
   220→  const nationalities = useMemo(() => {
   221→    const map: Record<string, number> = {}
   222→    for (const p of players) {
   223→      const nat = p.nationality || 'Unknown'
   224→      map[nat] = (map[nat] || 0) + 1
   225→    }
   226→    return Object.entries(map).sort((a, b) => b[1] - a[1]).slice(0, 10)
   227→  }, [players])
   228→
   229→  const uniqueTeams = useMemo(() => [...new Set(players.map(p => p.teamName).filter(Boolean))], [players])
   230→
   231→  // Similar players
   232→  const similarPlayers = useMemo(() => {
   233→    if (!selectedPlayer) return []
   234→    const stats = generateRadarStats(selectedPlayer)
   235→    return players
   236→      .filter(p => p.id !== selectedPlayer.id && p.position === selectedPlayer.position)
   237→      .map(p => {
   238→        const ps = generateRadarStats(p)
   239→        const diff = Math.abs(stats.Pace - ps.Pace) + Math.abs(stats.Shooting - ps.Shooting) +
   240→          Math.abs(stats.Passing - ps.Passing) + Math.abs(stats.Defending - ps.Defending) +
   241→          Math.abs(stats.Physical - ps.Physical) + Math.abs(stats.Dribbling - ps.Dribbling)
   242→        return { player: p, similarity: 100 - Math.round(diff / 6) }
   243→      })
   244→      .sort((a, b) => b.similarity - a.similarity)
   245→      .slice(0, 5)
   246→  }, [selectedPlayer, players])
   247→
   248→  // CSV Export
   249→  const exportCSV = useCallback(() => {
   250→    const headers = ['Name', 'Position', 'Team', 'Age', 'Goals', 'Assists', 'Rating', 'Market Value (€M)', 'Yellow Cards', 'Red Cards', 'Nationality']
   251→    const rows = filteredPlayers.map(p => [
   252→      p.name, p.position, p.teamName || '', p.age || '', p.goals, p.assists, p.rating, p.marketValue || '', p.yellowCards, p.redCards, p.nationality || ''
   253→    ])
   254→    const csv = [headers.join(','), ...rows.map(r => r.map(v => `"${v}"`).join(','))].join('\n')
   255→    const blob = new Blob([csv], { type: 'text/csv' })
   256→    const url = URL.createObjectURL(blob)
   257→    const a = document.createElement('a')
   258→    a.href = url
   259→    a.download = 'elastico-players.csv'
   260→    a.click()
   261→    URL.revokeObjectURL(url)
   262→    toast.success('Players exported to CSV')
   263→  }, [filteredPlayers])
   264→
   265→  const getPositionColor = (pos: string) => {
   266→    switch (pos) {
   267→      case 'GK': return 'text-yellow-400 bg-yellow-400/10 border-yellow-400/30'
   268→      case 'DEF': return 'text-blue-400 bg-blue-400/10 border-blue-400/30'
   269→      case 'MID': return 'text-emerald-400 bg-emerald-400/10 border-emerald-400/30'
   270→      case 'FWD': return 'text-red-400 bg-red-400/10 border-red-400/30'
   271→      default: return ''
   272→    }
   273→  }
   274→
   275→  const getRatingColor = (r: number) => {
   276→    if (r >= 8.0) return 'text-emerald-400'
   277→    if (r >= 7.0) return 'text-emerald-300'
   278→    if (r >= 6.0) return 'text-yellow-400'
   279→    return 'text-red-400'
   280→  }
   281→
   282→  return (
   283→    <motion.div
   284→      initial={{ opacity: 0, y: 20 }}
   285→      animate={{ opacity: 1, y: 0 }}
   286→      className="space-y-6"
   287→    >
   288→      {/* Header */}
   289→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   290→        <div>
   291→        <p className="text-muted-foreground text-sm">Search, analyze, and compare football players</p>
   292→        </div>
   293→        <Button variant="outline" size="sm" onClick={exportCSV} className="gap-2">
   294→          <Download className="size-4" /> Export CSV
   295→        </Button>
   296→      </div>
   297→
   298→      {/* Search & Filters */}
   299→      <div className="flex flex-col sm:flex-row gap-3">
   300→        <div className="relative flex-1">
   301→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   302→          <Input
   303→            placeholder="Search players..."
   304→            value={search}
   305→            onChange={(e) => { setSearch(e.target.value); setPage(0) }}
   306→            className="pl-9"
   307→          />
   308→        </div>
   309→        <Select value={positionFilter} onValueChange={(v) => { setPositionFilter(v); setPage(0) }}>
   310→          <SelectTrigger className="w-full sm:w-36">
   311→            <SelectValue placeholder="Position" />
   312→          </SelectTrigger>
   313→          <SelectContent>
   314→            <SelectItem value="all">All Positions</SelectItem>
   315→            <SelectItem value="GK">GK</SelectItem>
   316→            <SelectItem value="DEF">DEF</SelectItem>
   317→            <SelectItem value="MID">MID</SelectItem>
   318→            <SelectItem value="FWD">FWD</SelectItem>
   319→          </SelectContent>
   320→        </Select>
   321→        <Select value={teamFilter} onValueChange={(v) => { setTeamFilter(v); setPage(0) }}>
   322→          <SelectTrigger className="w-full sm:w-40">
   323→            <SelectValue placeholder="Team" />
   324→          </SelectTrigger>
   325→          <SelectContent>
   326→            <SelectItem value="all">All Teams</SelectItem>
   327→            {uniqueTeams.map(t => (
   328→              <SelectItem key={t} value={t!}>{t}</SelectItem>
   329→            ))}
   330→          </SelectContent>
   331→        </Select>
   332→        <Select value={sortBy} onValueChange={setSortBy}>
   333→          <SelectTrigger className="w-full sm:w-36">
   334→            <SelectValue placeholder="Sort by" />
   335→          </SelectTrigger>
   336→          <SelectContent>
   337→            <SelectItem value="rating">Rating</SelectItem>
   338→            <SelectItem value="goals">Goals</SelectItem>
   339→            <SelectItem value="assists">Assists</SelectItem>
   340→            <SelectItem value="marketValue">Market Value</SelectItem>
   341→            <SelectItem value="age">Age</SelectItem>
   342→            <SelectItem value="name">Name</SelectItem>
   343→          </SelectContent>
   344→        </Select>
   345→      </div>
   346→
   347→      <Tabs defaultValue="grid" className="space-y-4">
   348→        <TabsList className="flex flex-wrap gap-1 h-auto bg-card/50 p-1 rounded-lg">
   349→          <TabsTrigger value="grid" className="text-xs">Player Grid</TabsTrigger>
   350→          <TabsTrigger value="leaderboard" className="text-xs">Top Scorers</TabsTrigger>
   351→          <TabsTrigger value="radar" className="text-xs">Radar Charts</TabsTrigger>
   352→          <TabsTrigger value="compare" className="text-xs">Compare</TabsTrigger>
   353→          <TabsTrigger value="value" className="text-xs">Market Value</TabsTrigger>
   354→          <TabsTrigger value="positions" className="text-xs">By Position</TabsTrigger>
   355→          <TabsTrigger value="age" className="text-xs">Age Dist.</TabsTrigger>
   356→          <TabsTrigger value="nationality" className="text-xs">Nationality</TabsTrigger>
   357→          <TabsTrigger value="cards" className="text-xs">Cards</TabsTrigger>
   358→          <TabsTrigger value="subs" className="text-xs">Substitutions</TabsTrigger>
   359→          <TabsTrigger value="similar" className="text-xs">Similar</TabsTrigger>
   360→          <TabsTrigger value="form" className="text-xs">Form</TabsTrigger>
   361→        </TabsList>
   362→
   363→        {/* Player Cards Grid */}
   364→        <TabsContent value="grid">
   365→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4">
   366→            {pagedPlayers.map((player) => (
   367→              <motion.div
   368→                key={player.id}
   369→                whileHover={{ scale: 1.02, y: -4 }}
   370→                whileTap={{ scale: 0.98 }}
   371→                className="cursor-pointer"
   372→                onClick={() => setSelectedPlayer(player)}
   373→              >
   374→                <Card className="glass-card h-full overflow-hidden">
   375→                  <div className="h-2" style={{ backgroundColor: player.teamColor || '#00e676' }} />
   376→                  <CardContent className="p-4">
   377→                    <div className="flex items-start gap-3">
   378→                      <div className="w-14 h-14 rounded-full bg-muted flex items-center justify-center text-lg font-bold shrink-0 border-2 border-border">
   379→                        {player.number}
   380→                      </div>
   381→                      <div className="min-w-0 flex-1">
   382→                        <h3 className="text-sm font-semibold truncate">{player.name}</h3>
   383→                        <div className="flex items-center gap-2 mt-1">
   384→                          <Badge variant="outline" className={cn('text-[10px]', getPositionColor(player.position))}>
   385→                            {player.position}
   386→                          </Badge>
   387→                          <span className="text-[10px] text-muted-foreground truncate">{player.teamName}</span>
   388→                        </div>
   389→                      </div>
   390→                      <div className={cn('text-xl font-bold', player.rating > 0 ? getRatingColor(player.rating) : 'text-muted-foreground')}>
   391→                        {player.rating > 0 ? player.rating.toFixed(1) : 'N/A'}

   393→                    </div>
   394→                    <div className="grid grid-cols-3 gap-2 mt-4">
   395→                      <div className="text-center">
   396→                        <div className="text-lg font-bold text-primary">{player.goals}</div>
   397→                        <div className="text-[10px] text-muted-foreground">Goals</div>
   398→                      </div>
   399→                      <div className="text-center">
   400→                        <div className="text-lg font-bold text-primary">{player.assists}</div>
   401→                        <div className="text-[10px] text-muted-foreground">Assists</div>
   402→                      </div>
   403→                      <div className="text-center">
   404→                        <div className="text-lg font-bold text-primary">{player.yellowCards + player.redCards}</div>
   405→                        <div className="text-[10px] text-muted-foreground">Cards</div>
   406→                      </div>
   407→                    </div>
   408→                    {player.marketValue && (
   409→                      <div className="mt-3 pt-3 border-t border-border/50 flex items-center gap-1">
   410→                        <TrendingUp className="size-3 text-muted-foreground" />
   411→                        <span className="text-xs text-muted-foreground">€{player.marketValue}M</span>
   412→                      </div>
   413→                    )}
   414→                  </CardContent>
   415→                </Card>
   416→              </motion.div>
   417→            ))}
   418→          </div>
   419→          {/* Pagination */}
   420→          {totalPages > 1 && (
   421→            <div className="flex items-center justify-center gap-2 mt-4">
   422→              <Button variant="outline" size="icon" className="h-8 w-8" onClick={() => setPage(Math.max(0, page - 1))} disabled={page === 0}>
   423→                <ChevronLeft className="size-4" />
   424→              </Button>
   425→              <span className="text-sm text-muted-foreground">{page + 1} / {totalPages}</span>
   426→              <Button variant="outline" size="icon" className="h-8 w-8" onClick={() => setPage(Math.min(totalPages - 1, page + 1))} disabled={page >= totalPages - 1}>
   427→                <ChevronRight className="size-4" />
   428→              </Button>
   429→            </div>
   430→          )}
   431→        </TabsContent>
   432→
   433→        {/* Top Scorers Leaderboard */}
   434→        <TabsContent value="leaderboard">
   435→          <Card className="glass-card">
   436→            <CardHeader className="pb-2">
   437→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   438→                <Star className="size-4 text-yellow-400" /> Top Scorers Leaderboard
   439→              </CardTitle>
   440→            </CardHeader>
   441→            <CardContent>
   442→              <div className="max-h-96 overflow-y-auto">
   443→                <Table>
   444→                  <TableHeader>
   445→                    <TableRow>
   446→                      <TableHead className="w-10">#</TableHead>
   447→                      <TableHead>Player</TableHead>
   448→                      <TableHead className="text-center">Goals</TableHead>
   449→                      <TableHead className="text-center">Assists</TableHead>
   450→                      <TableHead className="text-center">Mins</TableHead>
   451→                      <TableHead className="text-center">G/90</TableHead>
   452→                      <TableHead className="text-center">Rating</TableHead>
   453→                    </TableRow>
   454→                  </TableHeader>
   455→                  <TableBody>
   456→                    {topScorers.map((p, i) => (
   457→                      <TableRow key={p.id} className="cursor-pointer hover:bg-accent/50" onClick={() => setSelectedPlayer(p)}>
   458→                        <TableCell className="font-bold text-muted-foreground">{i + 1}</TableCell>
   459→                        <TableCell>
   460→                          <div className="flex items-center gap-2">
   461→                            <span className="text-sm font-medium">{p.name}</span>
   462→                            <Badge variant="outline" className={cn('text-[9px]', getPositionColor(p.position))}>{p.position}</Badge>
   463→                          </div>
   464→                          <span className="text-[10px] text-muted-foreground">{p.teamName}</span>
   465→                        </TableCell>
   466→                        <TableCell className="text-center font-bold text-primary">{p.goals}</TableCell>
   467→                        <TableCell className="text-center">{p.assists}</TableCell>
   468→                        <TableCell className="text-center text-muted-foreground">{p.minutesPlayed || (p.appearances || 0) * 90}</TableCell>
   469→                        <TableCell className="text-center font-medium">
   470→                          {((p.goals / ((p.minutesPlayed || (p.appearances || 0) * 90) / 90))).toFixed(2)}
   471→                        </TableCell>
   472→                        <TableCell className={cn('text-center font-bold', p.rating > 0 ? getRatingColor(p.rating) : 'text-muted-foreground')}>{p.rating > 0 ? p.rating.toFixed(1) : '—'}</TableCell>
   473→                      </TableRow>
   474→                    ))}
   475→                  </TableBody>
   476→                </Table>
   477→              </div>
