    73→// ── Helper: get initials ──────────────────────────────────────────────────────
    74→
    75→function getInitials(name: string): string {
    76→  const parts = name.trim().split(/\s+/)
    77→  if (parts.length === 1) return parts[0].substring(0, 2).toUpperCase()
    78→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase()
    79→}

    81→// ── Helper: motivational message ───────────────────────────────────────────────
    82→
    83→function getMotivationalMessage(rank: number, total: number): { icon: React.ElementType; text: string; color: string } {
    84→  const pct = total > 0 ? rank / total : 1
    85→  if (rank <= 3) return { icon: Crown, text: 'You\'re among the elite! Keep dominating!', color: 'text-amber-400' }
    86→  if (rank <= 10) return { icon: Trophy, text: 'Top 10 territory — push for the podium!', color: 'text-emerald-400' }
    87→  if (pct <= 0.25) return { icon: TrendingUp, text: 'Great position! A few correct predictions could vault you up.', color: 'text-primary' }
    88→  if (pct <= 0.5) return { icon: Flame, text: 'Solid mid-table. Build a streak to climb!', color: 'text-orange-400' }
    89→  if (pct <= 0.75) return { icon: Target, text: 'Room to grow. Focus on high-confidence picks.', color: 'text-cyan-400' }
    90→  return { icon: Rocket, text: 'Every prediction counts. Start your climb today!', color: 'text-muted-foreground' }
    91→}

   106→        <span className="text-sm">🥈</span>

   117→  return (
   118→    <span className="inline-flex size-7 items-center justify-center rounded-full text-xs font-bold text-muted-foreground">
   119→      {rank}

   121→  )
   122→}
