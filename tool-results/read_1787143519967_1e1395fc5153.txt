   200→
   201→export function extractJumpDiffusionParams(residuals: number[]): JumpDiffusionParams {
   202→  /**
   203→   * Separates continuous diffusion from structural jump shocks.
   204→   * Uses a 2-standard-deviation threshold to identify outlier events
   205→   * (red cards, early injuries, unexpected tactical shifts).
   206→   */
   207→  if (residuals.length < 20) {
   208→    return { sigma: 0.25, jumpLambda: 0.10, jumpMu: -0.05, jumpSigma: 0.15 }
   209→  }
   210→
   211→  const threshold = 2.0 * (residuals.reduce((s, r) => s + r * r, 0) / residuals.length) ** 0.5
   212→  const normalShocks = residuals.filter(r => Math.abs(r) <= threshold)
   213→  const jumpShocks = residuals.filter(r => Math.abs(r) > threshold)
   214→
   215→  const sigma = normalShocks.length > 0
   216→    ? normalShocks.reduce((s, r) => s + r * r, 0) / normalShocks.length
   217→    : 0.25
   218→
   219→  const jumpLambda = jumpShocks.length / residuals.length
   220→  const jumpMu = jumpShocks.length > 0
   221→    ? jumpShocks.reduce((s, r) => s + r, 0) / jumpShocks.length
   222→    : 0.0
   223→  const jumpSigma = jumpShocks.length > 1
   224→    ? Math.sqrt(jumpShocks.reduce((s, r) => s + (r - jumpMu) ** 2, 0) / (jumpShocks.length - 1))
   225→    : 0.20
   226→
   227→  return {
   228→    sigma: Math.sqrt(sigma),
   229→    jumpLambda: clamp(jumpLambda, 0, 1),
   230→    jumpMu,
   231→    jumpSigma: clamp(jumpSigma, 0.01, 2),
   232→  }
   233→}

   235→// ── Injury Adjustment Layer ────────────────────────────────────────────────────
   236→
   237→export function applyInjuryAdjustments(
   238→  baseXg: number,
   239→  injuries: InjuryAdjustment[]
   240→): number {
   241→  /**
   242→   * Modifies projected expected goals based on missing players.
   243→   * Star player out: -15% xG
   244→   * Key player out: -8% xG
   245→   * Rotation player out: -3% xG
   246→   */
   247→  let adjustedXg = baseXg
   248→  for (const injury of injuries) {
   249→    if (injury.status === 'out') {
   250→      adjustedXg *= (1 - injury.xgImpact)
   251→    } else if (injury.status === 'doubtful') {
   252→      adjustedXg *= (1 - injury.xgImpact * 0.5)
   253→    } else if (injury.status === 'questionable') {
   254→      adjustedXg *= (1 - injury.xgImpact * 0.25)
   255→    }
   256→  }
   257→  return Math.max(0.1, adjustedXg)
   258→}

   260→// ── Core Stochastic Simulation Engine ──────────────────────────────────────────
   261→
   262→export function runStochasticSimulation(
   263→  input: MatchInput,
   264→  config: EngineConfig = DEFAULT_CONFIG
   265→): StochasticMatchResult {
   266→  /**
   267→   * Executes a high-iteration Monte Carlo simulation using a
   268→   * Bivariate Merton Jump-Diffusion Process with GARCH volatility scaling.
   269→   * 
   270→   * This models match outcomes as continuous stochastic differential
   271→   * processes interrupted by Poisson-distributed structural jumps,
   272→   * matching the methodology used by elite quantitative sports funds.
   273→   */
   274→  const { simulationRuns, garchEnabled, jumpDiffusionEnabled } = config
   275→
   276→  // Step 1: Calibrate base intensity parameters from xG + ELO
   277→  const eloDiff = input.homeElo - input.awayElo
   278→  const eloFactor = 1 / (1 + Math.pow(10, -eloDiff / 400))
   279→  const avgXg = (input.homeXg + input.awayXg) / 2
   280→
   281→  // Home advantage adjustment (+0.25 goals typical for home field)
   282→  let muHome = input.homeXg * (0.7 + 0.3 * eloFactor) + 0.25
   283→  let muAway = input.awayXg * (0.7 + 0.3 * (1 - eloFactor))
   284→
   285→  // Step 2: Apply injury adjustments
   286→  const homeInjuries = input.injuries?.filter(i => i.teamId === input.homeTeamId) || []
   287→  const awayInjuries = input.injuries?.filter(i => i.teamId === input.awayTeamId) || []
   288→  muHome = applyInjuryAdjustments(muHome, homeInjuries)
   289→  muAway = applyInjuryAdjustments(muAway, awayInjuries)
   290→
   291→  // Step 3: GARCH volatility calibration
   292→  const homeResiduals = input.historicalResiduals?.home || []
   293→  const awayResiduals = input.historicalResiduals?.away || []
   294→  const volScalarHome = garchEnabled ? calibrateGARCH(homeResiduals) : 1.0
   295→  const volScalarAway = garchEnabled ? calibrateGARCH(awayResiduals) : 1.0
   296→
   297→  // Step 4: Extract jump-diffusion parameters
   298→  const homeJd = jumpDiffusionEnabled
   299→    ? extractJumpDiffusionParams(homeResiduals)
   300→    : { sigma: 0.20, jumpLambda: 0.05, jumpMu: 0, jumpSigma: 0.15 }
   301→  const awayJd = jumpDiffusionEnabled
   302→    ? extractJumpDiffusionParams(awayResiduals)
   303→    : { sigma: 0.20, jumpLambda: 0.05, jumpMu: 0, jumpSigma: 0.15 }
   304→
   305→  // Step 5: Compute inter-team correlation
   306→  const rho = input.correlation ?? 0.12
   307→
   308→  // Step 6: Execute Monte Carlo simulation
   309→  let homeWins = 0
   310→  let awayWins = 0
   311→  let draws = 0
   312→  let over25 = 0
   313→  let over15 = 0
   314→  let over35 = 0
   315→  let btts = 0
   316→  const scorelineCounts: Record<string, number> = {}
   317→  let totalHomeGoals = 0
   318→  let totalAwayGoals = 0
   319→
   320→  // Pre-compute correlation transformation constants
   321→  const sqrtOneMinusRho2 = Math.sqrt(Math.max(0, 1 - rho * rho))
   322→
   323→  for (let i = 0; i < simulationRuns; i++) {
   324→    // Generate correlated Brownian motion paths via Cholesky decomposition
   325→    const zHomeRaw = normalRandom()
   326→    const zAwayRaw = normalRandom()
   327→    const zHome = zHomeRaw
   328→    const zAway = rho * zHomeRaw + sqrtOneMinusRho2 * zAwayRaw
   329→
   330→    // Apply GARCH volatility scaling to base intensities
   331→    const scaledMuHome = muHome * volScalarHome
   332→    const scaledMuAway = muAway * volScalarAway
   333→
   334→    // Merton Jump-Diffusion: continuous diffusion + Poisson jumps
   335→    const hSigma = homeJd.sigma
   336→    const aSigma = awayJd.sigma
   337→    const dt = 1.0 // One match unit
   338→
   339→    // Poisson jump processes for structural shocks
   340→    const jumpCountHome = poissonRandom(homeJd.jumpLambda * dt)
   341→    const jumpCountAway = poissonRandom(awayJd.jumpLambda * dt)
   342→
   343→    // Calculate jump magnitudes with drift + volatility
   344→    const jumpMagHome = jumpCountHome > 0
   345→      ? jumpCountHome * homeJd.jumpMu + Math.sqrt(jumpCountHome) * homeJd.jumpSigma * normalRandom()
   346→      : 0
   347→    const jumpMagAway = jumpCountAway > 0
   348→      ? jumpCountAway * awayJd.jumpMu + Math.sqrt(jumpCountAway) * awayJd.jumpSigma * normalRandom()
   349→      : 0
   350→
   351→    // Synthesize continuous diffusion with structural shocks
   352→    const stochasticHome = scaledMuHome *
   353→      Math.exp((hSigma * hSigma / -2.0) * dt + hSigma * Math.sqrt(dt) * zHome + jumpMagHome)
   354→    const stochasticAway = scaledMuAway *
   355→      Math.exp((aSigma * aSigma / -2.0) * dt + aSigma * Math.sqrt(dt) * zAway + jumpMagAway)
   356→
   357→    // Convert intensity paths to observable goal outputs via Poisson distribution
   358→    const goalsHome = poissonRandom(Math.max(0.01, stochasticHome))
   359→    const goalsAway = poissonRandom(Math.max(0.01, stochasticAway))
   360→
   361→    // Accumulate statistics
   362→    totalHomeGoals += goalsHome
   363→    totalAwayGoals += goalsAway
   364→
   365→    if (goalsHome > goalsAway) homeWins++
   366→    else if (goalsAway > goalsHome) awayWins++
   367→    else draws++
   368→
   369→    const total = goalsHome + goalsAway
   370→    if (total > 2.5) over25++
   371→    if (total > 1.5) over15++
   372→    if (total > 3.5) over35++
   373→    if (goalsHome > 0 && goalsAway > 0) btts++
   374→
   375→    const key = `${goalsHome}-${goalsAway}`
   376→    scorelineCounts[key] = (scorelineCounts[key] || 0) + 1
   377→  }
   378→
   379→  // Step 7: Compute Asian Handicap probabilities
   380→  const computeAH = (line: number) => {
   381→    let homeCovers = 0
   382→    let awayCovers = 0
   383→    // Re-simulate for AH (using same parameters, smaller run)
   384→    const ahRuns = Math.min(50000, simulationRuns)
   385→    for (let i = 0; i < ahRuns; i++) {
   386→      const zH = normalRandom()
   387→      const zA = rho * zH + sqrtOneMinusRho2 * normalRandom()
   388→      const sH = poissonRandom(Math.max(0.01, muHome * volScalarHome *
   389→        Math.exp((homeJd.sigma ** 2 / -2.0) + homeJd.sigma * zH)))
   390→      const sA = poissonRandom(Math.max(0.01, muAway * volScalarAway *
   391→        Math.exp((awayJd.sigma ** 2 / -2.0) + awayJd.sigma * zA)))
   392→      const diff = sH - sA - line
   393→      if (diff > 0) homeCovers++
   394→      else if (diff < 0) awayCovers++
   395→    }
   396→    return { home: homeCovers / ahRuns, away: awayCovers / ahRuns }
   397→  }
   398→
   399→  // Step 8: Determine confidence level
   400→  const maxProb = Math.max(homeWins, draws, awayWins) / simulationRuns
   401→  const confidence: StochasticMatchResult['confidence'] =
   402→    maxProb > 0.55 ? 'very-high' :
   403→    maxProb > 0.48 ? 'high' :
   404→    maxProb > 0.42 ? 'medium' : 'low'
   405→
   406→  // Step 9: Calculate volatility index (0-100)
   407→  const goalVariance = Array.from({ length: 100 }, () => {
   408→    const gH = poissonRandom(muHome)
   409→    const gA = poissonRandom(muAway)
   410→    return Math.abs(gH + gA - (muHome + muAway))
   411→  })
   412→  const avgDeviation = goalVariance.reduce((s, v) => s + v, 0) / 100
   413→  const volatilityIndex = clamp((avgDeviation / (muHome + muAway || 1)) * 100, 0, 100)
   414→
   415→  return {
   416→    matchProbabilities: {
   417→      homeVictory: homeWins / simulationRuns,
   418→      draw: draws / simulationRuns,
   419→      awayVictory: awayWins / simulationRuns,
   420→    },
   421→    totalsMarket: {
   422→      over25: over25 / simulationRuns,
   423→      under25: 1 - over25 / simulationRuns,
   424→      over15: over15 / simulationRuns,
   425→      over35: over35 / simulationRuns,
   426→    },
   427→    exoticScorelines: Object.fromEntries(
   428→      Object.entries(scorelineCounts)
   429→        .sort(([, a], [, b]) => b - a)
   430→        .slice(0, 10)
   431→        .map(([key, count]) => [key, count / simulationRuns])
   432→    ),
   433→    expectedMeans: {
   434→      home: totalHomeGoals / simulationRuns,
   435→      away: totalAwayGoals / simulationRuns,
   436→      total: (totalHomeGoals + totalAwayGoals) / simulationRuns,
   437→    },
   438→    asianHandicap: {
   439→      line0: computeAH(0),
   440→      lineHalf: computeAH(-0.5),
   441→      line1: computeAH(-1),
   442→      line15: computeAH(-1.5),
   443→    },
   444→    bothTeamsToScore: btts / simulationRuns,
   445→    confidence,
   446→    volatilityIndex,
   447→  }
   448→}

   450→// ── Kelly Criterion Engine ──────────────────────────────────────────────────────
   451→
   452→export function calculateKelly(
   453→  modelProbability: number,
   454→  bookmakerOdds: number,
   455→  bankroll: number,
   456→  fractionalKelly: number = 0.25
   457→): KellyResult {
   458→  /**
   459→   * Implements the Kelly Criterion for optimal bankroll allocation.
   460→   * Uses fractional Kelly (default 25%) for safer variance management.
   461→   * 
   462→   * Kelly Formula: f* = (p*b - q) / b
   463→   * Where: p = model probability, b = decimal odds - 1, q = 1 - p
   464→   */
   465→  const impliedProb = 1 / bookmakerOdds
   466→  const edge = modelProbability - impliedProb
   467→
   468→  if (edge <= 0) {
   469→    return {
   470→      action: 'NO_EDGE',
   471→      edgePercentage: -Math.abs(edge) * 100,
   472→      impliedProbability: impliedProb,
   473→      modelProbability,
   474→      suggestedFraction: 0,
   475→      suggestedWager: 0,
   476→      kellyFraction: 0,
   477→      odds: bookmakerOdds,
   478→    }
   479→  }
   480→
   481→  const b = bookmakerOdds - 1  // Net odds
   482→  const q = 1 - modelProbability
   483→  const rawKelly = (modelProbability * b - q) / b
   484→  const safeKelly = rawKelly * fractionalKelly
   485→  const wager = bankroll * safeKelly
   486→
   487→  // Risk classification
   488→  const action: KellyResult['action'] =
   489→    edge > 0.10 ? 'BET' :
   490→    edge > 0.03 ? 'BET' : 'RISKY'
   491→
   492→  return {
   493→    action,
   494→    edgePercentage: edge * 100,
   495→    impliedProbability: impliedProb,
   496→    modelProbability,
   497→    suggestedFraction: safeKelly,
   498→    suggestedWager: Math.round(wager * 100) / 100,
   499→    kellyFraction: rawKelly,
   500→    odds: bookmakerOdds,
   501→  }
   502→}

   504→// ── Multi-Asset Portfolio Allocation ────────────────────────────────────────────
   505→
   506→export function calculatePortfolioAllocation(
   507→  outcomes: { label: string; modelProb: number; odds: number }[],
   508→  bankroll: number,
   509→  correlationMatrix: number[][] = [
   510→    [1.00, 0.55, 0.15],
   511→    [0.55, 1.00, 0.20],
   512→    [0.15, 0.20, 1.00],
   513→  ]
   514→): PortfolioAllocation {
   515→  /**
   516→   * Implements a multi-asset Kelly Criterion with covariance adjustment.
   517→   * Accounts for correlation between bet outcomes to prevent overexposure
   518→   * on correlated positions (e.g., home win + over 2.5 goals).
   519→   */
   520→
   521→  // Invert correlation matrix for covariance-weighted allocation
   522→  const n = correlationMatrix.length
   523→  const covInv = invertMatrix(correlationMatrix)
   524→
   525→  const results = outcomes.map((out, i) => {
   526→    const impliedProb = 1 / out.odds
   527→    const edge = out.modelProb - impliedProb
   528→
   529→    // Covariance-adjusted edge
   530→    const adjustedEdge = covInv
   531→      ? covInv[i]?.reduce((sum, c, j) => sum + c * (outcomes[j].modelProb - 1 / outcomes[j].odds), 0) || edge
   532→      : edge
   533→
   534→    const b = out.odds - 1
   535→    const q = 1 - out.modelProb
   536→    const rawKelly = (out.modelProb * b - q) / b
   537→    const safeKelly = Math.max(0, rawKelly * 0.10) // 10% fractional Kelly
   538→
   539→    return {
   540→      label: out.label,
   541→      modelProb: out.modelProb,
   542→      marketOdds: out.odds,
   543→      edge: edge * 100,
   544→      kellyFraction: safeKelly,
   545→      wagerAmount: Math.round(bankroll * safeKelly * 100) / 100,
   546→      action: edge > 0.02 ? 'BET' as const : edge > 0 ? 'RISKY' as const : 'NO_EDGE' as const,
   547→    }
   548→  })
   549→
   550→  const totalExposure = results.reduce((s, r) => s + r.wagerAmount, 0)
   551→  const totalEV = results.reduce((s, r) => s + r.wagerAmount * r.edge / 100, 0)
   552→  const avgReturn = totalExposure > 0 ? totalEV / totalExposure : 0
   553→  const sharpeRatio = results.length > 1 ? avgReturn / (Math.sqrt(results.reduce((s, r) => s + (r.edge / 100 - avgReturn) ** 2, 0) / results.length) || 0.01) : 0
   554→
   555→  return {
   556→    outcomes: results,
   557→    totalBankroll: bankroll,
   558→    totalExposure: Math.round(totalExposure * 100) / 100,
   559→    totalExpectedValue: Math.round(totalEV * 100) / 100,
   560→    sharpeRatio: Math.round(sharpeRatio * 100) / 100,
   561→  }
   562→}

   564→// ── Matrix Inversion (3x3) ─────────────────────────────────────────────────────
   565→
   566→function invertMatrix(m: number[][]): number[][] | null {
   567→  if (m.length !== 3 || m[0].length !== 3) return null
   568→
   569→  const [a, b, c] = m[0]
   570→  const [d, e, f] = m[1]
   571→  const [g, h, i] = m[2]
   572→
   573→  const det = a * (e * i - f * h) - b * (d * i - f * g) + c * (d * h - e * g)
   574→  if (Math.abs(det) < 1e-10) return null
   575→
   576→  const invDet = 1 / det
   577→  return [
   578→    [(e * i - f * h) * invDet, (c * h - b * i) * invDet, (b * f - c * e) * invDet],
   579→    [(f * g - d * i) * invDet, (a * i - c * g) * invDet, (c * d - a * f) * invDet],
   580→    [(d * h - e * g) * invDet, (b * g - a * h) * invDet, (a * e - b * d) * invDet],
   581→  ]
   582→}

   584→// ── Market Signal Tracker ───────────────────────────────────────────────────────
   585→
   586→export function analyzeMarketSignals(
   587→  openingOdds: { home: number; draw: number; away: number },
   588→  currentOdds: { home: number; draw: number; away: number },
   589→  matchId: string,
   590→  homeTeam: string,
   591→  awayTeam: string
   592→): MarketSignal {
   593→  /**
   594→   * Tracks line movement velocity to detect sharp money action.
   595→   * 
   596→   * Key signals:
   597→   * - Steam Move: Rapid odds movement in one direction
   598→   * - Reverse Line Movement (RLM): Line moves AGAINST public betting
   599→   * - Sharp Action: Professional syndicate money detected
   600→   */
   601→
   602→  // Calculate line velocity (negative = odds dropping = money coming in)
   603→  const homeVelocity = (currentOdds.home - openingOdds.home) / openingOdds.home
   604→  const drawVelocity = (currentOdds.draw - openingOdds.draw) / openingOdds.draw
   605→  const awayVelocity = (currentOdds.away - openingOdds.away) / openingOdds.away
   606→
   607→  // Detect significant moves (>3% change is notable)
   608→  const threshold = 0.03
   609→  const homeSteam = homeVelocity < -threshold
   610→  const awaySteam = awayVelocity < -threshold
   611→  const drawSteam = drawVelocity < -threshold
   612→
   613→  // Detect RLM: odds drop on one side despite public likely betting the other
   614→  const rlmDetected = (homeVelocity < -threshold && awayVelocity > threshold) ||
   615→    (awayVelocity < -threshold && homeVelocity > threshold)
   616→
   617→  // Determine sharp action direction
   618→  const velocities = [
   619→    { dir: 'home' as const, vel: Math.abs(homeVelocity) },
   620→    { dir: 'draw' as const, vel: Math.abs(drawVelocity) },
   621→    { dir: 'away' as const, vel: Math.abs(awayVelocity) },
   622→  ]
   623→  velocities.sort((a, b) => b.vel - a.vel)
   624→
   625→  const sharpAction: MarketSignal['sharpAction'] =
   626→    velocities[0].vel > threshold ? velocities[0].dir : 'neutral'
   627→
   628→  // Confidence score based on magnitude of movement
   629→  const maxMove = Math.max(Math.abs(homeVelocity), Math.abs(drawVelocity), Math.abs(awayVelocity))
   630→  const confidence = clamp(maxMove * 500, 0, 100)
   631→
   632→  return {
   633→    matchId,
   634→    homeTeam,
   635→    awayTeam,
   636→    openingOdds,
   637→    currentOdds,
   638→    lineVelocity: {
   639→      home: Math.round(homeVelocity * 10000) / 100,
   640→      draw: Math.round(drawVelocity * 10000) / 100,
   641→      away: Math.round(awayVelocity * 10000) / 100,
   642→    },
   643→    sharpAction,
   644→    steamMove: homeSteam || awaySteam || drawSteam,
   645→    rlmDetected,
   646→    confidence: Math.round(confidence),
   647→    timestamp: new Date().toISOString(),
   648→  }
   649→}

   651→// ── xG Luck Adjustment ─────────────────────────────────────────────────────────
   652→
   653→export function adjustForLuck(
   654→  rawGoals: number,
   655→  xg: number,
   656→  sampleSize: number = 10
   657→): { luckAdjustedGoals: number; luckFactor: number } {
   658→  /**
   659→   * Removes "luck" from raw scores by regressing toward xG.
   660→   * Uses a Bayesian shrinkage estimator weighted by sample size.
   661→   * 
   662→   * Small sample → more regression toward xG (more luck adjustment)
   663→   * Large sample → trust raw goals more
   664→   */
   665→  const weight = Math.min(sampleSize / 30, 1) // Full weight at 30+ games
   666→  const luckAdjustedGoals = rawGoals * weight + xg * (1 - weight)
   667→  const luckFactor = rawGoals > 0 ? xg / rawGoals : 1
   668→
   669→  return {
   670→    luckAdjustedGoals: Math.round(luckAdjustedGoals * 100) / 100,
   671→    luckFactor: Math.round(luckFactor * 100) / 100,
   672→  }
   673→}

   675→// ── Complete Match Analysis Pipeline ────────────────────────────────────────────

   677→export interface FullMatchAnalysis {
   678→  matchInput: MatchInput
   679→  simulation: StochasticMatchResult
   680→  kellyResults: {
   681→    home: KellyResult
   682→    draw: KellyResult
   683→    away: KellyResult
   684→    over25: KellyResult
   685→    under25: KellyResult
   686→  }
   687→  portfolioAllocation: PortfolioAllocation
   688→  marketSignal: MarketSignal | null
   689→  luckAdjustment: {
   690→    home: ReturnType<typeof adjustForLuck>
   691→    away: ReturnType<typeof adjustForLuck>
   692→  }
   693→  recommendation: string
   694→  riskRating: 'low' | 'medium' | 'high' | 'very-high'
   695→}
   696→
   697→export function runFullMatchAnalysis(
   698→  input: MatchInput,
   699→  bankroll: number = 1000,
   700→  openingOdds?: { home: number; draw: number; away: number },
   701→  config: EngineConfig = DEFAULT_CONFIG
   702→): FullMatchAnalysis {
   703→  /**
   704→   * The complete ELASTICO analysis pipeline:
   705→   * 1. Luck-adjust historical data
   706→   * 2. Run stochastic simulation
   707→   * 3. Calculate Kelly Criterion for all markets
   708→   * 4. Generate portfolio allocation
   709→   * 5. Analyze market signals
   710→   * 6. Produce final recommendation
   711→   */
   712→
   713→  // Step 1: Luck adjustment
   714→  const homeLuck = adjustForLuck(
   715→    input.homeXg * 1.1, // Approximate actual goals from xG
   716→    input.homeXg
   717→  )
   718→  const awayLuck = adjustForLuck(
   719→    input.awayXg * 0.9,
   720→    input.awayXg
   721→  )
   722→
   723→  // Step 2: Run simulation with luck-adjusted inputs
   724→  const adjustedInput: MatchInput = {
   725→    ...input,
   726→    homeXg: homeLuck.luckAdjustedGoals,
   727→    awayXg: awayLuck.luckAdjustedGoals,
   728→  }
   729→  const simulation = runStochasticSimulation(adjustedInput, config)
   730→
   731→  // Step 3: Kelly Criterion for each market
   732→  const kellyResults = {
   733→    home: calculateKelly(
   734→      simulation.matchProbabilities.homeVictory,
   735→      input.bookmakerOdds.home,
   736→      bankroll,
   737→      config.kellyFraction
   738→    ),
   739→    draw: calculateKelly(
   740→      simulation.matchProbabilities.draw,
   741→      input.bookmakerOdds.draw,
   742→      bankroll,
   743→      config.kellyFraction
   744→    ),
   745→    away: calculateKelly(
   746→      simulation.matchProbabilities.awayVictory,
   747→      input.bookmakerOdds.away,
   748→      bankroll,
   749→      config.kellyFraction
   750→    ),
   751→    over25: calculateKelly(
   752→      simulation.totalsMarket.over25,
   753→      input.bookmakerOdds.home * 0.9, // Approximate over/under odds
   754→      bankroll,
   755→      config.kellyFraction
   756→    ),
   757→    under25: calculateKelly(
   758→      simulation.totalsMarket.under25,
   759→      input.bookmakerOdds.home * 1.1,
   760→      bankroll,
   761→      config.kellyFraction
   762→    ),
   763→  }
   764→
   765→  // Step 4: Portfolio allocation
   766→  const portfolioAllocation = calculatePortfolioAllocation([
   767→    { label: `Home Win (${input.homeTeam})`, modelProb: simulation.matchProbabilities.homeVictory, odds: input.bookmakerOdds.home },
   768→    { label: `Draw`, modelProb: simulation.matchProbabilities.draw, odds: input.bookmakerOdds.draw },
   769→    { label: `Away Win (${input.awayTeam})`, modelProb: simulation.matchProbabilities.awayVictory, odds: input.bookmakerOdds.away },
   770→  ], bankroll)
   771→
   772→  // Step 5: Market signals
   773→  const marketSignal = openingOdds
   774→    ? analyzeMarketSignals(openingOdds, input.bookmakerOdds, input.homeTeamId || '', input.homeTeam, input.awayTeam)
   775→    : null
   776→
   777→  // Step 6: Generate recommendation
   778→  const bestBet = [kellyResults.home, kellyResults.draw, kellyResults.away]
   779→    .filter(k => k.action === 'BET')
   780→    .sort((a, b) => b.edgePercentage - a.edgePercentage)[0]
   781→
   782→  const recommendation = bestBet
   783→    ? `RECOMMENDED: ${bestBet.edgePercentage.toFixed(1)}% edge detected. Suggested wager: $${bestBet.suggestedWager.toFixed(2)} (${(bestBet.suggestedFraction * 100).toFixed(2)}% of bankroll). Model probability: ${(bestBet.modelProbability * 100).toFixed(1)}% vs market implied: ${(bestBet.impliedProbability * 100).toFixed(1)}%.`
   784→    : 'No positive expected value bets found for this match. The market appears efficiently priced.'
   785→
   786→  // Risk rating
   787→  const maxEdge = Math.max(
   788→    kellyResults.home.edgePercentage,
   789→    kellyResults.draw.edgePercentage,
   790→    kellyResults.away.edgePercentage
   791→  )
   792→  const riskRating: FullMatchAnalysis['riskRating'] =
   793→    simulation.volatilityIndex > 70 ? 'very-high' :
   794→    simulation.volatilityIndex > 50 ? 'high' :
   795→    simulation.volatilityIndex > 30 ? 'medium' : 'low'
   796→
   797→  return {
   798→    matchInput: adjustedInput,
   799→    simulation,
   800→    kellyResults,
   801→    portfolioAllocation,
   802→    marketSignal,
   803→    luckAdjustment: { home: homeLuck, away: awayLuck },
   804→    recommendation,
   805→    riskRating,
   806→  }
   807→}

   815→// ── Dixon-Coles Correlation Enhancement ─────────────────────────────────────────
   816→
   817→export interface DixonColesParams {
   818→  rho: number  // Correlation parameter (typically -0.1 to -0.2)
   819→}
   820→
   821→/**
   822→ * Applies Dixon-Coles low-scoreline correction to Poisson probabilities.
   823→ * Adjusts P(0,0), P(1,0), P(0,1), P(1,1) to account for football's
   824→ * goal-scoring correlation: when one team scores, the game state shifts,
   825→ * making the other team's response correlated rather than independent.
   826→ *
   827→ * This enhances ELASTICO's existing independent Poisson sampling by
   828→ * introducing the mathematically-proven Dixon-Coles (1997) correction.
   829→ */
   830→export function dixonColesCorrection(
   831→  homeGoals: number,
   832→  awayGoals: number,
   833→  lambdaHome: number,
   834→  lambdaAway: number,
   835→  rho: number = -0.13
   836→): number {
   837→  if (rho === 0) return 1.0
   838→
   839→  const isLowScoreline =
   840→    (homeGoals === 0 && awayGoals === 0) ||
   841→    (homeGoals === 0 && awayGoals === 1) ||
   842→    (homeGoals === 1 && awayGoals === 0) ||
   843→    (homeGoals === 1 && awayGoals === 1)
   844→
   845→  if (!isLowScoreline) return 1.0
   846→
   847→  const pHome = poissonPMF(homeGoals, lambdaHome)
   848→  const pAway = poissonPMF(awayGoals, lambdaAway)
   849→  const independent = pHome * pAway
   850→
   851→  let correction: number
   852→  if (homeGoals === 0 && awayGoals === 0) {
   853→    correction = 1 - lambdaHome * lambdaAway * rho
   854→  } else if (homeGoals === 0 && awayGoals === 1) {
   855→    correction = 1 + lambdaHome * rho
   856→  } else if (homeGoals === 1 && awayGoals === 0) {
   857→    correction = 1 + lambdaAway * rho
   858→  } else {
   859→    correction = 1 - rho
   860→  }
   861→
   862→  const corrected = independent * Math.max(correction, 0)
   863→  return independent > 0 ? Math.min(corrected / independent, 1.5) : 1.0
   864→}
   865→
   866→function poissonPMF(k: number, lambda: number): number {
   867→  if (k < 0 || lambda <= 0) return k === 0 ? 1 : 0
   868→  return Math.exp(-lambda) * Math.pow(lambda, k) / factorial(k)
   869→}
   870→
   871→function factorial(n: number): number {
   872→  if (n <= 1) return 1
   873→  let result = 1
   874→  for (let i = 2; i <= n; i++) result *= i
   875→  return result
   876→}

   880→export interface EnsembleInput {
   881→  eloHomeProb: number
   882→  eloDrawProb: number
   883→  eloAwayProb: number
   884→  stochasticHomeProb: number
   885→  stochasticDrawProb: number
   886→  stochasticAwayProb: number
   887→  marketHomeProb?: number
   888→  marketDrawProb?: number
   889→  marketAwayProb?: number
   890→}
   891→
   892→export interface EnsembleResult {
   893→  homeWin: number
   894→  draw: number
   895→  awayWin: number
   896→  weights: { elo: number; stochastic: number; dixonColes: number; market: number }
   897→  modelAgreement: 'high' | 'moderate' | 'disagree'
   898→}
   899→
   900→/**
   901→ * Weighted ensemble combining ELO, Stochastic (Merton J-D), and optional Market implied probabilities.
   902→ * Weights: ELO 30%, Stochastic 45%, Market 25% (when available)
   903→ * Without market: ELO 35%, Stochastic 65%
   904→ */
   905→export function computeEnsemble(input: EnsembleInput): EnsembleResult {
   906→  const hasMarket = input.marketHomeProb !== undefined && input.marketHomeProb > 0
   907→
   908→  let wElo: number, wStochastic: number, wMarket: number
   909→  if (hasMarket) {
   910→    wElo = 0.30; wStochastic = 0.45; wMarket = 0.25
   911→  } else {
   912→    wElo = 0.35; wStochastic = 0.65; wMarket = 0
   913→  }
   914→
   915→  const home = wElo * input.eloHomeProb + wStochastic * input.stochasticHomeProb + wMarket * (input.marketHomeProb || 0)
   916→  const draw = wElo * input.eloDrawProb + wStochastic * input.stochasticDrawProb + wMarket * (input.marketDrawProb || 0)
   917→  const away = wElo * input.eloAwayProb + wStochastic * input.stochasticAwayProb + wMarket * (input.marketAwayProb || 0)
   918→
   919→  const total = home + draw + away
   920→  const h = home / total
   921→  const d = draw / total
   922→  const a = away / total
   923→
   924→  const eloFav = Math.max(input.eloHomeProb, input.eloDrawProb, input.eloAwayProb)
   925→  const stochFav = Math.max(input.stochasticHomeProb, input.stochasticDrawProb, input.stochasticAwayProb)
   926→  const ensembleFav = Math.max(h, d, a)
   927→
   928→  const modelsAgree = (
   929→    (eloFav === ensembleFav || Math.abs(eloFav - ensembleFav) < 0.05) &&
   930→    (stochFav === ensembleFav || Math.abs(stochFav - ensembleFav) < 0.05)
   931→  )
   932→
   933→  return {
   934→    homeWin: Math.round(h * 10000) / 10000,
   935→    draw: Math.round(d * 10000) / 10000,
   936→    awayWin: Math.round(a * 10000) / 10000,
   937→    weights: { elo: wElo, stochastic: wStochastic, dixonColes: 0, market: wMarket },
   938→    modelAgreement: modelsAgree ? 'high' : (Math.abs(eloFav - stochFav) < 0.15 ? 'moderate' : 'disagree'),
   939→  }
   940→}

   942→// ── Evaluation Metrics (Brier Score, Log Loss) ─────────────────────────────────
   943→
   944→export interface EvalMetrics {
   945→  brierScore: number
   946→  logLoss: number
   947→  accuracy: number
   948→  sampleSize: number
   949→}
   950→
   951→/**
   952→ * Computes Brier score and log loss for probability predictions.
   953→ * Brier: mean squared error [0,1]. Log loss: -log likelihood.
   954→ */
   955→export function computeEvalMetrics(
   956→  predictions: { predicted: number; actual: number }[]
   957→): EvalMetrics {
   958→  if (predictions.length === 0) {
   959→    return { brierScore: 0, logLoss: 0, accuracy: 0, sampleSize: 0 }
   960→  }
   961→
   962→  let brierSum = 0
   963→  let logLossSum = 0
   964→  let correct = 0
   965→
   966→  for (const p of predictions) {
   967→    brierSum += (p.predicted - p.actual) ** 2
   968→    const clampedP = Math.max(0.01, Math.min(0.99, p.predicted))
   969→    if (p.actual === 1) {
   970→      logLossSum += -Math.log(clampedP)
   971→    } else {
   972→      logLossSum += -Math.log(1 - clampedP)
   973→    }
   974→    if ((p.predicted > 0.5 && p.actual === 1) || (p.predicted <= 0.5 && p.actual === 0)) {
   975→      correct++
   976→    }
   977→  }
   978→
   979→  const n = predictions.length
   980→  return {
   981→    brierScore: Math.round((brierSum / n) * 100000) / 100000,
   982→    logLoss: Math.round((logLossSum / n) * 100000) / 100000,
   983→    accuracy: Math.round((correct / n) * 10000) / 10000,
   984→    sampleSize: n,
   985→  }
   986→}

   988→// ── Calibration / Reliability Bins ──────────────────────────────────────────────
   989→
   990→export interface CalibrationBin {
   991→  predictedRange: string
   992→  total: number
   993→  actualRate: number
   994→  avgPredicted: number
   995→  sampleError: number
   996→}
   997→
   998→/**
   999→ * Computes calibration/reliability diagram bins.
  1000→ * Well-calibrated: actualRate ≈ avgPredicted in every bin.
  1001→ */
  1002→export function computeCalibrationBins(
  1003→  predictions: { predicted: number; actual: number }[],
  1004→  nBins: number = 10
  1005→): CalibrationBin[] {
  1006→  if (predictions.length < nBins) return []
  1007→
  1008→  const bins: CalibrationBin[] = []
  1009→  const binSize = 1.0 / nBins
  1010→
  1011→  for (let i = 0; i < nBins; i++) {
  1012→    const lower = i * binSize
  1013→    const upper = (i + 1) * binSize
  1014→    const binPreds = predictions.filter(p => lower <= p.predicted && p.predicted < upper)
  1015→
  1016→    if (binPreds.length === 0) continue
  1017→
  1018→    const total = binPreds.length
  1019→    const actual = binPreds.reduce((s, p) => s + p.actual, 0)
  1020→    const avgPred = binPreds.reduce((s, p) => s + p.predicted, 0) / total
  1021→    const actualRate = actual / total
  1022→    const sampleError = Math.sqrt(actualRate * (1 - actualRate) / total)
  1023→
  1024→    bins.push({
  1025→      predictedRange: `${lower.toFixed(2)}-${upper.toFixed(2)}`,
  1026→      total,
  1027→      actualRate: Math.round(actualRate * 10000) / 10000,
  1028→      avgPredicted: Math.round(avgPred * 10000) / 10000,
  1029→      sampleError: Math.round(sampleError * 10000) / 10000,
  1030→    })
  1031→  }
  1032→
  1033→  return bins
  1034→}

  1038→export interface PredictionSignals {
  1039→  signals: string[]
  1040→  risks: string[]
  1041→}
  1042→
  1043→/**
  1044→ * Generates human-readable prediction signals and risk factors.
  1045→ * Maps to UI explainability: "What? Why? How strong? What could change it?"
  1046→ */
  1047→export function generatePredictionSignals(opts: {
  1048→  eloDiff: number
  1049→  homeXg: number
  1050→  awayXg: number
  1051→  hasOdds: boolean
  1052→  hasXgData: boolean
  1053→  modelAgreement: 'high' | 'moderate' | 'disagree'
  1054→  volatilityIndex: number
  1055→  homeForm?: string
  1056→  awayForm?: string
  1057→  injuryCount?: number
  1058→}): PredictionSignals {
  1059→  const signals: string[] = []
  1060→  const risks: string[] = []
  1061→
  1062→  const { eloDiff, homeXg, awayXg, hasOdds, hasXgData, modelAgreement, volatilityIndex } = opts
  1063→
  1064→  if (Math.abs(eloDiff) > 200) {
  1065→    const fav = eloDiff > 0 ? 'Home' : 'Away'
  1066→    signals.push(`Strong ELO advantage for ${fav.toLowerCase()} team (+${Math.abs(Math.round(eloDiff))} pts)`)
  1067→  } else if (Math.abs(eloDiff) > 80) {
  1068→    const fav = eloDiff > 0 ? 'Home' : 'Away'
  1069→    signals.push(`Moderate ELO advantage for ${fav.toLowerCase()} team (+${Math.abs(Math.round(eloDiff))} pts)`)
  1070→  }
  1071→
  1072→  if (hasXgData) {
  1073→    const xgRatio = homeXg / (awayXg || 0.01)
  1074→    if (xgRatio > 1.4) {
  1075→      signals.push(`Home xG significantly higher (${homeXg.toFixed(2)} vs ${awayXg.toFixed(2)})`)
  1076→    } else if (xgRatio < 0.7) {
  1077→      signals.push(`Away xG significantly higher (${awayXg.toFixed(2)} vs ${homeXg.toFixed(2)})`)
  1078→    }
  1079→  }
  1080→
  1081→  if (modelAgreement === 'high') {
  1082→    signals.push('All prediction models agree on the most likely outcome')
  1083→  } else if (modelAgreement === 'disagree') {
  1084→    risks.push('Prediction models disagree on the most likely outcome')
  1085→  }
  1086→
  1087→  if (volatilityIndex > 60) {
  1088→    risks.push(`High match volatility index (${Math.round(volatilityIndex)}) — wider range of outcomes possible`)
  1089→  }
  1090→
  1091→  if (!hasOdds) {
  1092→    risks.push('No market odds available — market model excluded from ensemble')
  1093→  }
  1094→  if (!hasXgData) {
  1095→    risks.push('No xG data available — predictions rely on ELO and form only')
  1096→  }
  1097→
  1098→  if (opts.homeForm && opts.homeForm.length >= 3) {
  1099→    const recentWins = (opts.homeForm.match(/W/g) || []).length
  1100→    if (recentWins >= 4) signals.push(`Home team in excellent form (${opts.homeForm.slice(0, 5)})`)
  1101→    const recentLosses = (opts.homeForm.match(/L/g) || []).length
  1102→    if (recentLosses >= 3) risks.push(`Home team in poor form (${opts.homeForm.slice(0, 5)})`)
  1103→  }
  1104→  if (opts.awayForm && opts.awayForm.length >= 3) {
  1105→    const recentWins = (opts.awayForm.match(/W/g) || []).length
  1106→    if (recentWins >= 4) signals.push(`Away team in excellent form (${opts.awayForm.slice(0, 5)})`)
  1107→    const recentLosses = (opts.awayForm.match(/L/g) || []).length
  1108→    if (recentLosses >= 3) risks.push(`Away team in poor form (${opts.awayForm.slice(0, 5)})`)
  1109→  }
  1110→
  1111→  if (opts.injuryCount && opts.injuryCount > 0) {
  1112→    risks.push(`${opts.injuryCount} player(s) unavailable — injury adjustments applied`)
  1113→  }
  1114→
  1115→  return { signals, risks }
  1116→}
