    39→interface EspnTeam {
    40→  id?: string; name?: string; abbreviation?: string; logo?: string; color?: string
    41→}
    42→interface EspnMatch {
    43→  id: string; homeTeam: EspnTeam; awayTeam: EspnTeam
    44→  homeScore: number; awayScore: number; status: string; minute?: number
    45→  competition: string; competitionCode?: string; date?: string; venue?: string
    46→}
    47→interface StandingRow {
    48→  rank: number; name: string; code: string; logo: string; color: string
    49→  played: number; wins: number; draws: number; losses: number; points: number
    50→}
    51→
    52→function useLeagueStandings(competition = 'PL') {
    53→  const [rows, setRows] = useState<StandingRow[]>([])
    54→  const [loading, setLoading] = useState(true)
    55→  const [source, setSource] = useState('')
    56→
    57→  useEffect(() => {
    58→    setLoading(true)
    59→    // Use /api/standings which returns football-data.org data (real) with ESPN fallback
    60→    fetch(`/api/standings?competition=${competition}`)
    61→      .then(r => r.ok ? r.json() : { data: [], standings: [] })
    62→      .then((data: any) => {
    63→        const items = (data.standings || []).slice(0, 10).map((t: any, i: number) => ({
    64→          rank: t.position ?? t.position ?? i + 1,
    65→          name: t.team || t.name || '?',
    66→          code: t.code || t.tla || '',
    67→          logo: t.crest || t.logo || '',
    68→          color: t.color || '',
    69→          played: t.played ?? 0,
    70→          wins: t.won ?? 0, draws: t.drawn ?? 0, losses: t.lost ?? 0,
    71→          points: t.points ?? 0,
    72→        }))
    73→        setRows(items)
    74→        setSource(data.source || 'unknown')
    75→      })
    76→      .catch(() => setRows([]))
    77→      .finally(() => setLoading(false))
    78→  }, [competition])
    79→
    80→  return { rows, loading, source }
    81→}
