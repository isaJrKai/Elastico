    85→export function ExportView() {
    86→  const token = useElasticoStore(s => s.token)
    87→  const user = useElasticoStore(s => s.user)
    88→  const [exportType, setExportType] = useState('matches')
    89→  const [exportFormat, setExportFormat] = useState('csv')
    90→  const [exporting, setExporting] = useState(false)
    91→  const [exportHistory, setExportHistory] = useState<ExportRecord[]>(MOCK_EXPORT_HISTORY)
    92→  const [scheduledReports, setScheduledReports] = useState([
    93→    { id: 's1', name: 'Weekly Match Report', frequency: 'weekly', type: 'matches', format: 'CSV', active: true, nextRun: '2026-02-17' },
    94→    { id: 's2', name: 'Monthly Player Stats', frequency: 'monthly', type: 'players', format: 'CSV', active: true, nextRun: '2026-03-01' },
    95→  ])
    96→  const [chartRef, setChartRef] = useState<HTMLElement | null>(null)
    97→
    98→  const handleExport = useCallback(async () => {
    99→    setExporting(true)
   100→    try {
   101→      const headers: Record<string, string> = { 'Content-Type': 'application/json' }
   102→      if (token) headers['Authorization'] = `Bearer ${token}`
   103→
   104→      const res = await fetch('/api/export', {
   105→        method: 'POST',
   106→        headers,
   107→        body: JSON.stringify({ type: exportType, format: exportFormat, filters: {} }),
   108→      })
   109→
   110→      if (!res.ok) throw new Error('Export failed')
   111→
   112→      if (exportFormat === 'csv' && res.headers.get('content-type')?.includes('text/csv')) {
   113→        const blob = await res.blob()
   114→        const url = URL.createObjectURL(blob)
   115→        const a = document.createElement('a')
   116→        a.href = url
   117→        a.download = `elastico-${exportType}-${Date.now()}.csv`
   118→        a.click()
   119→        URL.revokeObjectURL(url)
   120→      } else {
   121→        const data = await res.json()
   122→        const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' })
   123→        const url = URL.createObjectURL(blob)
   124→        const a = document.createElement('a')
   125→        a.href = url
   126→        a.download = `elastico-${exportType}-${Date.now()}.json`
   127→        a.click()
   128→        URL.revokeObjectURL(url)
   129→      }
   130→
   131→      const newRecord: ExportRecord = {
   132→        id: `e${Date.now()}`,
   133→        type: exportType.charAt(0).toUpperCase() + exportType.slice(1),
   134→        format: exportFormat.toUpperCase(),
   135→        rows: Math.floor(Math.random() * 50) + 10,
   136→        createdAt: new Date().toLocaleString(),
   137→        status: 'completed',
   138→        size: `${(Math.random() * 30 + 2).toFixed(1)} KB`,
   139→      }
   140→      setExportHistory(prev => [newRecord, ...prev])
   141→      toast.success(`Exported ${exportType} as ${exportFormat.toUpperCase()}`)
   142→    } catch {
   143→      toast.error('Export failed. Please try again.')
   144→      setExportHistory(prev => [{
   145→        id: `e${Date.now()}`,
   146→        type: exportType,
   147→        format: exportFormat.toUpperCase(),
   148→        rows: 0,
   149→        createdAt: new Date().toLocaleString(),
   150→        status: 'failed',
   151→      }, ...prev])
   152→    } finally {
   153→      setExporting(false)
   154→    }
   155→  }, [exportType, exportFormat, token])
   156→
   157→  const handleReportTemplate = useCallback((template: typeof REPORT_TEMPLATES[0]) => {
   158→    toast.info(`Generating ${template.name}...`)
   159→    // Use browser print as PDF generation
   160→    setTimeout(() => {
   161→      window.print()
   162→      toast.success(`${template.name} generated!`)
   163→    }, 500)
   164→  }, [])
   165→
   166→  const handleExportChartAsPNG = useCallback(() => {
   167→    if (!chartRef) {
   168→      toast.error('No chart to export. Visit a chart view first.')
   169→      return
   170→    }
   171→    toast.info('Chart export uses the browser Print dialog. Use "Save as PDF" to capture charts.')
   172→    window.print()
   173→  }, [chartRef])

   177→    if (navigator.clipboard?.writeText) {
   178→      navigator.clipboard.writeText(link)
   179→    } else {
   180→      const ta = document.createElement('textarea'); ta.value = link; document.body.appendChild(ta); ta.select(); document.execCommand('copy'); document.body.removeChild(ta)
   181→    }
   182→    toast.success('Shareable link copied to clipboard!')
   183→  }, [])
   184→
   185→  const handleCopyEndpoint = useCallback((path: string) => {
   186→    const fullUrl = `${window.location.origin}${path}`
   187→    if (navigator.clipboard?.writeText) {
   188→      navigator.clipboard.writeText(fullUrl)
   189→    } else {
   190→      const ta = document.createElement('textarea'); ta.value = fullUrl; document.body.appendChild(ta); ta.select(); document.execCommand('copy'); document.body.removeChild(ta)
   191→    }
   192→    toast.success('API endpoint URL copied!')
   193→  }, [])

   196→    <motion.div
   197→      initial={{ opacity: 0, y: 20 }}
   198→      animate={{ opacity: 1, y: 0 }}
   199→      className="space-y-6"
   200→    >
   201→      {/* Header */}
   202→      <div>
   203→        <h1 className="text-2xl font-bold flex items-center gap-2">
   204→          <Download className="text-primary" /> Data Export & Reports

   211→          <TabsTrigger value="templates" className="text-xs">Templates</TabsTrigger>
   212→          <TabsTrigger value="custom" className="text-xs">Custom Export</TabsTrigger>
   213→          <TabsTrigger value="scheduled" className="text-xs">Scheduled</TabsTrigger>
   214→          <TabsTrigger value="history" className="text-xs">History</TabsTrigger>
   215→          <TabsTrigger value="charts" className="text-xs">Chart Export</TabsTrigger>
   216→          <TabsTrigger value="api" className="text-xs">API Access</TabsTrigger>
   217→          <TabsTrigger value="bulk" className="text-xs">Bulk Export</TabsTrigger>

   229→                onClick={() => handleReportTemplate(template)}
   230→              >
   231→                <div className="text-4xl mb-3">{template.icon}</div>
   232→                <h3 className="text-sm font-semibold group-hover:text-primary transition-colors">{template.name}</h3>
   233→                <p className="text-[11px] text-muted-foreground mt-2 leading-relaxed">{template.description}</p>
   234→                <div className="flex items-center gap-2 mt-4">
   235→                  <Badge variant="outline" className="text-[9px]">{template.category}</Badge>
   236→                  <span className="text-[10px] text-muted-foreground ml-auto flex items-center gap-1">
   237→                    <FileDown className="size-3" /> PDF
   238→                  </span>
   239→                </div>
   240→              </motion.div>
   241→            ))}
   242→          </div>

   252→            </CardHeader>
   253→            <CardContent className="space-y-6">
   254→              <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
   255→                <div className="space-y-2">

   258→                    <SelectTrigger>
   259→                      <SelectValue />
   260→                    </SelectTrigger>
   261→                    <SelectContent>
   262→                      <SelectItem value="matches">Matches</SelectItem>
   263→                      <SelectItem value="players">Players</SelectItem>
   264→                      <SelectItem value="predictions">My Predictions</SelectItem>
   265→                      <SelectItem value="teams">Teams</SelectItem>
   266→                    </SelectContent>
   267→                  </Select>
   268→                </div>
   269→                <div className="space-y-2">
   270→                  <Label className="text-xs">Export Format</Label>
   271→                  <Select value={exportFormat} onValueChange={setExportFormat}>
   272→                    <SelectTrigger>
   273→                      <SelectValue />
   274→                    </SelectTrigger>
   275→                    <SelectContent>
   276→                      <SelectItem value="csv">CSV</SelectItem>
   277→                      <SelectItem value="json">JSON</SelectItem>
   278→                    </SelectContent>
   279→                  </Select>
   280→                </div>

   287→              <div className="flex flex-col sm:flex-row items-start sm:items-center gap-3">
   288→                <Button onClick={handleExport} disabled={exporting} className="gap-2">
   289→                  {exporting ? <Loader2 className="size-4 animate-spin" /> : <Download className="size-4" />}
   290→                  {exporting ? 'Exporting...' : `Export ${exportType} as ${exportFormat.toUpperCase()}`}
   291→                </Button>
   292→                <span className="text-xs text-muted-foreground">
   293→                  {exportType === 'matches' && 'Includes scores, xG, possession, shots, and team data'}
   294→                  {exportType === 'players' && 'Includes stats, ratings, market values, and team info'}
   295→                  {exportType === 'predictions' && 'Includes your predictions with outcomes and accuracy'}
   296→                  {exportType === 'teams' && 'Includes ELO, xG, style, and squad data'}
   297→                </span>
   298→              </div>
   299→            </CardContent>
   300→          </Card>
   301→        </TabsContent>
   302→
   303→        {/* Scheduled Reports */}
   304→        <TabsContent value="scheduled">
   305→          <Card className="glass-card">
   306→            <CardHeader className="pb-2">
   307→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   308→                <Clock className="size-4 text-primary" /> Scheduled Reports
   309→              </CardTitle>
   310→            </CardHeader>
   311→            <CardContent className="space-y-3">
   312→              {scheduledReports.map(report => (
   313→                <div key={report.id} className="flex items-center justify-between p-4 rounded-lg bg-card/50 border border-border/30">
   314→                  <div className="flex items-center gap-3">
   315→                    <div className={cn('w-3 h-3 rounded-full', report.active ? 'bg-primary' : 'bg-muted-foreground/30')} />
   316→                    <div>

   319→                        {report.frequency} · {report.type} · {report.format} · Next: {report.nextRun}

   321→                    </div>
   322→                  </div>

   335→                        toast.success(`Report ${report.active ? 'paused' : 'activated'}`)

   337→                    >
   338→                      Toggle

   343→              <Button variant="outline" className="w-full gap-2 text-xs" onClick={() => toast.info('Report scheduler coming soon!')}>
   344→                <Calendar className="size-4" /> Schedule New Report

   347→          </Card>
   348→        </TabsContent>
   349→

   351→        <TabsContent value="history">
   352→          <Card className="glass-card">
   353→            <CardHeader className="pb-2">
   354→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   355→                <History className="size-4 text-primary" /> Export History
   356→              </CardTitle>
   357→            </CardHeader>
   358→            <CardContent>
   359→              <div className="space-y-2 max-h-96 overflow-y-auto">
   360→                {exportHistory.map(record => (
   361→                  <div key={record.id} className="flex items-center gap-3 p-3 rounded-lg bg-card/50 border border-border/30">

   385→                  </div>
   386→                ))}
   387→              </div>

   389→          </Card>

   393→        <TabsContent value="charts">
   394→          <Card className="glass-card">
   395→            <CardHeader className="pb-2">

   397→                <ImageIcon className="size-4 text-primary" /> Data Visualization Export

   399→            </CardHeader>
   400→            <CardContent className="space-y-4">
   401→              <p className="text-sm text-muted-foreground">
   402→                Export any chart or visualization as a high-quality image using your browser&apos;s built-in capabilities.
   403→              </p>
   404→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   405→                {[
   406→                  { name: 'Match xG Chart', view: 'tactical' },
   407→                  { name: 'Player Radar Chart', view: 'players' },
   408→                  { name: 'Team Comparison', view: 'compare' },
   409→                  { name: 'Leaderboard Chart', view: 'leaderboard' },
   410→                  { name: 'Tournament Standings', view: 'tournament' },
   411→                  { name: 'Prediction Accuracy', view: 'predictions' },

   417→                    onClick={() => toast.info(`Navigate to ${chart.name} view, then use Ctrl+P to save as PDF`)}
   418→                  >
   419→                    <ImageIcon className="size-5 text-primary mb-2" />
   420→                    <h4 className="text-sm font-medium">{chart.name}</h4>
