   125→// ── Helper: format date ───────────────────────────────────────────────────────
   126→
   127→function formatDate(dateStr: string | null): string {
   128→  if (!dateStr) return 'TBD'
   129→  const d = new Date(dateStr)
   130→  const now = new Date()
   131→  const diffMs = d.getTime() - now.getTime()
   132→  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24))
   133→
   134→  if (diffDays === 0) return 'Today'
   135→  if (diffDays === 1) return 'Tomorrow'
   136→  if (diffDays === -1) return 'Yesterday'
   137→  if (diffDays > 0 && diffDays <= 7) return `In ${diffDays} days`
   138→
   139→  return d.toLocaleDateString('en-US', {
   140→    month: 'short',
   141→    day: 'numeric',
   142→    hour: '2-digit',
   143→    minute: '2-digit',
   144→  })
   145→}
   146→
   147→function formatTimeAgo(dateStr: string | null): string {
   148→  if (!dateStr) return ''
   149→  const d = new Date(dateStr)
   150→  const now = new Date()
   151→  const diffMs = now.getTime() - d.getTime()
   152→  const diffMins = Math.floor(diffMs / 60000)
   153→  const diffHours = Math.floor(diffMs / 3600000)
   154→  const diffDays = Math.floor(diffMs / 86400000)
   155→
   156→  if (diffMins < 1) return 'Just now'
   157→  if (diffMins < 60) return `${diffMins}m ago`
   158→  if (diffHours < 24) return `${diffHours}h ago`
   159→  if (diffDays < 7) return `${diffDays}d ago`
   160→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   161→}

   196→function CustomTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ value: number; name: string; color: string }>; label?: string }) {

   198→  return (
   199→    <div className="glass-card rounded-lg p-3 text-sm shadow-lg border border-border/50">
   200→      <p className="text-muted-foreground text-xs mb-1">{label}</p>
   201→      {payload.map((entry, i) => (
   202→        <p key={i} className="font-semibold" style={{ color: entry.color }}>
   203→          {entry.name}: {typeof entry.value === 'number' ? entry.value.toFixed(1) : entry.value}
   204→        </p>
   205→      ))}
   206→    </div>
