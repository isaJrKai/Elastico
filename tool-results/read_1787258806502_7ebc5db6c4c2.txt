    47→    47→interface StandingRow {

    51→    51→
    52→    52→function useLeagueStandings(competition = 'PL') {
    53→    53→  const [rows, setRows] = useState<StandingRow[]>([])
    54→    54→  const [loading, setLoading] = useState(true)
    55→    55→  const [source, setSource] = useState('')
    56→    56→
    57→    57→  useEffect(() => {
    58→    58→    setLoading(true)
    59→    59→    // Use /api/standings which returns football-data.org data (real) with ESPN fallback
    60→    60→    fetch(`/api/standings?competition=${competition}`)
    61→    61→      .then(r => r.ok ? r.json() : { data: [], standings: [] })
    62→    62→      .then((data: any) => {
    63→    63→        const items = (data.standings || []).slice(0, 10).map((t: any, i: number) => ({
    64→    64→          rank: t.position ?? t.position ?? i + 1,
    65→    65→          name: t.team || t.name || '?',
    66→    66→          code: t.code || t.tla || '',
    67→    67→          logo: t.crest || t.logo || '',
    68→    68→          color: t.color || '',
    69→    69→          played: t.played ?? 0,
    70→    70→          wins: t.won ?? 0, draws: t.drawn ?? 0, losses: t.lost ?? 0,
    71→    71→          points: t.points ?? 0,
    72→    72→        }))
    73→    73→        setRows(items)
    74→    74→        setSource(data.source || 'unknown')
    75→    75→      })
    76→    76→      .catch(() => setRows([]))
    77→    77→      .finally(() => setLoading(false))
    78→    78→  }, [competition])
    79→    79→
    80→    80→  return { rows, loading, source }
    81→    81→}
