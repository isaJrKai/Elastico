   310→# Changes from existing architecture:
   311→#   1. REDUCED hidden_size: 64 → 48
   312→#      Reason: ~4000 training sequences. 64*2*2=256 FC input with
   313→#      128→64→3 head = ~50K params just in FC. Too many for this data size.
   314→#      48*2*2=192 FC input with 64→3 head = ~14K params. More appropriate.
   315→#   2. REDUCED FC head: 256→128→64→3 → 192→64→3
   316→#      Reason: Removed one FC layer. Less overfitting risk.
   317→#   3. INCREASED dropout: 0.3 → 0.4
   318→#      Reason: XGBoost showed severe overfitting. LSTM with sequences
   319→#      will overfit even more without strong regularization.
   320→#   4. PRESERVED: BiLSTM, attention, batch norm, gradient clipping,
   321→#      LR scheduling, early stopping, Xavier/orthogonal init.
   322→#
   323→
   324→import torch
   325→import torch.nn as nn
   326→
   327→
   328→class FootballLSTM_v1(nn.Module):
   329→    """Two-branch BiLSTM with attention for football prediction.
   330→
   331→    Architecture:
   332→        Home seq (B, T, 15) → BiLSTM → Attention → home_ctx (B, 96)
   333→        Away seq (B, T, 15) → BiLSTM → Attention → away_ctx (B, 96) [shared weights]
   334→        Concat → FC(192→64) → BN → ReLU → Dropout → FC(64→3)
   335→
   336→    Changes from lstm_engine.py v1:
   337→      hidden_size: 64 → 48 (fewer params for ~4K sequences)
   338→      FC head: 256→128→64→3 → 192→64→3 (one fewer layer)
   339→      dropout: 0.3 → 0.4 (stronger regularization)
   340→    """
   341→
   342→    def __init__(self, input_size=15, hidden_size=48, num_layers=2,
   343→                 dropout=0.4, n_classes=3):
   344→        super().__init__()
   345→        self.input_size = input_size
   346→        self.hidden_size = hidden_size
   347→        self.num_layers = num_layers
   348→        self.dropout_rate = dropout
   349→        bidir_hidden = hidden_size * 2  # 96
   350→
   351→        # Shared BiLSTM
   352→        self.lstm = nn.LSTM(
   353→            input_size=input_size,
   354→            hidden_size=hidden_size,
   355→            num_layers=num_layers,
   356→            batch_first=True,
   357→            bidirectional=True,
   358→            dropout=dropout if num_layers > 1 else 0.0,
   359→        )
   360→
   361→        # Attention
   362→        self.attention = nn.Linear(bidir_hidden, 1)
   363→
   364→        # FC head
   365→        fc_input = bidir_hidden * 2  # 192
   366→        self.fc = nn.Sequential(
   367→            nn.Linear(fc_input, 64),
   368→            nn.BatchNorm1d(64),
   369→            nn.ReLU(),
   370→            nn.Dropout(dropout),
   371→            nn.Linear(64, n_classes),
   372→        )
   373→
   374→        self._init_weights()
   375→
   376→    def _init_weights(self):
   377→        for name, param in self.named_parameters():
   378→            if 'lstm' in name and 'weight_ih' in name:
   379→                nn.init.xavier_uniform_(param)
   380→            elif 'lstm' in name and 'weight_hh' in name:
   381→                nn.init.orthogonal_(param)
   382→            elif 'lstm' in name and 'bias' in name:
   383→                nn.init.zeros_(param)
   384→                n = param.size(0)
   385→                start, end = n // 4, n // 2
   386→                nn.init.constant_(param.data[start:end], 1.0)
   387→            elif 'weight' in name and param.ndim >= 2:
   388→                nn.init.xavier_uniform_(param)
   389→            elif 'bias' in name:
   390→                nn.init.zeros_(param)
   391→
   392→    def _attend(self, lstm_out):
   393→        """Softmax attention: (B, T, H*2) → (B, H*2)"""
   394→        attn_logits = self.attention(lstm_out)       # (B, T, 1)
   395→        attn_weights = torch.softmax(attn_logits, dim=1)  # (B, T, 1)
   396→        context = torch.sum(attn_weights * lstm_out, dim=1)  # (B, H*2)
   397→        return context
   398→
   399→    def forward(self, x_home, x_away):
   400→        """
   401→        x_home: (B, T, 15)  x_away: (B, T, 15)
   402→        Returns: logits (B, 3)
   403→        """
   404→        home_out, _ = self.lstm(x_home)
   405→        home_ctx = self._attend(home_out)
   406→
   407→        away_out, _ = self.lstm(x_away)
   408→        away_ctx = self._attend(away_out)
   409→
   410→        combined = torch.cat([home_ctx, away_ctx], dim=1)
   411→        return self.fc(combined)
   412→

   419→def train_lstm(model, train_loader, val_loader, config):
   420→    """Train LSTM with early stopping, gradient clipping, LR scheduling."""
   421→    device = torch.device('cpu')
   422→    model = model.to(device)
   423→
   424→    # Class weights for imbalance
   425→    y_all = []
   426→    for _, _, yb in train_loader:
   427→        y_all.append(yb.numpy())
   428→    y_all = np.concatenate(y_all)
   429→    counts = np.bincount(y_all, minlength=3).astype(float)
   430→    total = counts.sum()
   431→    class_weights = torch.tensor(total / (3 * counts), dtype=torch.float32).to(device)
   432→    print(f'  Class weights: {class_weights.tolist()}')
   433→
   434→    optimizer = torch.optim.Adam(model.parameters(), lr=config['lr'],
   435→                                  weight_decay=config['weight_decay'])
   436→    scheduler = torch.optim.lr_scheduler.ReduceLROnPlateau(
   437→        optimizer, factor=config['lr_factor'], patience=config['lr_patience'],
   438→        min_lr=config['lr_min'], mode='min')
   439→    criterion = nn.CrossEntropyLoss(weight=class_weights)
   440→
   441→    best_val_loss = float('inf')
   442→    best_state = None
   443→    patience_counter = 0
   444→    history = {'train_loss': [], 'val_loss': [],
   445→               'train_acc': [], 'val_acc': [], 'lr': []}
   446→
   447→    for epoch in range(config['max_epochs']):
   448→        # --- Train ---
   449→        model.train()
   450→        train_loss_sum = 0.0
   451→        train_correct = 0
   452→        train_total = 0
   453→        for xh, xa, yb in train_loader:
   454→            xh, xa, yb = xh.to(device), xa.to(device), yb.to(device)
   455→            optimizer.zero_grad()
   456→            logits = model(xh, xa)
   457→            loss = criterion(logits, yb)
   458→            loss.backward()
   459→            torch.nn.utils.clip_grad_norm_(model.parameters(), config['grad_clip'])
   460→            optimizer.step()
   461→            train_loss_sum += loss.item() * yb.size(0)
   462→            train_correct += (logits.argmax(1) == yb).sum().item()
   463→            train_total += yb.size(0)
   464→
   465→        train_loss = train_loss_sum / train_total
   466→        train_acc = train_correct / train_total
   467→
   468→        # --- Validate ---
   469→        model.eval()
   470→        val_loss_sum = 0.0
   471→        val_correct = 0
   472→        val_total = 0
   473→        with torch.no_grad():
   474→            for xh, xa, yb in val_loader:
   475→                xh, xa, yb = xh.to(device), xa.to(device), yb.to(device)
   476→                logits = model(xh, xa)
   477→                loss = criterion(logits, yb)
   478→                val_loss_sum += loss.item() * yb.size(0)
   479→                val_correct += (logits.argmax(1) == yb).sum().item()
   480→                val_total += yb.size(0)
   481→
   482→        val_loss = val_loss_sum / val_total
   483→        val_acc = val_correct / val_total
   484→        current_lr = optimizer.param_groups[0]['lr']
   485→
   486→        scheduler.step(val_loss)
   487→
   488→        history['train_loss'].append(train_loss)
   489→        history['val_loss'].append(val_loss)
   490→        history['train_acc'].append(train_acc)
   491→        history['val_acc'].append(val_acc)
   492→        history['lr'].append(current_lr)
   493→
   494→        if (epoch + 1) % 10 == 0 or epoch == 0:
   495→            print(f'    Epoch {epoch+1:3d}: train_loss={train_loss:.4f} train_acc={train_acc:.4f} '
   496→                  f'val_loss={val_loss:.4f} val_acc={val_acc:.4f} lr={current_lr:.6f}')
   497→
   498→        # Early stopping
   499→        if val_loss < best_val_loss:
   500→            best_val_loss = val_loss
   501→            best_state = copy.deepcopy(model.state_dict())
   502→            patience_counter = 0
   503→        else:
   504→            patience_counter += 1
   505→            if patience_counter >= config['patience']:
   506→                print(f'    Early stopping at epoch {epoch+1}')
   507→                break
   508→
   509→    # Restore best model
   510→    if best_state is not None:
   511→        model.load_state_dict(best_state)
   512→
   513→    return model, history

   520→def compute_ece(y_true, y_prob, n_bins=15):
   521→    """Expected Calibration Error."""
   522→    n_classes = y_prob.shape[1]
   523→    ece = 0.0
   524→    for c in range(n_classes):
   525→        y_binary = (y_true == c).astype(int)
   526→        prob_c = y_prob[:, c]
   527→        edges = np.linspace(0, 1, n_bins + 1)
   528→        for i in range(n_bins):
   529→            mask = (prob_c >= edges[i]) & (prob_c < edges[i + 1])
   530→            if i == n_bins - 1:
   531→                mask = (prob_c >= edges[i]) & (prob_c <= edges[i + 1])
   532→            n_bin = mask.sum()
   533→            if n_bin == 0:
   534→                continue
   535→            ece += (n_bin / len(y_true)) * abs(prob_c[mask].mean() - y_binary[mask].mean())
   536→    return ece / n_classes

   618→def compute_prediction_correlation(proba_a, proba_b, y_true):
   619→    """Correlation between two models' predictions."""
   620→    p_a = proba_a.flatten()
   621→    p_b = proba_b.flatten()
   622→    corr = float(np.corrcoef(p_a, p_b)[0, 1])
   623→    pred_a = np.argmax(proba_a, axis=1)
   624→    pred_b = np.argmax(proba_b, axis=1)
   625→    agreement = float((pred_a == pred_b).mean())
   626→ both_correct = float(((pred_a == y_true) & (pred_b == y_true)).mean())
   627→    # When they disagree, who is right more often?
   628→    disagree_mask = pred_a != pred_b
   629→    if disagree_mask.sum() > 0:
   630→        a_right = (pred_a[disagree_mask] == y_true[disagree_mask]).mean()
   631→        b_right = (pred_b[disagree_mask] == y_true[disagree_mask]).mean()
   632→    else:
   633→        a_right = b_right = 0.0
   634→    return {
   635→        'probability_correlation': corr,
   636→        'prediction_agreement': agreement,
   637→        'both_correct_rate': both_correct,
   638→        'disagree_rate': float(disagree_mask.mean()),
   639→        'lstm_right_on_disagree': float(a_right),
   640→        'xgb_right_on_disagree': float(b_right),
   641→    }

   644→def train_xgb_football_only(df_train, df_test, football_features):
   645→    """Train a football-only XGBoost on the same temporal split for direct comparison."""
   646→    import xgboost as xgb
   647→
   648→    X_train = df_train[football_features].values.astype(np.float32)
   649→    y_train = df_train['result_encoded'].values.astype(np.int32)
   650→    X_test = df_test[football_features].values.astype(np.float32)
   651→    y_test = df_test['result_encoded'].values.astype(np.int32)
   652→
   653→    # Filter complete cases
   654→    train_mask = ~np.isnan(X_train).any(axis=1)
   655→    test_mask = ~np.isnan(X_test).any(axis=1)
   656→    X_train, y_train = X_train[train_mask], y_train[train_mask]
   657→    X_test, y_test = X_test[test_mask], y_test[test_mask]
   658→
   659→    counts = np.bincount(y_train, minlength=3)
   660→    total = counts.sum()
   661→    sw = {i: total / (3 * c) for i, c in enumerate(counts) if c > 0}
   662→    sample_weights = np.array([sw.get(y, 1.0) for y in y_train])
   663→
   664→    dtrain = xgb.DMatrix(X_train, label=y_train, weight=sample_weights,
   665→                          feature_names=football_features)
   666→    dtest = xgb.DMatrix(X_test, label=y_test, feature_names=football_features)
   667→
   668→    params = {'objective': 'multi:softprob', 'num_class': 3,
   669→              'eval_metric': 'mlogloss', 'max_depth': 5, 'eta': 0.05,
   670→              'subsample': 0.8, 'colsample_bytree': 0.8,
   671→              'min_child_weight': 5, 'gamma': 0.1,
   672→              'lambda': 1.5, 'alpha': 0.1, 'seed': 42, 'nthread': 4}
   673→
   674→    bst = xgb.train(params, dtrain, num_boost_round=1000,
   675→                    evals=[(dtrain, 'train'), (dtest, 'test')],
   676→                    early_stopping_rounds=100, verbose_eval=False)
   677→
   678→    proba = bst.predict(dtest)
   679→    pred = np.argmax(proba, axis=1)
   680→    return proba, pred, y_test
