   170→def _safe_div(numerator: float, denominator: float, default: float = 0.0) -> float:
   171→    """Divide with safe fallback for zero denominators."""
   172→    if denominator == 0 or not math.isfinite(denominator):
   173→        return default
   174→    result = numerator / denominator
   175→    return result if math.isfinite(result) else default
   176→
   177→
   178→def _safe_avg(values: list[float] | np.ndarray, default: float = 0.0) -> float:
   179→    """Mean of a numeric sequence, returning *default* for empty input."""
   180→    arr = np.asarray(values, dtype=float)
   181→    if arr.size == 0:
   182→        return default
   183→    return float(np.nanmean(arr))
   184→
   185→
   186→def _linear_trend(values: list[float] | np.ndarray, default: float = 0.0) -> float:
   187→    """Slope of a simple linear fit (most recent → oldest = positive trend)."""
   188→    arr = np.asarray(values, dtype=float)
   189→    if arr.size < 2:
   190→        return default
   191→    x = np.arange(len(arr), dtype=float)
   192→    valid = np.isfinite(arr)
   193→    if valid.sum() < 2:
   194→        return default
   195→    x = x[valid]
   196→    y = arr[valid]
   197→    slope = np.polyfit(x, y, 1)[0]
   198→    return float(slope) if math.isfinite(slope) else default

   201→def _compute_streaks(
   202→    results: list[str] | list[int] | None,
   203→) -> tuple[int, int, int]:
   204→    """Return (win_streak, unbeaten_streak, loss_streak) from recent results.
   205→
   206→    *results* is ordered **most recent first**.  Each element is one of:
   207→    - 'W' / 1  → win
   208→    - 'D' / 0  → draw
   209→    - 'L' / -1 → loss
   210→    """
   211→    if not results:
   212→        return 0, 0, 0
   213→    win_s = unbeaten_s = loss_s = 0
   214→    for r in results:
   215→        if r in ('W', 1):
   216→            win_s += 1
   217→            unbeaten_s += 1
   218→        elif r in ('D', 0):
   219→            unbeaten_s += 1
   220→            break
   221→        elif r in ('L', -1):
   222→            break
   223→        else:
   224→            break
   225→    for r in results:
   226→        if r in ('L', -1):
   227→            loss_s += 1
   228→        else:
   229→            break
   230→    return win_s, unbeaten_s, loss_s
   231→
   232→
   233→def _form_points(results: list[str] | list[int] | None) -> float:
   234→    """Sum of 3/1/0 for W/D/L over recent results (max 15 for 5 matches)."""
   235→    if not results:
   236→        return 0.0
   237→    mapping = {'W': 3, 'D': 1, 'L': 0, 1: 3, 0: 1, -1: 0}
   238→    return sum(mapping.get(r, 0) for r in results)

   241→def _parse_date(value: Any) -> Optional[datetime]:
   242→    """Coerce *value* to a ``datetime``.  Returns ``None`` on failure."""
   243→    if isinstance(value, datetime):
   244→        return value
   245→    if isinstance(value, str):
   246→        for fmt in ("%Y-%m-%d", "%Y-%m-%dT%H:%M:%S", "%Y-%m-%d %H:%M:%S"):
   247→            try:
   248→                return datetime.strptime(value, fmt)
   249→            except ValueError:
   250→                continue
   251→    if isinstance(value, (int, float)):
   252→        try:
   253→            return datetime.fromtimestamp(value)
   254→        except (OSError, ValueError):
   255→            pass
   256→    return None

   285→    # ------------------------------------------------------------------
   286→    # Public API

   289→    def build_match_features(
   290→        self,
   291→        match_data: dict[str, Any],
   292→        home_stats: dict[str, Any],
   293→        away_stats: dict[str, Any],
   294→        h2h_data: list[dict[str, Any]] | None = None,
   295→        odds: dict[str, Any] | None = None,
   296→    ) -> dict[str, float]:
   297→        """Build a feature vector (50 values) for a **single** match prediction.
   298→
   299→        Parameters
   300→        ----------
   301→        match_data : dict
   302→            Must contain ``date`` (str / datetime / timestamp).
   303→        home_stats, away_stats : dict
   304→            Per-team season statistics.  Expected keys (all optional):
   305→
   306→            ELO: ``elo``, ``elo_history`` (list[float])
   307→            Goals: ``goals_scored``, ``goals_conceded``,
   308→                   ``goals_scored_home``, ``goals_conceded_home`` (or ``_away``)
   309→            xG: ``xg_for``, ``xg_for_home`` (or ``_away``)
   310→            Form: ``recent_results`` (list[str] 'W'/'D'/'L', most-recent first),
   311→                  ``recent_goals_scored`` (list[float]),
   312→                  ``last_match_date`` (str/datetime)
   313→            Shots: ``shots_total``, ``shots_on_target``
   314→            Venue: ``home_wins``, ``home_draws``, ``home_losses``,
   315→                   ``away_wins``, ``away_draws``, ``away_losses``,
   316→                   ``home_goals``, ``away_goals``,
   317→                   ``clean_sheets_home``, ``clean_sheets_away``,
   318→                   ``matches_played_home``, ``matches_played_away``
   319→        h2h_data : list[dict], optional
   320→            List of previous head-to-head match dicts.  Each needs
   321→            ``home_goals`` and ``away_goals`` (or ``score`` like '2-1').
   322→        odds : dict, optional
   323→            ``home_odds``, ``draw_odds``, ``away_odds`` (decimal).
   324→
   325→        Returns
   326→        -------
   327→        dict[str, float]
   328→            Mapping of all 50 feature names to float values.
   329→        """
   330→        feats: dict[str, float] = {}
   331→
   332→        # 1. ELO features ─────────────────────────────────────────────
   333→        feats.update(self._elo_features(home_stats, away_stats))
   334→
   335→        # 2. Goal features ────────────────────────────────────────────
   336→        feats.update(self._goal_features(home_stats, away_stats))
   337→
   338→        # 3. Form features ────────────────────────────────────────────
   339→        feats.update(self._form_features(home_stats, away_stats))
   340→
   341→        # 4. Head-to-head features ─────────────────────────────────────
   342→        feats.update(self._h2h_features(h2h_data))
   343→
   344→        # 5. Home/away split features ──────────────────────────────────
   345→        feats.update(self._venue_features(home_stats, away_stats))
   346→
   347→        # 6. Market / odds features ────────────────────────────────────
   348→        feats.update(self._market_features(odds))
   349→
   350→        # 7. Temporal features ─────────────────────────────────────────
   351→        feats.update(self._temporal_features(match_data, home_stats, away_stats))
   352→
   353→        # 8. Advanced features ─────────────────────────────────────────
   354→        feats.update(self._advanced_features(home_stats, away_stats))
   355→
   356→        # Guarantee ordered output matching FEATURE_NAMES
   357→        return {name: feats.get(name, 0.0) for name in FEATURE_NAMES}
   358→
   359→    # ------------------------------------------------------------------
   360→    def build_training_dataset(
   361→        self,
   362→        matches: list[dict[str, Any]],
   363→        window: int = 10,
   364→    ) -> pd.DataFrame:
   365→        """Build a training dataset from historical matches.
   366→
   367→        Each match in *matches* must contain:
   368→
   369→        - ``home_team``, ``away_team``
   370→        - ``home_goals``, ``away_goals``  (for target + stats)
   371→        - ``date``
   372→        - ``home_odds``, ``draw_odds``, ``away_odds``  (optional)
   373→
   374→        The method computes a sliding-window of *window* previous matches
   375→        per team to derive form, streak, and trend features.  Missing
   376→        historical data is back-filled with league-average defaults.
   377→
   378→        Parameters
   379→        ----------
   380→        matches : list[dict]
   381→            Historical match records.  Must be **chronologically sorted**
   382→            (oldest first) for correct sliding-window behaviour.
   383→        window : int
   384→            Number of recent matches to use for rolling form stats.
   385→
   386→        Returns
   387→        -------
   388→        pd.DataFrame
   389→            One row per match.  Columns are the 50 features plus
   390→            ``target`` (0=away, 1=draw, 2=home).
   391→        """
   392→        if not matches:
   393→            return pd.DataFrame(columns=[*FEATURE_NAMES, "target"])
   394→
   395→        # Index: team_id → list of (date, goals_scored, goals_conceded, result)
   396→        team_history: dict[str, list[tuple]] = defaultdict(list)
   397→        # team_id → list of home-only / away-only records
   398→        team_home_history: dict[str, list[tuple]] = defaultdict(list)
   399→        team_away_history: dict[str, list[tuple]] = defaultdict(list)
   400→
   401→        rows: list[dict[str, float]] = []
   402→
   403→        for match in matches:
   404→            home_team = str(match.get("home_team", ""))
   405→            away_team = str(match.get("away_team", ""))
   406→            hg = int(match.get("home_goals", 0))
   407→            ag = int(match.get("away_goals", 0))
   408→
   409→            # Derive result codes (from home perspective)
   410→            if hg > ag:
   411→                home_result, away_result = "W", "L"
   412→            elif hg < ag:
   413→                home_result, away_result = "L", "W"
   414→            else:
   415→                home_result, away_result = "D", "D"
   416→
   417→            match_date = _parse_date(match.get("date")) or datetime.now()
   418→
   419→            # ── Build per-team stats dicts from rolling history ───────
   420→            home_stats = self._stats_from_history(
   421→                team_history[home_team],
   422→                team_home_history[home_team],
   423→                team_away_history[home_team],
   424→                window=window,
   425→            )
   426→            away_stats = self._stats_from_history(
   427→                team_history[away_team],
   428→                team_home_history[away_team],
   429→                team_away_history[away_team],
   430→                window=window,
   431→            )
   432→
   433→            # Set last-match date for rest-day calculation
   434→            if team_history[home_team]:
   435→                home_stats["last_match_date"] = team_history[home_team][-1][0].isoformat()
   436→            if team_history[away_team]:
   437→                away_stats["last_match_date"] = team_history[away_team][-1][0].isoformat()
   438→
   439→            # ── Head-to-head (use previous matches between these teams) ─
   440→            h2h: list[dict[str, Any]] = []
   441→            for prev in matches:
   442→                if prev is match:
   443→                    continue
   444→                if (
   445→                    str(prev.get("home_team")) == home_team
   446→                    and str(prev.get("away_team")) == away_team
   447→                ) or (
   448→                    str(prev.get("home_team")) == away_team
   449→                    and str(prev.get("away_team")) == home_team
   450→                ):
   451→                    phg = int(prev.get("home_goals", 0))
   452→                    pag = int(prev.get("away_goals", 0))
   453→                    # Normalise to current home perspective
   454→                    if str(prev.get("home_team")) == home_team:
   455→                        h2h.append({"home_goals": phg, "away_goals": pag})
   456→                    else:
   457→                        h2h.append({"home_goals": pag, "away_goals": phg})
   458→
   459→            # ── Odds ──────────────────────────────────────────────────
   460→            odds: dict[str, Any] | None = None
   461→            if match.get("home_odds") and match.get("draw_odds") and match.get("away_odds"):
   462→                odds = {
   463→                    "home_odds": float(match["home_odds"]),
   464→                    "draw_odds": float(match["draw_odds"]),
   465→                    "away_odds": float(match["away_odds"]),
   466→                }
   467→
   468→            # ── Build features ────────────────────────────────────────
   469→            feats = self.build_match_features(
   470→                match_data={"date": match_date},
   471→                home_stats=home_stats,
   472→                away_stats=away_stats,
   473→                h2h_data=h2h if h2h else None,
   474→                odds=odds,
   475→            )
   476→
   477→            # Target: 0=away, 1=draw, 2=home
   478→            if hg > ag:
   479→                target = 2
   480→            elif hg < ag:
   481→                target = 0
   482→            else:
   483→                target = 1
   484→
   485→            feats["target"] = float(target)
   486→            rows.append(feats)
   487→
   488→            # ── Update histories for future windows ───────────────────
   489→            team_history[home_team].append(
   490→                (match_date, hg, ag, home_result)
   491→            )
   492→            team_history[away_team].append(
   493→                (match_date, ag, hg, away_result)
   494→            )
   495→            team_home_history[home_team].append(
   496→                (match_date, hg, ag, home_result)
   497→            )
   498→            team_away_history[away_team].append(
   499→                (match_date, ag, hg, away_result)
   500→            )
   501→
   502→        df = pd.DataFrame(rows)
   503→        # Ensure column order
   504→        cols = [*FEATURE_NAMES, "target"]
   505→        for c in cols:
   506→            if c not in df.columns:
   507→                df[c] = 0.0
   508→        return df[cols]
   509→

   525→    def _elo_features(
   526→        self, home_stats: dict[str, Any], away_stats: dict[str, Any]
   527→    ) -> dict[str, float]:
   528→        elo_home = float(home_stats.get("elo", 1500.0))
   529→        elo_away = float(away_stats.get("elo", 1500.0))
   530→        elo_diff = elo_home - elo_away
   531→        elo_haa = elo_diff + self.home_advantage_elo
   532→
   533→        # Form trend: slope of recent ELO history
   534→        home_history = home_stats.get("elo_history", [])
   535→        elo_trend = _linear_trend(home_history, default=0.0)
   536→
   537→        return {
   538→            "elo_home": elo_home,
   539→            "elo_away": elo_away,
   540→            "elo_diff": elo_diff,
   541→            "elo_home_advantage_adjusted": elo_haa,
   542→            "elo_form_trend": elo_trend,
   543→        }

   546→
   547→    def _goal_features(
   548→        self, home_stats: dict[str, Any], away_stats: dict[str, Any]
   549→    ) -> dict[str, float]:
   550→        lag = self.league_avg_goals
   551→
   552→        h_gs_home = float(home_stats.get("goals_scored_home", lag))
   553→        h_gs_away = float(home_stats.get("goals_scored_away", lag))
   554→        h_gs = float(home_stats.get("goals_scored", lag))
   555→        h_gc_home = float(home_stats.get("goals_conceded_home", lag))
   556→        h_gc_away = float(home_stats.get("goals_conceded_away", lag))
   557→        h_gc = float(home_stats.get("goals_conceded", lag))
   558→
   559→        a_gs_home = float(away_stats.get("goals_scored_home", lag))
   560→        a_gs_away = float(away_stats.get("goals_scored_away", lag))
   561→        a_gs = float(away_stats.get("goals_scored", lag))
   562→        a_gc_home = float(away_stats.get("goals_conceded_home", lag))
   563→        a_gc_away = float(away_stats.get("goals_conceded_away", lag))
   564→        a_gc = float(away_stats.get("goals_conceded", lag))
   565→
   566→        gd = h_gs - h_gc
   567→
   568→        xg_h = float(home_stats.get("xg_for_home", lag))
   569→        xg_a = float(away_stats.get("xg_for_away", lag))
   570→        xg_diff = xg_h - xg_a
   571→
   572→        return {
   573→            "avg_goals_scored_home": h_gs_home,
   574→            "avg_goals_scored_away": a_gs_away,
   575→            "avg_goals_conceded_home": h_gc_home,
   576→            "avg_goals_conceded_away": a_gc_away,
   577→            "avg_goals_scored": h_gs,
   578→            "avg_goals_conceded": h_gc,
   579→            "goal_difference": gd,
   580→            "xg_for_home": xg_h,
   581→            "xg_for_away": xg_a,
   582→            "xg_diff": xg_diff,
   583→        }

   613→    # ---------- 4. Head-to-head ----------------------------------------
   614→
   615→    def _h2h_features(
   616→        self, h2h_data: list[dict[str, Any]] | None
   617→    ) -> dict[str, float]:
   618→        if not h2h_data:
   619→            return {
   620→                "h2h_home_win_pct": 0.0,
   621→                "h2h_away_win_pct": 0.0,
   622→                "h2h_draw_pct": 0.0,
   623→                "h2h_avg_goals": self.league_avg_goals * 2,
   624→            }
   625→
   626→        n = len(h2h_data)
   627→        home_wins = sum(1 for m in h2h_data if m.get("home_goals", 0) > m.get("away_goals", 0))
   628→        away_wins = sum(1 for m in h2h_data if m.get("home_goals", 0) < m.get("away_goals", 0))
   629→        draws = n - home_wins - away_wins
   630→
   631→        total_goals_list = [m.get("home_goals", 0) + m.get("away_goals", 0) for m in h2h_data]
   632→        avg_goals = _safe_avg(total_goals_list, default=self.league_avg_goals * 2)
   633→
   634→        return {
   635→            "h2h_home_win_pct": _safe_div(home_wins, n),
   636→            "h2h_away_win_pct": _safe_div(away_wins, n),
   637→            "h2h_draw_pct": _safe_div(draws, n),
   638→            "h2h_avg_goals": avg_goals,
   639→        }

   641→    # ---------- 5. Venue splits ----------------------------------------
   642→
   643→    def _venue_features(
   644→        self, home_stats: dict[str, Any], away_stats: dict[str, Any]
   645→    ) -> dict[str, float]:
   646→        lag = self.league_avg_goals
   647→
   648→        # Home team's home record
   649→        hw = int(home_stats.get("home_wins", 0))
   650→        hd = int(home_stats.get("home_draws", 0))
   651→        hl = int(home_stats.get("home_losses", 0))
   652→        h_home_total = hw + hd + hl
   653→        h_home_goals = float(home_stats.get("home_goals", lag))
   654→
   655→        # Away team's away record
   656→        aw = int(away_stats.get("away_wins", 0))
   657→        ad = int(away_stats.get("away_draws", 0))
   658→        al = int(away_stats.get("away_losses", 0))
   659→        a_away_total = aw + ad + al
   660→        a_away_goals = float(away_stats.get("away_goals", lag))
   661→
   662→        return {
   663→            "home_win_pct": _safe_div(hw, h_home_total),
   664→            "away_win_pct": _safe_div(aw, a_away_total),
   665→            "home_goals_avg": h_home_goals,
   666→            "away_goals_avg": a_away_goals,
   667→        }

   669→    # ---------- 6. Market / odds features ------------------------------
   670→
   671→    def _market_features(
   672→        self, odds: dict[str, Any] | None
   673→    ) -> dict[str, float]:
   674→        default_odds = 1.0
   675→        if odds is None:
   676→            return {
   677→                "implied_prob_home": 1.0 / 3.0,
   678→                "implied_prob_draw": 1.0 / 3.0,
   679→                "implied_prob_away": 1.0 / 3.0,
   680→                "overround": 1.0,
   681→            }
   682→
   683→        ho = float(odds.get("home_odds", default_odds))
   684→        do = float(odds.get("draw_odds", default_odds))
   685→        ao = float(odds.get("away_odds", default_odds))
   686→
   687→        # Guard against zero
   688→        ho = max(ho, 1.01)
   689→        do = max(do, 1.01)
   690→        ao = max(ao, 1.01)
   691→
   692→        ip_h = _safe_div(1.0, ho, 1.0 / 3.0)
   693→        ip_d = _safe_div(1.0, do, 1.0 / 3.0)
   694→        ip_a = _safe_div(1.0, ao, 1.0 / 3.0)
   695→        overround = ip_h + ip_d + ip_a
   696→
   697→        return {
   698→            "implied_prob_home": ip_h,
   699→            "implied_prob_draw": ip_d,
   700→            "implied_prob_away": ip_a,
   701→            "overround": overround,
   702→        }

   704→    # ---------- 7. Temporal features -----------------------------------
   705→
   706→    def _temporal_features(
   707→        self,
   708→        match_data: dict[str, Any],
   709→        home_stats: dict[str, Any],
   710→        away_stats: dict[str, Any],
   711→    ) -> dict[str, float]:
   712→        match_date = _parse_date(match_data.get("date")) or datetime.now()
   713→        dow = float(match_date.weekday())
   714→        month = float(match_date.month)
   715→        is_weekend = 1.0 if dow >= 5 else 0.0
   716→
   717→        # Rest days
   718→        h_last = _parse_date(home_stats.get("last_match_date"))
   719→        a_last = _parse_date(away_stats.get("last_match_date"))
   720→
   721→        default_rest = 7.0
   722→        h_rest = default_rest
   723→        a_rest = default_rest
   724→        if h_last:
   725→            delta = (match_date - h_last).days
   726→            h_rest = float(max(delta, 0)) if delta is not None else default_rest
   727→        if a_last:
   728→            delta = (match_date - a_last).days
   729→            a_rest = float(max(delta, 0)) if delta is not None else default_rest
   730→
   731→        return {
   732→            "day_of_week": dow,
   733→            "month": month,
   734→            "is_weekend": is_weekend,
   735→            "days_since_last_match_home": h_rest,
   736→            "days_since_last_match_away": a_rest,
   737→        }

   739→    # ---------- 8. Advanced features -----------------------------------
   740→
   741→    def _advanced_features(
   742→        self, home_stats: dict[str, Any], away_stats: dict[str, Any]
   743→    ) -> dict[str, float]:
   744→        lag = self.league_avg_goals
   745→
   746→        # Attacking / defensive strength (relative to league average)
   747→        h_gs = float(home_stats.get("goals_scored", lag))
   748→        h_gc = float(home_stats.get("goals_conceded", lag))
   749→        a_gs = float(away_stats.get("goals_scored", lag))
   750→        a_gc = float(away_stats.get("goals_conceded", lag))
   751→
   752→        atk_h = _safe_div(h_gs, lag, 1.0)
   753→        atk_a = _safe_div(a_gs, lag, 1.0)
   754→        def_h = _safe_div(h_gc, lag, 1.0)
   755→        def_a = _safe_div(a_gc, lag, 1.0)
   756→
   757→        # Poisson lambda: attack * opponent defence * league average
   758→        poisson_h = atk_h * def_a * lag
   759→        poisson_a = atk_a * def_h * lag
   760→
   761→        # Shot conversion rates
   762→        h_shots = float(home_stats.get("shots_total", 0.0))
   763→        h_sot = float(home_stats.get("shots_on_target", 0.0))
   764→        h_matches = max(float(home_stats.get("matches_played", 1)), 1.0)
   765→        a_shots = float(away_stats.get("shots_total", 0.0))
   766→        a_sot = float(away_stats.get("shots_on_target", 0.0))
   767→        a_matches = max(float(away_stats.get("matches_played", 1)), 1.0)
   768→
   769→        h_conv = _safe_div(h_sot, h_shots, 0.33)
   770→        a_conv = _safe_div(a_sot, a_shots, 0.33)
   771→
   772→        # Clean-sheet percentage
   773→        h_cs = float(home_stats.get("clean_sheets_home", 0.0))
   774→        h_home_m = max(float(home_stats.get("matches_played_home", 1)), 1.0)
   775→        a_cs = float(away_stats.get("clean_sheets_away", 0.0))
   776→        a_away_m = max(float(away_stats.get("matches_played_away", 1)), 1.0)
   777→
   778→        return {
   779→            "attacking_strength_home": atk_h,
   780→            "attacking_strength_away": atk_a,
   781→            "defensive_strength_home": def_h,
   782→            "defensive_strength_away": def_a,
   783→            "poisson_lambda_home": poisson_h,
   784→            "poisson_lambda_away": poisson_a,
   785→            "shot_conversion_rate_home": h_conv,
   786→            "shot_conversion_rate_away": a_conv,
   787→            "clean_sheet_pct_home": _safe_div(h_cs, h_home_m),
   788→            "clean_sheet_pct_away": _safe_div(a_cs, a_away_m),
   789→        }

   794→
   795→    @staticmethod
   796→    def _stats_from_history(
   797→        overall: list[tuple],
   798→        home_only: list[tuple],
   799→        away_only: list[tuple],
   800→        window: int = 10,
   801→    ) -> dict[str, Any]:
   802→        """Derive a stats dict from rolling match history.
   803→
   804→        Each tuple is ``(date, goals_scored, goals_conceded, result_char)``.
   805→        Uses the last *window* entries for form/streak, and full history
   806→        for season averages.
   807→        """
   808→        stats: dict[str, Any] = {}
   809→
   810→        if not overall:
   811→            return stats
   812→
   813→        # Full-season averages (all matches)
   814→        gs_all = [m[1] for m in overall]
   815→        gc_all = [m[2] for m in overall]
   816→        stats["goals_scored"] = _safe_avg(gs_all)
   817→        stats["goals_conceded"] = _safe_avg(gc_all)
   818→        stats["matches_played"] = float(len(overall))
   819→
   820→        # Venue-split averages
   821→        if home_only:
   822→            stats["goals_scored_home"] = _safe_avg([m[1] for m in home_only])
   823→            stats["goals_conceded_home"] = _safe_avg([m[2] for m in home_only])
   824→            stats["home_wins"] = sum(1 for m in home_only if m[3] == "W")
   825→            stats["home_draws"] = sum(1 for m in home_only if m[3] == "D")
   826→            stats["home_losses"] = sum(1 for m in home_only if m[3] == "L")
   827→            stats["home_goals"] = _safe_avg([m[1] for m in home_only])
   828→            stats["matches_played_home"] = float(len(home_only))
   829→            stats["clean_sheets_home"] = float(
   830→                sum(1 for m in home_only if m[2] == 0)
   831→            )
   832→
   833→        if away_only:
   834→            stats["goals_scored_away"] = _safe_avg([m[1] for m in away_only])
   835→            stats["goals_conceded_away"] = _safe_avg([m[2] for m in away_only])
   836→            stats["away_wins"] = sum(1 for m in away_only if m[3] == "W")
   837→            stats["away_draws"] = sum(1 for m in away_only if m[3] == "D")
   838→            stats["away_losses"] = sum(1 for m in away_only if m[3] == "L")
   839→            stats["away_goals"] = _safe_avg([m[1] for m in away_only])
   840→            stats["matches_played_away"] = float(len(away_only))
   841→            stats["clean_sheets_away"] = float(
   842→                sum(1 for m in away_only if m[2] == 0)
   843→            )
   844→
   845→        # Recent form (sliding window)
   846→        recent = overall[-window:] if len(overall) >= window else overall
   847→        stats["recent_results"] = [m[3] for m in recent]
   848→        stats["recent_goals_scored"] = [m[1] for m in recent]
   849→
   850→        return stats
