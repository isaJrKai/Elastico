  1499→

  1505→
  1506→    const factorial = (n: number): number => {
  1507→      if (n <= 1) return 1;
  1508→      let res = 1;
  1509→      for (let i = 2; i <= n; i++) res *= i;
  1510→      return res;
  1511→    };
  1512→    
  1513→    const poissonProb = (k: number, mean: number): number => {
  1514→      return (Math.exp(-mean) * Math.pow(mean, k)) / factorial(k);
